!======================================================================
! examples/vz_tani2023: 谷(2023)の Mettman Ridge CB1 の人工降雨実験(MREX、1992 年 5〜6 月)の再現
!
!   CB1(860 m2、平均勾配約 40°のゼロ次谷)。設定は論文 Ⅳ.1 と表−2(README の対応表)。
!   土層厚は線形分布 0〜200 cm(81 区分)。初期条件は ψ = −100 cm(nest 5-4 の観測)。
!   地表面の境界条件は降雨強度の 80 %(20 % を蒸発散として差し引く。precip_MREX.txt
!   に適用済み)。流出強度は排水から深部浸透 0.3 mm/h を引き、その 1/2 ずつを上流堰・
!   下流堰に配分する(compare_tani2023.py)。図の降雨は約 12 時間ごとの平均の棒。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 60.0                 ! 時間刻み (s)。全セルが VZ で地表流はない
  t0 = 0                    ! 時刻 0 = 1992-05-27 00:00(図−11 の時間軸の左端)
  tt = 777600
  dt_disp = 3600
  dt_file = 777600
  dt_recrd = 600            ! vz_zone.csv・vz_column.csv の記録間隔 (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_precip = 'data/precip_MREX.txt'   ! extract_tani2023.py が生成
  fn_vz = '-'
  dir_data = 'data/grid_mr'  ! make_case.py が生成
  dir_result = 'result_mrex'
  f_check_cfl = 0
/

&list_initial
  f_htype = 0
  h0 = 0.0
/

&list_geoinfo
  nx = 81
  ny = 1
  dx = 1.0                  ! 1 区分 = 1 m2(流出強度は面積に依らない)
  dy = 1.0
  f_ztype = 0
  z0 = 0.0
  f_masktype = 1
  fn_mask = 'mask.txt'
  f_rntype = 0
  rn0 = 0.05
  f_sdtype = 1
  fn_sd = 'sd.txt'          ! 土層厚 (m)。0, 0.025, ..., 2.00
/

&list_bound
/

&list_vz
  fn_vz_zone = 'zone.txt'   ! 区画 1 = CB1
  dt_vz = 60.0
  vz_out_depth = 0.225, 0.475, 0.75, 1.025, 2.0
/

&list_vz_column
  ! MR の土壌の小杉パラメータ(表−2。Ebel et al. 2007 の関数関係に小杉式を当てはめた値)。
  ! Ks = 3.4e-2 cm/s = 1224 mm/h
  vz_ks_mmh(1) = 1224.0
  vz_thetas(1) = 0.500
  vz_thetar(1) = 0.180
  vz_psim(1) = -0.250       ! ψm = −25.0 cm
  vz_sigma(1) = 0.80
  vz_depth(1) = 10.0        ! 長さ 10 m の土壌柱、刻み 2.5 cm、下端は自由排水(Ⅱ.2)
  vz_dz(1) = 0.025
  vz_psi0(1) = -1.0         ! 初期条件: すべての深さで ψ = −100 cm
/

&list_vz_zone
  vz_zone_col(1) = 1, vz_out_kind(1) = 2    ! 排水は系外へ(深部浸透と堰への配分は比較時)
/
