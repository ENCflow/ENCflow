!======================================================================
! sewer_hybrid ラン B: 幹線+枝管の同一インスタンス(sy = 幹線適合)
!
!   目的: 単一の管路連続体層インスタンス内で「幹線1本(セル列)+
!   枝管網(面)」をセル別マップ(cap/cnd/zbot)の不均質だけで表せるか、
!   およびスカラー gwc_sy の歪みを実地に確かめる(設計議論の実証実験)。
!
!   系: 勾配 0.005 の都市斜面(400x210 m, dx=10 m)。全セルに枝管網
!   (D=0.4 m, 埋設 2 m)、j=11 の i=2..38 に幹線(D=1.0 m, 埋設 5 m)。
!   枡は全セル 1 個、幹線末端 i=38 は開口 10 個(吐口の代用)。
!   降雨 60 mm/h x 1 h(系の管内貯留 ~16 mm を大きく超え、
!   サーチャージ・噴出が起きる規模)。地表は不浸透(f_gwvertical=0)。
!
!   ラン比較(パラメータはこのファイルとの差分のみ):
!     A: sy=0.0314(枝管 D=0.4 に整合。幹線の水頭応答は過大 ~2.9倍)
!     B: sy=0.0911(幹線セルに整合。枝管の水頭応答は過小 ~2.9倍)
!     C: 幹線なし(全セル枝管、sy=0.0314)= 幹線の効果の対照
!   検証: 閉領域+降雨なので S_total(終了時) = 積算雨量 60 mm。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 0.02                 ! 時間刻み (s)(管路層の陽解法 dt 上限が支配)
  t0 = 0
  tt = 10800                ! 3 時間(降雨 1 h + 排水 2 h)
  dt_disp = 300             ! 画面表示間隔 (s)
  dt_file = 1800            ! 分布ファイル出力間隔 (s)
  dt_recrd = 60             ! 計測出力間隔 (s)
  fn_geoinfo = '-'
  fn_enc = '-'
  fn_gwflow = '-'
  fn_precip = '-'
  fn_record = '-'
  dir_data = 'data_sewer'
  dir_result = 'result_B'
  f_disp_debug = 1          ! S 列を全有効桁で表示(水収支の照合用)
  f_out_e = 1
  f_out_hg = 1              ! Hgc 場の出力
/

&list_enc
  f_adaptive_runge = 1
/

&list_geoinfo
  nx = 40
  ny = 21
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z.txt'            ! 勾配 0.005 の東向き斜面
  f_rntype = 0
  rn0 = 0.03                ! 市街地相当の粗度
/

&list_bound
/                           ! 全周閉境界(質量保存の検証)

&list_precip
  prtype = 1                ! 一様時系列
  dt_prupdate = 1
  prval(:,1) =  0, 60       ! 60 mm/h
  prval(:,2) = 60, 60
  prval(:,3) = 61,  0       ! 1 時間で終了
/

!======================================================================
! 地下水設定: 管路連続体層のみ(都市プリセット: 不浸透+下水道)
!======================================================================
&list_gwflow
  f_gwvertical = 0          ! 不浸透(都市域)
  f_gwlateral = 0
  f_gwlayer2 = 0
  f_gwconduit = 1
  dt_gwflow = 0             ! 毎ステップ更新
/

&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! sqrt 乱流則
  fn_gwc_cnd = 'cnd_T.txt'  ! 通水能密度(幹線 2.40 / 枝管 0.208 m2/s。
                            !   gen_data.py で管諸元から換算)
  fn_gwc_cap = 'cap_T.txt'  ! 貯留容量(幹線セル 0.091 / 枝管 0.0126 m)
  fn_gwc_bot = 'bot_T.txt'  ! 管底(幹線 z-5 / 枝管 z-2)
  gwc_sy = 0.0911           ! ラン B: 幹線セル(D=1.0 合算)に整合する貯留係数
  gwc_slot_sy = 0.03        ! 疑似スロット(dt 上限との折り合い)
  gwc_sat0 = 0.0            ! 初期は空管
  fn_gwc_inlet = 'inlet.txt' ! 枡 1 個/セル、幹線末端は 10 個(吐口代用)
  gwc_cw = 2.66
  gwc_co = 0.15
  gwc_leak_layer = 0        ! 層間交換なし(都市プリセット)
  gwc_eps_h = 0.1           ! sqrt 則の線形化幅 (m)
/

!======================================================================
! 計測: 幹線末端(吐口)・幹線中間・幹線から遠い枝管域
!======================================================================
&list_record
  pbxytype = 0
  pbxy(:,1) = 38, 11        ! 吐口セル
  pbxy(:,2) = 20, 11        ! 幹線中間
  pbxy(:,3) = 20, 5         ! 幹線から遠い枝管域
  flxytype = 0
  flxy(:,1) = 39, 15, 39, 7 ! 吐口直下流を横切る測線(東向きが正)
/
