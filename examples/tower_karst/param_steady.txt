!======================================================================
! タワーカルスト — 準定常強制(年平均の涵養)で 1 Myr を走らせる構成
!   param.txt(代表豪雨の反復、6 kyr)の係数はそのままに、降雨を年平均の
!   0.228 mm/h の定常に置き換え、morfac を 6e6 に上げて 60 日 = 0.99 Myr の
!   地形時間を得る。豪雨内の湛水の正帰還(morfac の上限の原因)がないので
!   溶食は線形・局所に戻り、大きな morfac が許される。失うもの: 豪雨時の
!   裾の湛水(側方溶食のノッチ)。残るもの: 塔の流出水が裾へ集まって浸透する
!   集中(裸岩の浸透能は年浸透率が豪雨と同じになる 0.0057 mm/h)と、塔の岩面の
!   表面流源、崩落。
! 実行: python3 make_terrain.py --steady
!       ./encflow param_steady.txt > Screen_steady.log
!       python3 plot_results.py param_steady.txt steady | tee analysis_steady.txt
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0: ENC)
  f_govequation = 0         ! 基礎方程式(0: 動的波)
  dt = 10.0                 ! 時間刻み (s)(定常の薄い水膜なので豪雨の 2 s より大きくできる)
  t0 = 0                    ! 開始時刻 (s)
  tt_c = "60d"              ! 終了時刻 = 60 日(地形時間 60 d × 6e6 = 0.99 Myr)
  dt_disp_c = "1h"          ! 画面表示間隔(Log の S_surf が湛水の時系列)
  dt_file_c = "10d"         ! 分布出力間隔 = 10 日(地形時間 164 kyr 毎)
  dt_recrd_c = "1d"         ! プローブ出力間隔(プローブは使わない)
  fn_geoinfo = '-'          ! 地理情報の namelist は本ファイル内
  fn_precip = '-'           ! 降雨の namelist は本ファイル内
  fn_gwflow = '-'           ! 地下水(浸透・管路層)の namelist は本ファイル内
  fn_geomorph = '-'         ! 地形変化の namelist は本ファイル内
  fn_enc = '-'              ! ENC 計算条件は本ファイル内(既定)
  dir_data = '.'            ! 入力データのディレクトリ
  dir_result = 'result_steady' ! 結果ディレクトリ
  f_out_z = 1               ! 時間発展する地盤高 Z0001... を出力
  f_out_h = 1               ! 湛水深 H0001...
  f_out_hg = 1              ! 土層・管路層の貯留 Hg / Hgc
  f_out_hmax = 1            ! 最大湛水深 H9999
  f_check_cfl = 0           ! CFL 検査なし(適応 RK に任せる)
/
&list_enc
  f_adaptive_runge = 1      ! 適応的ルンゲクッタ
/
&list_geoinfo
  lx = 1000                 ! 領域の x 長 (m)
  ly = 1000                 ! 領域の y 長 (m)
  nx = 100                  ! x 方向格子数(Δx = 10 m)
  ny = 100                  ! y 方向格子数(Δy = 10 m)
  f_ztype = 1               ! 地盤高をファイルから読む
  fn_z = 'z.txt'            ! 平野 + 裸岩の丘(make_terrain.py)
  f_rntype = 0              ! 粗度は固定値
  rn0 = 0.05                ! マニング粗度
  f_sdtype = 1              ! 土層厚をファイルから読む
  fn_sd = 'sd.txt'          ! 平野 1 m、裸岩 1e-3 m
  sy0 = 0.2                 ! 比産出率(平野の土層の貯留容量 0.2 m)
/
&list_precip
  prtype = 1                ! 降雨は時系列(一様)
  dt_prupdate = 1           ! 降雨の更新間隔 (min)
  prval(1:2,1) = 0, 0.228   ! (時刻 (min), 強度 (mm/h)): 年平均の雨 2000 mm / 8766 h = 0.228 mm/h を
  prval(1:2,2) = 9999999, 0.228 !   定常に与える(豪雨の反復ではなく年平均の涵養で Myr を走らせる)
/
&list_gwflow
  f_gwvertical = 2          ! 鉛直浸透は Green-Ampt(土層へ。溶食の水量)
  f_gwlateral = 0           ! 土層の側方流なし
  f_gwconduit = 1           ! 管路連続体層 = 土層の水の抜け先(岩盤の亀裂網)
  dt_gwflow = 0             ! 毎ステップ更新
/
&list_gwflow_greenampt
  fn_gw_ksv = 'ksv_steady.txt' ! 飽和透水係数 (mm/h): 平野 40、裸岩 0.0057(年平均の雨の下で豪雨と同じ
                            !   年浸透率 2.5 % = 裸岩の雨のほとんどが裾へ流れ下る。make_terrain.py --steady)
/
&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! sqrt 乱流則(側方通水なしのため未使用)
  gwc_cnd_m2s = 0.0         ! 側方通水なし(各セルの管路層は貯留のみ = 地下へ消える水の受け皿)
  gwc_cap = 2000.0          ! 貯留容量 (m 柱状)(満管にならない)
  gwc_depth = 1000.0        ! 水頭底 = z − 1000 m(水頭が地表へ達しない)
  gwc_sy = 1.0              ! 不圧貯留係数 1
  gwc_slot_sy = 0.001       ! 疑似スロット(未使用)
  gwc_inlet = 0.0           ! 吸込穴なし(取水は浸透能で上限 = 暴走しない)
  gwc_leak_layer = 1        ! 土層(層 1)と交換: 土層の水が亀裂網へ抜ける
  gwc_leak_mmh = 20.0       ! 交換能 20 mm/h(土層の貯留 0.2 m を半日で排水)
/
&list_geomorph
  dt_geomorph = 600.0       ! 地形更新間隔 (s) = 10 分
  morfac = 6.0e6            ! 地形時間の加速係数(1 日 = 6e6 日 = 16.4 kyr。定常強制なので豪雨内の
                            !   湛水の正帰還がなく、溶食は線形・局所 = 大きな morfac が許される)
                            !   1 豪雨の間に 9 m 級になり流れが発散した)
  f_dissol = 1              ! 石灰岩の溶食
  dissol_ceq = 300.0        ! 有効溶解能 (mg/L CaCO3)(湿潤熱帯の土層下の涵養水)
  dissol_rho = 2500.0       ! 岩の密度 (kg/m3)
  dissol_src = 3            ! 浸透量 + 裸岩で発生する流出水 × fsurf(表面流源 = 塔の岩面の溶食)
  dissol_fsurf = 0.3        ! 流出水の溶解能の比(裸岩の雨水 ≈ 土層下の涵養水の 0.3 倍)
  dissol_sdsurf = 0.1       ! 土層がこれより薄いセルを裸岩とみなす (m)
  dissol_sink = 0           ! 吸込穴項なし
  dissol_fres = 0.1         ! 不溶性残渣の体積率(溶けた岩の 1 割が土層に残る)
  dissol_lat = -1.0         ! 側方溶食: 濡れ面の面積比で湛水に接する岩壁へ自動配分
  dissol_dzmax = 0.3        ! 安全弁: 1 セル・1 更新の溶食の上限 (m)(平野は 1 更新 0.027 m、裾の集中は数倍)
  f_rockfall = 1            ! 崩落と崖錐: 限界高さ H_c を超えた自由面(裾を切られた塔の壁)が
                            !   崩れ、岩屑が崖錐 ht(出力 Ht)として裾に積もる
  rf_bedrock = 1            ! 基岩の崩落: 土層のない裸岩の塔(sd = 1e-3 m)も崩れる(溶食の規約と組)
  rf_hc = 10.0              ! 自由面の限界高さ H_c (m)。塔の壁 20 m の半分 = 裾の穴で露出面が 10 m を超えると崩れる
  rf_rc = 10.0              ! 自由面の高さを測る窓の半径 (m)(窓内の最低点との落差 = 1 セル窓)
  rf_phi = 35.0             ! 崖錐の安息角 (deg)
  rf_relax = 0.5            ! 1 更新で緩和する超過高さの割合
  rf_ftalus = 0.3           ! 崖錐に覆われたセルの乾式侵食の倍率(本例題は f_splash なしのため不使用)
  fluv_dzmax = 1.0          ! 1 エッジ・1 更新の崩落の変動上限 (m)
/
