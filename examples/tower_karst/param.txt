!======================================================================
! タワーカルスト(平野に残る裸岩の塔)の形成実験
!
! 現象: 湿潤熱帯のカルストでは、土層・沖積層を経て攻撃性の高くなった水が
!   平野と塔の裾を速く溶かし(裾のノッチ・足元の洞)、塔は裸岩で雨水を
!   すぐ捨てるので溶け残る。急になった岩壁は崩落して後退し、岩屑が裾に
!   積もる。平野が下がるほど塔の比高が増す(fenglin)。
! 設計: 平野(z ≈ 0、土層 1 m、Green-Ampt 40 mm/h)に裸岩の丘(高さ 40 m・
!   半径 90 m の釣鐘形 × 4 + 小丘 2。土層 1e-3 m、浸透能 1 mm/h)を置き、
!   代表豪雨 40 mm/h × 1 h を 1 日周期で反復する。丘の雨はほぼ全量が表面流
!   となって裾に流れ下り、平野(特に裾のセル)で浸透する。溶食は浸透量に
!   比例(dissol_src = 1)するので平野と裾が下がり、裾の湛水に接する岩壁は
!   側方溶食(自動配分)で削られ、急になった壁は崩落(φ_rock = 45°)で
!   岩屑(土層)として裾へ落ちる。吸込穴はなく、取水は浸透能で上限が
!   決まるため豪雨内の正帰還が暴走しない → morfac = 1e4 が使える。
! 地形時間: 1 日 = 1e4 豪雨(年 50 回なら 200 年)、60 日 = 12 kyr。
!   平野の面的溶食 ≈ k_d(c_eq 300 mg/L → 1.2e-4)× 40 mm × 1e4 /日 ≈ 5 cm/日
!   = 3 m/12 kyr(≈ 250 mm/kyr: 熱帯カルストの文献域の上端)。
! 期待される挙動: 平野が一様に下がり、丘の裾の環(集水)が速く下がって
!   堀(moat)ができる。裾の急化 → 崩落 → 壁の後退 → 岩屑の裾野。塔の
!   頂はほぼ残り、比高が増す。小丘は先に消える(fengcong → fenglin)。
! 時間: dt = 2 s、地形更新 10 分毎。約 1〜2 時間(4 スレッド)。
! 実行: python3 make_terrain.py && ./encflow param.txt > Screen.log
!======================================================================
&list_sysparam

  f_govequation = 0         ! 基礎方程式(0: 動的波)

  t0 = 0                    ! 開始時刻 (s)
  tt_c = "60d"              ! 終了時刻 = 60 日(地形時間 12 kyr)
  t_cycle_c = "1d"          ! 降雨の反復周期 = 1 日(代表豪雨 1 回)
  dt_disp_c = "1h"          ! 画面表示間隔(Log の S_surf が湛水の時系列)
  dt_file_c = "10d"         ! 分布出力間隔 = 10 日(地形時間 2 kyr 毎)
  dt_recrd_c = "1d"         ! プローブ出力間隔(プローブは使わない)
  fn_geoinfo = '-'          ! 地理情報の namelist は本ファイル内

  fn_geomorph = '-'         ! 地形変化の namelist は本ファイル内
  fn_enc = '-'              ! ENC 計算条件は本ファイル内(既定)
  dir_data = '.'            ! 入力データのディレクトリ
  dir_result = 'result'     ! 結果ディレクトリ
  f_out_z = 1               ! 時間発展する地盤高 Z0001... を出力

  f_out_hg = 1              ! 土層・管路層の貯留 Hg / Hgc

  f_check_cfl = 0           ! CFL 検査なし(適応 RK に任せる)

  f_adaptive_runge = 1      ! 適応的ルンゲクッタ
/
&list_geoinfo
  lx = 1000                 ! 領域の x 長 (m)
  ly = 1000                 ! 領域の y 長 (m)
  nx = 100                  ! x 方向格子数(Δx = 10 m)
  ny = 100                  ! y 方向格子数(Δy = 10 m)
  f_ztype = 1               ! 地盤高をファイルから読む
  fn_z = 'z.txt'            ! 平野 + 裸岩の丘(make_terrain.py)
  f_rntype = 0              ! 粗度は固定値
  rn0 = 0.05                ! マニング粗度
  f_sdtype = 1              ! 土層厚をファイルから読む
  fn_sd = 'sd.txt'          ! 平野 1 m、裸岩 1e-3 m

  prval(1:2,3) = 60, 0      ! 止む(以後 1 日周期で反復 = t_cycle_c)

  dt_gwflow = 0             ! 毎ステップ更新
/
&list_gwflow_greenampt
  fn_gw_ksv = 'ksv.txt'     ! 飽和透水係数 (mm/h): 平野 40、裸岩 1

  gwc_cap = 2000.0          ! 貯留容量 (m 柱状)(満管にならない)
  gwc_depth = 1000.0        ! 水頭底 = z − 1000 m(水頭が地表へ達しない)
  gwc_sy = 1.0              ! 不圧貯留係数 1
  gwc_slot_sy = 0.001       ! 疑似スロット(未使用)
  gwc_inlet = 0.0           ! 吸込穴なし(取水は浸透能で上限 = 暴走しない)
  gwc_leak_layer = 1        ! 土層(層 1)と交換: 土層の水が亀裂網へ抜ける
  gwc_leak_mmh = 20.0       ! 交換能 20 mm/h(土層の貯留 0.2 m を半日で排水)

&list_geomorph

  morfac = 1.0e4            ! 地形時間の加速係数(1 日 = 1e4 豪雨 = 200 年)
  f_dissol = 1              ! 石灰岩の溶食
  dissol_ceq = 300.0        ! 有効溶解能 (mg/L CaCO3)(湿潤熱帯の土層下の涵養水)
  dissol_rho = 2500.0       ! 岩の密度 (kg/m3)
  dissol_src = 1            ! 面的溶食の水量 = 浸透量
  dissol_sink = 0           ! 吸込穴項なし
  dissol_fres = 0.1         ! 不溶性残渣の体積率(溶けた岩の 1 割が土層に残る)
  dissol_lat = -1.0         ! 側方溶食: 濡れ面の面積比で湛水に接する岩壁へ自動配分
  f_rockfall = 1            ! 岩盤の崩落: 基岩面が隣より tan60° × 距離を超えて高い分を岩屑に
  rockfall_phi = 45.0       ! 岩盤の安定勾配角 (deg)。Δx = 10 m では tanφ × 10 m = 10 m を超える段差で崩れる
                            !   (60° では 17 m の段差が要り、裾の切り込みでは届きにくい)
  rockfall_relax = 0.5      ! 1 更新で緩和する超過高さの割合
  rockfall_bulk = 1.5       ! 岩屑のかさ比
  fluv_dzmax = 1.0          ! 1 セル・1 更新の崩落量の上限 (m)
/
