!======================================================================

!======================================================================

!======================================================================
! 流木: 上流から来る流木の供給を、上流の区間(x = 300〜600 m)の河道に置いた立木ストックで
!   代用する。水深が 2 m、流速が 1 m/s を超えると(流量 95 m3/s 前後から)流木になる。
!   橋の 4 セルが捕捉セル。捕捉率と集積の抵抗の係数は与件・校正値(README.md)
!======================================================================
&list_driftwood
  fn_dwstock = 'stock.txt'  ! 立木ストック 0.03 m3/m2(列 61〜120 の河道。180 m3)
  dw_dlog = 0.3             ! 流木の代表直径 (m)
  dw_sglog = 0.5            ! 材比重 → 喫水 0.15 m
  dw_wrec = 3.0e-5          ! 水理的流失のレート (m/s。0.03 m3/m2 が 1000 秒で全量)
  dw_hrec = 2.0             ! 流失が始まる水深 (m)
  dw_vrec = 1.0             ! 流失が始まる流速 (m/s)
  dw_wstop = 0.01           ! 接地堆積のレート (m/s)
  dw_vstop = 0.0            ! 流速による堆積はなし(喫水より浅い所で止まる)
  fn_dwtrap = 'trap_0.3.txt' ! 捕捉率 E = 0.3(橋の河道の 4 セル。ほかは 0)
  dw_jamgrow = 0.09         ! 集積が進むほど捕捉率を上げ、0.09 m3/m2(基本のケースと同じ値)に達したセルは、届いた流木をすべて止める
  f_dwjam = 1               ! 集積の抵抗を流れに作用させる(dw_jamcd = 1.0 は既定)
  dw_jamphi = 0.2           ! 集積体の充填率(既定 0.3)
/
