!======================================================================
! levee_breach: 解像堤防(DEM の標高で表した堤防)を規定底面運動で破堤させる
!   直線河道(幅 50 m)+ 幅 20 m・高さ 3 m の堤防 + 堤内地の理想化地形に
!   洪水流量(100 → 700 m3/s)を流し、水位がピークに達した 70 分から 90 分に
!   かけて堤防セル 5 行 × 2 列を堤内地の地盤高まで下げる(中央の行が 10 分で
!   堤底に達し、両側へ拡幅)。破堤のスケジュールは make_breach.py が
!   fn_bedmotion の入力(窓付きの累積変位ファイルとスナップショット表)に
!   変換する。組み方の理由・結果の図表は README.md(機能の説明は
!   users_guide/bedmotion.md)。
!
!   実行:  make && ./encflow param.txt
!          ./encflow param_vwall.txt     (比較: 仮想壁面 + &list_channel_breach)
!          python3 plot_results.py       → figs/*.png と数表
!   見どころ(result/):
!     - probes/probe0003.csv の z 列  破堤セルの地盤高の時刻歴(堤防が下がる)
!     - fluxes/flux0001.csv           破堤口を通る流量
!     - H9999                         堤内地の最大浸水深
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.5                  ! 時間刻み (s)(√(gh) ≈ 7 m/s、Δx = 10 m)
  t0 = 0
  tt = 10800                ! 3 時間
  dt_disp = 600
  dt_file = 600
  dt_recrd = 60
  fn_geoinfo = '-'
  fn_enc = '-'
  fn_boundary = '-'
  fn_record = '-'
  fn_bedmotion = '-'        ! 規定底面運動 = 破堤モデル
  dir_result = 'result'
  f_out_h = 1
  f_out_z = 1               ! 破堤後の地形 Z0001…
  f_out_hmax = 1
  f_out_vv = 1
  f_check_cfl = 0
/

&list_enc
  f_exflux_reduction = 1    ! 乾床前縁(堤内地への浸水先端)の定石
                            ! 移流は既定(f_advection_scheme = 3)。2026-10-18 以前のコードは
                            !   この河道で 5 分周期の流量振動を自励したため 1 にしていた(README「注意」)
/

!======================================================================
! 地理情報(dx = 10 m。make_terrain.py が生成。堤防は z の標高そのもの)
!======================================================================
&list_geoinfo
  nx = 120
  ny = 80
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z.txt'
  f_rntype = 1
  fn_rn = 'rn.txt'          ! 河道 0.03、堤内地 0.05
/

!======================================================================
! 境界: 北辺の河道セル 5 つから流入(洪水波形)、南辺の河道出口 5 セルは
!   等流水深の水位(Manning、2 次元計算なので側壁摩擦なし R = h: B 50 m、
!   n 0.03、S 1/1000 で Q 100 → 1.47 m、700 → 4.72 m。河床 −3.795 m)を
!   流量波形に同期して規定、堤内地の南辺・東辺は自由流出
!======================================================================
&list_bound_edge
  f_bc_s = 1
  f_bc_e = 1
/
&list_bound_stage
  stage_cell(:,1,1) = 16, 80
  stage_cell(:,2,1) = 17, 80
  stage_cell(:,3,1) = 18, 80
  stage_cell(:,4,1) = 19, 80
  stage_cell(:,5,1) = 20, 80
  stage_val(:,1,1) =   0, -2.3    ! (時刻 min, 水位 m)
  stage_val(:,2,1) =  30, -2.3
  stage_val(:,3,1) =  60,  0.9
  stage_val(:,4,1) = 120,  0.9
  stage_val(:,5,1) = 150, -2.3
  stage_val(:,6,1) = 180, -2.3
/
&list_bound_inflow
  inflow_cell(:,1,1) = 16, 1
  inflow_cell(:,2,1) = 17, 1
  inflow_cell(:,3,1) = 18, 1
  inflow_cell(:,4,1) = 19, 1
  inflow_cell(:,5,1) = 20, 1
  inflow_val(:,1,1) =   0, 100.0  ! (時刻 min, 流量 m3/s)
  inflow_val(:,2,1) =  30, 100.0
  inflow_val(:,3,1) =  60, 700.0  ! ピーク(等流水深 ≈ 4.7 m、天端の 1.3 m 下)
  inflow_val(:,4,1) = 120, 700.0
  inflow_val(:,5,1) = 150, 100.0
  inflow_val(:,6,1) = 180, 100.0
  inflow_dist(1) = 2
/

!======================================================================
! 規定底面運動 = 破堤スケジュール(make_breach.py の出力)
!   python3 make_breach.py --z z.txt --cols 21 22 --rows 48 52 --floor-col 23 \
!       --t0 70 --t1 90 --tdeep 10 --out bm
!======================================================================
&list_bedmotion
  fn_bmlist = 'bm/bmlist.txt'
/

!======================================================================
! 計測: プローブ(1 河道 上流 j=40、2 河道 破堤口前 j=50、3 破堤セル (21,50)、
!   4 堤内地 法尻 (25,50)、5 堤内地 中央 (60,50)、6 堤内地 遠方 (100,50))
!   測線(1 列 23 の全行(堤内地へ渡る全流量。正が東向き)、2 河道 下流 j=70、3 河道 上流 j=30)
!======================================================================
&list_record
  pbxytype = 0
  pbxy(:,1) = 18, 40
  pbxy(:,2) = 18, 50
  pbxy(:,3) = 21, 50
  pbxy(:,4) = 25, 50
  pbxy(:,5) = 60, 50
  pbxy(:,6) = 100, 50
  flxytype = 0
  flxy(:,1) = 23, 80, 23, 1 
  flxy(:,2) = 14, 70, 22, 70
  flxy(:,3) = 14, 30, 22, 30
/
