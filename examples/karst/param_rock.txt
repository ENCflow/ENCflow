!======================================================================
! カルスト地形の形成実験 — 側方溶食+岩盤の崩落(param_lat.txt との対照)
!   param_lat.txt(側方溶食の自動配分)に岩盤の崩落 f_rockfall を加える。
!   基岩面が隣接セルの地表より tan60° × 距離を超えて高い岩壁は、超過分の
!   半分を 1 更新ごとに岩屑(受け手の土層 sd。かさ比 1.5)として落とす。
!   窪みの壁が崩れて漏斗形になり、床は岩屑に覆われる(developer.md §87、
!   docs/karst_plan.md §2.6)。
! 実行: ./encflow param_rock.txt > Screen_rock.log(約 1 時間)
!       python3 plot_results.py param_rock.txt | tee analysis_rock.txt
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0: ENC)
  f_govequation = 0         ! 基礎方程式(0: 動的波)
  dt = 2.0                  ! 時間刻み (s)
  t0 = 0                    ! 開始時刻 (s)
  tt_c = "60d"              ! 終了時刻 = 60 日(地形時間 1.5 kyr)
  t_cycle_c = "1d"          ! 降雨の反復周期 = 1 日(代表豪雨 1 回)
  dt_disp_c = "1h"          ! 画面表示間隔(Log の S_surf が湛水の時系列)
  dt_file_c = "10d"         ! 分布出力間隔 = 10 日(地形時間 250 年毎に Z・H)
  dt_recrd_c = "1d"         ! プローブ出力間隔(プローブは使わない)
  fn_geoinfo = '-'          ! 地理情報の namelist は本ファイル内
  fn_precip = '-'           ! 降雨の namelist は本ファイル内
  fn_gwflow = '-'           ! 地下水(浸透・管路層)の namelist は本ファイル内
  fn_geomorph = '-'         ! 地形変化の namelist は本ファイル内
  fn_enc = '-'              ! ENC 計算条件は本ファイル内(既定)
  dir_data = '.'            ! 入力データのディレクトリ
  dir_result = 'result_rock' ! 結果ディレクトリ(側方溶食+岩盤の崩落)
  f_out_z = 1               ! 時間発展する地盤高 Z0001... を出力
  f_out_h = 1               ! 湛水深 H0001...(出力時刻 = 豪雨開始直前 = ほぼ排水済み)
  f_out_hg = 1              ! 土層・管路層の貯留 Hg / Hgc
  f_out_hmax = 1            ! 最大湛水深 H9999
  f_check_cfl = 0           ! CFL 検査なし(適応 RK に任せる)
/
&list_enc
  f_adaptive_runge = 1      ! 適応的ルンゲクッタ
/
&list_geoinfo
  lx = 1000                 ! 領域の x 長 (m)
  ly = 1000                 ! 領域の y 長 (m)
  nx = 100                  ! x 方向格子数(Δx = 10 m)
  ny = 100                  ! y 方向格子数(Δy = 10 m)
  f_ztype = 1               ! 地盤高をファイルから読む
  fn_z = 'z.txt'            ! 起伏のある石灰岩台地(make_terrain.py)
  f_rntype = 0              ! 粗度は固定値
  rn0 = 0.05                ! マニング粗度(草地・裸岩)
  sd0 = 0.5                 ! 土層厚 (m)(テラロッサ)
  sy0 = 0.2                 ! 比産出率(土層の貯留容量 0.1 m)
/
&list_precip
  prtype = 1                ! 降雨は時系列(一様)
  dt_prupdate = 1           ! 降雨の更新間隔 (min)
  prval(1:2,1) = 0, 40      ! (時刻 (min), 強度 (mm/h)): t = 0 に 40 mm/h
  prval(1:2,2) = 59.9, 40   ! 1 時間続く
  prval(1:2,3) = 60, 0      ! 止む(以後 1 日周期で反復 = t_cycle_c)
  prval(1:2,4) = 9999999, 0
/
&list_gwflow
  f_gwvertical = 2          ! 鉛直浸透は Green-Ampt(土層へ。面的溶食の水量)
  f_gwlateral = 0           ! 土層の側方流なし
  f_gwconduit = 1           ! 管路連続体層 = 洞窟・亀裂網(吸込穴の受け皿)
  dt_gwflow = 0             ! 毎ステップ更新
/
&list_gwflow_greenampt
  gw_ksv_mmh = 10.0         ! 飽和透水係数 10 mm/h(< 40 mm/h → 3/4 は表面流となり窪みへ)
/
&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! sqrt 乱流則(側方通水なしのため未使用)
  gwc_cnd_m2s = 0.0         ! 側方通水なし(各セルの管路層は貯留のみ = 地下へ消える水の受け皿)
  gwc_cap = 2000.0          ! 貯留容量 (m 柱状)(満管にならない = 地下へ消える水の受け皿。
                            !   大きな吸込穴は 1 日に数 m を飲み込む)
  gwc_depth = 1000.0        ! 水頭底 = z − 1000 m(水頭が計算中に地表へ達しない深さ)
  gwc_sy = 1.0              ! 不圧貯留係数 1(空洞 = 貯留 1 m で水頭 1 m。小さいと吸込穴の水頭が
                            !   地表に達して流入が止まり、窪みの湛水が残る)
  gwc_slot_sy = 0.001       ! 疑似スロット貯留係数(未使用)
  gwc_inlet = 1.0e-4        ! 吸込穴の密度 1e-4 個/m2(1 ha に 1 個 = 1 セルに 0.01 個)
  gwc_cw = 2.66             ! 堰係数(湛水深 0.1 m で 1 個あたり 0.084 m3/s)
  gwc_co = 0.15             ! オリフィス係数(満管時のみ)
  gwc_leak_layer = 1        ! 土層(層 1)と交換: 土層の水がエピカルスト経由で洞窟へ抜ける
  gwc_leak_mmh = 10.0       ! 交換能 10 mm/h(土層の貯留 0.1 m を約半日で排水)
/
&list_geomorph
  dt_geomorph = 600.0       ! 地形更新間隔 (s) = 10 分
  morfac = 1000.0           ! 地形時間の加速係数(1 日 = 1000 豪雨 = 25 年。上限の目安は本文)
  f_dissol = 1              ! 石灰岩の溶食
  dissol_ceq = 100.0        ! 有効溶解能 (mg/L CaCO3)(土壌 CO2 を経た涵養水)
  dissol_rho = 2500.0       ! 岩の密度 (kg/m3)
  dissol_src = 1            ! 面的溶食の水量 = 浸透量
  dissol_sink = 1           ! 吸込穴流入項(ドリーネ)
  dissol_fsink = 1.0        ! 吸込穴流入水の溶解能の比(土層を経た水と同じとみなす)
  dissol_fres = 0.1         ! 不溶性残渣の体積率(溶けた岩の 1 割が土層に残る)
  fn_dissol = 'sol.txt'     ! 可溶率(縁の尾根 0、内部 1)
  dissol_lat = -1.0         ! 側方溶食: 湛水の溶解能を濡れ面の面積比 hΣL/(hΣL + Δx²) で
                            !   水面より高い隣接セル(岩壁)へ自動配分(湛水 5 m なら約 0.8)
  f_rockfall = 1            ! 岩盤の崩落: 基岩面が隣より tan60° × 距離を超えて高い分を岩屑として落とす
  rockfall_phi = 60.0       ! 岩盤の安定勾配角 (deg)
  rockfall_relax = 0.5      ! 1 更新で緩和する超過高さの割合
  rockfall_bulk = 1.5       ! 岩屑のかさ比
  fluv_dzmax = 1.0          ! 1 セル・1 更新の崩落量の上限 (m)
/
