!======================================================================
! coastal_compound_flood: 海は堤防を越えていないのに、街が浸水する(基本ケース compound)
!   豪雨 + 高潮 + 高い地下水位が重なる複合浸水。高潮は防潮堤(+3.5 m)を越えないが、
!   潮位が低地の地盤より高い間は、吐口のフラップが閉じて下水道の水が海へ出られない。
!   斜面に降った雨は下水道を通って低地へ集まり、枡から噴き出して低地にたまる。
!   地下水位が高いと雨が土にしみ込めず、流れ出る水が増える。
!   町: 800 m × 500 m(dx = 20 m)。西から海、防潮堤、幅 120 m の低地(+0.9 m。窪地の底
!   +0.64 m)、勾配 1/250 の斜面(東端 +3.6 m)。下水道は全セルの枝管網 + 中央の幹線 +
!   海側の端のポンプ井(吐口と機場)。雨は 3 時間で 100 mm(ピーク 50 mm/h = 計画降雨)、
!   高潮は 2.5 時間で +2.0 m まで上がり 2.5 時間で下がる。両者のピークは 5.5 時間に重なる。
!   このファイルの 3 つの要因を 1 つずつ外したケースと、雨の時刻をずらしたケースの設定は、
!   gen_data.py がこのファイルの行を差し替えて作る(param_<ケース>.txt)。
!   詳細と結果は README.md。入力マップ data/ は gen_data.py が作る
!======================================================================
&list_sysparam
  dt = 0.5                  ! 時間刻み (s)
  tt = 43200                ! 計算終了時刻 (s)。12 時間
  dt_disp = 300             ! 画面・Log の表示の間隔 (s)。S 列(水収支)の時系列に使う
  dt_file = 900             ! 分布ファイルの出力の間隔 (s)。15 分
  dt_recrd = 60             ! プローブの出力の間隔 (s)
  fn_geoinfo = '-'          ! 地理情報(このファイルの &list_geoinfo)
  fn_tide = '-'             ! 潮位
  fn_precip = '-'           ! 降水
  fn_gwflow = '-'           ! 地下水(土への浸透と、下水道 = 管路連続体層)
  fn_structure = '-'        ! 構造物(機場のポンプ)
  fn_record = '-'           ! 計測(プローブ)
  dir_data = 'data'         ! 入力マップのディレクトリ
  dir_result = 'result_compound'   ! 結果のディレクトリ
  f_disp_debug = 1          ! S 列を全有効桁で表示する(水収支の照合用)
  f_out_e = 1               ! 水位 E の分布を出力する
  f_out_hg = 1              ! 地下の貯留 Hg(土)・Hgc(管路)の分布を出力する
/

!======================================================================
! 地理情報(入力マップは gen_data.py が作る)
!======================================================================
&list_geoinfo
  nx = 43                   ! 東西のセル数(西の 3 列が海、残り 40 列が陸)
  ny = 25                   ! 南北のセル数
  dx = 20.0                 ! 格子間隔 (m)。東西
  dy = 20.0                 ! 同上(南北)
  f_ztype = 1               ! 地盤高をファイルで与える
  fn_z = 'z.txt'            ! 地盤高 (m)
  fn_sw = 'sw.txt'          ! 海域マスク(西の 3 列。潮位を与えるセル)
  rn0 = 0.04                ! 粗度係数(建物を含む市街地の等価な値)
  seawall0 = 3.5            ! 防潮堤の天端 (m)。海に接する陸のセルの海側の辺に壁が立つ
  sd0 = 1.0                 ! 土層厚 (m)。浸透した水が入る層
  sy0 = 0.2                 ! 比湧水量(土層にためられる水 = sd0 × sy0 = 200 mm)
/

!======================================================================
! 潮位(分, m)。高潮の偏差だけを与える(天文潮と波は扱わない)。
! 3.0 時間から上がり始め、5.5 時間に +2.0 m、8.0 時間に平常の 0 m へ戻る
!======================================================================
&list_tide
  titype = 2                ! 一様な時系列
  hsea0 = 1.0               ! 海のセルの見かけの水柱の厚さ (m)
  dt_tiupdate = 1           ! 潮位の更新の間隔 (分)
  tival(1:2,1) =    0, 0.0
  tival(1:2,2) =  180, 0.0
  tival(1:2,3) =  330, 2.0
  tival(1:2,4) =  480, 0.0
/

!======================================================================
! 降水(分, mm/h)。3 時間で 100 mm。ピーク 50 mm/h の中央が高潮のピーク(330 分)に重なる
!======================================================================
&list_precip
  prtype = 1                ! 一様な時系列
  dt_prupdate = 1           ! 降雨の更新の間隔 (分)
  prval(:,1) =  240,  0.0
  prval(:,2) =  300, 50.0
  prval(:,3) =  360, 50.0
  prval(:,4) =  420,  0.0
/

!======================================================================
! 地下水: 土への浸透(Green-Ampt)と、下水道(管路連続体層)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! 鉛直の浸透は Green-Ampt
  f_gwlateral = 0           ! 土の中の横の流れは解かない(半日の出来事では小さい)
  f_gwconduit = 1           ! 管路連続体層(下水道)
  dt_gwflow = 0             ! 毎ステップ更新する
  gw_sat0 = 0.9             ! 土層の初期の飽和度。0.9 = 地下水面が地表の 0.1 m 下(残りの空き 20 mm)
/

&list_gwflow_greenampt
  gw_ksv_mmh = 20.0         ! 町の平均の浸透能 (mm/h)。計画降雨 50 mm/h の 4 割がしみ込む値
  gw_psif = 0.0             ! 湿潤前線の吸引は使わない(浸透能は一定)
/

&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! 管の中の流れは √勾配に比例(乱流の管路)
  fn_gwc_cnd = 'cnd.txt'    ! 通水能密度 (m2/s)。枝管網 0.104、幹線 11.7(1 セル幅の線の換算)
  fn_gwc_cap = 'cap.txt'    ! 貯留容量 (m)。枝管網 0.0126、幹線のセル 0.138、ポンプ井 0.625
  fn_gwc_bot = 'bot.txt'    ! 管底の標高 (m)。枝管網は地表 − 1.5 m、幹線は地表 − 2.5 m
  fn_gwc_sy = 'sy.txt'      ! 満管までの貯留係数(容量 ÷ 管の高さ。ポンプ井は水面の面積 ÷ セルの面積)
  fn_gwc_slot_sy = 'slot.txt' ! 満管の後の疑似スロットの貯留係数(枝管網は sy/20、幹線は sy/10)
  fn_gwc_inlet = 'inlet.txt'  ! 枡の密度 (個/m2)。全セル 0.01(1 セルに 4 個)
  fn_gwc_outfall = 'outfall.txt' ! 吐口の Cd·A (m2)。ポンプ井のセルに 3.0。海に隣り合うので受け水頭は潮位
  gwc_eps_h = 0.02          ! √則を直線でつなぐ水頭差 (m)。勾配 1/1000 より緩い所が直線になる
  gwc_nsubmax = 100         ! 管の中の流れを細かく刻む回数の上限(既定)
/

!======================================================================
! 機場: ポンプ井の水を防潮堤の外へ出す(吐口のセルを書かない = 域外へ排水)
!======================================================================
&list_struct_pump
  pump_in_cell(1:2,1,1) = 4, 13      ! 取水のセル = ポンプ井
  f_pump_src(1) = 1                  ! 管路連続体層から汲む
  f_pump_ref(1) = 0                  ! 運転の基準は管路の水頭 (m)
  pump_rule(1:2,1,1) = -1.2, 0.0     ! ポンプ井の水位が −1.2 m より低ければ止まる
  pump_rule(1:2,2,1) = -0.8, 1.5     ! −0.8 m より高ければ全力の 1.5 m3/s(潮位によらない)
/

!======================================================================
! 計測(プローブ。セル番号)
!======================================================================
&list_record
  pbxytype = 0              ! プローブの位置をセル番号 (ix, iy) で与える
  pbxy(:,1) = 4, 13         ! ポンプ井のセル
  pbxy(:,2) = 7, 7          ! 窪地の底
  pbxy(:,3) = 8, 13         ! 低地の幹線の上
  pbxy(:,4) = 24, 7         ! 斜面の中ほど
/
