!======================================================================
! tsunami_meander: 蛇行水路(蛇行度 1.10)の波状段波を 1 層非静水圧で計算する
!   小林ら (2019・2021) の実験水路(蛇行度 1.10、幅 0.80 m、蛇行長 7.50 m、最大偏角 35°、勾配 1/600。上流水槽つき)。
!   ゲートの上流に水深 0.285 m、下流に 0.147 m の静水を置き、t = 0 にゲートが開いたとして(初期水位の段差)、
!   波状段波(段波のフルード数 1.32)を進ませる。1 層非静水圧補正を入れた計算で、段波の先端が波列に分かれる。
!   格子(2 cm)・地形・初期水位・プローブは make_init.py が書いた
!   geoinfo_s110.txt・initial_s110_ub.txt・record_s110.txt。
!   計測との比較と図は plot_results.py(README)。dir_result = result_ub_nh
!======================================================================
&list_sysparam
  dt = 0.002                ! 時間刻み (s)。クーラン数 0.2〜0.3
  tt = 6.5                  ! 計算終了時刻 (s)。段波は下流端に届かない
  dt_disp = 0.5             ! 画面・Log の表示間隔 (s)
  dt_file = 0.5             ! 分布(水位)の出力間隔 (s)
  dt_recrd = 0.01           ! プローブの出力間隔 (s)
  fn_geoinfo = 'geoinfo_s110.txt'
  fn_initial = 'initial_s110_ub.txt'
  fn_enc = '-'
  fn_record = 'record_s110.txt'
  dir_result = 'result_ub_nh'
  f_out_h = 0               ! 水深の分布は出さない
  f_out_e = 1               ! 水位の分布(水面の形の図に使う)
  f_disp_vv = 1             ! 画面に最大流速を出す
/

&list_enc
  f_nonhydrostatic = 1      ! 1 層非静水圧補正(砕波スイッチ f_nh_breaking は使わない。README の「砕波スイッチ」)
  nh_hmin = 0.01            ! これより浅いセルは静水圧のまま (m)。既定の 0.1 m は実規模の値で、この水路の水深と同じ大きさなので下げる
/
