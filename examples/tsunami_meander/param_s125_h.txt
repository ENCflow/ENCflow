!   ゲートの上流に水深 0.285 m、下流に 0.147 m の静水を置き、t = 0 にゲートが開いたとして(初期水位の段差)、
!   波状段波(段波のフルード数 1.32)を進ませる。静水圧(浅水流方程式)の計算で、段波は不連続のまま進む。
!   格子(2 cm)・地形・初期水位・プローブは make_init.py が書いた
!   geoinfo_s125.txt・initial_s125_ub.txt・record_s125.txt。
!   計測との比較と図は plot_results.py(README)。dir_result = result_s125_h

  dt = 0.002                ! 時間刻み (s)。クーラン数 0.2〜0.3
  tt = 6.5                  ! 計算終了時刻 (s)。段波は下流端に届かない
  dt_disp = 0.5             ! 画面・Log の表示間隔 (s)
  dt_file = 0.5             ! 分布(水位)の出力間隔 (s)
  dt_recrd = 0.01           ! プローブの出力間隔 (s)
  fn_geoinfo = 'geoinfo_s125.txt'
  fn_initial = 'initial_s125_ub.txt'
  fn_enc = '-'
  fn_record = 'record_s125.txt'
  dir_result = 'result_s125_h'
  f_out_h = 0               ! 水深の分布は出さない
  f_out_e = 1               ! 水位の分布(水面の形の図に使う)
  f_disp_vv = 1             ! 画面に最大流速を出す
/

&list_enc
  f_hcap_upwind = 2         ! エッジ水深を上流側の水深にする(段波の先端の 1 セル幅の跳ねを抑える)
