!======================================================================
! source_area: 全ケース共通の設定(降雨・計測・流域の出口)。単独では実行しない
!   各 param の fn_precip / fn_record / fn_boundary がこのファイルを指す
!   (グループは名前で読まれるので 1 ファイルにまとめられる)。
!   セル番号は make(make_catchment.py)が切り出した格子(192 × 143)のもの。
!   出口を変えたら data/catchment.txt の outlet_ij・gauge_ij に合わせる
!======================================================================

!----------------------------------------------------------------------
! 降雨: 10 mm/h × 12 時間 = 120 mm を、36 時間あけて 2 回(全域一様)。
!   1 回目(0〜12 h)は乾いた土に、2 回目(48〜60 h)は 1 回目の水が
!   残る土に降る。時系列は点の間を直線で補間するので、立ち上がりと
!   終わりを 1 分の傾斜にしている
!----------------------------------------------------------------------
&list_precip
  prtype = 1                ! 一様な時系列
  prval(:,1) =    0, 10.0   ! (時刻 min, 雨量強度 mm/h)
  prval(:,2) =  720, 10.0
  prval(:,3) =  721,  0.0
  prval(:,4) = 2880,  0.0
  prval(:,5) = 2881, 10.0
  prval(:,6) = 3600, 10.0
  prval(:,7) = 3601,  0.0
/

!----------------------------------------------------------------------
! 計測: 出口のすぐ上流の谷筋のセル(測るセル)の水深と、その列を横切る測線の
!   流量。出口セルへ入る水はすべてこの列から来るので、測線の流量が流域からの
!   流出の全量になる
!----------------------------------------------------------------------
&list_record
  pbxytype = 0              ! セル番号 (i, j)
  pbxy(:,1) = 189, 33       ! 測るセル(出口セルの西隣)
  flxytype = 0              ! 両端セルの番号 (右岸 i, j, 左岸 i, j)。東向きが正
  flxy(:,1) = 189, 34, 189, 30
/

!----------------------------------------------------------------------
! 流域の出口: 谷筋の東端の 1 セルを、河床より低い水位の規定で常に空にする
!   (完全排水口。領域マスクの外は壁なので、水はここからだけ出ていく)
!----------------------------------------------------------------------
&list_bound_stage
  stage_cell(:,1,1) = 190, 33
  stage_eta(1) = -999.0
/
