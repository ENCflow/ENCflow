!======================================================================
! source_area 貯留関数法のタンク: 流域を 1 つのタンク(S = K·q^P)とみなすと、
!   流出はどうなるか(追補「貯留関数法との関係」の比較用。実行は make sfm)
!   貯留関数法の区画流出(fn_sfm)のタンク 3 つに、過程計算と同じ雨を与える。
!   流域の地形は使わない: 3 × 8 セルの平らな格子を 3 つの区画に分け、区画ごとに
!   タンクを 1 つ置いて、流出は系外へ出す(地表に水は流れない)。見るのは
!   sfm_zone.csv の q(流出強度 mm/h)と S(貯留 mm)。
!   K と P は、この例題のほかの計算の「貯留と流出の関係」から決めた値で、
!   観測で校正した値ではない(求め方は README の追補。plot_results.py が、
!   結果から求め直した値を表に並べる)。雨 30 mm/h のケースは param_sfm_r30.txt
!======================================================================
&list_sysparam
  dt = 5.0                  ! 時間刻み (s)。地表に水がないので、タンクの計算には効かない
  tt_c = "96h"              ! 計算終了時刻(4 日。過程計算と同じ)
  dt_disp_c = "12h"         ! 画面表示の間隔
  dt_file_c = "96h"         ! 分布ファイル出力の間隔(分布は使わない)
  dt_recrd_c = "10m"        ! sfm_zone.csv の記録間隔
  fn_geoinfo = '-'
  fn_precip = 'param_common.txt'    ! 降雨は過程計算と同じ(共通設定)
  fn_sfm = '-'
  dir_data = 'data_sfm'
  dir_result = 'result_sfm'
/

&list_geoinfo
  nx = 3
  ny = 8                    ! MPI で 4 つに分けても 1 ランクに 2 行
  dx = 10.0
  dy = 10.0
  f_ztype = 0
  z0 = 0.0                  ! 平らな地面(標高は結果に効かない)
/

!----------------------------------------------------------------------
! 貯留関数法: 区画ごとにタンク 1 つ。S [mm] = K·q[mm/h]^P、dS/dt = 雨 − q。
!   初期の流出強度は既定の 0.1 mm/h(初期の貯留は K·0.1^P)
!----------------------------------------------------------------------
&list_sfm
  fn_sfm_zone = 'zone.txt'  ! 区画番号のラスタ(行 1〜3 が区画 1、4〜6 が 2、7〜8 が 3)
/

&list_sfm_zone
  ! 区画 1: 過程計算と同じ貯留。P は既定の 0.6(書かない)。K は、過程計算が雨 10 mm/h と
  !         釣り合ったときに流域にある水 45.1 mm から、K = 45.1 ÷ 10^0.6
  sfm_k(1) = 11.3
  sfm_out_kind(1) = 2       ! 流出は系外へ出す
  ! 区画 2: 過程計算の 2 つの状態(雨と釣り合ったときの 45.1 mm と 10 mm/h、2 回目の雨の
  !         直前の 5.2 mm と 0.09 mm/h)を通る K と P
  sfm_k(2) = 15.6
  sfm_p(2) = 0.46
  sfm_out_kind(2) = 2
  ! 区画 3: VZ 流出の土壌柱(深さ 0.5 m。make vz)の、雨のあとの貯留と排水の関係に
  !         合わせた K と P
  sfm_k(3) = 37.0
  sfm_p(3) = 0.167
  sfm_out_kind(3) = 2
/
