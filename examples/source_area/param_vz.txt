!======================================================================
! source_area VZ 流出の土壌柱: 鉛直の不飽和浸透だけで、流出がどれだけ遅れるか
!   (追補「鉛直不飽和浸透(VZ)との関係」の比較用。実行は make vz)
!   谷 (2023) の VZ 流出(fn_vz)の参照柱 1 本に、過程計算と同じ雨を与える。
!   流域の地形は使わない: 3 × 8 セルの平らな格子の全セルを 1 つの区画にし、
!   区画の排水は系外へ出す(地表に水は流れない)。見るのは vz_column.csv の、
!   深さ vz_out_depth を下向きに通った水の累積 D(その時間差分が排水)。
!   柱の設定(&list_vz_column)は書かない。土壌は VZ 流出の試算用の既定
!   (谷 (2023) の竜ノ口山の土壌)、初期条件は 0.1 mm/h の定常排水になる
!   (採用した値は画面に (default) を付けて表示される)。この流域の土壌の
!   値ではない。雨 30 mm/h のケースは param_vz_r30.txt
!======================================================================
&list_sysparam
  dt = 5.0                  ! 時間刻み (s)。地表に水がないので、柱の計算には効かない
  tt_c = "96h"              ! 計算終了時刻(4 日。過程計算と同じ)
  dt_disp_c = "12h"         ! 画面表示の間隔
  dt_file_c = "96h"         ! 分布ファイル出力の間隔(分布は使わない)
  dt_recrd_c = "10m"        ! vz_column.csv の記録間隔
  fn_geoinfo = '-'
  fn_precip = 'param_common.txt'    ! 降雨は過程計算と同じ(共通設定)
  fn_vz = '-'
  dir_data = 'data_vz'
  dir_result = 'result_vz'
/

&list_geoinfo
  nx = 3
  ny = 8                    ! MPI で 4 つに分けても 1 ランクに 2 行
  dx = 10.0
  dy = 10.0
  f_ztype = 0
  z0 = 0.0                  ! 平らな地面(標高は結果に効かない)
  f_sdtype = 0
  sd0 = 0.5                 ! 土層厚 (m)。過程計算と同じ。区画の排水(vz_zone.csv)を読む深さ
/

!----------------------------------------------------------------------
! VZ 流出: 参照柱 1 本(深さ 10 m・層厚 2.5 cm の既定)を全セルが共有する
!----------------------------------------------------------------------
&list_vz
  fn_vz_zone = 'zone.txt'   ! 区画番号のラスタ(全セル 1)
  dt_vz = 60.0              ! 柱を進める同期間隔 (s)
  vz_out_depth = 0.25, 0.5, 1.0     ! vz_column.csv に出す深さ (m)
/

&list_vz_zone
  vz_out_kind(1) = 2        ! 区画の排水は系外へ出す
/
