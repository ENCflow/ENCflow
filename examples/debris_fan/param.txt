!======================================================================
! examples/debris_fan: 斜面崩壊 → 土石流が谷を下る → 扇状地の住宅地へ(格子 5 m・谷の向き 0°)
!   1997 年の出水市針原川の土石流のような、急な谷壁の崩壊から始まる土石流の模式ケース。
!   北岸の谷壁(35°)の崩壊土塊 10 万 m³ が濃度 0.45 の土石流になって谷へ流れ込み、
!   谷(勾配 0.22 → 0.12)を 750 m 下って扇頂から扇状地(0.12 → 0.03)に広がり、住宅地に堆積する。
!   ねらいは格子への依存の確認: 同じ地形を格子間隔 20・10・5 m と、格子に対する谷の向き
!   0°・22.5°・45° で計算して比べる(このファイルが雛形。他のケースの param_*.txt は
!   make_init.py が格子・時間刻み・入出力の行を差し替えて書く)。
!   入力は make(make_init.py)が data_dx5_r0/ に作る。README を参照
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.05                 ! 時間刻み (s)。dt = 0.01 Δx(格子によらずクーラン数がそろう)
  t0 = 0
  tt = 600                  ! 10 分(土石流は 3 分ほどで止まる。残りは水の流下)
  dt_disp = 60
  dt_file = 5               ! 分布出力 5 秒おき(前縁の位置の時間変化を読む)
  dt_recrd = 600            ! (このケースでは計測なし)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  fn_boundary = ''          ! 辺はすべて壁(水は扇状地の下流端の吸い込みへ)
  dir_data = 'data_dx5_r0'
  dir_result = 'result_dx5_r0'
  f_output_mode = 2         ! bil(単精度バイナリ)。出力の量を抑える
  f_out_z = 1               ! 地盤高 Z(堆積・侵食)
  f_out_hs = 1              ! 土砂柱状量 Hs(流動深 = H + Hs)
  f_out_dmax = 1            ! 最大流動深 D9999(土石流の浸水深)
  f_out_fmax = 1            ! 最大流体力 F9999 = (ρm/ρw)(h+hs)V²(家屋被害の指標)
  f_out_vvmax = 1           ! 最大流速 V9999
/

&list_enc
  f_gravity_correction = 2  ! 急勾配の重力補正(既定の 2。斜面に沿う摩擦と混合体流動の抵抗則)
  p_diagratio = 0.5857864376  ! 斜めのエッジの通過幅の配分(既定 = 8 本の通過幅が等しい値。
                            !   4 近傍の比較ケースだけ 0 にする)
/

&list_initial
  f_htype = 1
  fn_hinit = 'hinit.txt'    ! 初期水深: 崩壊域だけに水(崩壊土塊が流れ出す濃度 0.45 に足りない水)
/

&list_geoinfo
  lx = 1850.0
  ly = 810.0
  nx = 370
  ny = 162
  f_ztype = 1
  fn_z = 'z.txt'            ! 地盤高(谷沿いの座標の解析式を格子のセル中心で標本化)
  f_masktype = 1
  fn_mask = 'mask.txt'      ! 谷沿いの長方形(外は領域外)
  fn_sw = 'sw.txt'          ! 扇状地の下流端の外側の帯(標高 0 m の吸い込み)
  f_rntype = 0
  rn0 = 0.04                ! 清水域のマニングの粗度(土石流の抵抗は f_dbres の構成則)
  f_sdtype = 1
  fn_sd = 'sd.txt'          ! 土層厚 = 崩壊深(崩壊域の下と、それ以外は不侵食の基盤。
                            !   堆積すると増える)
  sy0 = 0.4                 ! (地下水計算なし。土層の間隙率の参考値)
/

&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ更新
  morfac = 1.0              ! f_debris は 1 必須
  f_debris = 1
  f_dbed = 3                ! 侵食・堆積: 高橋・中川 (1991)(石礫型。慣性で走る間は堆積しない)
  f_dbres = 3               ! 抵抗則: 高橋・中川 (1991) 石礫型(降伏 tanα' = 0.45 + ダイラタント項)
  db_phi = 35.0             ! 内部摩擦角 (deg)。平衡濃度に使う
  db_d50 = 0.3              ! 代表粒径 d_L (m)。石礫型の先頭部の巨礫(推奨範囲 0.05〜0.3 m の上限)
  db_delte = 0.0001         ! 侵食係数 δe(高橋・中川の洞谷への適用値)
  db_deltd = 0.5            ! 堆積係数 δd(同)
  f_dbstop = 1              ! 低速で止まった土砂を地形に固定する
  db_vstop = 0.05
  db_wstop = 0.05
  f_dbwet = 1               ! 間隙水の授受: 堆積物は流れの水で飽和する(崩壊土塊の間隙水が戻る)
  db_satbed = 1.0
  fluv_porosity = 0.4       ! 崩壊土塊・堆積物の空隙率 λ(C* = 0.6。make_init.py の LAMBDA と対)
  fn_dbinit = 'dbinit.txt'  ! 崩壊深 (m)(体積 10 万 m³ に規格化)
  db_reltime = 0.0          ! 計算開始と同時に流動化
  db_relsat = 1.0           ! 崩壊土塊は飽和(間隙水 λD を地表水に加える)
/
