!======================================================================
! timberyard: 高潮による貯木場からの材木の流入(ケース far)
!   貯木場を市街地から離す(木材港の考え方)。同じ 6 万 m3 を港の沖側(防潮壁から
!   210〜340 m)に置く。流れ出し始める水位は param.txt と同じ +1.0 m。
!   共通: 1,200 m × 600 m(dx = 10 m)。西から港(−3 m)、貯木場(−2 m)、防潮壁(+1.5 m。
!   中央に幅 120 m の開口部 +0.5 m = 陸閘・水門が開いている所)、市街地(+0.5 → +2.0 m。
!   建物空隙率 0.6)。港・貯木場・市街地は通常セル+初期水位で、海域マスクは高潮の
!   水位を与える西端の 1 列だけ。材木はラワン原木級(直径 0.6 m・比重 0.5 → 喫水 0.3 m)。
!   詳細と結果は README.md。入力は make_terrain.py が作る
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0
  tt = 14400                ! 4 時間(高潮の上げ〜引きまで)
  dt_disp = 600
  dt_file = 600
  dt_recrd = 300
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_tide = '-'
  fn_driftwood = '-'
  fn_enc = '-'
  dir_result = 'result_far'
  f_out_h = 1
  f_out_vv = 1
  f_out_hd = 1              ! 流動材木の柱状材積 Hd0001…
  f_out_wd = 1              ! 堆積材木 Wd0001… + 最大到達量 Wd9999
  f_out_hmax = 1
  f_out_vvmax = 1
  f_out_fmax = 1            ! 最大流体力 F9999(水のみ = u²h)
  f_check_cfl = 0
/

&list_enc
  f_exflux_reduction = 1    ! 乾床前縁(市街地への浸水先端)の定石
/

!======================================================================
! 初期条件(水位一定 T.P. 0 m。港湾・貯木場が湛水し、陸は乾燥)
!======================================================================
&list_initial
  f_htype = 2               ! 初期水位固定
  e0 = 0.0
/

!======================================================================
! 地理情報(dx = 10 m。make_terrain.py が生成)
!======================================================================
&list_geoinfo
  nx = 120
  ny = 60
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z.txt'
  f_rntype = 0
  rn0 = 0.03
  fn_gv = 'gv.txt'          ! 市街地の建物空隙率 0.6(建物占有 40%)
  f_masktype = 2            ! 領域マスクを海域マスクから自動生成
  fn_sw = 'sw.txt'          ! 西端 1 列のみ海域(潮位強制)
/

!======================================================================
! 高潮(沖の水位の一様時系列。分単位: 天文潮 +1.0 m → ピーク +3.5 m → 引き。
! 台風による水位の上昇だけを与え、波浪は扱わない)
!======================================================================
&list_tide
  titype = 2
  hsea0 = 2.0               ! 海セルの見かけ水柱厚 (m)
  dt_tiupdate = 1.0
  tival(1:2,1) =   0, 0.0
  tival(1:2,2) =  60, 1.0   ! 1 時間で天文潮位まで
  tival(1:2,3) = 150, 3.5   ! 高潮ピーク(防潮壁 +1.5 m を大きく越える)
  tival(1:2,4) = 180, 3.5   ! ピーク維持 30 分
  tival(1:2,5) = 240, 0.5   ! 引き潮
/

!======================================================================
! 流木(貯木場の材木。発生 = 高潮による係留切れ・浮上)
!======================================================================
&list_driftwood
  fn_dwstock = 'stock_far.txt' ! 港の沖側(防潮壁から 210〜340 m)に 0.77 m3/m2(計 6 万 m3)
  dw_dlog = 0.6             ! ラワン原木級の代表直径 (m)
  dw_sglog = 0.5            ! 材比重 → 喫水 0.3 m(浮力平衡から自動導出)
  dw_wrec = 2.0e-4          ! 流失レート(高潮継続中に全量流失する規模)
  dw_hrec = 4.0             ! 流失の水深閾値 = 港の平常水深 3.0 m + 1.0 m
                            !   (param.txt と同じく水位 +1.0 m で流失が始まる)
  dw_vrec = 0.0             ! 流速閾値なし(浮けば流失)
  dw_wstop = 0.005          ! 接地堆積レート (m/s)
  dw_vstop = 0.02           ! 低速堆積の流速閾値(市街地で滞留したら堆積)
/
