!======================================================================
! timberyard: 高潮による貯木場からの材木流入デモ(伊勢湾台風型)
!   海岸の貯木場に浮かぶ材木が、高潮の水位上昇で係留を失って流出し、
!   防潮壁を越えた氾濫流に乗って市街地へ流入・堆積する現象の理想化
!   実験。手順・考え方の解説は users_guide/driftwood.md の例題節。
!
!   実行:  python3 make_terrain.py && ./encflow param.txt
!   見どころ(result/):
!     - Hd0001…  流動材木の移動(高潮ピーク時に開口部から市街地へ)
!     - Wd9999   材木の最大到達量マップ(= 市街地への漂着分布)
!     - F9999    最大流体力 u²h(家屋被害の実務指標。材木衝突力の
!                後処理評価には V9999 と代表材諸元を用いる — ガイド参照)
!     - driftwood.csv  材積台帳(流出・堆積・貯木場残)
!
!   設定の要点:
!     - 貯木場・市街地は通常セル+初期水位(f_htype=2)。海域マスクは
!       西端の潮位強制列だけ(海セルには材木を置けないため)
!     - 発生は水理的流失を「高潮による係留切れ・浮上」として使う:
!       dw_hrec = 3.0 m(平常の貯木場水深 2.0 m より高い = 高潮で
!       水位が上がって初めて流失)、dw_vrec = 0、dw_wrec は高潮の
!       継続時間内に全量が流失するレート
!     - 材はラワン原木級(直径 0.6 m・比重 0.5 → 喫水 0.3 m)
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0
  tt = 14400                ! 4 時間(高潮の上げ〜引きまで)
  dt_disp = 600
  dt_file = 1800
  dt_recrd = 300
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_tide = '-'
  fn_driftwood = '-'
  fn_enc = '-'
  dir_result = 'result'
  f_out_h = 1
  f_out_vv = 1
  f_out_hd = 1              ! 流動材木の柱状材積 Hd0001…
  f_out_wd = 1              ! 堆積材木 Wd0001… + 最大到達量 Wd9999
  f_out_hmax = 1
  f_out_vvmax = 1
  f_out_fmax = 1            ! 最大流体力 F9999(水のみ = u²h)
  f_check_cfl = 0
/

&list_enc
  f_exflux_reduction = 1    ! 乾床前縁(市街地への浸水先端)の定石
/

!======================================================================
! 初期条件(水位一定 T.P. 0 m。港湾・貯木場が湛水し、陸は乾燥)
!======================================================================
&list_initial
  f_htype = 2               ! 初期水位固定
  e0 = 0.0
/

!======================================================================
! 地理情報(dx = 10 m。make_terrain.py が生成)
!======================================================================
&list_geoinfo
  nx = 120
  ny = 60
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z.txt'
  f_rntype = 0
  rn0 = 0.03
  fn_gv = 'gv.txt'          ! 市街地の建物空隙率 0.6(建物占有 40%)
  f_masktype = 2            ! 領域マスクを海域マスクから自動生成
  fn_sw = 'sw.txt'          ! 西端 1 列のみ海域(潮位強制)
/

!======================================================================
! 高潮(一様時系列。分単位: 天文潮 +1.0 m → ピーク +3.5 m → 引き)
!======================================================================
&list_tide
  titype = 2
  hsea0 = 2.0               ! 海セルの見かけ水柱厚 (m)
  dt_tiupdate = 1.0
  tival(1:2,1) =   0, 0.0
  tival(1:2,2) =  60, 1.0   ! 1 時間で天文潮位まで
  tival(1:2,3) = 150, 3.5   ! 高潮ピーク(防潮壁 +1.5 m を大きく越える)
  tival(1:2,4) = 180, 3.5   ! ピーク維持 30 分
  tival(1:2,5) = 240, 0.5   ! 引き潮
/

!======================================================================
! 流木(貯木場の材木。発生 = 高潮による係留切れ・浮上)
!======================================================================
&list_driftwood
  fn_dwstock = 'stock.txt'  ! 貯木場セルのみ 0.5 m3/m2(計 6 万 m3)
  dw_dlog = 0.6             ! ラワン原木級の代表直径 (m)
  dw_sglog = 0.5            ! 材比重 → 喫水 0.3 m(浮力平衡から自動導出)
  dw_wrec = 2.0e-4          ! 流失レート(高潮継続中に全量流失する規模)
  dw_hrec = 3.0             ! 流失の水深閾値 = 平常水深 2.0 m + 1.0 m
                            !   (高潮で水位が上がって初めて流失が始まる)
  dw_vrec = 0.0             ! 流速閾値なし(浮けば流失)
  dw_wstop = 0.005          ! 接地堆積レート (m/s)
  dw_vstop = 0.02           ! 低速堆積の流速閾値(市街地で滞留したら堆積)
/
