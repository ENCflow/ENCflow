!======================================================================
! estuary_barrage: 4 ケース共通の設定(浅水流・境界・降雨・地下水)。単独では実行しない
!   各 param の fn_enc / fn_boundary / fn_precip / fn_gwflow がこの
!   ファイルを指す(グループは名前で読まれるので 1 ファイルにまとめられる)。
!   値は現象を見やすくするための模式値(実河川の代表値ではない)。
!======================================================================

!----------------------------------------------------------------------
! 浅水流(ENC)の調整
!----------------------------------------------------------------------
&list_enc
  f_adaptive_runge = 1      ! 適応 Runge-Kutta
/

!----------------------------------------------------------------------
! 境界: 辺は壁、西端の河道 3 セルから河川流量を与える
!----------------------------------------------------------------------
&list_bound_edge
/

&list_bound_inflow
  inflow_cell(:,1,1) = 1, 11      ! 西端の河道セル (i, j)
  inflow_cell(:,2,1) = 1, 12
  inflow_cell(:,3,1) = 1, 13
  ! 最初の 1.5 日は出水 200 m3/s(塩水くさびを河道から押し出したまま、
  ! 河川水と降雨で帯水層を淡水で満たす)。2 日目に平水 20 m3/s へ下げ、
  ! 以後は低水期(塩水くさびが遡上する)
  inflow_val(:,1,1) =     0, 200.0    ! (時刻 min, 総流量 m3/s)
  inflow_val(:,2,1) =  2160, 200.0
  inflow_val(:,3,1) =  2880,  20.0
  inflow_val(:,4,1) = 99999,  20.0
  inflow_dist(1) = 2                  ! 通水能按分
/

!----------------------------------------------------------------------
! 降雨: 最初の 42 時間だけ一様 50 mm/h(陸の帯水層を淡水で満たす前処理)
!----------------------------------------------------------------------
&list_precip
  prtype = 1                ! 一様時系列
  dt_prupdate = 10          ! 更新間隔 (min)
  prval(:,1) =     0, 50.0  ! (時刻 min, 雨量強度 mm/h)
  prval(:,2) =  2520, 50.0
  prval(:,3) =  2521,  0.0
  prval(:,4) = 99999,  0.0
/

!----------------------------------------------------------------------
! 地下水: Green-Ampt の鉛直浸透+層1の側方流(f_salt_gw=1 の前提)
!----------------------------------------------------------------------
&list_gwflow
  f_gwvertical = 2          ! Green-Ampt(河床からの浸透もこれが担う)
  f_gwlateral = 1           ! 非線形 Boussinesq
/

&list_gwflow_greenampt
  gw_ksv_mmh = 360.0        ! 鉛直飽和透水係数 (mm/h) = 1e-4 m/s(河床を含む全域)
  gw_psif = 0.0             ! 毛管圧力水頭 0 = 一定浸透能
/

&list_gwflow_lateral
  gw_ksh_mmh = 36000.0      ! 側方飽和透水係数 (mm/h) = 1e-2 m/s。礫混じりの砂。
                            ! 20 日で岸の帯水層の変化が見える速さにするための値
                            ! (1e-4〜1e-3 m/s の帯水層では同じ変化に年単位かかる)
/
