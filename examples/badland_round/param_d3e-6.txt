!======================================================================
! 丸い峰のバッドランド(稜線が尖らず、低い丘ほど丸くなる地形)の成因実験
!
! 現象: 膨潤性粘土(スメクタイト質泥岩。米国 Chinle 層・Chadron 層など)の
!   バッドランドでは、雨水は表面のポップコーン状風化殻に吸われて表面流が
!   ほとんど生じず、風化殻の膨潤・収縮によるクリープ(拡散型の土砂移動)が
!   斜面を下る。拡散型の過程は凸形(丸い)の頂部を作り、リル・雨滴の
!   移流型(下刻)過程は尖った稜線と直線〜凹形の斜面を作る。両者の比が
!   長さ尺度 L = D / (P kt S^(h-1)) を決め、L より小さい丘は全体が丸い
!   ドームになり、L より大きい丘は頂部だけが丸く側面は直線のまま刻まれる
!   (小さい・低い丘ほど丸い、の説明)。
! 設計: 平地(z = 0 = 侵食基準面)に側面 tanθ = 0.8 の円錐丘を 4 つ
!   (H = 24 / 15 / 9 / 5 m)置き、一様降雨 10 mm/h を全量浸透させて
!   (Green-Ampt 36 mm/h > 10 mm/h、表面流ゼロ)、乾式斜面侵食 f_splash
!   (移流型。examples/badland と同じ係数)とクリープ f_creep(拡散型)を
!   重ねる。本ファイルは creep_d = 3e-6 m2/s。
!   param_d3e-7 / d1e-6 / d3e-6.txt は creep_d だけを変えた派生
!   (L ≈ 2.4 / 8 / 24 m)。
! 期待される挙動: クリープなしでは稜線が尖ったまま丘が縮む。クリープが
!   強いほど頂部が丸くなり、H = 5・9 m の小さい丘から順に全体がドーム化する。
!   裾にはクリープの堆積による緩いエプロンができる。
! 時間: dt = 2 s、地形更新 2 s 毎 × morfac 500 → 地形時間は実時間の 500 倍。
!   24 時間 = T_geo 4.3e7 s(10 mm/h の降雨 1.4 年分 ≈ 430 m の雨)。
!   2 時間毎(T_geo 3.6e6 s)に Z を出力。約 2 分(4 スレッド)。
! 実行: python3 make_terrain.py && ../../bin/encflow param_d0.txt(約 2 分)
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0: ENC)
  f_govequation = 0         ! 基礎方程式(0: 動的波。地表流は生じないので結果に影響しない)
  dt = 2.0                  ! 時間刻み (s)(水はないので浸透と地形の更新周期で決める)
  t0 = 0                    ! 開始時刻 (s)
  tt = 86400                ! 終了時刻 (s) = 24 h(地形時間 4.3e7 s)
  dt_disp = 7200            ! 画面表示間隔 (s)
  dt_file = 7200            ! 分布出力間隔 (s) = 2 h(地形時間 3.6e6 s 毎に Z)
  dt_recrd = 86400          ! プローブ出力間隔 (s)(プローブは使わない)
  fn_geoinfo = '-'          ! 地理情報の namelist は本ファイル内
  fn_precip = '-'           ! 降雨の namelist は本ファイル内
  fn_gwflow = '-'           ! 地下水(浸透)の namelist は本ファイル内
  fn_geomorph = '-'         ! 地形変化の namelist は本ファイル内
  fn_enc = '-'              ! ENC 計算条件は本ファイル内(既定)
  dir_data = '.'            ! 入力データのディレクトリ
  dir_result = 'result_d3e-6' ! 結果ディレクトリ(creep_d = 3e-6)
  f_out_z = 1               ! 時間発展する地盤高 Z0001... を出力
  f_check_cfl = 0           ! CFL 検査なし(地表流なし)
/
&list_enc
/                           ! ENC の計算条件は既定
&list_geoinfo
  lx = 176                  ! 領域の x 長 (m)
  ly = 80                   ! 領域の y 長 (m)
  nx = 176                  ! x 方向格子数(Δx = 1 m)
  ny = 80                   ! y 方向格子数(Δy = 1 m)
  f_ztype = 1               ! 地盤高をファイルから読む
  fn_z = 'z.txt'            ! 円錐丘 4 つ(make_terrain.py)
  f_rntype = 0              ! 粗度は固定値
  rn0 = 0.03                ! マニング粗度(地表流はないので影響なし)
  sd0 = 100.0               ! 土層厚 (m)。丘の高さより十分厚く(基準面は spl_zbase で与える)
  sy0 = 0.2                 ! 比産出率(浸透水の貯留に使う)
/
&list_precip
  prtype = 1                ! 降雨は時系列(一様)
  prval(1:2,1) = 0, 10      ! t = 0 s に 10 mm/h
  prval(1:2,2) = 9999999, 10 ! 以後ずっと 10 mm/h
/
&list_gwflow
  f_gwvertical = 2          ! 鉛直浸透は Green-Ampt
  f_gwlateral = 0           ! 地下水の側方流なし
/
&list_gwflow_greenampt
  gw_ksv_mmh = 36.0         ! 飽和透水係数 36 mm/h(> 降雨 10 mm/h → 全量浸透、表面流ゼロ)
/
&list_geomorph
  dt_geomorph = 2.0         ! 地形更新間隔 (s)
  morfac = 500.0            ! 地形時間の加速係数(実効刻み dts = 1000 s)
  f_creep = 1               ! クリープ(拡散)あり
  creep_d = 3e-6             ! クリープ拡散係数 (m2/s)(L = D/(P kt S^0.5) ≈ 24.1 m)
                            !   (陽解法の安定条件 D×dts×2/Δx² ≤ 0.5 → D ≤ 2.5e-4)
  f_splash = 1              ! 乾式斜面侵食(雨滴+サブグリッドリル。移流型の下刻)
  spl_kr = 0.0002           ! 雨滴侵食係数(無次元。examples/badland と同じ)
  spl_kt = 0.05             ! サブグリッドリル侵食係数(無次元)
  spl_cb = 32.0             ! リル項の凹地形増幅長 (m)(谷の正のフィードバック)
  spl_h = 1.5               ! 勾配べき指数
  spl_dzmax = 0.2           ! 1 更新の侵食上限 (m)(暴走の安全弁)
  spl_lc = 2.0              ! 曲率の相関長 (m)(谷間隔 ≈ 4.4〜6 ℓ)
  spl_zbase = 0.0           ! 侵食基準面 = 平地の高さ(平地は削れない)
/
