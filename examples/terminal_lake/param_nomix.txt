!======================================================================
! terminal_lake / nomix: 湖の縮小と再上昇(混合なし・溶解度の上限なし)
!   放物面のすり鉢の湖(半径 5 km で深さ 10 m。gen_data.py)。初期水位 8 m、塩分 10 g/L。蒸発 1 m/年。
!   流入を、蒸発と釣り合う Q0 = 1.991 m3/s の 5 % に絞って 12 年(湖はほぼ干上がる)、その後 Q0 に戻して
!   13 年(水位が戻る)。流入水の塩分は 0.5 g/L。
!   水平の混合も溶解度も与えない(移流と蒸発による濃縮だけ)。流入口のまわりが淡水、縁が極端に濃くなり、
!   塩は干上がった湖底の水柱の柱状量として残る。水が戻ると全量が瞬時に溶ける。
!   時間の圧縮(docs/users_guide/time.md): 圧縮率 M = 1000。蒸発・流入・塩の負荷・混合係数・溶解の
!   係数を M 倍にし、25 年を 25 年 / M で計算する。結果の時刻は M 倍して読む。
!======================================================================
&list_sysparam
  dt = 5                  ! 時間刻み (s)
  tt = 789050.0            ! 計算終了時刻 (s) = 25 年 / M
  dt_disp = 7890.0          ! Log.txt の間隔(実時間の 0.25 年ごと)
  dt_file = 31560.0         ! 分布出力の間隔(実時間の 1 年ごと)
  dt_recrd = 1578.0         ! wq.csv・evap.csv・プローブの間隔(実時間の 0.05 年ごと)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_boundary = '-'
  fn_evap = '-'
  fn_wq = '-'
  fn_record = '-'
  dir_data = 'data_lake'
  dir_result = 'result_nomix'
  f_output_mode = 2         ! 分布は bil
/

&list_geoinfo
  nx = 61
  ny = 61
  dx = 200.0
  dy = 200.0
  f_ztype = 1
  fn_z = 'z.txt'            ! 放物面のすり鉢(gen_data.py)
  rn0 = 0.03
/

&list_initial
  f_htype = 2               ! 水位の固定値
  e0 = 8                  ! 初期水位 (m)。流入 Q0 と蒸発が釣り合う水位
/

&list_bound_source
  fn_src_cell(1) = 'inflow_cells.txt'   ! 流入(川の水)を湖の中心の 5 × 5 セル(1 km 四方)へ直接入れる
  src_val(:,1,1) = 0.0, 9.955106e+01          ! (min, m3/s) = 0.05 × Q0 × M
  src_val(:,2,1) = 6312.0000, 9.955106e+01
  src_val(:,3,1) = 6312.0833, 1.991021e+03   ! 12 年目の出力の直後に、流入を元(Q0 × M)に戻す
  src_val(:,4,1) = 9.9e9, 1.991021e+03
/

&list_evap
  f_evmodel = 1             ! 一定(年平均値を直接与える)
  evap0 = 2.737851e+03    ! 蒸発 (mm/day) = 1 m/年 × M
/

&list_wq
  wq_c0 = 10000             ! 湖の初期の濃度 (mg/L)
  fn_wq_pt_cell(1) = 'inflow_cells.txt'   ! 流入水の塩(流入と同じセルへの点源)
  wq_pt_series(:,1,1) = 0.0, 4.977553e+04    ! (経過日, g/s) = 流入 × 流入水の濃度
  wq_pt_series(:,2,1) = 4.38333333, 4.977553e+04
  wq_pt_series(:,3,1) = 4.38339120, 9.955106e+05
  wq_pt_series(:,4,1) = 9.9e9, 9.955106e+05
/

&list_record
  pbxytype = 0
  pbxy(:,1) = 31, 31        ! 湖の中心(流入を入れる範囲の中)
  pbxy(:,2) = 31, 38        ! 中心から 1.4 km
/
