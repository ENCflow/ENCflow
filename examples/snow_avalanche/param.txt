!======================================================================
! examples/snow_avalanche: 沢を下る流れ型雪崩(基本ケース。格子 5 m・沢の向き 0°)
!   ボウル状の急斜面(34〜37°)で面発生した厚さ 1 m・1.7 万 m³ の雪が、沢(走路)を 700 m 下りながら
!   沢底の雪を取り込んで育ち、沢の出口から扇状地(堆積区)に広がって止まる模式ケース。
!   土砂流動モデルの援用(users_guide/geomorph_applications.md の雪崩): 土層厚 sd = 連行できる雪の厚さ、
!   土砂柱状量 hs と地表水 h の和 = 流れている雪の厚さ、標高 z = 雪面。
!   ねらい: (1) 走路の雪の量(0・1・2 m)で、雪崩の大きさと到達距離がどう変わるか、
!           (2) 扇状地の誘導堤が、雪崩の向きと堆積の場所をどう変えるか。
!   このファイルが雛形で、ほかのケースの param_*.txt は make_init.py が格子・入出力・差し替えの行を
!   書き換えて作る。入力は make(make_init.py)が data_*/ に作る。README を参照
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.025                ! 時間刻み (s)。dt = 0.005 Δx(格子によらずクーラン数がそろう)
  t0 = 0
  tt = 240                  ! 4 分(雪崩は 1 分で扇状地に広がり、1 分半でほぼ止まる)
  dt_disp = 20
  dt_file = 5               ! 分布出力 5 秒おき(先頭の位置と流れている雪の量の時間変化を読む)
  dt_recrd = 240            ! (このケースでは計測なし)
  dd = 0.0001               ! 限界水深 (m)。既定の 1 mm から 0.1 mm に下げる。キャリア(地表水 h = 雪の厚さの
                            !   1 割)がこの値を切ったセルは乾いたセルとして扱われ、流れている雪がその場で
                            !   地形に固定される。1 mm のままだと、厚さ 1 cm 未満の雪が流れの縁と先頭で
                            !   固定され、結果が時間刻みに左右される(README の「キャリアと限界水深」)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  fn_boundary = ''          ! 辺はすべて壁(雪崩は領域の端に届かない)
  dir_data = 'data_dx5_r0_t1'
  dir_result = 'result_base'
  f_output_mode = 2         ! bil(単精度バイナリ)。出力の量を抑える
  f_out_z = 1               ! 雪面の標高 Z(連行で下がり、堆積で上がる)
  f_out_hs = 1              ! 土砂柱状量 Hs(流れている雪の厚さ = H + Hs)
  f_out_vv = 1              ! 流速の大きさ V
  f_out_dmax = 1            ! 最大流動深 D9999(流れた雪の最大の厚さ)
  f_out_vvmax = 1           ! 最大流速 V9999
/

&list_enc
  f_gravity_correction = 2  ! 急勾配の重力補正(既定の 2。Voellmy 則を斜面に沿う速さと地盤に直角な厚さで評価する)
/

&list_initial
  f_htype = 0
  h0 = 0.0                  ! 初期の水なし(キャリアは発生・連行のときに雪と一緒に入る)
/

&list_geoinfo
  lx = 1810.0
  ly = 650.0
  nx = 362
  ny = 130
  f_ztype = 1
  fn_z = 'z.txt'            ! 雪面の標高(沢沿いの座標の解析式を格子のセル中心で標本化。地山 + sd)
  f_masktype = 1
  fn_mask = 'mask.txt'      ! 沢沿いの長方形(外は領域外)
  f_rntype = 0
  rn0 = 0.03                ! マニングの粗度(雪の流れの抵抗は Voellmy 則が置き換えるので、結果に効かない)
  f_sdtype = 1
  fn_sd = 'sd.txt'          ! 土層厚 = 連行できる雪の厚さ(発生区の破断深と、沢底の雪の帯。ほかは 0 =
                            !   連行されない面。止まった雪が堆積すると増える)
/

&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ更新
  morfac = 1.0              ! f_debris は 1 必須
  f_debris = 1
  f_dbres = 4               ! 抵抗則: Voellmy(見かけ摩擦 μ + 乱流抵抗 g V²/(ξ h))
  db_mu = 0.25              ! 摩擦係数 μ(RAMMS の表の、中規模 2.5〜6 万 m³・流路型・標高 1500 m 以上・
  db_xi = 1750.0            !   再現期間 300 年の値。ξ は m/s²)
  f_dbed = 4                ! 連行: 速度に比例(E = δe |V|。連行だけで、堆積は f_dbstop)
  db_delte = 0.05           ! 連行係数 δe(35 m/s の先頭が 1.75 m/s で削る = 密度 200 kg/m³ の雪で
                            !   350 kg/(m² s)。観測された先頭の取り込みの上限の桁)
  f_dbmom = 1               ! 連行した雪を静止から加速する分を数える(流速が u·m/(m + Δm) に下がる。
                            !   0 にすると、取り込んだ雪は流れの速さを持って加わる)
  db_er = 0.0               ! Hungr の連行則(f_dbed = 5: E = E_r·h_t·|V|)の成長率 E_r (1/m)。補足のケースが使う
  db_hent = 0.0             ! 連行する流動深の下限 (m。0 = 下限なし)。補足のケースが使う
  db_cmin = 0.02            ! 連行・Voellmy 則を適用する濃度の下限(この計算の濃度は一定の 0.9)
  f_dbstop = 1              ! 低速で止まった雪を地形(雪面 z)に固定する
  db_vstop = 0.05           ! 止まったとみなす速さ (m/s)
  db_wstop = 0.1            ! 固定の速さ (m/s。堆積の厚さで)
  fluv_porosity = 0.1       ! キャリアの割合 λ(雪の厚さのうち地表水 h として運ぶ分。固体の分は 1 − λ)
  f_dbwet = 1               ! キャリアを雪と一緒に授受する: 連行した雪にも同じ割合のキャリアが付き、
  db_satbed = 1.0           !   止まった雪がキャリアを引き取る(db_satbed = db_relsat にそろえる)
  fn_dbinit = 'dbinit.txt'  ! 発生区の破断深 (m)(体積 1.7 万 m³ に規格化)
  db_reltime = 0.0          ! 計算開始と同時に発生
  db_relsat = 1.0           ! 発生した雪のキャリア(λ × 破断深を地表水に置く。h + hs = 破断深)
/
