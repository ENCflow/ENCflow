!======================================================================
! landslide_tsunami: C. 海底地滑り: 水中斜面(水深 12〜36 m)の土塊を可動化 → 引き波先行の津波
!   【非静水圧版・加速度項なし】1 層非静水圧補正(f_nonhydrostatic=1、底面勾配項)だけで、動く底面の
!   加速度項は入れない。伝播の分散性だけを見る対照(底層との自励振動が小さい)
!   水中の斜面 x∈[220,290] m の土塊(平面すべり)を t=1 s に可動化する。
!   水面が平らでも駆動 (1−r)∇z が残るので底層が斜面を下り、スカーで
!   水面が下がり(引き波)、堆積側で上がる。陸上の過程はない。
! 【共通の幾何】600 m × 200 m、dx=dy=5 m。x<200 m は tanθ=0.4 の斜面
!   (汀線 x=200、静水面 e0=0)、水中も同勾配で下り x≥300 m は平坦な湖底
!   z=−40 m。西辺は山側(壁)、東・北・南は長波放射境界(「解く海」の型:
!   海域マスクを使わず初期水位 f_htype=2 で水域を表す)。
!   崩壊は平面すべり(すべり面勾配 0.25、下端で地表に抜ける。体積 19,110 m3)。
!   プローブ: 汀線前後(x=188, 212)、斜面下端(298)、湖央(398)、対岸側(548)、
!   北寄り(398, y=28)。詳細は README.md。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! ENC 格子
  f_govequation = 0         ! 完全な動的浅水方程式
  dt = 0.05                 ! 時間刻み (s)。c≈20 m/s, dx=5 m → CFL≈0.2
  t0 = 0
  tt = 120                  ! 計算時間 (s)
  dt_disp = 5               ! 画面表示間隔 (s)
  dt_file = 5               ! 分布ファイル出力間隔 (s)
  dt_recrd = 0.5            ! プローブ出力間隔 (s)
  fn_geoinfo = '-'          ! 同一ファイルから読む
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'         ! 土砂・地形変化(土石流モデル)を有効化
  fn_boundary = '-'         ! 辺境界(放射)を有効化
  fn_record = '-'           ! プローブを有効化
  dir_result = 'result_C_nh0'
  f_check_cfl = 0           ! CFL 超過で止めない(突入時の一時的な超過を許す)
  f_out_e = 1               ! 水位 E
  f_out_z = 1               ! 地盤高 Z(堆積の時間発展)
  f_out_vv = 1              ! 流速 V
  f_disp_vv = 1             ! 画面に最大流速
  f_disp_cn = 1             ! 画面に最大クーラン数
/

&list_enc
  f_gravity_correction = 1  ! 急斜面(tanθ=0.4)なので重力項の勾配補正を有効に
  f_nonhydrostatic = 1      ! 1 層非静水圧補正(developer.md §69)
  nh_solver = 2             ! CG
  nh_tol = 1.0e-6
  f_nh_slope = 1            ! 底面勾配項(水中斜面 tanθ = 0.4。§69.5)
  f_nh_bottom = 0           ! 加速度項なし(分散と底面勾配項だけ。底層との自励振動を避ける対照)
/

&list_initial
  f_htype = 2               ! 初期水位固定(湖の静水面)
  e0 = 0.0                  ! 静水面標高 (m)。z<0 のセルが湖
  h0 = 0.0
/

&list_geoinfo
  lx = 600.0
  ly = 200.0
  nx = 120
  ny = 40
  f_ztype = 1
  fn_z = 'z.txt'            ! 斜面+湖(make_init.py z)
  f_rntype = 0
  rn0 = 0.03                ! 清水のマニング粗度
  sd0 = 12.0                ! 土層厚(崩壊深 D=10 の供給源。f_dbed=0 では交換なし)
  sy0 = 0.2
/

&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ
  morfac = 1.0              ! イベント計算(f_bedslide は 1 必須)
  fluv_porosity = 0.4       ! λ(かさ体積の間隙率。台帳の表示に使用)
  f_bedslide = 1            ! 動く底層
  fn_bsinit = 'submarine.txt' ! 水中の崩壊深分布(make_init.py submarine)
  bs_reltime = 1.0          ! t=1 s に可動化
  bs_rho = 2000.0           ! 土塊のかさ密度 → r = ρw/ρs = 0.5
  f_bsres = 1               ! Voellmy(慣性なし)
  bs_mu = 0.15              ! 摩擦係数 μ
  bs_xi = 500.0             ! 乱流係数 ξ (m/s²)
  bs_vstop = 0.05           ! 停止判定の速度閾値 (m/s)
  bs_tstop = 15.0           ! 低速が 15 s 続いたセルだけ固定
  bs_eps_s = 0.02
  bs_cfl = 0.4
/
