!======================================================================
! landslide_tsunami: B0. 土石流(混合体)のみ — 底層への引き渡しなし(比較用)
!   B と同じ設定で f_bedslide=0。混合体は水面が平らになると駆動を失い、
!   汀線から 50 m 以内に積み上がって水中を走らない(一層モデルの限界)。
!   突入直後の柱高増加による造波だけが出る。
! 【共通の幾何】600 m × 200 m、dx=dy=5 m。x<200 m は tanθ=0.4 の斜面
!   (汀線 x=200、静水面 e0=0)、水中も同勾配で下り x≥300 m は平坦な湖底
!   z=−40 m。西辺は山側(壁)、東・北・南は長波放射境界(「解く海」の型:
!   海域マスクを使わず初期水位 f_htype=2 で水域を表す)。
!   崩壊は平面すべり(すべり面勾配 0.25、下端で地表に抜ける。体積 19,110 m3)。
!   プローブ: 汀線前後(x=188, 212)、斜面下端(298)、湖央(398)、対岸側(548)、
!   北寄り(398, y=28)。詳細は README.md。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! ENC 格子
  f_govequation = 0         ! 完全な動的浅水方程式
  dt = 0.05                 ! 時間刻み (s)。c≈20 m/s, dx=5 m → CFL≈0.2
  t0 = 0
  tt = 120                  ! 計算時間 (s)
  dt_disp = 5               ! 画面表示間隔 (s)
  dt_file = 5               ! 分布ファイル出力間隔 (s)
  dt_recrd = 0.5            ! プローブ出力間隔 (s)
  fn_geoinfo = '-'          ! 同一ファイルから読む
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'         ! 土砂・地形変化(土石流モデル)を有効化
  fn_boundary = '-'         ! 辺境界(放射)を有効化
  fn_record = '-'           ! プローブを有効化
  dir_result = 'result_B0'
  f_check_cfl = 0           ! CFL 超過で止めない(突入時の一時的な超過を許す)
  f_out_e = 1               ! 水位 E
  f_out_z = 1               ! 地盤高 Z(堆積の時間発展)
  f_out_hs = 1              ! 土砂柱状量 Hs
  f_out_vv = 1              ! 流速 V
  f_out_dmax = 1            ! 最大流動深 D9999
  f_disp_vv = 1             ! 画面に最大流速
  f_disp_cn = 1             ! 画面に最大クーラン数
/

&list_enc
  f_gravity_correction = 1  ! 急斜面(tanθ=0.4)なので重力項の勾配補正を有効に
/

&list_initial
  f_htype = 2               ! 初期水位固定(湖の静水面)
  e0 = 0.0                  ! 静水面標高 (m)。z<0 のセルが湖
  h0 = 0.0
/

&list_geoinfo
  lx = 600.0
  ly = 200.0
  nx = 120
  ny = 40
  f_ztype = 1
  fn_z = 'z.txt'            ! 斜面+湖(make_init.py z)
  f_rntype = 0
  rn0 = 0.03                ! 清水のマニング粗度
  sd0 = 12.0                ! 土層厚(崩壊深 D=10 の供給源。f_dbed=0 では交換なし)
  sy0 = 0.2
/

&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ
  morfac = 1.0              ! イベント計算(f_debris は 1 必須)
  f_debris = 1              ! 土石流モデル
  f_dbed = 0                ! 交換なし(等価流体モード: 移流・停止のみ)
  f_dbres = 4               ! Voellmy(降伏 μ + 乱流項 gV²/(ξ h))
  db_mu = 0.15              ! 摩擦係数 μ(岩屑なだれの慣用域 0.1〜0.2)
  db_xi = 500.0             ! 乱流係数 ξ (m/s²)
  db_cmin = 0.02            ! これ未満の濃度はマニング水(湖内で希釈された裾)
  db_vstop = 0.05           ! 降伏・停止判定の速度閾値 (m/s)
  f_dbstop = 1              ! 低速凝集(停止した土砂を湖底 z へ固定)
  db_wstop = 0.05           ! 河床転換レート (m/s)
  fluv_porosity = 0.4       ! λ(C* = 0.6)
  fn_dbinit = 'slide.txt'   ! 崩壊深分布(make_init.py slide)。混合体として流動化
  db_reltime = 1.0          ! t=1 s に一斉流動化
  db_relsat = 0.2           ! 崩壊土塊の飽和度(疑似間隙水 h = λ・relsat・D = 0.8 m)
  f_bedslide = 0            ! 引き渡しなし(比較用)
/
