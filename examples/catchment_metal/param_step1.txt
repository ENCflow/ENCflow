!======================================================================
! catchment_metal ステップ 1: 地表水と物質(溶存・保存性)
!   雨 → 地表流出 → 河道 → 海。浸透(Green-Ampt)は土層に貯まるだけで
!   戻らない(地下水の側方流動なし)。物質は鉱山の排水(点源)と農地(面源)
!   から入る溶存態の保存性物質(減衰・沈降・吸着なし)。浸透する水に
!   同伴して地下へ入った分は、このステップでは地下に留まる。
!   共通: 20 km × 12 km、250 m 格子、流域 214 km2。180 日(雨季 100 日+乾季)。
!   詳細と結果は README.md。データは make(gen_data.py)が data/ に作る
!======================================================================
&list_sysparam
  dt = 10.0                 ! 時間刻み (s)
  tt_c = "180d"             ! 計算終了時刻(180 日)
  dt_disp_c = "5d"          ! 画面表示の間隔
  dt_file_c = "1d"          ! 分布ファイル出力の間隔
  dt_recrd_c = "1h"         ! プローブ・測線・wq.csv の記録間隔
  fn_geoinfo = '-'
  fn_channel = '-'
  fn_gwflow = '-'
  fn_wq = '-'
  fn_enc = 'param_common.txt'       ! 共通設定(浅水流・境界・降雨・計測)
  fn_boundary = 'param_common.txt'
  fn_precip = 'param_common.txt'
  fn_record = 'param_common.txt'
  dir_data = 'data'
  dir_result = 'result_step1'
  f_check_cfl = 0
  f_out_hg = 1              ! 地下貯留 Hg と地下の物質量 Cg
  f_output_mode = 2         ! 分布は bil(単精度。低い濃度まで桁を保つ)
/

&list_geoinfo
  nx = 80
  ny = 48
  dx = 250.0
  dy = 250.0
  f_ztype = 1               ! 地盤高ファイル
  fn_z = 'z_notank.txt'     ! ため池なしの地形
  f_masktype = 1            ! 領域マスクファイル(流域+入り江)
  fn_mask = 'mask.txt'
  fn_sw = 'sw.txt'          ! 海域マスク(入り江。海面 0 m = 流域の出口)
  f_rntype = 0
  rn0 = 0.06                ! 斜面の粗度(畑地・草地)
  fn_rw = 'rw_notank.txt'   ! 河道マスク
  rn0_rw = 0.035            ! 河道の粗度
  sd0 = 4.0                 ! 土層厚 (m)
  sy0 = 0.1                 ! 有効間隙率(地下の貯留容量 0.4 m)
/

&list_channel
  fn_width = 'width_notank.txt'     ! サブグリッド河道幅(本川 15 m、支川 8 m)
/

&list_gwflow
  f_gwvertical = 2          ! Green-Ampt の浸透
  f_gwlateral = 0           ! 地下水の側方流動なし(浸透した水は戻らない)
/

&list_gwflow_greenampt
  gw_ksv_mmh = 10.0         ! 鉛直飽和透水係数 (mm/h)。砂壌土(河床も同じ)
  gw_psif = 0.0             ! 一定浸透能
/

!----------------------------------------------------------------------
! 水質: 溶存態の保存性物質(重金属の溶存態を想定。単位は g と mg/L)
!----------------------------------------------------------------------
&list_wq
  wq_pt_cell(:,1,1) = 24, 9         ! 鉱山の排水口(内部ソースと同じセル)
  wq_pt_load(1) = 0.025             ! 負荷 (g/s) = 排水 0.05 m3/s × 0.5 mg/L
  fn_wq_ar_cell(1) = 'farm_cells.txt'     ! 農地 290 セル(1812 ha)
  wq_ar_series(:,1,1) =  0.0,   0.002     ! (経過日, kg/ha/day) 施用期 60 日
  wq_ar_series(:,2,1) = 60.0,   0.002
  wq_ar_series(:,3,1) = 60.001, 0.0
  wq_ar_series(:,4,1) = 999.0,  0.0
/
