!======================================================================
! qanat: カナート(カレーズ/フォガラ)の理想化ケース
!
!   扇状地面(勾配 0.015)+段丘崖(勾配 0.05、x=300..360)+低地の
!   理想化断面(400 m x 210 m、dx=10 m)の地下に、層2(風化基岩層
!   d2=8 m, sy2=0.2, 初期飽和度 0.9)の帯水層を置き、初期地下水位を
!   地表下 1.8 m とする。中央行 j=11 の i=2..35 に管路連続体層で
!   坑道(cap=0.1 m 柱状 = 断面 1 m2、床勾配 0.004)をセル列として
!   与える。坑道床は出口セル i=35(崖の中腹 z=1.25 m)で地表に達する。
!   出口を崖に開けることで水面上の導水部を i=32..34 の 3 セルに抑える
!   (実カナートが崖・段丘に口を開ける構図。導水部が長いと坑道→層2の
!   漏水が集水と釣り合い、出口まで水が届かないことを一様斜面の試行で
!   確認済み — それ自体が「導水部を掘り抜き・裏込めで守る」実務の理由)。
!
!   期待される挙動:
!    (1) i=2..31 の集水部で層2水頭 > 坑道水頭となり浸入(gwc_leak_mmh=30
!        → 約 0.83 L/s/セル、計 25 L/s 程度)
!    (2) 坑内を sqrt 乱流則で出口へ導水(充満率に応じた通水)
!    (3) 出口セルで満管化 → オリフィス噴出により地表へ湧出し、
!        崖を下って低地へ流下・湛水する
!    (4) 導水部(i=32..34)では坑道→層2 の漏水損失が出る
!   検証: 降雨・蒸発・境界なしの閉領域なので Log の S_total は一定
!   (質量保存)。S_surf の増加 = 湧出の累積。出口下流の測線流量と
!   プローブ水深で湧出の時系列を確認する。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 28800                ! 計算終了時刻 (s) = 8 時間
  dt_disp = 600             ! 画面表示間隔 (s)
  dt_file = 7200            ! 分布ファイル出力間隔 (s)
  dt_recrd = 60             ! 計測出力間隔 (s)
  fn_geoinfo = '-'          ! 地理情報は本ファイル内
  fn_enc = '-'              ! ENC 設定は本ファイル内
  fn_gwflow = '-'           ! 地下水設定は本ファイル内
  fn_record = '-'           ! 計測設定は本ファイル内
  dir_data = 'data_qanat'   ! 分布データの置き場
  dir_result = 'result_open'
  f_disp_debug = 1          ! S 列を全有効桁で表示(水収支の照合用)
  f_out_e = 1               ! 水位場 E の出力
  f_out_hg = 1              ! 地下水場 Hg / Hg2 / Hgc の出力
/

&list_enc
  f_adaptive_runge = 1      ! 適応 Runge(浅い流れの安定化)
/

&list_geoinfo
  nx = 40                   ! 格子数(東西)
  ny = 21                   ! 格子数(南北)
  dx = 10.0                 ! 格子間隔 (m)
  dy = 10.0
  f_ztype = 1               ! 地盤高はファイル
  fn_z = 'z.txt'            ! 一様斜面(勾配 0.02)
  f_rntype = 0              ! 粗度は固定値
  rn0 = 0.03                ! Manning 粗度
  f_sdtype = 0              ! 土層厚は固定値
  sd0 = 1.0                 ! 土層厚 (m)(層2の上に薄い土層)
  sy0 = 0.2                 ! 土層の比湧水量
/

&list_bound
/                           ! 境界条件なし = 全周閉境界(質量保存の検証)

!======================================================================
! 地下水設定: 層2(帯水層)+ 管路連続体層(坑道)
!======================================================================
&list_gwflow
  f_gwvertical = 0          ! 鉛直浸透なし(降雨なしのため不要)
  f_gwlateral = 0           ! 層1側方なし(層1は薄い乾いた土層のまま)
  f_gwlayer2 = 1            ! 層2 = 扇状地の帯水層
  f_gwconduit = 1           ! 管路連続体層 = カナート坑道
  dt_gwflow = 0             ! 毎ステップ更新(dts = dt = 0.5 s)
/

&list_gwflow_layer2
  gw2_depth = 8.0           ! 帯水層厚 (m)
  gw2_sy = 0.2              ! 比湧水量
  gw2_infil_mmh = 10.0      ! 層1→2 浸透能(層1が乾いているため実質不使用)
  gw2_ksh_mmh = 3600.0      ! 側方飽和透水係数 (mm/h) = 1e-3 m/s(扇状地礫層相当。
                            !   小さいと坑道周りの水位降下で集水が止まる)
  gw2_sat0 = 0.9            ! 初期飽和度 → 初期地下水位 = z - 1.8 m
/

&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! sqrt 乱流則(坑内の自由水面流の近似)
  fn_gwc_cnd = 'gwc_cnd.txt'    ! 通水能密度 0.05 m2/s(坑道セル列のみ)
  fn_gwc_cap = 'gwc_cap.txt'    ! 貯留容量 0.1 m 柱状(断面 1 m2 相当)
  fn_gwc_bot = 'gwc_bot.txt'    ! 坑道床標高(出口で地表に一致)
  gwc_sy = 0.1              ! 不圧貯留係数 = cap/坑道高(1 m)
  gwc_slot_sy = 0.02        ! 疑似スロット(被圧枝。dt 上限との折り合い)
  gwc_sat0 = 0.5            ! 初期充満率(満管までの助走の短縮)
  fn_gwc_inlet = 'gwc_inlet.txt' ! 出口セル(i=36,j=11)のみ開口 0.01 個/m2
  fn_gwc_outfall = 'gwc_outfall.txt' ! 坑口 (35,11) を陸側開放吐口に(自由流出)
  gwc_cw = 2.66             ! 堰係数(既定)
  gwc_co = 0.15             ! オリフィス係数 Cd·A(既定)
  gwc_leak_layer = 2        ! 層2と交換(集水・漏水)
  gwc_leak_mmh = 30.0       ! 交換能 (mm/h) → 集水部 1 セルあたり約 0.8 L/s
  gwc_eps_h = 0.05          ! sqrt 則の線形化幅 (m)
/

!======================================================================
! 計測: 出口の噴出と下流の地表流
!======================================================================
&list_record
  pbxytype = 0
  pbxy(:,1) = 35, 11        ! 出口セル(湧出点)
  pbxy(:,2) = 37, 11        ! 出口下流(崖下)の地表流
  pbxy(:,3) = 20, 11        ! 集水部中央(地下水位の低下確認)
  flxytype = 0
  flxy(:,1) = 36, 15, 36, 7 ! 出口直下流を南→北に横切る測線(東向き流れが正)
/
