!======================================================================
! qanat: カナート(カレーズ/フォガラ)と揚水井(ケース well2)
!   カナート+揚水井(2 L/s。150 日目から)。扇状地の帯水層に上端から一定の
!   涵養(10 L/s)が入り、坑道が地下水を集めて段丘崖の出口から流し出す。
!   揚水井(坑道から 40 m 北)が同じ帯水層から汲む
!   共通: 扇状地面(勾配 0.015)+段丘崖(勾配 0.05)+平野の 500 m x 210 m(dx = 10 m)。
!   帯水層は層2(厚さ 8 m、比湧水量 0.2、側方透水係数 5e-4 m/s)。東端の列は扇状地の
!   末端の排水路(地盤高 -0.5 m)で、カナートに集まらなかった地下水はそこで湧き出す。
!   坑道は中央の行の i = 2〜35(床勾配 0.004)を管路連続体層のセル列で表し、
!   出口(崖の中腹)は開放の吐口で地表へ流れ出し、平野を流れて東の辺から出ていく。
!   詳細と結果は README.md。入力は gen_data.py が data_qanat/ に作る
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 10.0                 ! 時間刻み (s)。出口から平野を流れる水の Courant 数で決まる
                            !   (坑道の安定条件は自動で細分される)
  t0 = 0
  tt_c = '300d'             ! 計算終了時刻(300 日。地下水位が釣り合うまで 150 日 + 揚水の比較 150 日)
  dt_disp_c = '1d'          ! 画面表示・Log の間隔(全水量の収支から湧出量を求める)
  dt_file_c = '10d'         ! 分布ファイル出力間隔
  dt_recrd_c = '6h'         ! 計測出力間隔
  fn_geoinfo = '-'
  fn_enc = '-'
  fn_precip = '-'           ! 涵養(上端の帯の雨。すべて地下へしみ込む)
  fn_gwflow = '-'
  fn_record = '-'
  fn_boundary = '-'
  dir_data = 'data_qanat'
  dir_result = 'result_well2'
  f_disp_debug = 1          ! S 列を全有効桁で表示(水収支から湧出量を求める)
  f_out_hg = 1              ! 地下水場 Hg / Hg2 / Hgc の出力
/

&list_enc
  f_adaptive_runge = 1
/

&list_geoinfo
  nx = 50
  ny = 21
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z.txt'            ! 扇状地面 + 段丘崖 + 平野 + 排水路
  f_rntype = 0
  rn0 = 0.03
  f_sdtype = 0
  sd0 = 1.0                 ! 土層厚 (m)(帯水層の上の薄い土層)
  sy0 = 0.2
/

&list_initial
  f_htype = 0
  h0 = 0.0                  ! 地表は乾いている
/

&list_bound_edge
  f_bc_e = 1                ! 東の辺: 自由流出(出口から出た水と排水路の湧水が出ていく)
/

&list_precip
  prtype = 2                ! 分布 × 倍率
  fn_prmap = 'recharge.txt' ! 涵養の帯(x < 50 m)だけ mm/day の値。他は 0
  prval(:,1) = 0, 1.0
  prval(:,2) = 1.0e7, 1.0
/

!======================================================================
! 地下水: 土層(涵養の通り道)+ 層2(帯水層)+ 管路連続体層(坑道)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! Green-Ampt(涵養の帯の浸透)
  f_gwlateral = 0           ! 土層の側方流なし(土層は涵養を層2へ渡すだけ)
  f_gwlayer2 = 1            ! 層2 = 扇状地の帯水層
  f_gwconduit = 1           ! 管路連続体層 = カナートの坑道
  f_gwpump = 1              ! 井戸揚水
  dt_gwflow = 0
/

&list_gwflow_greenampt
  fn_gw_ksv = 'gw_ksv.txt'  ! 浸透能: 涵養の帯 100 mm/h、他は 0
  gw_psif = 0.0
/

&list_gwflow_layer2
  gw2_depth = 8.0           ! 帯水層の厚さ (m)
  gw2_sy = 0.2              ! 比湧水量
  gw2_infil_mmh = 50.0      ! 土層 → 帯水層の浸透能 (mm/h)(涵養をそのまま渡せる値)
  gw2_ksh_mmh = 1800.0      ! 側方の透水係数 (mm/h) = 5e-4 m/s(扇状地の砂礫層)
  gw2_sat0 = 0.9            ! 初期の飽和度 → 初期の地下水位 = 地表下 1.8 m
/

&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! sqrt 則(坑内の自由水面流の近似)
  fn_gwc_cnd = 'gwc_cnd.txt'    ! 通水能密度 0.5 m2/s(坑道のセル列だけ)
  fn_gwc_cap = 'gwc_cap.txt'    ! 貯留容量 0.1 m 柱状(断面 1 m2 相当)
  fn_gwc_bot = 'gwc_bot.txt'    ! 坑道の床の標高(出口で地表に一致)
  gwc_sy = 0.1              ! 不圧の貯留係数 = 容量 / 坑道の高さ(1 m)
  gwc_slot_sy = 0.05        ! 満水時の疑似スロット(坑道は満水にならない。安定条件を緩める)
  gwc_sat0 = 0.0            ! 初期の充水率(坑道は空から始める)
  fn_gwc_outfall = 'gwc_outfall.txt' ! 出口(35, 11)の開放の吐口(地表へ流れ出る)
  gwc_leak_layer = 2        ! 帯水層と水をやり取りする
  fn_gwc_leak = 'gwc_leak.txt'  ! 交換能 (mm/h): 集水区間(x < 165 m)30 = 坑道 1 セル(長さ
                            !   10 m)あたり 0.83 L/s、下流の導水区間 0(漏れない)。
                            !   地下水位が坑道の水位より高ければ集水、低ければ漏水
  gwc_eps_h = 0.1           ! sqrt 則の線形化幅 (m)
/

&list_gwflow_pump
  gwp_cell(1:2,1,1) = 8, 7  ! 揚水井のセル(坑道の上流部の脇。坑道から 40 m 北)
  gwp_layer(1) = 2          ! 帯水層(層2)から汲む
  gwp_val(1:2,1,1) = 0, 0            ! (分, m3/s)。同じ時刻を 2 回書くと階段になる
  gwp_val(1:2,2,1) = 216000, 0       ! 150 日目まで止めておく
  gwp_val(1:2,3,1) = 216000, 0.0020  ! 150 日目から 2 L/s
  gwp_val(1:2,4,1) = 1.0e7, 0.0020
/

&list_record
  pbxytype = 0
  pbxy(:,1) = 5, 11         ! 坑道の上端付近(母井戸)の地下水位
  pbxy(:,2) = 20, 11        ! 坑道の中ほど
  pbxy(:,3) = 8, 7          ! 揚水井のセル
  flxytype = 0
  flxy(:,1) = 36, 14, 36, 8 ! 出口のすぐ下流で流れを横切る測線(東向きが正)= 湧出量
  flxy(:,2) = 49, 21, 49, 1 ! 排水路の手前で平野を横切る測線(出口から来た地表流)
  flxy(:,3) = 50, 21, 50, 1 ! 東の辺(排水路から出ていく水の全量)
/
