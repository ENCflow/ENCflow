!======================================================================
! tsunami_town: 津波による木造家屋の破壊と瓦礫の流動・堆積(ケース wall3c)
!   津波 +4 m・防潮堤を +3 m にかさ上げし、開口部も閉じる(+3.0 m)。
!   共通: 1,200 m × 600 m(dx = 10 m)。西から海(−5 m)、浜、防潮堤(中央に幅 120 m の
!   開口部 +0.5 m = 陸閘・水門・道路の切れ目)、市街地(+0.5 → +3.0 m。建物空隙率 0.6、
!   家屋ストックは木造率込みで海側 100 % → 内陸 40 %)。海・浜・市街地は通常セル+初期
!   水位で、海域マスクは津波の水位を与える西端の 1 列だけ。家屋は浸水深で壊れる。
!   詳細と結果は README.md。入力は make_terrain.py が作る
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0
  tt = 3600                 ! 1 時間(津波の来襲〜引き波〜排水)
  dt_disp = 300
  dt_file = 300
  dt_recrd = 60
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_tide = '-'
  fn_bldgdebris = '-'
  fn_enc = '-'
  dir_result = 'result_wall3c'
  f_out_h = 1
  f_out_vv = 1
  f_out_hbd = 1             ! 流動瓦礫の柱状材積 Bf0001…
  f_out_wbd = 1             ! 堆積瓦礫 Bd0001… + 最大到達量 Bd9999
  f_out_bds = 1             ! 家屋の破壊率 Bs0001…
  f_out_fdmax = 1           ! 流木・瓦礫込み最大流体力 Fd9999
  f_out_hmax = 1
  f_out_vvmax = 1
  f_out_fmax = 1            ! 最大流体力 F9999(水のみ = u²h)
  f_check_cfl = 0
/

&list_enc
  f_exflux_reduction = 1    ! 乾床前縁(市街地への浸水先端)の定石
/

!======================================================================
! 初期条件(水位一定 T.P. 0 m。海が湛水し、陸は乾燥)
!======================================================================
&list_initial
  f_htype = 2               ! 初期水位固定
  e0 = 0.0
/

!======================================================================
! 地理情報(dx = 10 m。make_terrain.py が生成)
!======================================================================
&list_geoinfo
  nx = 120
  ny = 60
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z_wall3c.txt'     ! 防潮堤 +3.0 m、開口部も +3.0 m
  f_rntype = 0
  rn0 = 0.03
  fn_gv = 'gv.txt'          ! 市街地の建物空隙率 0.6(建物占有 40 %)
  f_masktype = 2            ! 領域マスクを海域マスクから自動生成
  fn_sw = 'sw.txt'          ! 西端 1 列のみ海域(潮位強制)
/

!======================================================================
! 津波(沖の水位の一様時系列。分単位: 5 分後から 10 分で +4.0 m に達し 10 分維持、
! 10 分で引く。先行波だけの理想化。発生・伝播は扱わず、沖の波形を与える)
!======================================================================
&list_tide
  titype = 2
  hsea0 = 2.0               ! 海セルの見かけ水柱厚 (m)
  dt_tiupdate = 1.0
  tival(1:2,1) =   0, 0.0
  tival(1:2,2) =   5, 0.0
  tival(1:2,3) =  15, 4.0   ! 津波の押し波(10 分で +4.0 m)
  tival(1:2,4) =  25, 4.0   ! 維持 10 分
  tival(1:2,5) =  35, 0.0   ! 引き波
  tival(1:2,6) =  60, 0.0
/

!======================================================================
! 家屋破壊・瓦礫(浸水深判定。空隙率への帰還あり)
!======================================================================
&list_bldgdebris
  fn_bdstock = 'bdstock.txt' ! 家屋ストック (m3/m2。木造率込み。計 11.7 万 m3)
  fn_bdfrac = 'bdfrac.txt'  ! 破壊可能割合(木造率。空隙率帰還の上限)
  bd_dlog = 0.3             ! 瓦礫の代表寸法 (m)
  bd_sg = 0.5               ! 見かけ比重 → 喫水 0.15 m
  f_bdcrit = 1              ! 浸水深判定
  bd_hcrit = 1.0            ! 浸水深閾値 (m。破壊の開始)
  bd_hcrit2 = 3.0           ! 第2閾値 (m。破壊可能率 1 = 被害関数の幅)
  bd_wdes = 5.0e-4          ! 破壊レート (m/s。0.3 m3/m2 が 10 分で全壊)
  bd_fsink = 0.3            ! 沈下率(瓦・基礎など 3 割はその場で堆積)
  bd_wstop = 0.01           ! 接地堆積レート (m/s)
  bd_vstop = 0.05           ! 低速堆積の流速閾値(停滞したら堆積)
  f_bdgv = 1                ! 破壊率に応じて空隙率を上げる(帰還あり)
/
