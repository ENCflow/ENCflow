!======================================================================
! tsunami_town(比較ケース): 荷重判定+流木(浜の材木が流失して破壊力に加わる)。本体は param.txt
!   海岸の低平な市街地に津波(先行波のみの理想化波形)が遡上し、木造家屋を
!   破壊して瓦礫を内陸へ運ぶ理想化実験。組み方の理由・結果の図表・
!   感度ケースの解説は README.md(機能の説明は users_guide/bldgdebris.md)。
!
!   実行:  python3 make_terrain.py && ./encflow param.txt
!          python3 plot_results.py   → figs/*.png と数表
!   比較ケース: param_oneway.txt(空隙率帰還なし f_bdgv=0)、
!          param_load.txt(荷重判定 f_bdcrit=2)、
!          param_load_dw.txt(荷重判定+流木 = 流木による破壊力の増加)
!   見どころ(result/):
!     - Bs0001…  家屋の破壊率(最終フレームが被害率マップ)
!     - Bd9999   瓦礫の最大到達量マップ、Bf0001… 流動瓦礫の移動
!     - Fd9999   流木・瓦礫込みの最大流体力(F9999 は水のみ)
!     - bldgdebris.csv  材積台帳(破壊・堆積・残ストック)
!
!   設定の要点:
!     - 海・浜・市街地は通常セル+初期水位(f_htype=2)。海域マスクは
!       西端の潮位強制列だけ(海セルには瓦礫・流木を置けないため)
!     - 家屋ストック fn_bdstock は木造率込み(海側 100 % → 内陸 40 %)、
!       破壊可能割合 fn_bdfrac で空隙率帰還の上限を与える
!     - 破壊判定は浸水深(f_bdcrit=1。閾値 1.0〜3.0 m の線形ランプ =
!       東北津波の木造家屋被害関数の幅の近似)
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0
  tt = 3600                 ! 1 時間(津波の来襲〜引き波〜排水)
  dt_disp = 300
  dt_file = 300
  dt_recrd = 60
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_tide = '-'
  fn_bldgdebris = '-'
  fn_enc = '-'
  dir_result = 'result_load_dw'
  fn_driftwood = '-'
  f_out_hd = 1
  f_out_wd = 1
  f_out_h = 1
  f_out_vv = 1
  f_out_hbd = 1             ! 流動瓦礫の柱状材積 Bf0001…
  f_out_wbd = 1             ! 堆積瓦礫 Bd0001… + 最大到達量 Bd9999
  f_out_bds = 1             ! 家屋の破壊率 Bs0001…
  f_out_fdmax = 1           ! 流木・瓦礫込み最大流体力 Fd9999
  f_out_hmax = 1
  f_out_vvmax = 1
  f_out_fmax = 1            ! 最大流体力 F9999(水のみ = u²h)
  f_check_cfl = 0
/

&list_enc
  f_exflux_reduction = 1    ! 乾床前縁(市街地への浸水先端)の定石
/

!======================================================================
! 初期条件(水位一定 T.P. 0 m。海が湛水し、陸は乾燥)
!======================================================================
&list_initial
  f_htype = 2               ! 初期水位固定
  e0 = 0.0
/

!======================================================================
! 地理情報(dx = 10 m。make_terrain.py が生成)
!======================================================================
&list_geoinfo
  nx = 120
  ny = 60
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z.txt'
  f_rntype = 0
  rn0 = 0.03
  fn_gv = 'gv.txt'          ! 市街地の建物空隙率 0.6(建物占有 40 %)
  f_masktype = 2            ! 領域マスクを海域マスクから自動生成
  fn_sw = 'sw.txt'          ! 西端 1 列のみ海域(潮位強制)
/

!======================================================================
! 津波(一様時系列。分単位: 5 分後から 10 分で +4.0 m に達し 10 分維持、
! 10 分で引く。先行波のみの理想化)
!======================================================================
&list_tide
  titype = 2
  hsea0 = 2.0               ! 海セルの見かけ水柱厚 (m)
  dt_tiupdate = 1.0
  tival(1:2,1) =   0, 0.0
  tival(1:2,2) =   5, 0.0
  tival(1:2,3) =  15, 4.0   ! 津波の押し波(10 分で +4.0 m)
  tival(1:2,4) =  25, 4.0   ! 維持 10 分
  tival(1:2,5) =  35, 0.0   ! 引き波
  tival(1:2,6) =  60, 0.0
/

!======================================================================
! 家屋破壊・瓦礫(浸水深判定。空隙率への帰還あり)
!======================================================================
&list_bldgdebris
  fn_bdstock = 'bdstock.txt' ! 家屋ストック (m3/m2。木造率込み。計 11.7 万 m3)
  fn_bdfrac = 'bdfrac.txt'  ! 破壊可能割合(木造率。空隙率帰還の上限)
  bd_dlog = 0.3             ! 瓦礫の代表寸法 (m)
  bd_sg = 0.5               ! 見かけ比重 → 喫水 0.15 m
  f_bdcrit = 2              ! 荷重判定 (h+(1+s)hs+sg木·hd+sg瓦礫·hbd)·V²
  bd_fcrit = 1.0            ! 荷重閾値 (m3/s2。破壊の開始)
  bd_fcrit2 = 4.0           ! 第2閾値 (m3/s2。破壊可能率 1)
  bd_wdes = 5.0e-4          ! 破壊レート (m/s。0.3 m3/m2 が 10 分で全壊)
  bd_fsink = 0.3            ! 沈下率(瓦・基礎など 3 割はその場で堆積)
  bd_wstop = 0.01           ! 接地堆積レート (m/s)
  bd_vstop = 0.05           ! 低速堆積の流速閾値(停滞したら堆積)
  f_bdgv = 1                ! 破壊率に応じて空隙率を上げる(帰還あり)
/

!======================================================================
! 流木(浜・貯木場の材木。津波で浮いて流失し、荷重に算入される)
!======================================================================
&list_driftwood
  fn_dwstock = 'dwstock.txt' ! 浜のセルのみ 1.0 m3/m2(計 6 万 m3)
  dw_dlog = 0.6             ! 代表直径 (m)
  dw_sglog = 0.5            ! 材比重 → 喫水 0.3 m
  dw_wrec = 5.0e-4          ! 流失レート(押し波の間に全量流失)
  dw_hrec = 3.0             ! 流失の水深閾値 (m。浜の平常水深 0〜2 m より高い = 津波で
                            !   水位が上がって初めて流失。timberyard と同じ流儀)
  dw_vrec = 0.0             ! 流速閾値なし
  dw_wstop = 0.005          ! 接地堆積レート (m/s)
  dw_vstop = 0.05           ! 低速堆積の流速閾値 (m/s)
/
