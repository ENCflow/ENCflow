!======================================================================
! examples/debris_fan_flume: 混合粒径の材料の土石流がつくる扇状地と、そのあとの清水による侵食
!   高橋・中川・山敷 (1991) の水路実験(京都大学防災研究所年報 34 B-2)の計算。
!   幅 10 cm・勾配 18° の水路に敷いた材料(厚さ 10 cm、長さ 3 m、飽和)を、600 cc/s の給水が侵食して
!   土石流になり、水路の出口につないだ勾配 5° の氾濫台に扇状地をつくる。
!   このファイルはケース form(形成の段。混合粒径 5 階、格子 5 cm)。ほかのケース(単一粒径、格子 2.5 cm、
!   清水 300 cc/s を 10 分流す侵食の段)の param_<ケース>.txt は、make_init.py がこのファイルの行を
!   差し替えて書く。入力は make(make_init.py)が data_dx5/ に作る。README を参照
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.004                ! 時間刻み (s)。dt = 0.08 Δx
  t0 = 0
  tt = 40                   ! 給水 20 秒と、そのあとの 20 秒
  dt_disp = 5
  dt_file = 5               ! 分布出力 5 秒おき
  dt_recrd = 0.1            ! 水路の出口の計測点・流量測線 0.1 秒おき
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  fn_boundary = '-'
  fn_record = '-'
  dir_data = 'data_dx5'
  dir_result = 'result_form'
  dir_save = 'save_form'
  f_state_save = 1          ! 終わりの状態を保存する(侵食の段の初期条件)
  f_state_restore = 0
  f_check_cfl = 0
  f_output_mode = 2         ! bil(単精度バイナリ)
  f_out_z = 1               ! 地盤高 Z(堆積・侵食)
  f_out_hs = 1              ! 土砂柱状量 Hs(流動深 = H + Hs)
  f_out_bmix = 1            ! 交換層の粒径階ごとの割合 Fa1_〜Fa5_(平均粒径 Dm は常に出る)
/

&list_enc
  f_gravity_correction = 2  ! 急勾配の重力補正(既定)
/

&list_initial
  f_htype = 0
  h0 = 0.0                  ! 乾いた水路と氾濫台(材料の間隙水は db_satbed で与える)
/

&list_geoinfo
  lx = 10.0
  ly = 1.4
  nx = 200
  ny = 28
  f_ztype = 1
  fn_z = 'z.txt'            ! 地盤高: 水路 18°(材料の区間は床 + 鉛直の厚さ 10.5 cm)、氾濫台 5°
  f_masktype = 1
  fn_mask = 'mask.txt'      ! 水路の幅の外(x < 8.3 m)は領域外 = 側壁
  f_rntype = 0
  rn0 = 0.018               ! マニングの粗度(粒径 3 mm の粗度。固定床と、材料を貼った氾濫台)
  f_sdtype = 1
  fn_sd = 'sd.txt'          ! 土層厚: 材料の区間だけ 10.5 cm、ほかは 0(固定床)
  sy0 = 0.35                ! (地下水計算なし。空隙率の参考値)
/

&list_bound_edge
  f_bc_e = 1                ! 氾濫台の三方(東・北・南の辺)は自由流出
  f_bc_n = 1
  f_bc_s = 1
/

&list_bound_inflow          ! 水路の上流端(西辺)の幅 2 セルから給水。600 cc/s を 20 秒
  inflow_cell(:,1,1) = 1, 14
  inflow_cell(:,2,1) = 1, 15
  inflow_val(:,1,1) = 0, 0.0006
  inflow_val(:,2,1) = 0.333333, 0.0006
  inflow_val(:,3,1) = 0.333433, 0
  inflow_val(:,4,1) = 9999, 0
/

!======================================================================
! 地形変化: 土石流(高橋・中川の石礫型)+ 掃流砂 + 混合粒径
!======================================================================
&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ更新
  morfac = 1.0              ! f_debris は 1 必須
  f_debris = 1
  f_dbed = 3                ! 侵食・堆積: 高橋・中川 (1991)
  f_dbres = 3               ! 抵抗則: 高橋・中川 (1991)
  db_phi = 35.0             ! 内部摩擦角 (deg)。tanφ = 0.70(論文の値)
  db_d50 = 0.00304          ! 代表粒径 d_L (m)。材料の平均粒径(論文の値)
  db_delte = 0.0007         ! 飽和床の侵食係数 δe(論文の値)
  db_deltd = 0.5            ! 堆積係数 δd(式の形が論文と違う。README)
  f_dbimm = 1               ! 緩い勾配(氾濫台の 5°)は掃流状集合流動の平衡濃度と堆積式
  db_immexp = 2             ! 掃流状集合流動の抵抗を (d_L/h)² で
  f_dbstop = 1              ! 低速で止まった土砂を河床へ戻す
  db_vstop = 0.05
  db_wstop = 0.05
  f_dbwet = 1               ! 間隙水: 侵食で材料の間隙水が流れに入り、堆積で流れの水が間隙に入る
  db_satbed = 1.0           ! 材料は飽和
  fluv_porosity = 0.35      ! 空隙率 λ(C* = 0.65。論文の値)
  f_fluvial = 1             ! 掃流砂(混合粒径は掃流砂が前提)
  f_qbform = 1              ! 流砂量式: 芦田・道上
  fluv_d50 = 0.00304        ! 単一粒径のケースの粒径 (m)(混合粒径では使わない)
  f_bedmix = 1              ! 混合粒径(粒径階 × 交換層・下層)。土石流は粒径階ごとの組成を運ぶ
  bmix_d = 0.00017, 0.00055, 0.0014, 0.0045, 0.008    ! 粒径階の粒径 (m)。細かい順
  bmix_frac0 = 0.18, 0.17, 0.22, 0.25, 0.18           ! 割合(平均粒径 3.0 mm)
  bmix_la = 0.008           ! 交換層厚 (m)。最大の粒径階の粒径
/

&list_record                ! 水路の出口の 1 セル上流の計測点(幅の 2 セル)と流量測線
