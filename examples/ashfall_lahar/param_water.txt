!======================================================================
! ashfall_lahar: 降灰 → 降雨 → 泥流(ラハール)の発生・流下・堆積(ケース water)
!   ケース ash と同じ地形・浸透能・降雨で、土砂を運ばない(侵食しない)場合。
!   泥流と、同じ雨の清水の洪水の違いを見る対照
!   共通: 西の山体斜面(tanθ=0.35、谷底 15 m・谷壁 0.25 の谷)→ x=300 m の谷出口
!   → 扇状地(tanθ=0.08)。降灰は火口側(西)で厚く東へ薄い(0.30 → 0.05 m、
!   扇状地 0.03 m)。降雨 30 mm/h × 60 分。降灰層は土層厚 sd として与え、
!   地表 z にも上乗せする(z − sd = 元地形。灰を使い切ると侵食が止まる)。
!   詳細と結果は README.md。入力は make(make_init.py)が作る
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.1                  ! 時間刻み (s)。dx=5 m、流速 ~10 m/s で CFL ≈ 0.3
  t0 = 0
  tt = 5400                 ! 90 分(無降雨 10 分 + 降雨 60 分 + 減水 20 分)
  dt_disp = 600
  dt_file = 120             ! 分布出力 2 分おき
  dt_recrd = 10             ! プローブ・フラックス測線 10 s おき
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_gwflow = '-'           ! 浸透(Green-Ampt の定数浸透能)
  fn_geomorph = ''          ! 土砂プロセスなし
  fn_boundary = '-'
  fn_record = '-'
  dir_result = 'result_water'
  f_out_z = 1               ! 地盤高 Z(侵食・堆積の時間発展)
  f_check_cfl = 0
  dd = 0.0002                ! 移動限界水深 (m)。斜面のシート流(深さ 1 mm 前後)を解くため
                            ! 既定の 1 mm から下げる(既定のままだと格子スケールの縞が出る)
/

&list_enc
  f_gravity_correction = 1  ! 急勾配の重力補正(土石流ケースの標準)
/

&list_initial
  f_htype = 0
  h0 = 0.0                  ! 乾いた地表(水は降雨で供給)
/

&list_geoinfo
  lx = 600.0
  ly = 150.0
  nx = 120
  ny = 30
  f_ztype = 1
  fn_z = 'z.txt'            ! 降灰後の地表(元地形 + 降灰厚)
  f_rntype = 0
  rn0 = 0.05                ! 降灰面の粗度(清水域のマニング)
  f_sdtype = 1
  fn_sd = 'ash.txt'         ! 降灰厚 = 侵食できる土層
  sy0 = 0.5                 ! 灰の間隙率(しみ込んだ水を貯める容量 = sd × sy0)
/

&list_precip
  prtype = 1                ! 一様分布の時系列(時刻 min, 強度 mm/h)
  prval(1:2,1) =  0,  0     ! 値は時刻の間で線形補間される。同じ時刻を
  prval(1:2,2) = 10,  0     !   2 回書くと階段(矩形の雨)になる
  prval(1:2,3) = 10, 30     ! 10 分から 30 mm/h
  prval(1:2,4) = 70, 30
  prval(1:2,5) = 70,  0     ! 70 分で止む
/

&list_gwflow
  f_gwvertical = 2          ! Green-Ampt(gw_psif = 0 で定数浸透能)
/

&list_gwflow_greenampt
  gw_ksv_mmh = 5.0          ! 皮膜のできた細粒の灰の浸透能 (mm/h)。噴火前の 100 mm/h の 1/20
  gw_psif = 0.0
/

&list_bound_edge
  f_bc_e = 1                ! 東辺(扇状地の末端): 自由流出
/

&list_record
  pbxytype = 1              ! 実座標 (x, y) で指定
  pbxy(:,1) = 202.5, 77.5   ! 斜面の谷筋(谷出口の 100 m 上流)
  pbxy(:,2) = 302.5, 77.5   ! 谷出口(扇頂)
  pbxy(:,3) = 452.5, 77.5   ! 扇状地の中ほど
  flxy(:,1) = 50, 30, 50, 1  ! フラックス測線 1: 谷の中(x = 247.5 m。堰堤・堆積域の上流)。東向きが正
  flxy(:,2) = 71, 30, 71, 1  ! フラックス測線 2: 扇状地(x = 352.5 m。扇頂の堆積域の下流)
/
