!======================================================================
! ashfall_lahar: 降灰 → 降雨 → 泥流(ラハール)の発生・流下・堆積
!   西の山体斜面(tanθ=0.35、V 字谷)に火口側で厚く東へ薄くなる降灰
!   (0.30 → 0.05 m)が載り、その後の降雨(40 mm/h × 60 分)で斜面の
!   表面流が降灰層を侵食して濃度を上げ、谷出口(x=300 m)から扇状地
!   (tanθ=0.05)へ泥流として流下・堆積する。発生源の指定は不要
!   (侵食起動型: 表面流 → E-D → 混合体)。降灰層は土層厚 sd として与え、
!   地表 z にも上乗せする(z − sd = 元地形。灰を使い切ると侵食が止まる)。
!   対照 param_noash.txt(降灰なし・清水のみ)と並べて、降灰が泥流を
!   生むことと、扇状地の堆積・水量の違いを見る(README・plot_results.py)。
!   浸透は無視(新鮮な降灰のクラストは浸透能が低い = 安全側の概算)。
!   抵抗則は Voellmy(μ 小): 細粒の灰スラリーは間隙水圧が高く見かけ摩擦が
!   小さい。江頭層流則(f_dbres=2)は d50=1 mm では h/d が大きすぎて抵抗が
!   消え速度が発散する(README「設計の経緯」)。
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.1                  ! 時間刻み (s)。dx=5 m、流速 ~10 m/s で CFL ≈ 0.3
  t0 = 0
  tt = 5400                 ! 90 分(無降雨 10 分 + 降雨 60 分 + 減水 20 分)
  dt_disp = 600
  dt_file = 300             ! 出力 5 分おき(18 フレーム)
  dt_recrd = 10             ! プローブ 10 s おき
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_geomorph = '-'
  fn_boundary = '-'
  fn_record = '-'
  dir_result = 'result_lahar'
  f_out_z = 1               ! 地盤高 Z(侵食・堆積の時間発展)
  f_out_hs = 1              ! 土砂柱状量 Hs(濃度の可視化)
  f_check_cfl = 0
/

&list_enc
  f_gravity_correction = 1  ! 急勾配の重力補正(土石流ケースの標準)
/

&list_initial
  f_htype = 0
  h0 = 0.0                  ! 乾いた地表(水は降雨で供給)
/

&list_geoinfo
  lx = 600.0
  ly = 300.0
  nx = 120
  ny = 60
  f_ztype = 1
  fn_z = 'z.txt'            ! 降灰後の地表(元地形 + 降灰厚。make_init.py z)
  f_rntype = 0
  rn0 = 0.05                ! 降灰面の粗度(清水域のマニング)
  f_sdtype = 1
  fn_sd = 'ash.txt'         ! 降灰厚 = 侵食できる土層(make_init.py ash)
  sy0 = 0.5
/

&list_precip
  prtype = 1                ! 一様分布の時系列(時刻 min, 強度 mm/h)
  prval(1:2,1) =  0,  0
  prval(1:2,2) = 10, 40     ! 10 分から 40 mm/h
  prval(1:2,3) = 70,  0     ! 70 分で止む
/

&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ更新
  morfac = 1.0              ! f_debris は 1 必須
  f_debris = 1
  f_dbed = 1                ! 高橋型の平衡濃度緩和(δe・δd で速度を校正)
  db_deltd = 0.01           ! 堆積係数(既定 0.05。小さいほど扇状地の奥まで運ばれる)
  f_dbres = 4               ! Voellmy(細粒・高間隙水圧のラハールは見かけ摩擦が小さい)
  db_mu = 0.05              ! 見かけ摩擦係数(扇状地勾配 0.08 を流下できる)
  db_xi = 500.0             ! 乱流係数 (m/s²)
  db_phi = 30.0             ! 火山灰の内部摩擦角 (deg)
  db_d50 = 0.001            ! 灰の代表粒径 (m)。細粒
  db_erest = 0.85           ! 粒子の反発係数
  fluv_porosity = 0.5       ! 新鮮な降灰層の空隙率(C* = 0.5)
  f_dbstop = 1              ! 停止した土砂を地形に固定(堆積の形が残る)
  db_vstop = 0.05
  db_wstop = 0.05
  f_dbwet = 1               ! 侵食で降灰層の間隙水を取り込む(増水の一次効果)
  db_satbed = 0.8           ! 降灰層の飽和度(雨で湿った灰)
/

&list_bound_edge
  f_bc_e = 1                ! 東辺(扇状地の末端): 自由流出
/

&list_record
  pbxytype = 1              ! 実座標 (x, y) で指定
  pbxy(:,1) = 202.5, 152.5  ! 斜面の谷筋(谷出口の 100 m 上流)
  pbxy(:,2) = 302.5, 152.5  ! 谷出口(扇頂)
  pbxy(:,3) = 452.5, 152.5  ! 扇状地の中ほど
/
