!======================================================================
! examples/glacier_valley 補足: 侵食係数を既定の 10 倍にして、6000 年で 6 万年分の侵食を見る
!   本番(param_glacial.txt)と同じ地形・気候・氷河で、侵食係数 gl_kg だけを 10 倍にする。
!   氷河は数百年で地形に追いつくので、侵食を 10 倍にすることは、気候を一定にしたまま
!   時間を 10 倍に進めることにほぼ等しい(侵食の速さそのものが実測に合うという意味では
!   ない)。この段の最終の地盤高に湖の段(param_lakes.txt)をかける
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 30                   ! 時間刻み (s)
  t0 = 0.0
  tt = 946740               ! 計算時間 (s) = 地形時間 6000 年 / gl_morfac
  dt_disp = 157800          ! 地形時間 1000 年ごと
  dt_file = 157800
  dt_recrd = 157800
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_meteo = '-'
  fn_snow = '-'
  fn_glacier = '-'
  fn_boundary = '-'
  dir_data = 'data'
  dir_result = 'result_glacial_x10'
  dir_save = 'save_glacial_x10'
  f_state_save = 1
  f_out_z = 1               ! 地盤高 Z の時間変化(侵食)を出力
  f_check_cfl = 0
  f_output_mode = 1         ! テキスト
/

&list_enc
/

&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報(24 km × 8 km、Δx = 200 m。make_terrain.py が data/z.txt を作る)
!======================================================================
&list_geoinfo
  lx = 24000.0
  ly = 8000.0
  nx = 120
  ny = 40
  f_ztype = 1
  fn_z = 'z.txt'
  f_rntype = 0
  rn0 = 0.05
/

!======================================================================
! 降水(年 1500 mm = 0.1711 mm/h を一定に。標高による増え方は &list_meteo)
!======================================================================
&list_precip
  prtype = 1
  dt_prupdate = 3600.0
  prval(1:2,1) = 0.0, 0.1711157
  prval(1:2,2) = 1.0e9, 0.1711157
/

!======================================================================
! 気温(年平均の一定気候。季節はない。0 ℃ の高さ = 平衡線 2680 m)
!======================================================================
&list_meteo
  temp0 = 17.42             ! 標高 0 m の気温 (℃) = 0.65 ℃/100 m × 2680 m
  f_temp_lapse = 1
  temp_lapse = 0.65         ! 気温減率 (℃/100 m)
  temp_zref = 0.0
  f_prec_lapse = 1
  prec_lapse = 5.0          ! 降水は平衡線から 100 m 上がるごとに 5% 増える
  prec_zref = 2680.0
/

!======================================================================
! 積雪(0 ℃ を境に雨と雪を分ける。季節がないので、平衡線より上では雪が
! 年中積もり、下では雪は降らない)
!======================================================================
&list_snow
  snow_t_snow = 0.0
  snow_t_rain = 0.0
  snow_t_melt = 0.0
  snow_ddf = 4.0
/

!======================================================================
! 氷河(雪崩・氷化・流動・滑動・侵食。侵食の結合長 gl_lc は既定 500 m)
!======================================================================
&list_glacier
  dt_glacier_c = "30 s"     ! 毎ステップ更新(1 回 = 地形時間 0.19 年)
  gl_morfac = 200000.0      ! 地形時間の加速
  gl_tfirn_yr = 0.001       ! 氷化の時定数(計算時間の年。地形時間では 200 年)
  gl_ddfi = 6.0             ! 氷面の度日係数 (mm/℃/day)
  f_glflow = 1
  f_glslide = 1
  gl_as = 1.0e-13           ! Weertman 滑動係数(既定)
  f_glero = 1
  gl_kg = 1.0e-3            ! 侵食係数(既定 1e-4 の 10 倍)
  gl_lexp = 1.0
  f_glava = 1               ! 雪崩(急な斜面の雪を下へ移す)
  gl_ava_tanc = 0.6
  gl_nsubmax = 1000000      ! 1 回の更新の SIA の内部反復の上限
/
