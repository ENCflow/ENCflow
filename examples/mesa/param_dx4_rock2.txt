!======================================================================
! メサ・ビュートの形成実験(キャップロック + 厚く硬い中間硬層 = ベンチの対照)
!   param_cap.txt の中間硬層(z = 8〜10 m、倍率 0.3。ベンチにならなかった)を
!   5 m 厚・倍率 0.02(キャップより硬い) に変え(param_bench2)、格子を Δx = 4 m に粗くして
!   崖 20 m が 1 セルの段になる構成。崩落と崖錐(H_c = 6 m、崖錐の侵食性 0.01 倍)。
! 現象: 硬いキャップロック(砂岩・玄武岩)が軟層(泥岩)を覆う台地では、軟層の
!   斜面が速く削れ、キャップの縁が崖として後退し、台地(メサ)が縮んで孤立丘
!   (ビュート)→ 尖塔になる。段(ベンチ)は中間の硬層でできる。
! 設計: 平地(z = 0 = 侵食基準面)に高さ 20 m の台地 2 つ(半径 30 m / 12 m、縁 63°)。
!   水平層理 spl_strata: 上 3 m(z >= 17)がキャップ(倍率 0.05)、z = 8〜10 m に中間の
!   硬層(倍率 0.3)、ほかは軟層(倍率 1)。降雨 10 mm/h は全量浸透(表面流ゼロ)。
!   乾式斜面侵食 f_splash(examples/badland_round と同じ係数)+ 弱いクリープ
!   (D = 3e-7、裾の崖錐)。対照 param_nocap.txt は層理なし(全層倍率 1)。
! 期待: キャップの縁のセルは急崖に面して S が大きく、倍率 0.05 でも削れて 1 セルずつ
!   後退する(崖の平行後退)。軟層は露出するとすぐ基準面まで下がり、垂直に近い崖が
!   保たれる。小さい台地が先にビュート→消失。キャップなしでは縁が丸まり円錐状に縮む。
! 時間: dt = 2 s、地形更新 2 s × morfac 500。48 h = T_geo 8.6e7 s。4 h 毎に Z を出力。
! 実行: make && python3 make_terrain.py && ./encflow param_dx4_rock2.txt(約 30 秒)
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0: ENC)
  f_govequation = 0         ! 基礎方程式(地表流は生じない)
  dt = 2.0                  ! 時間刻み (s)
  t0 = 0                    ! 開始時刻 (s)
  tt = 172800               ! 終了時刻 (s) = 48 h(地形時間 8.6e7 s)
  dt_disp = 14400           ! 画面表示間隔 (s)
  dt_file = 14400           ! 分布出力間隔 (s) = 4 h(地形時間 7.2e6 s 毎)
  dt_recrd = 172800         ! プローブ出力間隔 (s)(使わない)
  fn_geoinfo = '-'          ! 以下の namelist は本ファイル内
  fn_precip = '-'
  fn_gwflow = '-'
  fn_geomorph = '-'
  fn_enc = '-'
  dir_data = '.'            ! 入力データのディレクトリ
  dir_result = 'result_dx4_rock2' ! 結果ディレクトリ
  f_out_z = 1               ! 地盤高 Z0001... を出力
  f_check_cfl = 0           ! CFL 検査なし(地表流なし)
/
&list_enc
/                           ! ENC の計算条件は既定
&list_geoinfo
  lx = 200                  ! 領域の x 長 (m)
  ly = 100                  ! 領域の y 長 (m)
  nx = 50                   ! x 方向格子数(Δx = 4 m: 崖 20 m が 1 セルの段)
  ny = 25                   ! y 方向格子数
  f_ztype = 1               ! 地盤高をファイルから読む
  fn_z = 'z_dx4.txt'        ! 台地 2 つ(make_terrain.py --dx 4)
  f_rntype = 0              ! 粗度は固定値
  rn0 = 0.03                ! マニング粗度(地表流なし)
  sd0 = 100.0               ! 土層厚 (m)。基準面は spl_zbase で与える
  sy0 = 0.2                 ! 比産出率
/
&list_precip
  prtype = 1                ! 一様降雨の時系列
  prval(1:2,1) = 0, 10      ! 10 mm/h
  prval(1:2,2) = 9999999, 10 ! ずっと 10 mm/h
/
&list_gwflow
  f_gwvertical = 2          ! Green-Ampt
  f_gwlateral = 0           ! 側方流なし
/
&list_gwflow_greenampt
  gw_ksv_mmh = 36.0         ! 浸透能 36 mm/h > 10 mm/h → 全量浸透
/
&list_geomorph
  dt_geomorph = 2.0         ! 地形更新間隔 (s)
  morfac = 500.0            ! 地形時間の加速(dts = 1000 s)
  f_creep = 1               ! 弱いクリープ(裾の崖錐)
  creep_d = 3e-7            ! クリープ拡散係数 (m2/s)(L ≈ 2.4 m)
  f_splash = 1              ! 乾式斜面侵食
  spl_kr = 0.0002           ! 雨滴侵食係数
  spl_kt = 0.05             ! サブグリッドリル侵食係数
  spl_cb = 32.0             ! 凹地形増幅長 (m)
  spl_h = 1.5               ! 勾配べき指数
  spl_dzmax = 0.2           ! 1 更新の侵食上限 (m)
  spl_lc = 2.0              ! 曲率の相関長 (m)
  spl_zbase = 0.0           ! 侵食基準面 = 平地
  f_rockfall = 1            ! 崩落と崖錐(§48.7)
  rf_hc = 6.0               ! 自由面(崖)の限界高さ H_c (m)(Δx = 4 m の縁の段は 8 m(tanθ = 2)。15 m では崩落しない)
  rf_phi = 35.0             ! 崖錐の安息角 (deg)
  rf_relax = 0.5            ! 1 更新の緩和率
  rf_ftalus = 0.01           ! 崖錐に覆われたセルの乾式侵食の倍率(崖錐の除去速度 = 崖の後退を律する)
  fluv_dzmax = 0.5          ! 1 エッジ・1 更新の崩落の変動上限 (m)
  spl_strata(1:2,1) = 0.0, 1.0     ! z >= 0: 軟層(倍率 1)
  spl_strata(1:2,2) = 5.0, 0.02    ! z >= 5: 中間の硬層(5 m 厚・倍率 0.02(キャップより硬い)。z < 10 まで)
  spl_strata(1:2,3) = 10.0, 1.0    ! z >= 10: 軟層
  spl_strata(1:2,4) = 17.0, 0.05   ! z >= 17: キャップロック(倍率 0.05)
/
