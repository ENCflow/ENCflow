!   2 倍(96 h = T_geo 1.7e8 s)にする。岩屑が脚部に溜まって軟層の面が支えられ(垂直のまま)、
!   後退が岩屑の除去待ちになる領域が現れるかを見る。
! 実行: make && python3 make_terrain.py && ./encflow param_uc_slow.txt(約 16 分)
!======================================================================

  f_govequation = 0         ! 基礎方程式(地表流は生じない)
  dt = 2.0                  ! 時間刻み (s)
  t0 = 0                    ! 開始時刻 (s)
  tt = 345600               ! 終了時刻 (s) = 96 h(地形時間 1.7e8 s)
  dt_disp = 28800           ! 画面表示間隔 (s)
  dt_file = 28800           ! 分布出力間隔 (s) = 8 h(地形時間 1.4e7 s 毎)
  dt_recrd = 345600         ! プローブ出力間隔 (s)(使わない)
  fn_geoinfo = '-'          ! 以下の namelist は本ファイル内
  fn_precip = '-'

  fn_geomorph = '-'
  fn_enc = '-'
  dir_data = '.'            ! 入力データのディレクトリ
  dir_result = 'result_uc_slow' ! 結果ディレクトリ
  f_out_z = 1               ! 地盤高 Z0001... を出力
  f_check_cfl = 0           ! CFL 検査なし(地表流なし)
/
&list_enc
/                           ! ENC の計算条件は既定
&list_geoinfo
  lx = 200                  ! 領域の x 長 (m)
  ly = 100                  ! 領域の y 長 (m)
  nx = 200                  ! x 方向格子数(Δx = 1 m)
  ny = 100                  ! y 方向格子数
  f_ztype = 1               ! 地盤高をファイルから読む
  fn_z = 'z.txt'            ! 台地 2 つ(make_terrain.py)
  f_rntype = 0              ! 粗度は固定値
  rn0 = 0.03                ! マニング粗度(地表流なし)
  sd0 = 100.0               ! 土層厚 (m)。基準面は spl_zbase で与える
  sy0 = 0.2                 ! 比産出率
/
&list_precip
  prtype = 1                ! 一様降雨の時系列
  prval(1:2,1) = 0, 10      ! 10 mm/h
  prval(1:2,2) = 9999999, 10 ! ずっと 10 mm/h

&list_gwflow
  f_gwvertical = 2          ! Green-Ampt
  f_gwlateral = 0           ! 側方流なし
/
&list_gwflow_greenampt
  gw_ksv_mmh = 36.0         ! 浸透能 36 mm/h > 10 mm/h → 全量浸透
/
&list_geomorph
  dt_geomorph = 2.0         ! 地形更新間隔 (s)
  morfac = 500.0            ! 地形時間の加速(dts = 1000 s)
  f_creep = 1               ! 弱いクリープ(裾の崖錐)
  creep_d = 3e-7            ! クリープ拡散係数 (m2/s)(L ≈ 2.4 m)
  f_splash = 1              ! 乾式斜面侵食
  spl_kr = 0.0002           ! 雨滴侵食係数
  spl_kt = 0.05             ! サブグリッドリル侵食係数
  spl_cb = 32.0             ! 凹地形増幅長 (m)
  spl_ksat = 0.2            ! 増幅の飽和曲率 κ_s (1/m)
  spl_h = 1.5               ! 勾配べき指数
  spl_dzmax = 0.2           ! 1 更新の侵食上限 (m)
  spl_lc = 2.0              ! 曲率の相関長 (m)
  spl_zbase = 0.0           ! 侵食基準面 = 平地
  f_rockfall = 1            ! 崩落と崖錐(§48.7)
  rf_hc = 15.0              ! 自由面の限界高さ H_c (m)(窓内の最低点との落差に対する閾値。崖 20 m の面全体が対象)

  rf_relax = 0.5            ! 1 更新の緩和率
  rf_ftalus = 0.03         ! 崖錐に覆われたセルの乾式侵食の倍率(崖錐の除去速度 = 崖の後退を律する)
  rf_fface = 0.0            ! 支え方式: 崖錐に脚部を覆われた面は削れない(倍率 0)。崩落は柱の崩落(§48.7)
  rf_uc = 0.5               ! 限界張り出し量 (m): 脚部の岩盤がこれだけ削られると縁の柱が落ちる
  fluv_dzmax = 0.5          ! 1 エッジ・1 更新の崩落の変動上限 (m)
  spl_strata(1:2,1) = 0.0, 1.0     ! z >= 0: 軟層(倍率 1)
  spl_strata(1:2,2) = 5.0, 0.02    ! z >= 5: 中間の硬層(5 m 厚・倍率 0.02(キャップより硬い)。z < 10 まで)
  spl_strata(1:2,3) = 10.0, 1.0    ! z >= 10: 軟層
  spl_strata(1:2,4) = 17.0, 0.05   ! z >= 17: キャップロック(倍率 0.05)
/
