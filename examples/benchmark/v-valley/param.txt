!======================================================================
! examples/benchmark/v-valley: 水路のない V 字谷の降雨流出(斜面の勾配 0.03・格子 10 m・向き 0°・8 近傍)
!   長さ 1000 m・幅 1620 m の谷(谷の勾配 0.02、斜面の勾配 0.01〜0.04、n = 0.1)に 10.8 mm/h の雨を
!   60 分降らせ、下流端の流量を比べる。谷底が格子の軸と 45° をなすとき、4 近傍の格子では谷底の
!   セルの 4 つの隣がすべて高くなり(疑似的な窪地。深さ (√2/2)(S_S − S_V) Δx)、水が溜まって
!   流出が遅れる。ENC の 8 近傍(斜めのエッジ)ではこの窪地が生じないことを確かめる例題。
!   地形・マスクは make_inputs.py が data_s03_dx10_r0/ に作る。このファイルが雛形で、斜面の勾配・
!   格子 20・5 m・向き 45°・4 近傍の param_*.txt は make_inputs.py が格子・時間刻み・計測・入出力の
!   行と 4 近傍の 1 行だけを差し替えて書く。README を参照。
!   浅水流の設定はすべて既定値(力学波・急勾配の重力補正 2・MUSCL・開口の振り替え 3・
!   適応的ルンゲクッタ)
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! ENC 格子
  f_govequation = 0         ! 力学波(既定)
  dt = 1                    ! 時間刻み (s)。dt = 0.1 Δx(クーラン数 0.5 以下)
  t0 = 0
  tt = 21600                ! 6 時間(雨は 60 分で止む。n = 0.1 の薄い流れは遅く、谷が排水し切るまで見る)
  dt_disp = 60              ! 1 分ごとの Log(S 列を水収支に使う)
  dt_file = 1800            ! 分布出力 30 分おき(水深 H。45° の 4 近傍で水が溜まる場所を見る)
  dt_recrd = 60             ! 測線・プローブの出力 1 分ごと
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_precip = '-'
  fn_record = '-'
  fn_enc = '-'
  dir_data = 'data_s03_dx10_r0'
  dir_result = 'result_s03_dx10_r0'
  f_disp_debug = 1          ! S 列を全有効桁で(降雨量と流出量の収支を照合する)
/

&list_enc
  p_diagratio = 0.5857864376  ! 斜めのエッジへの通過幅の配分(既定 = 8 本の通過幅が等しい値)。
                            !   4 近傍の比較ケースは 0
/

&list_initial
  f_htype = 1
  fn_hinit = 'hinit.txt'    ! 初期水深: 谷は乾いた状態、海セルだけ 1 m(水位 1 m の吸い込み)
/

&list_geoinfo
  nx = 104                  ! 格子数(谷 1000 × 1620 m + 下流端の先の海 2 セル + 余白 1 セル)
  ny = 164
  dx = 10
  dy = 10
  f_ztype = 1
  fn_z = 'z.txt'            ! 地盤高 z = 3 + 0.02 (1000 − s) + S_S |n|(セル中心で評価)
  f_rntype = 0
  rn0 = 0.1                 ! マニングの粗度(一様)
  f_masktype = 1
  fn_mask = 'mask.txt'      ! 谷の長方形と下流端の先の海(外は領域外 = 壁)
  fn_sw = 'sw.txt'          ! 海域マスク(下流端の先 2 セル・全幅。標高 0 m・水深 1 m に固定され、入った水は消える)
/

&list_precip
  prtype = 1                ! 一様な雨の時系列(時刻 (min)、強度 (mm/h)。直線補間)
  prval(:,1) = 0, 10.8
  prval(:,2) = 59.99, 10.8
  prval(:,3) = 60, 0.0      ! 60 分で止む(値の更新は 1 分ごとで、ステップの終わりの時刻で評価する)
/

&list_record
  pbxytype = 0              ! セル番号で指定(西端の列・北端の行が 1)
  pbxy(:,1) = 101, 82       ! 下流端の谷底のセル(水深の時系列)
  pbxy(:,2) = 51, 82        ! 谷の中央(s = 500 m)の谷底のセル
  flxytype = 0
  flxy(:,1) = 101, 163, 101, 2   ! 下流端の計測面: 右岸側の尾根から左岸側の尾根まで(全幅。
                            !   エッジ流量の符号付き和。始点→終点の右側 = 下流が正)
/
