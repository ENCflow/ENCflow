!======================================================================
! examples/benchmark/v-shaped: V 字流域の降雨流出(格子 10 m・向き 0°・8 近傍)
!   両側 800 m の斜面(勾配 0.05、n = 0.015)が幅 20 m・勾配 0.02・n = 0.15 の水路に集まる
!   1000 m の流域に、10.8 mm/h の雨を 90 分降らせ、出口の流量を運動学的波の理論解と比べる。
!   Tada (2026, Hydrological Research Letters) の検証問題で、Overton & Brakensiek (1973)・
!   Di Giammarco ら (1996) 以来の降雨流出の標準問題。地形・粗度・マスクは make_inputs.py が
!   data_dx10_r0/ に作る。このファイルが雛形で、格子 20・5 m、向き 45°、4 近傍、拡散波・
!   運動波などの param_*.txt は make_inputs.py が格子・時間刻み・計測・入出力の行と
!   比較の 1 行だけを差し替えて書く。README を参照。
!   浅水流の設定はすべて既定値(力学波・急勾配の重力補正 2・MUSCL・開口の振り替え 3・
!   適応的ルンゲクッタ)。書いてあるのは比較ケースで差し替える行だけ
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! ENC 格子
  f_govequation = 0         ! 力学波(既定)。比較ケースは 1(拡散波)・2(運動波)
  dt = 1                    ! 時間刻み (s)。dt = 0.1 Δx(クーラン数 0.5 以下)
  t0 = 0
  tt = 10800                ! 3 時間(雨は 90 分で止み、逓減まで見る)
  dt_disp = 60              ! 1 分ごとの Log(S 列を水収支に使う)
  dt_file = 1800            ! 分布出力 30 分おき(水深 H。45° の 4 近傍で水が止まる場所を見る)
  dt_recrd = 60             ! 測線・プローブの出力 1 分ごと
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_precip = '-'
  fn_record = '-'
  fn_enc = '-'
  dir_data = 'data_dx10_r0'
  dir_result = 'result_dx10_r0'
  f_disp_debug = 1          ! S 列を全有効桁で(降雨量と流出量の収支を 0.1 % の桁で照合する)
/

&list_enc
  p_diagratio = 0.5857864376  ! 斜めのエッジへの通過幅の配分(既定 = 8 本の通過幅が等しい値)。
                            !   4 近傍の比較ケースは 0
  f_opening_dynamic = 3     ! 塞がれた開口の振り替え(既定)。通過幅の調整の解説の比較ケースは 2
/

&list_initial
  f_htype = 1
  fn_hinit = 'hinit.txt'    ! 初期水深: 流域は乾いた状態、海セルだけ 1 m(水位 1 m の吸い込み)
/

&list_geoinfo
  nx = 108                  ! 格子数(流域 1000 × 1620 m + 出口の先の延長と海 6 セル + 余白 1 セル)
  ny = 164
  dx = 10
  dy = 10
  f_ztype = 1
  fn_z = 'z.txt'            ! 地盤高(流域の座標の式をセル中心で評価。水路は上流端で 1 m、出口で 21 m 低い)
  f_rntype = 1
  fn_rn = 'rn.txt'          ! マニングの粗度(斜面 0.015、水路 0.15)
  f_masktype = 1
  fn_mask = 'mask.txt'      ! 流域の長方形と出口の先の延長・海(外は領域外 = 壁)
  fn_sw = 'sw.txt'          ! 海域マスク(出口の先 3 セル。標高 0 m・水深 1 m に固定され、入った水は消える)
/

&list_precip
  prtype = 1                ! 一様な雨の時系列(時刻 (min)、強度 (mm/h)。直線補間)
  prval(:,1) = 0, 10.8
  prval(:,2) = 89.99, 10.8
  prval(:,3) = 90, 0.0      ! 90 分で止む(値の更新は 1 分ごとで、ステップの終わりの時刻で評価する)
/

&list_record
  pbxytype = 0              ! セル番号で指定(西端の列・北端の行が 1)
  pbxy(:,1) = 101, 82       ! 出口の水路のセル(水深の時系列。理論の等流水深と比べる)
  pbxy(:,2) = 51, 81        ! 斜面の脚(水路に接する斜面のセル)の中央
  flxytype = 0
  flxy(:,1) = 101, 163, 101, 2   ! 出口の計測面: 流域の下流端の列を右岸側の尾根から左岸側の尾根まで
                            !   (全幅。エッジ流量の符号付き和。始点→終点の右側 = 下流が正)
  flxy(:,2) = 2, 81, 101, 81     ! 斜面の脚: 左岸側の斜面の水路に接する列を上流端から出口まで
                            !   (斜面から水路へ入る流量。右側 = 水路側が正)
/
