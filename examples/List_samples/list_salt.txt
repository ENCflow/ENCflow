!======================================================================
!
! 淡塩2層設定 &list_salt の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_salt = "list_salt.txt"
!   のように指定して使用します(fn_salt 未指定なら淡塩2層は完全に
!   不活性 = 状態も計算も追加されません)。
!
!   地表水と浅層地下水(土層)に「塩水層」(常に最下層。鋭利界面)を
!   導入し、SWI2(MODFLOW)型の準静的近似で塩水くさび・海水浸入・
!   淡水レンズを扱います。水深 h・地下貯留 Hg は淡塩の合計のままで、
!   塩水層厚(Hss / Hgs)はその内数です。密度差は塩水層の駆動水頭
!   Φs = 水位 + ε×界面標高 にのみ現れます(ε = (ρs−ρf)/ρf)。
!   海域セル(&list_tide の海域マスク)は自動的に「全塩水・水位 =
!   潮位」の境界になります。設計は docs/swi_plan.md、決定は
!   developer.md §47。
!
!======================================================================
&list_salt
  ! 地表の塩水層(浅水流の移流への相乗り+底層重力流)
  f_salt_surf = 1

  ! 地下(土層)の塩水 zone(側方流+海側境界)。
  ! &list_gwflow の f_gwlateral=1 が前提(K_sh・sy0 を共有)
  f_salt_gw = 1

  ! 淡水・塩水の密度 (kg/m3)。ε = (ρs−ρf)/ρf を決める
  salt_rhof = 1000.0
  salt_rhos = 1025.0

  ! 塩水層のバロトロピック移流分担率 α [0,1]
  ! (底層は流れの分担が小さい効果。1 = 濃度比例 = 完全混合相当)
  salt_alpha = 1.0

  ! 底層重力流の界面 Manning 粗度
  salt_ni = 0.025

  ! 初期の地表塩水層厚(既存 h の内数として指定。min(h, 値) に切り詰め)
  salt_hss0 = 0.0
  !fn_salt_hss0 = "hss0.txt"

  ! 初期の地下塩水貯留(塩水の追加投入 = Hg にも加算される)
  salt_hgs0 = 0.0
  !fn_salt_hgs0 = "hgs0.txt"

  ! 数値パラメータ(通常変更不要)
  salt_eps = 1.0e-3         ! 乾燥判定の正則化量 (m)
  salt_eps_h = 1.0e-2       ! 重力流 sqrt 則の線形化幅(水頭差 m)
  !salt_diagratio = 0.5857864 ! 対角方向の通過幅配分 [0,1]
  salt_nsubmax = 200        ! 地表重力流の適応サブサイクル上限

/
