!======================================================================
!
! 河道条件設定 &list_channel の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_channel = "list_channel.txt"
!   のように指定して使用します(fn_channel 未指定なら河道水理モデル
!   =堤防(仮想壁面)は完全に不活性)。
!
!   河道の水理構造モデル(堤防、サブグリッド河道幅)の
!   パラメータをここに集約します。線引きは:
!     - セルの地形・物性を作る量 = &list_geoinfo
!       (fn_rw: 河道マスク、depth_rw / fn_depth_rw: 河床掘り込み、
!        rn0_rw: 河道の粗度係数)
!     - エッジの通水構造と水理モード = 本ファイル
!   堤防の指定には &list_geoinfo の fn_rw(河道マスク)が必須です。
!
!   堤防高さは次のいずれか一方で指定します(両方指定はエラー):
!     fn_bank : 河道セル位置の堤防高さ分布ファイル(-900 以下は堤防なし)
!     bank0   : 全河道セルに適用する一律固定値
!
!   fn_bank の注意: 壁エッジは河道セル×堤内地セルの全エッジ(対角含む)に
!   立ち、天端には河道側セルの値が使われます。堤内地セルと角(対角)だけで
!   接する河道セルにも高さを与えないと、対角リンクから漏水します
!   (海岸堤防 fn_seawall の陸側セル持ちとは逆。users_guide/channel.md)
!
!======================================================================

&list_channel
  ! ---- 堤防(仮想壁面)。fn_bank または bank0 の指定で有効化 ----
  !fn_bank = "bank_height.txt"  ! 堤防高さ分布ファイル(河道セルのみ有効。
                                !   -900 以下は堤防なし。bank0 と排他)
  bank0 = 1.0                   ! 堤防高さの一律固定値(-900 以下は未指定扱い)

  f_bank_datum = 1              ! 堤防高さの基準
                                !   0: 河床(掘り込み後 z)基準
                                !   1: 堤内地セル標高基準(既定の推奨。
                                !      隣接堤内地セルにわたり f_bank_aggr で集約)
                                !   2: 絶対標高
  f_bank_aggr = 0               ! 天端の集約方法(f_bank_datum=1 のみ有効)
                                !   0: 平均(ノイズ抑制の既定)
                                !   1: 最小(越流開始の物理に忠実)
                                !   2: 最大(堤防機能の安全側)
  f_bank_mode = 0               ! 堤防の水理モード
                                !   0: 越流のみ(単純堤防。双方向とも天端まで
                                !      不透過、超えたら本間公式の堰越流)
                                !   1: 樋門(逆止弁。堤内→河道のみ通常計算、
                                !      河道→堤内は天端まで不透過)
                                !   2: 強制排水(天端以下では堤内の全水深で
                                !      段落ち。過去計算との互換・極端シナリオ
                                !      実験用。実務の排水機場は fn_structure の
                                !      &list_struct_pump を使うこと)
  f_bank_opening = 1            ! 堤防時の開口補正(既定 1)
                                !   壁で塞がれた斜め開口の通過幅シェアを
                                !   河道—河道の法線エッジへ振り替え、狭い
                                !   河道の過小通水(幅1セルで開口 41.4%)を
                                !   補正する。0 で旧挙動(A/B 比較用)

  ! ---- サブグリッド河道幅。fn_width の指定で有効化 ----
  !   セルサイズ未満の川幅の通水能と貯留を表現する(矩形断面、
  !   河道セルの水位は河道内水位の1本。セル内の非河道部の貯留は
  !   表現しない)。fn_bank / bank0 無指定なら「高さ0・堤内地セル
  !   標高基準」の堤防を自動有効化する。幅がセルサイズ以上のセルは
  !   自動的に従来の解像河道(掘り込み+壁)と同じ扱いになる
  !fn_width = "channel_width.txt"  ! 河道幅分布 (m。河道セルのみ有効。
                                   !   0 以下は幅情報なし=解像扱い)
  f_channel_advection = 1       ! 河道セルを含むエッジの移流項
!  p_sect_m = 0.5                ! 河道断面の形状指数 m(σ(h)=(h/D)^m。§26)。
                                 !   0=矩形=従来(既定)、0.5=放物線的、1=三角形的。
                                 !   遷移深さ D は堤防有効セルで天端−河床、なければ
                                 !   掘り込み深さ depth_rw/fn_depth_rw(どちらも無いと
                                 !   p_sect_m>0 はエラー停止)。低水〜中水の水位・逓減の
                                 !   再現が改善し、h≥D では従来の矩形力学に退化する
                                !   (1:通常(既定), 0:落とす=強い幅
                                !    不均質下での安定化オプション)
/

!----------------------------------------------------------------------
! 破堤(グループ不在=破堤なし。堤防の有効化が前提)
!   サイト=隣接する河道セル (ic,jc) と堤内地セル (il,jl) の対で、
!   その境界エッジ1本の仮想壁面の実効天端を時系列で変える:
!     実効天端 = 堤内地盤高 + 割合 ×(天端高 − 堤内地盤高)
!   割合は 0〜1(1=天端高のまま、0=堤内地盤高まで決壊)。時系列は
!   (時刻 min, 割合)の組で、間は線形補間・範囲外は端値保持。
!   幅1セル河道でも堤内地セル側の指定で左右岸を区別できる。
!   複数セル区間の決壊は複数サイトで指定する(決壊幅はエッジ開口幅の
!   粒度)。基準の堤内地盤高は計算開始時の値で固定(侵食に追従しない)
!----------------------------------------------------------------------
!&list_channel_breach
!  br_cell(:,1) = 120, 45, 120, 44         ! サイト1: ic, jc, il, jl
!  br_series(:,1,1) = 0.0,  1.0            ! (時刻 min, 割合 0〜1)
!  br_series(:,2,1) = 60.0, 1.0            !   60分までは無破堤
!  br_series(:,3,1) = 70.0, 0.2            !   60〜70分で 2 割まで決壊
!  br_series(:,4,1) = 90.0, 0.0            !   90分で堤内地盤高まで全壊
!  br_cell(:,2) = ...                      ! サイト2以降も同様
!/
