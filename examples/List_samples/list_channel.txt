!======================================================================
!
! 河道条件設定 &list_channel の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_channel = "list_channel.txt"
!   のように指定して使用します(fn_channel 未指定なら河道水理モデル
!   =堤防(仮想壁面)は完全に不活性)。
!
!   河道の水理構造モデル(堤防、将来のサブグリッド河道幅)の
!   パラメータをここに集約します。線引きは:
!     - セルの地形・物性を作る量 = &list_geoinfo
!       (fn_rw: 河道マスク、depth_rw / fn_depth_rw: 河床掘り込み、
!        rn0_rw: 河道の粗度係数)
!     - エッジの通水構造と水理モード = 本ファイル
!   堤防の指定には &list_geoinfo の fn_rw(河道マスク)が必須です。
!
!   堤防高さは次のいずれか一方で指定します(両方指定はエラー):
!     fn_bank : 河道セル位置の堤防高さ分布ファイル(-900 以下は堤防なし)
!     bank0   : 全河道セルに適用する一律固定値
!
!======================================================================

&list_channel
  ! ---- 堤防(仮想壁面)。fn_bank または bank0 の指定で有効化 ----
  !fn_bank = "bank_height.txt"  ! 堤防高さ分布ファイル(河道セルのみ有効。
                                !   -900 以下は堤防なし。bank0 と排他)
  bank0 = 1.0                   ! 堤防高さの一律固定値(-900 以下は未指定扱い)

  f_bank_datum = 1              ! 堤防高さの基準
                                !   0: 河床(掘り込み後 z)基準
                                !   1: 堤内地セル標高基準(既定の推奨。
                                !      隣接堤内地セルにわたり f_bank_aggr で集約)
                                !   2: 絶対標高
  f_bank_aggr = 0               ! 天端の集約方法(f_bank_datum=1 のみ有効)
                                !   0: 平均(ノイズ抑制の既定)
                                !   1: 最小(越流開始の物理に忠実)
                                !   2: 最大(堤防機能の安全側)
  f_bank_mode = 0               ! 堤防の水理モード
                                !   0: 越流のみ(単純堤防。双方向とも天端まで
                                !      不透過、超えたら本間公式の堰越流)
                                !   1: 樋門(逆止弁。堤内→河道のみ通常計算、
                                !      河道→堤内は天端まで不透過)
                                !   2: 強制排水(天端以下では堤内の全水深で
                                !      段落ち。排水能力無制限のため適用は
                                !      排水機場が実在する区間に限ること)
