!======================================================================
!
! 沿岸の波浪場の設定 &list_wavefield の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_wavefield = "list_wavefield.txt"
!   のように指定して使用します(fn_wavefield 未指定なら完全に不活性)。
!
!   沖の波(波高・周期・進行方向・方向分散)を外縁の辺から入れ、浅水変形・
!   屈折・砕波・底面摩擦・防波堤や島の背後の陰を、位相平均・代表周期 1 個・
!   方向ビンのエネルギー平衡で求めます(users_guide/wavefield.md、
!   developer.md §88)。wf_flow = 1, 2 で流れに作用し、セットアップ・沿岸流が
!   生じます(段 W2。既定 0 は作用なし)。f_wf_current = 1 で流れが波を変えます
!   (ドップラー効果・流れによる屈折・阻止。段 W3)。地形には作用しません。
!
!   必要な入力:
!     - 入射の辺(wf_bc_side に 1 が 1 つ以上)
!     - 沖の波 wf_hm0・wf_trep・wf_dir(または時系列 wf_val)
!   沖は海底地形を持つ通常のセル(f_htype = 2 などで水を張る)で表します。
!   海域マスクの海セル(sw)は波が通りません
!
!   向きは「進む向き」(x 軸 = 東から反時計回り、北 = 90°)。SWAN・XBeach の
!   「来る向き」とは 180° 違います
!
!   併用できない機能(指定すると停止): STG 格子・ネスティング
!
!   出力: Wh(波高。不規則波は H_m0)・Wa(平均波向)。f_wf_outdiag = 1 で
!   Wu(底面軌道流速 u_rms)・Wb(砕波散逸)も、wf_flow >= 1 なら外力 Wfx・Wfy・
!   Stokes 輸送 Wsx・Wsy・ローラー Wr も(y 成分は流速 v と同じく南が正)。
!   台帳 result/wavefield.csv
!
!======================================================================
&list_wavefield
  ! 一時無効化(fn_wavefield を書いたまま 0 にすると完全不活性)
  f_wavefield = 1

  ! 外縁の辺 W, E, N, S の型(0: 吸収 / 1: 入射 / 2: 側方 Neumann)。既定は全辺 0。
  ! 直線海岸なら沖の辺を 1、海岸に交わる左右の辺を 2(外にも同じ海が続く)にする
  wf_bc_side = 1, 0, 2, 2

  ! 沖の波(固定値)
  wf_hm0 = 1.5              ! 波高 (m。不規則波は有義波高 H_m0、規則波は H)
  wf_trep = 8.0             ! 代表周期 (s。不規則波はエネルギー周期 T_m-1,0 の解釈。T_p だけなら T_p/1.1 が目安)
  wf_dir = 0.0              ! 進行方向 (deg。東 = 0、北 = 90)
  !wf_spread = 20.0         ! 方向分散 σθ (deg。0 = 単一方向、上限 48.8。既定 20、規則波は 0)

  ! 沖の波の時系列(固定値の代わり): 時刻 (min)・波高・周期・方向・方向分散。
  ! 行の間は線形補間(方向は短い向きの差で補間)。方向分散を省略した行は既定
  !wf_val(1:5,1) = 0,   1.0, 8.0, 350.0, 20.0
  !wf_val(1:5,2) = 360, 2.0, 10.0, 10.0, 15.0
  !wf_clock = 0             ! 時系列の時計 0: 水理時間 / 1: 地形時間(MORFAC 倍)

  ! 解法
  !wf_mode = 2              ! 1: 非定常(毎ステップ物理時間で) / 2: 準定常(dt_wave ごと。既定)
  !dt_wave = 3600.0         ! 準定常の更新間隔 (s)
  !wf_tol = 1.0e-4          ! 準定常の収束判定(1 回の反復での波高の相対変化)
  !wf_itmax = 1000          ! 準定常の反復の上限(既定 4 max(nx, ny))
  !f_wf_init = 1            ! 1: 開始時に準定常解 / 0: 静穏から(非定常で前線を見るとき)
  !wf_ntheta = 40           ! 方向ビン数(8 の倍数、8〜360。既定 40 = 9°)

  ! 波の種類・砕波・摩擦(既定は XBeach 定常モード・SnapWave の値)
  !wf_wavetype = 1          ! 1: 不規則波 / 2: 規則波(H <= γh の飽和砕波。検定用)
  !wf_break = 1             ! 1: Baldock ら(1998) / 2: Janssen & Battjes(2007)
  !wf_gamma = 0.78          ! 砕波指標 γ
  !wf_alpha = 1.0           ! 砕波の散逸係数 α
  !wf_gammax = 0.6          ! 水深による上限 H_rms <= γmax h(不規則波)
  !wf_fw = 0.0              ! 底面摩擦係数 f_w(0 = 摩擦なし)
  !wf_hmin = 0.1            ! 波浪セルの最小水深 (m)

  ! 流れへの作用(段 W2)。使うときは海岸に交わる左右の辺を &list_bound_edge の
  ! f_bc_* = 3(側方 Neumann)にする
  !wf_flow = 0              ! 0: 作用なし(既定) / 1: 外力だけ / 2: 外力 + GLM の Stokes 輸送 + 波・流れの摩擦(推奨)
  !f_wf_roller = 1          ! 1: ローラーのエネルギー平衡(wf_flow >= 1。既定 1)
  !wf_beta = 0.1            ! ローラーの散逸係数 β
  !wf_tramp = 100.0         ! 外力の立ち上げ時間 (s。0 で立ち上げなし)

  ! 流れの波への作用(段 W3)。河口の噴流・潮流・離岸流の上の波の変化と阻止
  !f_wf_current = 0         ! 0: なし(既定) / 1: ドップラー効果・流れによる屈折と輸送・阻止

  ! 出力
  !f_wf_outdiag = 0         ! 1: 底面軌道流速 Wu・砕波散逸 Wb も分布出力(wf_flow >= 1 なら外力・Stokes 輸送・ローラーも)
/
