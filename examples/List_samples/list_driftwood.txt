!======================================================================
! 流木設定サンプル(&list_driftwood。developer.md §50)
!   有効化はメインパラメータの fn_driftwood = "このファイル" または "-"。
!   流木をセルごとの柱状材積 (m3/m2) のラスタ場として概算する
!   (個別流木の追跡・丸太が掛かるかどうかの幾何的な判定は対象外)。
!   台帳: 立木ストック → (水理的流失/侵食連行) → 流動流木 →
!   (接地・乾燥) → 堆積流木(再流動も可)。捕捉率の分布 fn_dwtrap を与えると、
!   流動流木 → (構造物での捕捉) → 集積流木 が加わり、集積が流れを堰き上げる
!   (カルバートの流入口の集積は、&list_struct_culvert の culv_jam = 1)。
!   出力: f_out_hd(流動 Hd0001…)・f_out_wd(堆積 Wd0001… +
!   期間最大到達量 Wd9999)・f_out_wj(集積 Wj0001…)・result/driftwood.csv(材積台帳)。
!   ※ 閾値・レートは校正パラメータ(実務的な標準値は未確立)。
!   ※ 物性・閾値・レートには既定値があり、&list_driftwood を空にしても動く
!     (立木ストックは陸のセル一様 0.001 m3/m2。採用された既定値は実行開始時に
!     "(default)" 印付きで表示される。既定の根拠は developer.md §50.6)
!======================================================================
&list_driftwood
  f_dw = 1                          ! 0 でファイルを残したまま一時無効化

  ! ---- 立木ストック(一様か分布の排他。両方未指定なら陸のセル一様 0.001)----
  dw_stock0 = 0.01                  ! 一様な立木ストック (m3/m2)
  !fn_dwstock = "stock.txt"         ! ストック分布 (m3/m2。地形と同じ行列形式。
                                    !   森林簿等からの材積換算は前処理で)

  ! ---- 流木の代表諸元(既定 0.3 m / 0.5)----
  dw_dlog = 0.3                     ! 代表直径 (m)
  dw_sglog = 0.5                    ! 材比重 ρ木/ρ水 (0〜1)。喫水(浮遊/接地の
                                    !   切替水深)は円柱の浮力平衡から自動導出
                                    !   (Braudrick & Grant 2000。sg=0.5 で半没)

  ! ---- 発生(流木化)。両方未指定なら両方に既定、片方だけ指定なら他方は無効 ----
  dw_wrec = 1.0e-5                  ! 水理的流失レート (m/s = m3/m2/s。指定で有効化)
  dw_hrec = 0.5                     ! 流失の浸水深閾値 (m。土石流時は混合流動深で判定)
  dw_vrec = 1.0                     ! 流失の流速閾値 (m/s)
  dw_droot = 0.5                    ! 根系深 (m。指定で侵食連行を有効化 —
                                    !   累積侵食深がこれを超えたセルの立木は全量流失。
                                    !   掃流砂・土石流・斜面崩壊のどの侵食とも自動連動)

  ! ---- 停止・堆積 ----
  dw_wstop = 0.01                   ! 接地堆積レート (m/s。既定 0.01)
  !dw_vstop = 0.05                  ! 低速堆積の流速閾値 (m/s。既定 0 = 喫水判定のみ)

  ! ---- 再流動(既定なし = 堆積した流木は動かない(安全側))----
  !dw_wfloat = 0.01                 ! 再流動レート (m/s。>0 で有効化)
  !dw_rfloat = 1.5                  ! 浮遊余裕率(水深 > rfloat×喫水 で再流動。>1)
  !dw_vfloat = 0.1                  ! 再流動の流速閾値 (m/s。dw_vstop 以上)

  ! ---- 構造物での捕捉と集積(fn_dwtrap の指定で有効。橋・透過型堰堤など)----
  !fn_dwtrap = "trap.txt"           ! 捕捉率 E の分布(0〜1。捕捉セルを 1 回通過する流木の
                                    !   うち捕捉される割合。0 = 捕捉しない。地形と同じ行列形式。
                                    !   捕捉率は与件で、モジュールは掛かるかどうかを判定しない)
  !f_dwjam = 1                      ! 集積の流れへの作用(0: 台帳だけ、1: 集積したセルの抵抗)
  !dw_jamcd = 1.0                   ! 集積体の抗力係数 C_D(校正値。既定 1.0 は仮の値)
  !dw_jamphi = 0.3                  ! 集積体の充填率(実材積 / 見かけの体積。0〜1。校正値。
                                    !   抵抗の係数 2 C_D/(π d (1−φ)³) に (1−φ)³ で効く)
  !dw_jamgrow = 0.05                ! 集積量 (m3/m2) がこれを超えたセルは捕捉率 1
                                    !   (未指定 = 捕捉率は一定)
  ! カルバートの流入口の集積(開口がセルより狭い構造物)は、&list_struct_culvert の
  ! culv_jam(n) = 1 で有効にする(fn_dwtrap は不要。f_dwjam・dw_jamcd・dw_jamphi は共通)
/
