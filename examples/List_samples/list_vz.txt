!======================================================================
!
! 鉛直不飽和浸透(VZ)流出設定 &list_vz / &list_vz_column / &list_vz_zone の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_vz = "list_vz.txt"
!   のように指定して使用します(fn_vz 未指定なら完全に不活性)。
!
!   谷(2023)の VZ 流出モデル: 十分深い共有参照柱の 1 次元 Richards 式
!   (小杉式)を解き、VZ 区画の各セルの土層厚の深さで下向き流束を読み出して
!   面積積分し、区画の排水を出口セル(または系外)へ瞬間的に移します
!   (users_guide/vz.md、developer.md §87)。
!
!   必要な入力:
!     - 区画番号ラスタ fn_vz_zone(dir_data 下。0 = 適用外)
!     - 土層厚(&list_geoinfo の f_sdtype / sd0 / fn_sd。既定なし)
!     - 柱の小杉パラメータと初期条件(既定なし)
!     - 区画ごとの出口 vz_out_kind(必須)
!
!   初版で併用できない機能(指定すると停止): fn_gwflow・fn_snow・
!   fn_glacier・fn_geomorph・fn_wq・fn_salt・ネスティング・STG 格子
!
!   出力: vz_zone.csv(区画の累積水量と収支残差)・vz_column.csv(柱の
!   積算流束と指定深さの排水・貯留)。VZ 貯留は Log の S_grnd に算入
!
!======================================================================
&list_vz
  ! 一時無効化(fn_vz を書いたまま 0 にすると完全不活性)
  f_vz = 1

  ! 区画番号ラスタ(整数。0 = 適用外、1, 2, … = 区画番号の連番)
  fn_vz_zone = "vz_zone.asc"

  ! 柱を進める同期間隔 (s)。0 = 毎ステップ(費用大)。既定 60
  !dt_vz = 60.0

  ! 柱の上端流束(雨)の扱い
  !   0: 柱を参照するセルの地上雨量の一致を要求(違えば停止。既定)
  !   1: 柱を参照する全セルの面積平均(雨に分布があるとき)
  !f_vz_rain = 0

  ! 界面透水係数 1: 算術平均(既定) / 2: 調和平均
  !f_vz_kface = 1

  ! 数値制御(通常は既定のまま。結果の確認には許容を 1/4 にした再計算と比べる)
  !vz_dt_init = 1.0         ! 内部刻みの初期値 (s)
  !vz_dt_min = 0.001        ! 内部刻みの下限 (s。下回ると停止)
  !vz_dt_max = 3600.0       ! 内部刻みの上限 (s)
  !vz_tol_theta = 1.0e-3    ! 時間誤差の θ の尺度(θs − θr に対する比)
  !vz_tol_psi = 1.0e-3      ! 時間誤差の ψ の尺度(絶対 m と相対の共通値)
  !vz_tol_q = 1.0e-2        ! 時間誤差の界面積算流束の相対尺度
  !vz_mb_rate = 1.0e-11     ! 柱の非線形水収支誤差の予算 (m/s)
  !vz_maxit = 25            ! Picard 反復の上限

  ! 柱の診断(vz_column.csv)に出す深さ (m。最大 20)
  !vz_out_depth = 0.5, 1.0, 2.0

  ! 1: 分布出力の時刻(dt_file)に柱の鉛直分布 vzprof_<柱>_<番号>.csv を出力
  !f_vz_outprof = 0
/

&list_vz_column
  ! 共有参照柱(柱番号 c で添字。1, 2, … の連番。最大 50)
  ! 小杉式のパラメータに既定値はありません(土壌の保水・透水試験から与える)。
  ! 下の値は書式の例で、特定の土壌の推奨値ではありません
  vz_ks_mmh(1) = 100.0      ! 飽和透水係数 (mm/h)
  vz_thetas(1) = 0.5        ! 飽和含水率
  vz_thetar(1) = 0.1        ! 残留含水率
  vz_psim(1) = -0.5         ! 中央値の圧力水頭 ψm (m。負)
  vz_sigma(1) = 1.5         ! 対数標準偏差 σ

  ! 計算柱の深さ (m) と層厚 (m)(深さは層厚の整数倍。既定 10 m・2.5 cm)。
  ! 読み出す最大の土層厚より十分深くとる
  !vz_depth(1) = 10.0
  !vz_dz(1) = 0.025

  ! 初期条件(どちらか一方):
  !   vz_q0_mmh: 定常重力排水の流束 (mm/h)。K(ψ0) = q0 の一様な ψ0(降雨前の流出率)
  !   vz_psi0  : 一様な圧力水頭 (m。負)
  vz_q0_mmh(1) = 1.0
  !vz_psi0(1) = -1.0

  ! 下端 1: 自由排水(既定) / 2: 固定圧(検証用。vz_psibot で ψ を与える)
  !f_vz_bottom(1) = 1
  !vz_psibot(1) = 0.0
/

&list_vz_zone
  ! VZ 区画(区画番号 z で添字。区画番号ラスタの番号)
  ! 参照する柱(省略時は区画番号と同じ。複数区画が同じ柱を共有してよい)
  vz_zone_col(1) = 1

  ! 出口(必須) 1: 地表セルへ移す / 2: 系外へ出す
  vz_out_kind(1) = 1

  ! 出口セル (i, j)(vz_out_kind = 1。最大 99。VZ セル・海セル・水位規定セルは不可)
  vz_out_cell(:,1,1) = 120, 80
  !vz_out_cell(:,2,1) = 121, 80

  ! 出口の配分率(和 1。1 セルなら省略可。複数セルなら全て必須)
  !vz_out_w(1:2,1) = 0.6, 0.4
/
