!======================================================================
!
! 初期条件設定 &list_initial の例(全パラメータの一覧)
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_initial = "list_initial.txt"
!   のように指定して使用します(fn_initial 未指定なら全域が乾いた
!   状態 h=0 から始まります。降雨流出・河川流入で水を入れる計算では
!   それで十分なことも多い)。
!
!   より現実的な「流れている場」から始めたいときは、助走計算の保存
!   状態を初期条件に使えます(&list_sysparam の f_state_restore = 2。
!   docs/users_guide/restart.md)。解説は docs/users_guide/initial.md。
!
!======================================================================
&list_initial

  ! 初期水深/水位の与え方
  !  0: 水深の固定値 h0(既定)
  !  1: 水深の分布ファイル fn_hinit
  !  2: 水位の固定値 e0(h = max(e0 - z, 0)。湖・貯水池・海面の初期化に)
  !  3: 水位の分布ファイル fn_hinit
  f_htype = 0
  h0 = 1.0                  ! 初期水深固定値 (m)
  !e0 = 5.0                 ! 初期水位固定値 (m。z と同じ標高基準。f_htype=2)
  !fn_hinit = "hinit.txt"   ! 初期水深/水位の分布ファイル(f_htype=1, 3)

  ! 河道マスク(&list_geoinfo の fn_rw)セルへの初期水深の増分 (m)
  ! (平水の河道を濡らしておく)
  !h0_rw = 0.5

  ! 初期流速(f_uvtype=0: 一様値のみ)
  !f_uvtype = 0
  !u0 = 0.0                 ! x方向初期流速 (m/s)
  !v0 = 0.0                 ! y方向初期流速 (m/s)

  ! 窪地の充填(流出先のない凹みをあらかじめ埋めてから開始する)
  !  0: なし(既定)
  !  1: 全域の窪地を満水にする(流出開始までの助走の省略)
  !  2: 河道セルのみ満水にする(河道縦断の偽ピットの充填)
  !  3: 河道セルのみ満水にした後、その水を地盤高へ転換(地形で埋める。
  !     初期水量を水収支に持ち込みたくない場合)
  !  2・3 は河道マスク fn_rw が前提。海へ排水できる窪地は埋めない
  !f_fill_depres = 0

  ! 初期場を加工するユーザールーチン識別名(理想化実験用。
  ! 登録名は src/user_initial.f90 冒頭)
  !f_user_routine = "wave_hump"
