!======================================================================
!
! 地下水条件設定 &list_gwflow / &list_gwflow_bucket の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_gwflow = "list_gwflow.txt"
!   のように指定して使用します(fn_gwflow 未指定なら地下水計算は
!   完全に不活性)。
!
!   共通設定 &list_gwflow と、選択したモデルの固有設定
!   (バケツモデルなら &list_gwflow_bucket)を同じこのファイルに
!   書きます。固有設定グループは選択されたモデルだけが読むため、
!   使わないモデルのグループが残っていても無視されます。
!
!======================================================================
! 地下水共通設定
!======================================================================
&list_gwflow
  ! 地下水モデルの選択
  !  0:なし(fn_gwflow を書いたまま一時無効化できる)
  !  1:バケツモデル(&list_gwflow_bucket を使用)
  f_gwmodel = 1

  ! 地下水計算の更新時間間隔 (s)
  !  0 なら毎ステップ更新
  !  0 以外を指定する場合は時間刻み dt 以上とすること
  dt_gwflow = 0

/


!======================================================================
! バケツモデル固有設定 (f_gwmodel = 1)
!======================================================================
!   セルごとの地下貯留(容量 gw_capacity)へ一定浸透能 gw_infil_mmh で
!   表面水を移す最簡モデル。水平流なし・鉛直交換のみ。
!   各ステップの浸透量は「浸透能×dt・表面水深・貯留残容量」の最小値。
!======================================================================
&list_gwflow_bucket
  ! 浸透能 (mm/h)。0 より大きいこと
  gw_infil_mmh = 10.0

  ! 地下貯留容量(柱状換算水深) (m)。0 より大きいこと
  gw_capacity = 0.5

/
