!======================================================================
!
! 地下水条件設定 &list_gwflow とモデル固有設定の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_gwflow = "list_gwflow.txt"
!   のように指定して使用します(fn_gwflow 未指定なら地下水計算は
!   完全に不活性)。
!
!   共通設定 &list_gwflow と、選択したモデルの固有設定
!   (&list_gwflow_bucket / &list_gwflow_greenampt 等)を同じこの
!   ファイルに書きます。固有設定グループは選択されたモデルだけが
!   読むため、使わないモデルのグループが残っていても無視されます
!   (選択子を 0 に切り替えるだけで A/B 比較ができます)。
!
!   土層厚(sd)と比湧水量(sy0)は地理情報の一部として
!   &list_geoinfo 側で設定します(粗度係数と同じ流儀):
!     sd0 = 1.0          ! 土層厚固定値 (m)(f_sdtype=0 のとき)
!     f_sdtype = 1       ! 0:固定値 sd0, 1:ファイル fn_sd
!     fn_sd = "sd.txt"   ! 土層厚分布ファイル
!     sy0 = 0.2          ! 比湧水量(= 有効間隙率 n_e)
!   土層厚を必要とするモデル(Green-Ampt)を選んだときだけ
!   メモリが確保されます。
!
!======================================================================
! 地下水共通設定
!======================================================================
&list_gwflow
  ! 鉛直モデルの選択
  !  0:なし(fn_gwflow を書いたまま一時無効化できる)
  !  1:バケツモデル(&list_gwflow_bucket を使用)
  !  2:Green-Ampt モデル(&list_gwflow_greenampt を使用。
  !    &list_geoinfo の土層厚 sd・比湧水量 sy0 が必要)
  f_gwvertical = 2

  ! 側方モデルの選択
  !  0:なし(鉛直浸透のみ。側方流動の資源は一切確保されない)
  !  1以降は予約(未実装。指定するとエラー停止)
  f_gwlateral = 0

  ! 地下水計算の更新時間間隔 (s)
  !  0 なら毎ステップ更新
  !  0 以外を指定する場合は時間刻み dt 以上とすること
  !  (モデルには実効時間刻み dts = dt×間引き数が渡される)
  dt_gwflow = 0

/


!======================================================================
! バケツモデル固有設定 (f_gwvertical = 1)
!======================================================================
!   セルごとの地下貯留(容量 gw_capacity)へ一定浸透能 gw_infil_mmh で
!   表面水を移す最簡モデル。水平流なし・鉛直交換のみ。
!   各ステップの浸透量は「浸透能×dts・表面水深・貯留残容量」の最小値。
!======================================================================
&list_gwflow_bucket
  ! 浸透能 (mm/h)。0 より大きいこと
  gw_infil_mmh = 10.0

  ! 地下貯留容量(柱状換算水深) (m)。0 より大きいこと
  gw_capacity = 0.5

/


!======================================================================
! Green-Ampt モデル固有設定 (f_gwvertical = 2)
!======================================================================
!   Green-Ampt 式(ピストン近似)の浸透能
!     f_v = K_sv * (1 + psi_f * n_e / F)   (F: 累積浸透深)
!   で表面水を地下貯留へ移す。セルごとの貯留容量は土層厚×比湧水量
!   (&list_geoinfo の sd, sy0)。飽和したセルは浸透が止まり、
!   超過分は表面水として浅水方程式側が扱う。
!======================================================================
&list_gwflow_greenampt
  ! 鉛直飽和透水係数 (mm/h)。0 より大きいこと
  gw_ksv_mmh = 36.0

  ! 湿潤前線の毛管圧力水頭 (m)。0 以上(0 なら一定浸透能 K_sv に退化)
  gw_psif = 0.1

/
