!======================================================================
!
! 地下水条件設定 &list_gwflow とモデル固有設定の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_gwflow = "list_gwflow.txt"
!   のように指定して使用します(fn_gwflow 未指定なら地下水計算は
!   完全に不活性)。
!
!   共通設定 &list_gwflow と、選択したモデルの固有設定
!   (&list_gwflow_bucket / &list_gwflow_greenampt 等)を同じこの
!   ファイルに書きます。固有設定グループは選択されたモデルだけが
!   読むため、使わないモデルのグループが残っていても無視されます
!   (選択子を 0 に切り替えるだけで A/B 比較ができます)。
!
!   土層厚(sd)と比湧水量(sy0)は地理情報の一部として
!   &list_geoinfo 側で設定します(粗度係数と同じ流儀):
!     sd0 = 1.0          ! 土層厚固定値 (m)(f_sdtype=0 のとき)
!     f_sdtype = 1       ! 0:固定値 sd0, 1:ファイル fn_sd
!     fn_sd = "sd.txt"   ! 土層厚分布ファイル
!     sy0 = 0.2          ! 比湧水量(= 有効間隙率 n_e)
!   土層厚を必要とするモデル(Green-Ampt)を選んだときだけ
!   メモリが確保されます。
!
!======================================================================
! 地下水共通設定
!======================================================================
&list_gwflow
  ! 鉛直モデルの選択
  !  0:なし(fn_gwflow を書いたまま一時無効化できる)
  !  1:バケツモデル(&list_gwflow_bucket を使用)
  !  2:Green-Ampt モデル(&list_gwflow_greenampt を使用。
  !    &list_geoinfo の土層厚 sd・比湧水量 sy0 が必要)
  f_gwvertical = 2

  ! 側方モデルの選択
  !  0:なし(鉛直浸透のみ。側方流動の資源は一切確保されない)
  !  1:非線形 Boussinesq(&list_gwflow_lateral を使用。
  !    &list_geoinfo の土層厚 sd・比湧水量 sy0 が必要)
  f_gwlateral = 1

  ! 風化基岩層(第2地下水層)の有効化
  !  0:なし(層2の資源は一切確保されない)
  !  1:有効(&list_gwflow_layer2 を使用。バケツ鉛直モデルとは併用不可)
  !f_gwlayer2 = 1

  ! 管路連続体層(下水道網・岩盤亀裂網などの等価被圧連続体)の有効化
  !  0:なし(管路層の資源は一切確保されない)
  !  1:有効(&list_gwflow_conduit を使用)
  !f_gwconduit = 1

  ! 井戸揚水・地下水取水(シンク)の有効化
  !  0:なし(資源は一切確保されない)
  !  1:有効(&list_gwflow_pump を使用)
  !f_gwpump = 1

  ! 地下水計算の更新時間間隔 (s)
  !  0 なら毎ステップ更新
  !  0 以外を指定する場合は時間刻み dt 以上とすること
  !  (モデルには実効時間刻み dts = dt×間引き数が渡される)
  dt_gwflow = 0

/


!======================================================================
! バケツモデル固有設定 (f_gwvertical = 1)
!======================================================================
!   セルごとの地下貯留(容量 gw_capacity)へ一定浸透能 gw_infil_mmh で
!   表面水を移す最簡モデル。水平流なし・鉛直交換のみ。
!   各ステップの浸透量は「浸透能×dts・表面水深・貯留残容量」の最小値。
!   側方モデル(f_gwlateral=1)とは併用不可(容量の二重定義になるため
!   エラー停止。土層厚を容量とする定数浸透能が必要なら Green-Ampt の
!   gw_psif=0 を使うこと)。
!======================================================================
&list_gwflow_bucket
  ! 浸透能 (mm/h)。0 より大きいこと
  gw_infil_mmh = 10.0

  ! 地下貯留容量(柱状換算水深) (m)。0 より大きいこと
  gw_capacity = 0.5

/


!======================================================================
! Green-Ampt モデル固有設定 (f_gwvertical = 2)
!======================================================================
!   Green-Ampt 式(ピストン近似)の浸透能
!     f_v = K_sv * (1 + psi_f * n_e / F)   (F: 累積浸透深)
!   で表面水を地下貯留へ移す。セルごとの貯留容量は土層厚×比湧水量
!   (&list_geoinfo の sd, sy0)。飽和したセルは浸透が止まり、
!   超過分は表面水として浅水方程式側が扱う。
!======================================================================
&list_gwflow_greenampt
  ! 鉛直飽和透水係数 (mm/h)。0 より大きいこと
  gw_ksv_mmh = 36.0

  ! 湿潤前線の毛管圧力水頭 (m)。0 以上(0 なら一定浸透能 K_sv に退化)
  gw_psif = 0.1

  ! K_sv・psi_f の分布マップ(任意。土地利用別の浸透能・透水性舗装等)。
  ! 指定するとマップが一様値より優先される。マップの K_sv は 0 を許し、
  ! 0 のセルは不浸透(完全舗装)になる
  !fn_gw_ksv = "ksv.txt"      ! K_sv マップ (mm/h)
  !fn_gw_psif = "psif.txt"    ! psi_f マップ (m)

/

!======================================================================
! 非線形 Boussinesq 側方流固有設定 (f_gwlateral = 1)
!======================================================================
!   飽和帯の側方 Darcy 流(2次元・8近傍。ENC 本体と同じ対角配分)。
!   全水頭
!     H = (地盤高 - 土層厚) + 飽和帯厚 [+ 地表湛水深(飽和セルのみ)]
!   の勾配で駆動され、飽和容量を超えたセルの超過分は地表流へ渡される。
!   陽解法の安定条件
!     dt <= 0.5 * sy0 * dx*dy / (K_sh * max(土層厚) * Σ_k l_k/dr_k)
!   を初期化時に検査し、超える場合は上限値を表示して停止する。
!======================================================================
&list_gwflow_lateral
  ! 側方飽和透水係数 (mm/h)。0 より大きいこと
  gw_ksh_mmh = 360.0

  ! 乾燥判定・過大流出抑制の正則化厚 (m)。0 より大きいこと(既定 1e-3)
  gw_eps = 1.0e-3

  ! 対角方向の通過幅配分 [0,1](既定 2/(2+sqrt(2)) = ENC 本体の
  ! p_diagratio と同値。通常は変更不要。0 で4近傍相当)
  !gw_diagratio = 0.5857864

/


!======================================================================
! 風化基岩層固有設定 (f_gwlayer2 = 1)
!======================================================================
!   土層の直下に厚さ gw2_depth・比湧水量 gw2_sy の第2層を挟む
!   (容量 = gw2_depth×gw2_sy)。層1から一定浸透能 gw2_infil_mmh で
!   下方浸透し、gw2_ksh_mmh > 0 なら層2内を側方 Darcy 流
!   (水頭 = (地盤高 − 土層厚 − gw2_depth) + 飽和帯厚。地表湛水結合なし)
!   で移動する。層2の飽和超過は層1へ戻り、層1の超過は地表へ湧出する。
!   逓減の遅い成分(基底流)・長い滞留時間を持たせるための層。
!   蒸発散は層2に触れない。バケツ鉛直モデル(f_gwvertical=1)とは併用不可。
!   Hg2 場は f_out_hg=1 のとき出力される。リスタートは私有ファイル
!   gwflow_layer2.dat(save 時に f_gwlayer2 が有効だったことが必要)。
!======================================================================
&list_gwflow_layer2
  ! 層厚 (m)。0 より大きいこと
  gw2_depth = 2.0

  ! 比湧水量(有効間隙率)。(0,1]
  gw2_sy = 0.05

  ! 層1→2 の浸透能 (mm/h)。0 より大きいこと
  gw2_infil_mmh = 5.0

  ! 層2の側方飽和透水係数 (mm/h)。0 で側方なし(容量バッファとして動作)
  gw2_ksh_mmh = 360.0

  ! 初期飽和度 [0,1](既定 0 = 空。イベント計算では実質的に基底流量を
  ! 決める校正パラメータ。長期計算はスピンアップ→save→restore を推奨)
  gw2_sat0 = 0.0

/


!======================================================================
! 管路連続体層固有設定 (f_gwconduit = 1)
!======================================================================
!   下水道網・岩盤亀裂網・カルスト・農地暗渠などのサブグリッド管路網を、
!   セル別容量 cap と 8 方向通水能を持つ「人工被圧帯水層」として表す
!   (等価連続体近似。設計は docs/gwconduit_plan.md、決定は
!   developer.md §46)。満管で疑似スロットの被圧に切り替わり、
!   圧力流(サーチャージ)とマンホール噴出を表現する。
!   用途はスイッチの組み合わせ:
!     都市下水道 = 枡密度あり・層間交換なし(浸入水は gwc_leak_layer=1)
!     岩盤・カルスト = 枡密度なし(吸込穴があれば指定)・gwc_leak_layer=2
!   陽解法の安定条件は初期化時に検査される(上限は gwc_slot_sy が支配。
!   超える場合は dt_gwflow を下げるか slot_sy・eps_h を緩める)。
!   Hgc 場は f_out_hg=1 のとき出力。リスタートは私有ファイル
!   gwflow_conduit.dat(save 時に f_gwconduit が有効だったことが必要)。
!======================================================================
&list_gwflow_conduit
  ! 側方の流束則(1:線形, 2:sqrt 乱流管路流)
  f_gwc_fluxlaw = 2

  ! 通水能密度 (m2/s。満管時・単位動水勾配の流量/単位幅)。
  ! 一様値かマップ(fn_gwc_cnd)。0 で側方通水なし(貯留・交換のみ)
  gwc_cnd_m2s = 0.02
  !fn_gwc_cnd = "gwc_cnd.txt"

  ! 貯留容量(柱状換算 m = 管路総体積/セル面積)。一様値かマップ。
  ! 0 のセルは管路なし(貯留・通水・交換とも無効)
  gwc_cap = 0.01
  !fn_gwc_cap = "gwc_cap.txt"

  ! 水頭底(管底)標高: 一様埋設深(zbot = z − gwc_depth)か標高マップ
  gwc_depth = 3.0
  !fn_gwc_bot = "gwc_bot.txt"

  ! 貯留係数: 不圧枝(管内充水相当。(0,1])と被圧枝(疑似スロット。
  ! ≤ gwc_sy。小さいほど硬い圧力応答だが dt 上限が厳しくなる)
  gwc_sy = 0.05
  gwc_slot_sy = 0.001

  ! 初期充満率 [0,1](既定 0 = 空)
  gwc_sat0 = 0.0

  ! 雨水枡・人孔・吸込穴の密度 (個/m2)。一様値かマップ。
  ! 指定すると地表交換(堰式流入/オリフィス式噴出)が有効化される
  gwc_inlet = 0.01
  !fn_gwc_inlet = "gwc_inlet.txt"

  ! 地表交換の係数: 堰式(1 個あたり q = cw·h^1.5 m3/s)と
  ! オリフィス式(q = co·sqrt(2gΔH)。co = Cd×開口面積 m2)
  gwc_cw = 2.66
  gwc_co = 0.15

  ! 層間交換(0:なし, 1:土層と交換 = 浸入水・漏水,
  !           2:風化基岩層と交換 = 岩盤の涵養・漏出。要 f_gwlayer2=1)
  gwc_leak_layer = 0
  ! 層間交換能 (mm/h)。gwc_leak_layer > 0 のとき必須
  !gwc_leak_mmh = 5.0

  ! 数値パラメータ(通常変更不要)
  gwc_eps = 1.0e-3          ! 乾燥判定の正則化量 (m)
  gwc_eps_h = 1.0e-2        ! sqrt 則の線形化幅(水頭差 m)
  !gwc_diagratio = 0.5857864 ! 対角方向の通過幅配分 [0,1]

/

!======================================================================
! 井戸揚水・地下水取水固有設定 (f_gwpump = 1)
!======================================================================
!   セル集合+流量で指定した揚水量を地下貯留から汲み上げて系外へ除去
!   する(内部ソース &list_bound_source の地下版)。井戸は最大 50 個、
!   番号は 1 から連続。需要はセル集合へ等分配し、各セルの取水はその
!   セルの貯留で頭打ち(涸れ井戸。計算終了時に需要・実取水の総括が
!   表示される)。淡塩2層(f_salt_gw=1)併用時は淡水優先で汲む
!   (足りない分は塩水の混入)。注入(負の流量)は非対応(地表への
!   還元は &list_bound_source で)。
!======================================================================
&list_gwflow_pump
  ! 井戸1: セルをインラインで指定し、一定流量で取水
  gwp_cell(1:2,1,1) = 15, 22     ! 取水セル (i, j)(複数指定可 = 井戸群)
  !gwp_cell(1:2,2,1) = 16, 22    ! 2つ目のセル(需要は等分配)
  gwp_q0(1) = 0.02               ! 一定揚水量 (m3/s。正 = 取水)
  !gwp_layer(1) = 1              ! 取水層(1:土層(既定), 2:風化基岩層。
                                 !   2 は f_gwlayer2=1 が必要)

  ! 井戸2: セル一覧ファイル+流量時系列ファイルで指定
  !fn_gwp_cell(2) = 'well2.txt'  ! セル一覧(各行 "i j")
  !fn_gwp_val(2) = 'qwell2.txt'  ! 流量時系列(各行 "分 m3/s")
  !gwp_val(1:2,1,2) = 0.0, 0.01  ! (インライン時系列 (分, m3/s) も可)

/
