!======================================================================
! 溶岩流設定サンプル(&list_lavaflow。developer.md §51、docs/lava_plan.md)
!   有効化はメインパラメータの fn_lavaflow = "このファイル" または "-"。
!   他モジュールへの依存はない(水・気象は不要)。
!   状態は溶岩厚 s%hl(m)。場出力 Hl(m)、リスタートは私有ファイル
!   lavaflow.dat。固化分は地盤高 z の増加として現れる(sd は増えない =
!   侵食されない岩盤)。地表水は流動中の溶岩に不可視で z の上を流れる。
!   力学は等温の深さ平均 Bingham 粘性重力流(潤滑近似)。レオロジーは
!   ラン全体で1組(噴火口別の粘度指定はない — 温度拡張の将来枠)。
!======================================================================
&list_lavaflow
  f_lavaflow = 1            ! 0 でファイルを残したまま一時無効化

  ! ---- 更新間隔 ----
  !dt_lavaflow_c = "10 s"   ! 溶岩更新の間隔(未指定 = 毎ステップ)

  ! ---- レオロジー(直接入力。ラン全体で1組)----
  lv_rho = 2600.0           ! 密度 ρ (kg/m3)
  lv_visc = 1.0e4           ! 粘度 η (Pa s。必須。玄武岩質 1e2〜1e4、
                            !   安山岩質以上 >=1e5 が目安。η が小さいほど
                            !   陽解法のサブサイクル数が増える — plan §6)
  lv_tauy = 2000.0          ! 降伏応力 τ_y (Pa。0 = Newton 流体。
                            !   斜面 tanθ の停止厚 = τ_y/(ρg tanθ))

  ! ---- 固化(停止セルの hl → z 転換。0 = 固化なし)----
  lv_wsol = 0.005           ! 固化レート (m/s)
  lv_vsol = 5.0e-4          ! 停止判定の速度閾値 (m/s。lv_wsol>0 で必須。
                            !   目安 1e-4〜1e-3。小さいほど停止厚が
                            !   理論値 h∞ に近づくが停止判定まで長くかかる)

  ! ---- サブサイクル ----
  !lv_cfl = 0.4             ! 陽解法の安全係数 (0-1)
  !lv_nsubmax = 10000       ! 1回の更新のサブサイクル数上限(暴走ガード)

  ! ---- 噴火口(番号 n は 1 から連続。最大 20)----
  !   噴出率はセル集合全体の総量で、各セルへ均等分配(運動量なし)。
  !   割れ目噴火はセル列挙で表現する
  lv_cell(1:2,1,1) = 55, 40 ! 噴火口1のセル (i, j)
  lv_cell(1:2,2,1) = 55, 41
  lv_val(1:2,1,1) = 0, 10.0 ! 噴出率時系列(時刻(分), m3/s。線形補間。
  lv_val(1:2,2,1) = 60, 5.0 !   同時刻を2回書くと階段状に切替)
  lv_val(1:2,3,1) = 120, 0.0
  !lv_q0(1) = 10.0          ! 一定噴出率 (m3/s。lv_val の代わり)
  !fn_lv_cell(1) = "vent1_cells.txt"  ! セル集合ファイル(i j の行列挙。優先)
  !fn_lv_val(1) = "vent1_q.txt"       ! 時系列ファイル(分, m3/s。最優先)
/
