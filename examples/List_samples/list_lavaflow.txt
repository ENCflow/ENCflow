!======================================================================
! 溶岩流設定サンプル(&list_lavaflow。developer.md §51、docs/lava_plan.md)
!   有効化はメインパラメータの fn_lavaflow = "このファイル" または "-"。
!   等温モードは他モジュールに依存しない(水・気象は不要)。温度モード
!   (f_lvtemp = 1)は fn_meteo があればその気温を使う。
!   状態は溶岩厚 s%hl(m)。場出力 Hl(m)、リスタートは私有ファイル
!   lavaflow.dat。固化分は地盤高 z の増加として現れる(sd は増えない =
!   侵食されない岩盤。固化したセルにあった土層は溶岩の下に埋まり sd = 0)。
!   地表水は流動中の溶岩に不可視で z の上を流れる。
!   力学は深さ平均 Bingham 粘性重力流(潤滑近似)。既定は等温で、
!   f_lvtemp = 1 で溶岩の温度(冷えて止まる流れ)を計算する(lava_plan.md §8、
!   developer.md §51.7)。レオロジーはラン全体で1組(温度モードでは基準温度での値。
!   噴火口ごとの違いは噴出温度で表す)。
!======================================================================
&list_lavaflow
  f_lavaflow = 1            ! 0 でファイルを残したまま一時無効化

  ! ---- 更新間隔 ----
  !dt_lavaflow_c = "10 s"   ! 溶岩更新の間隔(未指定 = 毎ステップ)

  ! ---- レオロジー(直接入力。ラン全体で1組)----
  lv_rho = 2600.0           ! 密度 ρ (kg/m3)
  lv_visc = 1.0e4           ! 粘度 η (Pa s。既定 1e4 = 玄武岩質アア溶岩。
                            !   玄武岩質 1e2〜1e4、安山岩質以上 >=1e5 が目安。
                            !   η が小さいほど陽解法のサブサイクル数が増える
                            !   — plan §6)
  lv_tauy = 2000.0          ! 降伏応力 τ_y (Pa。0 = Newton 流体。
                            !   斜面 tanθ の停止厚 = τ_y/(ρg tanθ))

  ! ---- 固化(停止セルの hl → z 転換。0 = 固化なし)----
  lv_wsol = 0.005           ! 固化レート (m/s)
  lv_vsol = 5.0e-4          ! 停止判定の速度閾値 (m/s。lv_wsol>0 で読まれる。
                            !   既定 5e-4。目安 1e-4〜1e-3。小さいほど停止厚が
                            !   理論値 h∞ に近づくが停止判定まで長くかかる)

  ! ---- 温度(f_lvtemp = 1。冷えて止まる流れ。lava_plan.md §8)----
  !   温度モードの値には既定値がない(すべて明示する。未指定は停止)。以下の
  !   値は例(test/lava 構成 9 と同じ組)で、既定でも推奨値でもない。対象の
  !   溶岩に合わせて決める(ユーザーガイドの溶岩流の章「温度モード」)。
  !   温度モードは固化を温度で判定する(Tl <= lv_tsol で柱ごと z へ)ので、
  !   上の lv_wsol は 0 にする(lv_vsol は読まれない)
  !f_lvtemp = 1             ! 1: 温度を計算する(既定 0 = 等温)
  !f_lvtemp_scheme = 3      ! 熱の移流の面の温度 (既定 3 = MUSCL: 格子に斜めの流れの横断の数値拡散を
                            !   1 次風上の約 1/10 にする。1 = 送り手の温度(1 次風上。2026-11-05 より前の
                            !   既定)。developer.md §94)
  !lv_temp0(1) = 1150.0     ! 噴火口1の噴出温度 (°C。lv_tsol より高く)
  !lv_tref = 1150.0         ! 基準温度 (°C。lv_visc・lv_tauy はこの温度の値)
  !lv_bvisc = 0.04          ! 粘度の温度感度 β_η (1/K。ΔT 冷えて k 倍なら ln k/ΔT)
  !lv_btau = 0.04           ! 降伏応力の温度感度 β_τ (1/K)
  !lv_tsol = 1000.0         ! 固化温度 (°C)
  !lv_emis = 0.95           ! 有効放射率 (0-1。表面の殻の断熱を含めた有効値)
  !lv_hconv = 70.0          ! 対流の熱伝達係数 (W/m2/K)
  !lv_cp = 1150.0           ! 比熱 (J/kg/K。結晶化の潜熱を含めた有効値でよい)
  !lv_tair = 20.0           ! 気温 (°C。fn_meteo がないときだけ書く)

  ! ---- サブサイクル ----
  !lv_cfl = 0.4             ! 陽解法の安全係数 (0-1)
  !lv_nsubmax = 10000       ! 1回の更新のサブサイクル数上限(暴走ガード)
  !lv_diagratio = 0.0       ! 流動の斜めの配分 d (既定 0 = 4 近傍。1/3 = 8 近傍の 9 点と、塞がれた
                            !   斜めの配分の動的な振り替え。格子に斜めの細い谷の流れに。developer.md §96)

  ! ---- 噴火口(番号 n は 1 から連続。最大 20)----
  !   噴出率はセル集合全体の総量で、各セルへ均等分配(運動量なし)。
  !   割れ目噴火はセル列挙で表現する
  lv_cell(1:2,1,1) = 55, 40 ! 噴火口1のセル (i, j)
  lv_cell(1:2,2,1) = 55, 41
  lv_val(1:2,1,1) = 0, 10.0 ! 噴出率時系列(時刻(分), m3/s。線形補間。
  lv_val(1:2,2,1) = 60, 5.0 !   同時刻を2回書くと階段状に切替)
  lv_val(1:2,3,1) = 120, 0.0
  !lv_q0(1) = 10.0          ! 一定噴出率 (m3/s。lv_val の代わり)
  !fn_lv_cell(1) = "vent1_cells.txt"  ! セル集合ファイル(i j の行列挙。優先)
  !fn_lv_val(1) = "vent1_q.txt"       ! 時系列ファイル(分, m3/s。最優先)
/
