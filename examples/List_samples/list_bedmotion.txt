!======================================================================
! 規定底面運動設定サンプル(&list_bedmotion。developer.md §70、docs/bedmotion_plan.md)
!   有効化はメインパラメータの fn_bedmotion = "このファイル" または "-"。
!   地盤変位の時刻歴(スナップショット表 + 累積変位ラスタ)で z を強制する。
!   h は変えず水面 e = z + h が持ち上がる(断層津波の発生機構)。増分適用
!   なので geomorph・lavaflow と同居可。海域マスクのセルは動かない
!   (津波は「解く海」f_htype=2 で)。リスタートに私有状態なし。
!======================================================================
&list_bedmotion
  f_bedmotion = 1           ! 0 でファイルを残したまま一時無効化
  fn_bmlist = 'bmlist.txt'  ! スナップショット表: 1 行「時刻 ファイル名」(dir_data 相対。
                            !   時刻は t0 からの相対・昇順。ファイルは全域の累積変位 d (m)。
                            !   z と同じ形式。表の前は先頭の値、後は末尾の値を保持)
  !f_bminterp = 1           ! 時間補間(1: 線形, 0: 階段)
  !bm_tscale = 1.0          ! 表の時刻の単位 (s。60 で分)
/
