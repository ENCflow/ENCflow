!======================================================================
!
! 降水条件設定 &list_precip の例
!
!   以下に例示した複数のリスト &list_precip_X のうち、
!   有効にしたいものを &list_precip に修正して使用してください。
!   それ以外のものは無視されるのでそのまま残しておいて大丈夫です。
!
!======================================================================
! 降水条件設定
!======================================================================
&list_precip_0
  ! 降水データのタイプ
  !  0:降水なし
  !  1:設定ファイル内で一様分布の時系列
  !  2:分布ファイル × 設定ファイル内で倍率
  !  3:時系列分布ファイル(fn_maplistで指定)
  prtype = 0

  ! 降水なしの場合は、システムパラメータ&list_sysparam内で
  ! fn_precip="" と空文字で指定するか、fn_precip自体を省略してもよい
/


!======================================================================
! 降水条件設定
!======================================================================
&list_precip_1
  ! 降水データのタイプ
  !  0:降水なし
  !  1:設定ファイル内で一様分布の時系列
  !  2:分布ファイル × 設定ファイル内で倍率
  !  3:時系列分布ファイル(fn_maplistで指定)
  prtype = 1

  ! 降水分布の更新時間間隔 (min)
  dt_prupdate = 1

  ! 降水の時系列 (時刻(min), 雨量強度(mm/h)) for prtype=1
  !  最後の指定時刻以降は最後の雨量強度で継続
  prval(:,1) =    0,   0
  prval(:,2) =   10,  15
  prval(:,3) = 1440,  15
  prval(:,4) = 1450,   0

/


!======================================================================
! 降水条件設定
!======================================================================
&list_precip_2
  ! 降水データのタイプ
  !  0:降水なし
  !  1:設定ファイル内で一様分布の時系列
  !  2:分布ファイル × 設定ファイル内で倍率
  !  3:時系列分布ファイル(fn_maplistで指定)
  prtype = 2

  ! 降水分布の更新時間間隔 (min)
  dt_prupdate = 1

  ! 降水分布ファイル (mm/day) for prtype=2
  !  システムパラメータで指定したデータディレクトリからの相対パス
  fn_prmap = "precip/ProFloodPrecip_200y_TM_all_station_30m.bil"

  ! 分布ファイルに対する倍率 (時刻(min), 倍率(0.0～1.0)) for prtype=2
  !  指定する数は任意
  !  最後の指定時刻以降は最後の倍率で継続
  prval(:,1) =     0,   0
  prval(:,2) =    10,   1
  prval(:,3) =  1440,   1
  prval(:,4) =  1450,   0

/


!======================================================================
! 降水条件設定
!======================================================================
&list_precip_3
  ! 降水データのタイプ
  !  0:降水なし
  !  1:設定ファイル内で一様分布の時系列
  !  2:分布ファイル × 設定ファイル内で倍率
  !  3:時系列分布ファイル(fn_maplistで指定)
  prtype = 3

  ! 降水分布ファイルの時間間隔 (min)
  !  この時間間隔の間はファイルの雨量で一定
  !  降雨分布の更新時間間隔dt_prupdateは指定しても無効
  dt_maplist = 60

  ! 降雨分布ファイルのリスト
  !  システムパラメータのデータディレクトリからの相対パス
  !  リスト中のファイルもデータディレクトリからの相対パス
  !   リストファイルからの相対パスではないことに注意
  fn_maplist = "AnPrecip/An01.Prec_20191011.txt"
  !------------------------------------------
  ! リストの中身の例
  !------------------------------------------
  !AnPrecip/An01.Prec_20191011_01.bil
  !AnPrecip/An01.Prec_20191011_02.bil
  !AnPrecip/An01.Prec_20191011_03.bil
  !AnPrecip/An01.Prec_20191011_04.bil
  !------------------------------------------

/
