!======================================================================
! 家屋破壊・瓦礫設定サンプル(&list_bldgdebris。developer.md §63)
!   有効化はメインパラメータの fn_bldgdebris = "このファイル" または "-"。
!   木造家屋群を「破壊可能なストック」のラスタ場として扱い、氾濫流に
!   よる破壊 → 瓦礫化 → 流動・堆積をセルごとの柱状材積 (m3/m2) で概算する
!   (個別家屋・個別瓦礫の追跡、衝突力、閉塞の幾何判定は対象外)。
!   台帳: 家屋ストック → (破壊。沈下分はその場で堆積) → 流動瓦礫 →
!   (接地・低速) → 堆積瓦礫(再流動も可)。
!   流木モジュール(fn_driftwood)と同時に有効化すると、荷重判定で
!   流動流木が破壊力に算入される(流木 → 家屋破壊 → 瓦礫の連鎖)。
!   出力: f_out_hbd(流動 Bf0001…)・f_out_wbd(堆積 Bd0001… + 期間最大
!   到達量 Bd9999)・f_out_bds(破壊率 Bs0001…)・f_out_fdmax(流木・瓦礫
!   込み最大流体力 Fd9999)・result/bldgdebris.csv(材積台帳)。
!   ※ 閾値・レートは校正パラメータ(文献照合は developer.md §63.7)。
!   ※ 物性・閾値・レートには既定値があり、fn_gv のある地形では
!     &list_bldgdebris を空にしても動く(採用された既定値は実行開始時に
!     "(default)" 印付きで表示される。既定の根拠は developer.md §63.9)
!======================================================================
&list_bldgdebris
  f_bd = 1                          ! 0 でファイルを残したまま一時無効化

  ! ---- 家屋ストック(一様か分布の排他。両方未指定なら建物のあるセル
  !      (gv<1)にだけ 0.3 m3/m2 を置く)----
  !bd_stock0 = 0.3                  ! 一様な家屋ストック (m3/m2。瓦礫化可能材積。全セル)
  fn_bdstock = "bdstock.txt"        ! ストック分布 (m3/m2。地形と同じ行列形式。
                                    !   延床面積×構造材積原単位×木造率の集計は前処理で)
  !fn_bdfrac = "bdfrac.txt"         ! 建物占有のうち破壊可能な割合 fw (0〜1。省略時 1。
                                    !   RC 造・耐津波構造が残るセルで空隙率帰還の上限
                                    !   gv0 + (1−gv0)·fw を決める)

  ! ---- 瓦礫の代表諸元(既定 0.3 m / 0.5)----
  bd_dlog = 0.3                     ! 代表寸法 (m)
  bd_sg = 0.5                       ! 見かけ比重 (0〜1)。喫水(浮遊/接地の切替水深)は
                                    !   流木と同じ円柱の浮力平衡から自動導出

  ! ---- 破壊判定 ----
  f_bdcrit = 1                      ! 判定量 1: 浸水深 h+hs(被害関数と対応)
                                    !        2: 荷重 (h+(1+s)hs+sg木·hd+sg瓦礫·hbd)·V²
                                    !           (F9999 と同じ基底。流木・瓦礫が破壊力に加わる)
  bd_hcrit = 1.0                    ! 浸水深閾値 (m。f_bdcrit=1。既定 1.0)
  bd_hcrit2 = 3.0                   ! 浸水深の第2閾値 (m。閾値1で 0・閾値2で 1 の線形
                                    !   ランプをレートに掛ける = 被害関数の幅。両方未指定
                                    !   なら 1.0〜3.0、bd_hcrit だけ指定ならランプなし)
  !bd_fcrit = 2.0                   ! 荷重閾値 (m3/s2。f_bdcrit=2。既定 2.0)
  !bd_fcrit2 = 6.0                  ! 荷重の第2閾値 (m3/s2。既定 6.0。同上)
  bd_wdes = 5.0e-4                  ! 破壊レート (m/s = m3/m2/s。所要時間 = ストック÷レート。
                                    !   既定 5e-4 = 0.3 m3/m2 が 10 分で全壊)
  bd_fsink = 0.0                    ! 沈下率 (0〜1。破壊量のうちその場で堆積に直行する比率)

  ! ---- 停止・堆積(既定 0.01 / 0.05)----
  bd_wstop = 0.01                   ! 接地堆積レート (m/s)
  bd_vstop = 0.05                   ! 低速堆積の流速閾値 (m/s。0 = 水深(喫水)判定のみ)

  ! ---- 再流動(既定 bd_wfloat=0 = 堆積した瓦礫は動かない)----
  bd_wfloat = 0.0                   ! 再流動レート (m/s。0 = なし)
  bd_rfloat = 1.5                   ! 浮遊余裕率(再流動水深 = rfloat×喫水。1 より大)
  !bd_vfloat = 0.5                  ! 再流動の流速閾値 (m/s。bd_wfloat>0 で必須。bd_vstop 以上)

  ! ---- 空隙率への帰還 ----
  f_bdgv = 1                        ! 1: 破壊率 d に応じて gv = gv0 + (1−gv0)·fw·d
                                    !    (同じ水が広い面積に広がる = 破壊が流れへ返る)
                                    ! 0: 片方向結合(破壊率・瓦礫分布のみ。流れは不変)
/
