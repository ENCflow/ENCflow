!======================================================================
!
! 波浪条件設定 &list_wave の例(全パラメータの一覧)
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_wave = "list_wave.txt"
!   のように指定して使用します(fn_wave 未指定なら波浪の効果は完全に不活性 =
!   メモリ・CPU ゼロ追加)。沿岸漂砂は掃流砂の経路に乗るため、fn_geomorph の
!   f_fluvial = 1 が必須です。設計は docs/wave_plan.md、決定は developer.md §84.15、
!   使い方は docs/users_guide/geomorph.md「波浪による沿岸漂砂」。
!
!   波は解きません。沖の波(有義波高・周期・進行方向)を与件として、砕波帯の
!   セルで汀線方位を水深勾配から取り、沖波で書いた CERC 式(Ashton & Murray 2006)
!   の沿岸漂砂を岸平行のエッジ流束として河床に適用します。
!
!======================================================================
&list_wave
  f_wave_longshore = 1      ! 汀線帯の沿岸漂砂(0:無効, 1:有効)

  ! ---- 沖波(固定値。時系列 wave_val があればそちらが優先)----
  wave_h0 = 1.0             ! 有義波高 H0 (m)
  wave_t = 6.0              ! 周期 T (s)
  wave_dir = 135.0          ! 進行方向 (deg。x 軸(東)から反時計回り。北 = 90°、西 = 180°)
                            !   135 = 北西へ進む波(南東から来る)

  ! ---- 沖波の時系列(時刻, H0 (m), T (s), 方向 (deg)。線形補間、範囲外は端値)----
  !   時刻の単位は分。wave_clock = 1(既定)なら地形時間(= 水理時間 × morfac。
  !   年単位の波候を与える)、0 なら水理時間(嵐の継続時間を水理時間で与え、
  !   地形には morfac 倍で効く)
  !wave_val(1:4,1) = 0,    0.5, 5.0, 90.0
  !wave_val(1:4,2) = 600,  0.5, 5.0, 90.0
  !wave_val(1:4,3) = 620,  2.0, 8.0, 120.0    ! 嵐
  !wave_val(1:4,4) = 1200, 2.0, 8.0, 120.0
  !wave_val(1:4,5) = 1220, 0.5, 5.0, 90.0
  !wave_clock = 1

  ! ---- 係数(文献値の既定)----
  !wave_k2 = 0.34           ! 沖波 CERC 式の係数 K2 (m^{3/5} s^{-6/5})。K = 0.39、γ = 0.78 の換算値
  !wave_gamma = 0.78        ! 砕波指標 γ = H_b / h_b(砕波帯 = 水深 h <= H_b/γ)
  !wave_smooth = 2          ! 汀線方位を取る水深の平滑化半径(セル。1 または 2)
  !wave_smin = 1.0e-3       ! 汀線方位を定義する海底勾配の下限(これ未満の平坦部は対象外)
/
