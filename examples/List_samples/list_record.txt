!======================================================================
!
! 計測設定 &list_record の例(全パラメータの一覧)
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_record = "list_record.txt"
!   のように指定して使用します(fn_record 未指定なら計測なし)。
!   出力間隔は &list_sysparam の dt_recrd(時間窓は st_recrd/et_recrd)。
!
!   出力先: result/probes/probe0001.csv(1点1ファイル)、
!           result/fluxes/flux0001.csv(1測線1ファイル)、
!           result/summary.csv(全計測の集計)
!
!   セル番号は 1 スタート(西端の列・北端の行が 1)。QGIS 等の GIS は
!   ラスタを 0 から数えるので、GIS で拾った番号は行・列とも +1。
!   実座標(pbxytype/flxytype = 1)は x 東・y 北が正で、地理参照
!   (hdr/GeoTIFF 由来)があれば絶対座標、なければ領域南西端原点の
!   ローカル座標 (m) として解釈されます(解決されたセル番号を初期化時に
!   表示。セル境界ちょうどの座標は避けてセル内部を指すこと。§36)。
!   解説は docs/users_guide/record.md。
!
!======================================================================
&list_record

  ! ---- プローブ(点の時系列: z, h, u, v, |V|, q, hg, hs, sd ...)----
  ! 座標の指定法(全プローブ共通)
  !  0: セル番号 (ix, iy) で指定(既定)
  !  1: 実座標 (x, y) で指定(上記ヘッダの解釈規則)
  pbxytype = 0
  pbxy(:,1) = 150, 150           ! 1点目 (ix, iy)
  pbxy(:,2) = 200, 120           ! 2点目(必要なだけ続ける)

  ! ---- フラックス測線(線を横切る流量の時系列)----
  ! 座標の指定法(全測線共通)
  !  0: セル番号で指定(既定)。両端セル中心を結ぶ DDA 階段面で計測。
  !     両端に同一セルは指定不可(向きが決められないためエラー。
  !     1セル内の計測は実座標指定を使う)
  !  1: 実座標で指定。線分がかすめる各セルのセル内線分成分で積算。
  !     複数セルを跨がない短い測線も計測できる
  flxytype = 0

  ! 測線の始点・終点。符号は「始点から終点に向かって右側へ抜ける
  ! 流れが正」= 河川では右岸を始点・左岸を終点にすると下流向きが正
  flxy(:,1) = 100, 50, 100, 60   ! 1本目 (右岸ix, 右岸iy, 左岸ix, 左岸iy)
  !flxy(:,2) = 40.0, 55.0, 42.0, 56.5   ! 実座標指定(flxytype=1)の例

  ! 測線をファイルで指定する場合(1行1測線、カンマまたは空白区切りで
  ! 「右岸x 右岸y 左岸x 左岸y」。dir_data 起点の相対パス)
  !flxyfile = 1
  !fn_flxy = "transects.csv"

/
