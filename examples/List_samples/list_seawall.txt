!======================================================================
! 海岸堤防(仮想壁面の海岸応用)設定サンプル(developer.md §17.1)
!   キーは &list_geoinfo に書きます(海側マスクは fn_sw/fn_rw と同族の
!   地理情報のため)。以下の断片を使用中の &list_geoinfo に貼り込んで
!   使ってください。指定は河川堤防(fn_channel の fn_bank 系)と独立。
!
!   壁エッジ = 「天端を持つ陸側セル」と「海側セル」の間の全エッジ
!   (対角含む)。海側セルの集合 = fn_seaside の 1 セル ∪ sw>0 セル。
!   - 潮位ケース(fn_sw の海域マスクあり): fn_seaside は省略可
!   - 津波ケース(海を通常セルとして波を解く・sw なし): fn_seaside 必須
!   河道セル(rw>0)に天端は置けず、河口は自動的に開口になります。
!   越流は本間公式。天端は静的(地形変化に追従しない)。
!======================================================================
&list_geoinfo
  ! …(nx, ny, fn_z などの通常の地理情報設定)…

  ! ---- 海側マスク(0/1 ラスタ。sw>0 セルは常に海側扱い)----
  !fn_seaside = 'seaside.txt'

  ! ---- 天端(陸側=保護側セルが持つ。どちらか一方)----
  !fn_seawall = 'seawall.txt' ! 天端分布(陸側セル位置。-900 以下はなし。
                              !   塗らないセルが開口部になる)
  !seawall0 = 5.0             ! 一律天端値(海側に接する全陸セルへ自動付与
                              !   =最小入力で海岸線全周の壁)

  ! ---- 高さの基準 ----
  !f_seawall_datum = 2        ! 1:自セル地盤基準(z+値), 2:絶対標高(既定。
                              !   T.P.+x m の公表天端高をそのまま使える)

  ! ---- 水理モード(f_bank_mode と同義)----
  !f_seawall_mode = 0         ! 0:越流のみ(既定), 1:フラップ=陸閘・水門相当
                              !   (陸→海の排水のみ通す), 2:強制排水=排水機場相当
/
