!======================================================================
! 積雪・融雪設定サンプル(&list_snow。developer.md §31)
!   有効化はメインパラメータの fn_snow = "このファイル" または "-"。
!   気温が必須のため fn_meteo(&list_meteo)も指定すること。
!   標高減率(f_temp_lapse)を併用すると雪線の標高依存が自動で出る。
!   度日法: 降水は気温の二重閾値で雨/雪に線形分離され、雪分は
!   セル別の積雪水量 SWE に蓄積、融雪 M = ddf×max(T−t_melt, 0) が
!   地表水へ入る。SWE 場(Sw、mm)が dt_file 間隔で出力される。
!   リスタートは私有ファイル snow.dat(save 時に fn_snow が有効で
!   あったことが必要)。
!======================================================================
&list_snow
  f_snow = 1                ! 0 でファイルを残したまま一時無効化

  ! ---- 降雨/降雪の分離(二重閾値の線形混合)----
  snow_t_snow = 0.0         ! 全て雪になる気温 (℃)
  snow_t_rain = 2.0         ! 全て雨になる気温 (℃。同値なら単一閾値)

  ! ---- 度日融雪 ----
  snow_t_melt = 0.0         ! 融雪閾値 (℃)
  snow_ddf = 4.0            ! 度日係数 (mm/℃/day。文献値 3〜5 程度。
                            !   森林で小さく、開放地・融雪期後半で大きい)

  ! ---- 初期積雪(未指定なら 0 = 無雪から開始)----
  !snow_swe0 = 100.0        ! 一様初期 SWE (mm)
  !fn_snow_swe0 = "swe0.txt" ! 初期 SWE 分布 (mm。snow_swe0 と排他。
                            !   衛星・アメダス等の観測同化用)
/
