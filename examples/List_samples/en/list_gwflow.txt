! English mirror of examples/List_samples/list_gwflow.txt.
! The Japanese file is the master copy; edit it first.
!======================================================================
!
! Example of groundwater condition settings &list_gwflow and
! model-specific settings
!
!   Use this file by specifying, in the system parameters &list_sysparam,
!     fn_gwflow = "list_gwflow.txt"
!   (if fn_gwflow is not specified, the groundwater computation is
!   completely inactive).
!
!   Write the common settings &list_gwflow and the specific settings of
!   the selected model (&list_gwflow_bucket / &list_gwflow_greenampt
!   etc.) in this same file. A model-specific group is read only by the
!   selected model, so groups of unused models may remain and are ignored
!   (you can A/B compare just by switching the selector to 0).
!
!   Soil depth (sd) and specific yield (sy0) are configured on the
!   &list_geoinfo side as part of the geographic information (same style
!   as the roughness):
!     sd0 = 1.0          ! fixed soil depth (m) (when f_sdtype=0)
!     f_sdtype = 1       ! 0: fixed value sd0, 1: file fn_sd
!     fn_sd = "sd.txt"   ! soil depth distribution file
!     sy0 = 0.2          ! specific yield (= effective porosity n_e)
!   Memory is allocated only when a model that needs the soil depth
!   (Green-Ampt) is selected.
!
!   The material and calibration values have defaults ("default x" in
!   each item; rationale in developer.md §64). When unspecified, the
!   default is adopted and printed on screen with "(default)". The
!   minimal input is just the model selectors of &list_gwflow (the
!   model-specific groups may be empty; sd0 defaults to 1 m too).
!   Values for which 0 has a meaning (gw_psif, gw2_ksh_mmh,
!   gwc_cnd_m2s, gwc_inlet, ...) have no default.
!
!======================================================================
! Groundwater common settings
!======================================================================
&list_gwflow
  ! Choice of the vertical model
  !  0: none (fn_gwflow can stay written; temporarily disabled)
  !  1: bucket model (uses &list_gwflow_bucket)
  !  2: Green-Ampt model (uses &list_gwflow_greenampt. Requires the soil
  !     depth sd and specific yield sy0 of &list_geoinfo)
  f_gwvertical = 2

  ! Choice of the lateral model
  !  0: none (vertical infiltration only. No resources for lateral flow
  !     are allocated at all)
  !  1: nonlinear Boussinesq (uses &list_gwflow_lateral. Requires the
  !     soil depth sd and specific yield sy0 of &list_geoinfo)
  f_gwlateral = 1

  ! Enabling the weathered bedrock layer (second groundwater layer)
  !  0: none (no resources for layer 2 are allocated at all)
  !  1: enabled (uses &list_gwflow_layer2. Cannot be combined with the
  !     bucket vertical model)
  !f_gwlayer2 = 1

  ! Enabling the conduit continuum layer (equivalent confined continuum
  ! for sewer networks, fractured bedrock, etc.)
  !  0: none (no resources for the conduit layer are allocated at all)
  !  1: enabled (uses &list_gwflow_conduit)
  !f_gwconduit = 1

  ! Enabling well pumping / groundwater abstraction (sink)
  !  0: none (no resources are allocated at all)
  !  1: enabled (uses &list_gwflow_pump)
  !f_gwpump = 1

  ! Enabling infiltration suppression by frozen ground
  !  0: none (no resources are allocated at all)
  !  1: enabled (uses &list_gwflow_frost. Requires air temperature, fn_meteo)
  !f_gwfrost = 1

  ! Enable the streambed exchange limit
  !  0: none (no resources are allocated)
  !  1: enabled (uses &list_gwflow_bed. Requires the channel mask fn_rw)
  !f_gwbed = 1

  ! Update time interval of the groundwater computation (s)
  !  0 updates every step
  !  A nonzero value must be at least the time step dt
  !  (the model receives the effective time step dts = dt x thinning
  !  factor)
  dt_gwflow = 0

/


!======================================================================
! Bucket model specific settings (f_gwvertical = 1)
!======================================================================
!   The simplest model: moves surface water into a per-cell subsurface
!   storage (capacity gw_capacity) at a constant infiltration capacity
!   gw_infil_mmh. No horizontal flow; vertical exchange only.
!   The infiltration of each step is the minimum of "infiltration
!   capacity x dts, surface depth, and remaining storage capacity".
!   Cannot be combined with the lateral model (f_gwlateral=1) (that would
!   define the capacity twice, so it stops with an error. If you need a
!   constant infiltration capacity with the soil depth as the capacity,
!   use Green-Ampt with gw_psif=0).
!======================================================================
&list_gwflow_bucket
  ! Infiltration capacity (mm/h). Must be greater than 0 (default 10 =
  ! saturated conductivity of a sandy loam)
  gw_infil_mmh = 10.0

  ! Subsurface storage capacity (column-equivalent water depth) (m).
  ! Must be greater than 0 (default 0.2 = 1 m of soil x specific yield 0.2)
  gw_capacity = 0.2

/


!======================================================================
! Green-Ampt model specific settings (f_gwvertical = 2)
!======================================================================
!   Moves surface water into the subsurface storage at the infiltration
!   capacity of the Green-Ampt formula (piston approximation)
!     f_v = K_sv * (1 + psi_f * n_e / F)   (F: cumulative infiltration depth)
!   The per-cell storage capacity is soil depth x specific yield
!   (sd, sy0 of &list_geoinfo). Saturated cells stop infiltrating and
!   the excess is handled by the shallow water side as surface water.
!======================================================================
&list_gwflow_greenampt
  ! Vertical saturated hydraulic conductivity (mm/h). Must be greater
  ! than 0 (default 10 = sandy loam; values by texture in the user's
  ! guide "Recommended values by pattern")
  gw_ksv_mmh = 10.0

  ! Capillary pressure head at the wetting front (m). 0 or more (0
  ! degenerates to the constant infiltration capacity K_sv)
  gw_psif = 0.1

  ! Distribution maps of K_sv and psi_f (optional; land-use-dependent
  ! infiltration capacity, permeable pavement, etc.). A map takes
  ! precedence over the uniform value. The map K_sv may be 0; cells
  ! with 0 are impervious (fully paved)
  !fn_gw_ksv = "ksv.txt"      ! K_sv map (mm/h)
  !fn_gw_psif = "psif.txt"    ! psi_f map (m)

/

!======================================================================
! Nonlinear Boussinesq lateral flow specific settings (f_gwlateral = 1)
!======================================================================
!   Lateral Darcy flow of the saturated zone (2D, 8 neighbors; the same
!   diagonal share as the ENC core). Driven by the gradient of the total
!   head
!     H = (ground elevation - soil depth) + saturated thickness
!         [+ surface ponded depth (saturated cells only)];
!   the excess of cells exceeding the saturated capacity is passed to
!   the surface flow. The stability condition of the explicit scheme
!     dt <= 0.5 * sy0 * dx*dy / (K_sh * max(soil depth) * sum_k l_k/dr_k)
!   is checked at initialization; if exceeded, the upper bound is
!   displayed and the run stops.
!======================================================================
&list_gwflow_lateral
  ! Lateral saturated hydraulic conductivity (mm/h). Must be greater
  ! than 0 (default 360 = 1e-4 m/s; the effective value including
  ! macropore / pipe flow is 1-2 orders above the matrix)
  gw_ksh_mmh = 360.0

  ! Regularization thickness for the dryness judgment and the suppression
  ! of excessive outflow (m). Must be greater than 0 (default 1e-3)
  gw_eps = 1.0e-3

  ! Diagonal passage-width share [0,1] (default 2/(2+sqrt(2)) = the same
  ! value as p_diagratio of the ENC core. Normally no need to change.
  ! 0 is equivalent to 4 neighbors)
  !gw_diagratio = 0.5857864

  ! Transfer of blocked diagonal shares (1: the passage width of diagonals
  ! blocked by outside or sea cells goes to the axis edges, so the
  ! conductance along walls equals the interior value = default, 0: none)
  !f_gw_opening = 0

/


!======================================================================
! Weathered bedrock layer specific settings (f_gwlayer2 = 1)
!======================================================================
!   Inserts a second layer of thickness gw2_depth and specific yield
!   gw2_sy directly under the soil layer (capacity = gw2_depth x gw2_sy).
!   Water percolates down from layer 1 at the constant infiltration
!   capacity gw2_infil_mmh and, if gw2_ksh_mmh > 0, moves within layer 2
!   as lateral Darcy flow
!   (head = (ground elevation - soil depth - gw2_depth) + saturated
!   thickness. No coupling to surface ponding).
!   Saturation excess of layer 2 returns to layer 1, and the excess of
!   layer 1 exfiltrates to the surface. A layer for providing the
!   slowly-receding component (baseflow) and long residence times.
!   Evapotranspiration does not touch layer 2. Cannot be combined with
!   the bucket vertical model (f_gwvertical=1).
!   The Hg2 field is output when f_out_hg=1. Restart uses the private
!   file gwflow_layer2.dat (f_gwlayer2 must have been active at save
!   time).
!======================================================================
&list_gwflow_layer2
  ! Layer thickness (m). Must be greater than 0 (default 3 =
  ! representative weathered zone of a granitic headwater)
  gw2_depth = 3.0

  ! Specific yield (effective porosity). (0,1] (default 0.05)
  gw2_sy = 0.05

  ! Infiltration capacity from layer 1 to layer 2 (mm/h). Must be
  ! greater than 0 (default 1 = saturated conductivity of weathered
  ! rock, 2.8e-7 m/s)
  gw2_infil_mmh = 1.0

  ! Lateral saturated hydraulic conductivity of layer 2 (mm/h). 0 for no
  ! lateral flow (acts as a capacity buffer)
  gw2_ksh_mmh = 360.0

  ! Initial saturation [0,1] (default 0 = empty. In event runs this is
  ! effectively the calibration parameter that sets the baseflow. For
  ! long-term runs, spin-up -> save -> restore is recommended)
  gw2_sat0 = 0.0

/


!======================================================================
! Conduit continuum layer specific settings (f_gwconduit = 1)
!======================================================================
!   Represents subgrid conduit networks - sewer networks, fractured
!   bedrock, karst, farmland tile drains - as an "artificial confined
!   aquifer" with a per-cell capacity cap and 8-direction conveyances
!   (equivalent continuum approximation; design in
!   docs/gwconduit_plan.md, decisions in developer.md sec. 46). When a
!   cell becomes pipe-full it switches to a pseudo-slot confined state,
!   representing pressurized flow (surcharge) and manhole eruption.
!   Applications are switch combinations:
!     urban sewers   = inlet density given, no interlayer exchange
!                      (use gwc_leak_layer=1 for infiltration/inflow)
!     bedrock/karst  = no inlet density (specify it for sinkholes),
!                      with gwc_leak_layer=2
!   The stability condition of the explicit scheme is checked at
!   initialization (the limit is governed by gwc_slot_sy; if exceeded,
!   lower dt_gwflow or relax slot_sy / eps_h).
!   The Hgc field is output with f_out_hg=1. Restart uses the private
!   file gwflow_conduit.dat (f_gwconduit must have been active at save
!   time).
!======================================================================
&list_gwflow_conduit
  ! Lateral flux law (1: linear, 2: sqrt turbulent conduit flow)
  f_gwc_fluxlaw = 2

  ! Conveyance density (m2/s; pipe-full discharge per unit width at
  ! unit hydraulic gradient). Uniform value or a map (fn_gwc_cnd).
  ! 0 disables lateral conveyance (storage and exchanges only)
  gwc_cnd_m2s = 0.02
  !fn_gwc_cnd = "gwc_cnd.txt"

  ! Storage capacity (columnar m = total pipe volume / cell area).
  ! Uniform value or a map. Cells with 0 have no conduits (storage,
  ! conveyance, and exchanges are all disabled) (default 0.01 = about
  ! an urban branch-sewer network)
  gwc_cap = 0.01
  !fn_gwc_cap = "gwc_cap.txt"

  ! Head datum (invert) elevation: uniform burial depth
  ! (zbot = z - gwc_depth; default 3) or an elevation map
  gwc_depth = 3.0
  !fn_gwc_bot = "gwc_bot.txt"

  ! Storage coefficients: unconfined branch (in-pipe filling; (0,1];
  ! default 0.05 = cap / pipe height) and confined branch (pseudo slot;
  ! <= gwc_sy; default gwc_sy/50. Smaller = stiffer pressure response
  ! but a stricter dt limit, which subcycling absorbs automatically)
  gwc_sy = 0.05
  gwc_slot_sy = 0.001
  ! Per-cell maps of the storage coefficients (take precedence over the
  ! scalars when given. For trunk and branch pipes of different section
  ! heights in one run. sy = cap/pipe height; slot <= sy of each cell)
  !fn_gwc_sy = "gwc_sy.txt"
  !fn_gwc_slot_sy = "gwc_slot_sy.txt"

  ! Initial filling ratio [0,1] (default 0 = empty)
  gwc_sat0 = 0.0

  ! Density of stormwater inlets / manholes / sinkholes (1/m2).
  ! Uniform value or a map. Specifying it enables surface exchange
  ! (weir-type inflow / orifice-type eruption)
  gwc_inlet = 0.01
  !fn_gwc_inlet = "gwc_inlet.txt"

  ! Surface exchange coefficients: weir type (per inlet,
  ! q = cw h^1.5 m3/s) and orifice type (q = co sqrt(2g dH);
  ! co = Cd x opening area, m2)
  gwc_cw = 2.66
  gwc_co = 0.15

  ! Interlayer exchange (0: none, 1: with the soil layer =
  ! infiltration/exfiltration of sewers, 2: with the weathered bedrock
  ! layer = recharge/leakage of bedrock conduits; requires f_gwlayer2=1)
  gwc_leak_layer = 0
  ! Interlayer exchange capacity (mm/h). Read when gwc_leak_layer > 0
  ! (or a map; default 10 = about the vertical conductivity of the
  ! soil around the pipes)
  !gwc_leak_mmh = 10.0
  ! Map of the interlayer exchange capacity (mm/h). Takes precedence over
  ! the scalar when given. Cells with 0 have no exchange (lined reaches of
  ! qanats and culverts, sound sewer pipes)
  !fn_gwc_leak = "gwc_leak.txt"

  ! Map of the outfall orifice coefficient Cd*A (m2). Cells with a value
  ! are outfalls (a conduit, cap>0, is required). Open pipe ends with a
  ! built-in flap (no backflow), unconfined or confined alike. The
  ! receiver is decided by the cell's neighbourhood:
  !  - adjacent to a sea cell: receiving head = lowest water level (tide)
  !    of the adjacent sea cells. The discharge leaves the system (the
  !    cumulative amount is printed at the end). Drainage below sea level
  !    is combined with the conduit intake of &list_struct_pump (f_pump_src=1)
  !  - not adjacent to the sea: open land-side outfall (adit mouth,
  !    daylighting culvert outlet). Receiving head = surface water level
  !    of the cell itself. The discharge goes to surface flow (stays in
  !    the domain = conserved)
  !fn_gwc_outfall = "outfall.txt"

  ! Numerical parameters (normally no need to change)
  gwc_eps = 1.0e-3          ! regularization amount for the dry test (m)
  gwc_eps_h = 1.0e-2        ! linearization width of the sqrt law (m)
  !gwc_diagratio = 0.5857864 ! diagonal partitioning of the flux width [0,1]
  ! Upper limit of the number of subcycles of the lateral conveyance. A dt
  ! above the stability limit of the explicit scheme is automatically
  ! split into N = ceiling(dt/limit) substeps (N is printed at
  ! initialization). The run stops when N exceeds this limit (relax
  ! slot_sy/eps_h or raise the limit)
  !gwc_nsubmax = 100

/

!======================================================================
! Well pumping / groundwater abstraction settings (f_gwpump = 1)
!======================================================================
!   Withdraws the specified pumping rate from subsurface storage at a
!   set of cells and removes it from the system (the subsurface
!   counterpart of the internal source &list_bound_source). Up to 50
!   wells, numbered consecutively from 1. The demand is divided
!   equally over the cell set, and the withdrawal at each cell is
!   capped by that cell's storage (dry well; a per-well summary of
!   demand vs. actual withdrawal is printed at the end of the run).
!   Combined with the fresh/salt layers (f_salt_gw=1), pumping takes
!   fresh water first (the shortfall becomes salt contamination).
!   Injection (negative rates) is not supported (return flow to the
!   surface goes through &list_bound_source).
!======================================================================
&list_gwflow_pump
  ! Well 1: inline cells and a constant rate
  gwp_cell(1:2,1,1) = 15, 22     ! abstraction cell (i, j) (multiple cells = well field)
  !gwp_cell(1:2,2,1) = 16, 22    ! second cell (the demand is divided equally)
  gwp_q0(1) = 0.02               ! constant pumping rate (m3/s; positive = abstraction)
  !gwp_layer(1) = 1              ! abstraction layer (1: soil layer (default),
                                 !   2: weathered bedrock; 2 requires f_gwlayer2=1)

  ! Well 2: cell list file plus a rate time series file
  !fn_gwp_cell(2) = 'well2.txt'  ! cell list (each line "i j")
  !fn_gwp_val(2) = 'qwell2.txt'  ! rate time series (each line "min m3/s")
  !gwp_val(1:2,1,2) = 0.0, 0.01  ! (an inline time series (min, m3/s) also works)

/

!======================================================================
! Frozen ground settings (f_gwfrost = 1)
!======================================================================
!   Accumulates a freezing index FI (degC.day) per cell from the air
!   temperature (fn_meteo; lapse rate included) and multiplies the
!   vertical infiltration capacity (Green-Ampt and bucket) by the
!   reduction factor
!     fac = max(fro_fmin, 1 - FI/fro_fifull)
!   (meltwater running over frozen ground = snowmelt floods).
!   Freezing degree-days accumulate below fro_tf; otherwise thawing
!   degree-days (x fro_ct) reduce FI. With snow enabled (fn_snow),
!   fro_swe0 > 0 attenuates both freezing and thawing by
!   exp(-swe/fro_swe0) (snow insulation). The simplest degree-day
!   model - soil temperature and frost depth are not solved
!   (fro_fifull is the practical calibration point).
!   The reduction factor is output as the Ff field. The freezing index
!   is carried across restarts by the private file gwflow_frost.dat.
!======================================================================
&list_gwflow_frost
  ! Freezing index at which the factor reaches its minimum (degC.day).
  ! Must be greater than 0 (default 20 = freezing of the top 20-30 cm)
  fro_fifull = 20.0

  ! Lower bound of the reduction factor [0,1) (default 0 = fully
  ! frozen ground is impervious)
  fro_fmin = 0.05

  ! Freeze/thaw threshold air temperature (degC) (default 0)
  !fro_tf = 0.0

  ! Thawing efficiency (multiplier on thawing degree-days; default 1.
  ! Smaller values delay the spring thaw)
  !fro_ct = 1.0

  ! e-folding snow water equivalent of the snow insulation (m)
  ! (default 0 = no insulation. Requires fn_snow)
  !fro_swe0 = 0.1

  ! Cap on the freezing index (degC.day) (default 0 = no cap. About
  ! 1-2 x fro_fifull is recommended to keep the thaw from being
  ! unrealistically delayed after a long severe winter)
  !fro_fimax = 40.0

  ! Initial freezing index (degC.day) (default 0. Set it to start an
  ! event run of the melt season "already frozen")
  !fro_fi0 = 20.0

/


!======================================================================
! Streambed exchange limit specific settings (f_gwbed = 1)
!======================================================================
!   On the cells of the channel mask (fn_rw), gives the exchange between
!   surface water and soil-layer groundwater through the bed and banks a
!   limit per unit area k_bed (mm/h) (the counterpart of the streambed
!   conductance of MODFLOW RIV). It replaces the vertical infiltration
!   capacity of channel cells (gw_infil_mmh / gw_ksv_mmh) by k_bed and, in
!   the lateral flow model, caps the edge discharge between channel and
!   non-channel cells at k_bed x interface area in both directions (the
!   exfiltration of a gaining river is throttled too).
!   0 = impervious (concrete channel). Because 0 is meaningful there is no
!   default: either gwbed_k_mmh or fn_gwbed_k is required. Either the
!   vertical or the lateral model must be enabled.
!======================================================================
&list_gwflow_bed
  ! Uniform exchange limit of the channel cells (mm/h). >= 0 (0 =
  ! impervious). Guide: clogged sand bed 1-100, gravel bed 100-3600,
  ! concrete channel 0
  gwbed_k_mmh = 36.0

  ! Map of the exchange limit (mm/h). Takes precedence over the uniform
  ! value. Only the values on channel cells are used; cells with 0 do not
  ! exchange (lined reaches)
  !fn_gwbed_k = "gwbed_k.txt"

/
