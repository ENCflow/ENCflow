! Example of sediment and landform evolution settings &list_geomorph
! (list of all parameters)
!
!   Use this file by specifying, in the system parameters &list_sysparam,
!     fn_geomorph = "list_geomorph.txt"
!   (if fn_geomorph is not specified, landform evolution is completely
!   inactive).
!
!   The processes are superposed, not an exclusive choice. Only the
!   processes set to f_XXX = 1 are applied in order; processes at 0
!   allocate no resources.
!   The time-evolving terrain can be output with f_out_z = 1 of
!   &list_sysparam.
!   Verified examples: test/creep, fluvial, suspend, wash, debris, slide,
!   avalanche, sedinflow. Explanations in docs/users_guide/geomorph.md.
!
!======================================================================

  ! Debris flow / landslide (Takahashi-type equilibrium-concentration E-D)
  ! (0: off, 1: on)
  !   Represents a debris flow as a single-layer mixture of surface water
  !   h and sediment column hs (docs/debris_plan.md, developer.md Sec. 28).
  !   When enabled, the flow computation includes hs in the momentum
  !   (flow depth = h+hs) and the resistance law is switched by f_dbres.
  !   Exclusive with f_suspend (avoids double-counting E-D on the same hs).
  !   morfac=1 required. The movable extent and depth are given by the
  !   soil depth (sd0/f_sdtype/fn_sd of &list_geoinfo; sd=0 = non-erodible
  !   bedrock). Porosity and specific gravity are shared with fluv_*.
  !   Recommended: combine with f_gravity_correction = 1 of &list_enc on
  !   steep terrain
  f_debris = 0
  db_phi = 0.0             ! internal friction angle of the sediment (deg).
                           !   0 = default 35 (representative of gravelly
                           !   debris. Used by f_dbed=1-3 / f_dbres=1,2)
  f_dbed = 1               ! E-D closure (0: no exchange - equivalent-fluid mode
                           !   (advection and stopping only; for pyroclastic
                           !   flows and debris avalanches without bed erosion.
                           !   With f_dbstop=1 the equilibrium concentration is
                           !   treated as 0 = all hs of slow cells is fixed to
                           !   the bed at the db_wstop rate),
                           !   1: relaxation to the equilibrium concentration
                           !   (simple form; delta_e/delta_d.|V| rates),
                           !   2: Egashira-Ashida 1992 - E = |V|.C*.tan(theta-theta_e).
                           !   No flow-direction slope or extra parameters
                           !   (checked against the literature),
                           !   3: Takahashi-Nakagawa 1991 - saturated-bed erosion
                           !   (eq. 12) + deposition with inertia coefficient
                           !   (eq. 27). Rates delta_e/delta_d.|V|h/d (checked
                           !   against the literature. db_d50 required. Horadani
                           !   values delta_e=0.0001, delta_d=0.5),
                           !   4: velocity-proportional entrainment E = delta_e.|V| -
                           !   path entrainment for avalanches and debris
                           !   avalanches (entrainment only; deposition via
                           !   f_dbstop. The entrainable layer is sd (for
                           !   avalanches, the snow depth along the path). Cells
                           !   with C >= db_cmin only. See the avalanche
                           !   paragraph of users_guide/geomorph.md))

  f_dbres = 1              ! resistance law (0: Manning only, 1: Coulomb +

                           !   laminar resistance; checked against the
                           !   literature. db_d50/db_erest required),
                           !   3: Takahashi-Nakagawa 1991 stony type (yield
                           !   tan(alpha')=0.45 + dilatant term. Switches
                           !   automatically between stony / immature / mud
                           !   types by concentration and relative depth;
                           !   checked against the literature. db_d50 required),
                           !   4: Voellmy equivalent fluid (yield mu + turbulent
                           !   term gV^2/(xi h). The operational standard for
                           !   debris avalanches and pyroclastic flows.
                           !   db_mu/db_xi required. Typically combined with
                           !   f_dbed=0),
                           !   5: constant retarding stress tau_y + Manning
                           !   combined (proven for runout and deposition of
                           !   pyroclastic flows. db_tauy required))
  db_d50 = 0.0             ! representative grain size (m). Used by f_dbres=2,3 /
                           !   f_dbed=3. 0 = default 0.05
  db_erest = 0.0           ! particle restitution coefficient e (0-1]. Used by
                           !   f_dbres=2. 0 = default 0.85
  db_cmin = 0.02           ! lower concentration bound of the Egashira laminar

  f_dbstop = 0             ! stopping condition (0: none, 1: low-speed
                           !   consolidation - in cells with vv<db_vstop, the
                           !   sediment above equilibrium is fixed to the bed
                           !   (elevation z) at the db_wstop rate. Represents
                           !   natural-dam formation by stopped sediment)
  db_vstop = 0.05          ! velocity threshold of the stopping judgment (m/s)

  db_wstop = 0.0           ! bed conversion rate of low-speed consolidation (m/s).
                           !   Used with f_dbstop=1. 0 = default 0.05
  f_dbwet = 0              ! pore-water entrainment (0: ignore, 1: exchange the
                           !   pore water of the bed change dz with the surface
                           !   water - erosion: h += lambda.s_b.|dz|, deposition:
                           !   h -= lambda.s_b.dz (clamped by the remaining h).
                           !   The same ledger as i{c*+(1-c*)s_b} of
