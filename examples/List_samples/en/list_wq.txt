  ! ---- Distributed diffuse source (all active cells. A unit load
  !      distribution built from land use etc.) ----

  ! ---- Surface accumulation pool + washoff (buildup-washoff; a

  !   Buildup dB/dt = r*(1 - B/Bmax) (linear if wq_bd_max is omitted).

  !wq_wash_tauc = 0.05              ! critical shear stress (N/m2. Required

  !f_wq_settle = 0                  ! destination of settling (0: lost to the
