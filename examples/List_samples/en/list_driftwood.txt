!======================================================================
! Driftwood sample (&list_driftwood; developer.md §50)
!   Enable with fn_driftwood = "this file" or "-" in the main parameter
!   file. Estimates driftwood as a raster field of wood volume per cell
!   (m3/m2); individual-log tracking and the geometric judgement of
!   whether a log gets caught are out of scope.
!   Ledgers: standing stock -> (hydraulic washout / erosion
!   entrainment) -> floating wood -> (grounding / drying) -> deposited
!   wood (refloat optional). With the capture-ratio map fn_dwtrap,
!   floating wood -> (capture at structures) -> jammed wood is added and
!   the jam raises the water upstream (the jam at a culvert inlet is
!   enabled by culv_jam = 1 in &list_struct_culvert).
!   Output: f_out_hd (floating Hd0001...), f_out_wd (deposited
!   Wd0001... + maximum arrival Wd9999), f_out_wj (jammed Wj0001...),
!   result/driftwood.csv (volume ledger).
!   NOTE: thresholds and rates are calibration parameters (no
!   established standard values).
!   Properties, thresholds and rates have defaults: an empty
!   &list_driftwood runs (stock 0.001 m3/m2 uniformly on land cells;
!   adopted defaults are printed with a "(default)" mark at start-up;
!   rationale in developer.md §50.6).
!======================================================================
&list_driftwood
  f_dw = 1                          ! 0 disables temporarily, keeping the file

  ! ---- standing stock (uniform or map, exclusive; with neither given,
  !      0.001 uniformly on land cells) ----
  dw_stock0 = 0.01                  ! uniform standing stock (m3/m2)
  !fn_dwstock = "stock.txt"         ! stock map (m3/m2, same matrix format as
                                    !   the terrain; convert forest inventory
                                    !   data in preprocessing)

  ! ---- representative log properties (defaults 0.3 m / 0.5) ----
  dw_dlog = 0.3                     ! representative diameter (m)
  dw_sglog = 0.5                    ! wood specific gravity (0-1); the draft
                                    !   (float/ground switch depth) is derived from
                                    !   the cylinder buoyancy balance
                                    !   (Braudrick & Grant 2000; half submerged at 0.5)

  ! ---- recruitment; both default when neither is given, one given disables the other ----
  dw_wrec = 1.0e-5                  ! hydraulic washout rate (m/s = m3/m2/s; specifying it enables)
  dw_hrec = 0.5                     ! washout depth threshold (m; under debris flow judged by
                                    !   the mixture flow depth)
  dw_vrec = 1.0                     ! washout velocity threshold (m/s)
  dw_droot = 0.5                    ! rooting depth (m; enables erosion entrainment --
                                    !   full washout when the cumulative erosion depth
                                    !   exceeds it; couples automatically to bedload,
                                    !   debris-flow and slope-failure erosion)

  ! ---- stopping / deposition ----
  dw_wstop = 0.01                   ! grounding deposition rate (m/s; default 0.01)
  !dw_vstop = 0.05                  ! slow-flow deposition threshold (m/s; default 0
                                    !   = draft criterion only)

  ! ---- refloat (default off = deposited wood never moves; conservative) ----
  !dw_wfloat = 0.01                 ! refloat rate (m/s; >0 enables)
  !dw_rfloat = 1.5                  ! buoyancy margin (refloat when depth > rfloat*draft; >1)
  !dw_vfloat = 0.1                  ! refloat velocity threshold (m/s; >= dw_vstop)

  ! ---- capture and jamming at structures (enabled by fn_dwtrap; bridges, open check dams) ----
  !fn_dwtrap = "trap.txt"           ! map of the capture ratio E (0-1; the share of the wood
                                    !   passing a capture cell once that is caught; 0 = no
                                    !   capture; same matrix format as the terrain; the ratio is
                                    !   a given, the module does not judge whether a log is caught)
  !f_dwjam = 1                      ! effect of the jam on the flow (0: ledger only,
                                    !   1: drag in the cells holding the jam)
  !dw_jamcd = 1.0                   ! drag coefficient C_D of the jam (calibration value;
                                    !   the default 1.0 is provisional)
  !dw_jamphi = 0.3                  ! packing ratio of the jam (solid / bulk volume; 0-1;
                                    !   calibration value; acts through (1-phi)^3 in the drag
                                    !   coefficient 2 C_D/(pi d (1-phi)^3))
  !dw_jamgrow = 0.09                ! cells whose jammed amount (m3/m2) reaches this get a
                                    !   capture ratio of 1; until then the capture ratio rises from
                                    !   E to 1 in proportion to the jammed amount (not given =
                                    !   constant capture ratio). Guide: a multiple of
                                    !   dw_jamphi x dw_dlog (one layer of logs)
  ! The jam at a culvert inlet (structures whose opening is narrower than a cell) is enabled
  ! by culv_jam(n) = 1 in &list_struct_culvert (fn_dwtrap is not needed; f_dwjam, dw_jamcd
  ! and dw_jamphi are shared)
/
