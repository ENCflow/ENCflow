! Lava flow settings sample (&list_lavaflow; developer.md Sec. 51,
!   docs/lava_plan.md)
!   Enabled by fn_lavaflow = "this file" or "-" in the main parameters.
!   The isothermal mode depends on no other module (water and weather not
!   required). The temperature mode (f_lvtemp = 1) uses the air temperature
!   of fn_meteo when given.
!   The state is the lava thickness s%hl (m). Field output Hl (m);
!   restart uses the private file lavaflow.dat. Solidified lava appears
!   as an increase of the bed elevation z (sd not increased = non-erodible
!   bedrock; the soil layer that was in a cell where lava solidifies is
!   buried under the lava, sd = 0). Surface water does not see flowing lava
!   and flows over z.
!   Dynamics: depth-averaged Bingham viscous gravity current (lubrication
!   approximation). Isothermal by default; f_lvtemp = 1 computes the lava
!   temperature (a flow that stops as it cools; lava_plan.md Sec. 8,
!   developer.md Sec. 51.7). One rheology set per run (in the temperature
!   mode, the values at the reference temperature; per-vent differences are
!   given by the eruption temperatures).
!======================================================================
&list_lavaflow
  f_lavaflow = 1            ! 0 disables temporarily while keeping the file

  ! ---- update interval ----
  !dt_lavaflow_c = "10 s"   ! lava update interval (default = every step)

  ! ---- rheology (given directly; one set per run) ----
  lv_rho = 2600.0           ! density rho (kg/m3)
  lv_visc = 1.0e4           ! viscosity eta (Pa s; default 1e4 = basaltic
                            !   aa lava; basaltic 1e2-1e4, andesitic and up
                            !   >= 1e5. Smaller eta means more explicit
                            !   subcycles - plan Sec. 6)
  lv_tauy = 2000.0          ! yield stress tau_y (Pa; 0 = Newtonian fluid.
                            !   stopping thickness on a slope tan(theta)
                            !   = tau_y/(rho g tan(theta)))

  ! ---- solidification (stopped cells: hl -> z; 0 = none) ----
  lv_wsol = 0.005           ! solidification rate (m/s)
  lv_vsol = 5.0e-4          ! stop-detection velocity threshold (m/s;
                            !   read if lv_wsol > 0; default 5e-4; typically
                            !   1e-4-1e-3. Smaller values approach the
                            !   theoretical h-inf but take longer to trigger)

  ! ---- temperature (f_lvtemp = 1; a flow that stops as it cools;
  !      lava_plan.md Sec. 8) ----
  !   The temperature-mode values have no defaults (give all of them; a
  !   missing one stops the run). The values below are an example (the same
  !   set as test/lava configuration 9), neither defaults nor
  !   recommendations; set them for the lava of interest (User's Guide, lava
  !   chapter, "Temperature mode").
  !   The temperature mode judges solidification on temperature (the column
  !   moves to z when Tl <= lv_tsol), so set lv_wsol above to 0 (lv_vsol is
  !   not read)
  !f_lvtemp = 1             ! 1: compute the temperature (default 0 = isothermal)
  !f_lvtemp_scheme = 3      ! face temperature of the heat advection (default 3 = MUSCL: cuts the
                            !   crosswind numerical diffusion of a flow oblique to the grid to about
                            !   1/10 of first-order upwind; 1 = the donor's temperature (first-order
                            !   upwind; the default before 2026-11-05); developer.md Sec. 97)
  !lv_temp0(1) = 1150.0     ! eruption temperature of vent 1 (deg C; above lv_tsol)
  !lv_tref = 1150.0         ! reference temperature (deg C; lv_visc and lv_tauy are the values here)
  !lv_bvisc = 0.04          ! temperature sensitivity of the viscosity (1/K; ln k/dT if k times after cooling by dT)
  !lv_btau = 0.04           ! temperature sensitivity of the yield stress (1/K)
  !lv_tsol = 1000.0         ! solidification temperature (deg C)
  !lv_emis = 0.95           ! effective emissivity (0-1; includes the insulation by the surface crust)
  !lv_hconv = 70.0          ! convective heat transfer coefficient (W/m2/K)
  !lv_cp = 1150.0           ! specific heat (J/kg/K; may include the latent heat of crystallization)
  !lv_tair = 20.0           ! air temperature (deg C; write only when fn_meteo is not given)

  ! ---- subcycling ----
  !lv_cfl = 0.4             ! safety factor of the explicit scheme (0-1)
  !lv_nsubmax = 10000       ! cap on subcycles per update (runaway guard)
  !lv_diagratio = 0.3333333333   ! diagonal share d of the flow (default 1/3 = the 8-neighbour
                            !   9-point stencil with dynamic reassignment of blocked diagonal shares;
                            !   flows down narrow valleys oblique to the grid do not depend on the
                            !   orientation. 0 = 4-neighbour (the default before 2026-11-05);
                            !   developer.md Sec. 99)

  ! ---- vents (numbers n consecutive from 1; up to 20) ----
  !   The rate is the total over the cell set, distributed equally
  !   (no momentum). Represent fissure eruptions by listing cells.
  lv_cell(1:2,1,1) = 55, 40 ! cells (i, j) of vent 1
  lv_cell(1:2,2,1) = 55, 41
  lv_val(1:2,1,1) = 0, 10.0 ! effusion time series (time (min), m3/s;
  lv_val(1:2,2,1) = 60, 5.0 !   linear interpolation; duplicate a time
  lv_val(1:2,3,1) = 120, 0.0 !  for a step change)
  !lv_q0(1) = 10.0          ! constant effusion rate (m3/s; instead of lv_val)
  !fn_lv_cell(1) = "vent1_cells.txt"  ! cell-set file (rows of "i j"; takes precedence)
  !fn_lv_val(1) = "vent1_q.txt"       ! time-series file (min, m3/s; highest precedence)
/
