!
! Example of ENC computation settings &list_enc (list of all parameters)
!
!   Use this file by specifying, in the system parameters &list_sysparam,
!     fn_enc = "list_enc.txt"
!   (even without fn_enc, the core shallow water computation runs with
!   the defaults. This group is for tuning accuracy and cost).
!
!   Everything below states the default values explicitly. In ordinary
!   runs, about the only one worth considering changing is the adaptive
!   Runge-Kutta threshold p_adprunge_thresh
!   (see tutorial Step 2). Many of these switches change the results, so
!   comparison with the regression-test references is possible only at
!   the default values.
!   Explanations in docs/users_guide/swflow.md.
!
!======================================================================
&list_enc

  ! ---- Time integration (adaptive Runge-Kutta) ----
  !f_adaptive_runge = 1     ! recompute with 4-stage RK only the cells with
                            !   large velocity variation
  !p_adprunge_thresh = 1.5  ! velocity variation ratio threshold triggering
                            !   recomputation (1.1 and up; smaller = more
                            !   accurate and more costly)

  ! ---- Fluxes and stabilization ----
  !f_gravity_correction = 1 ! gravity correction (corrects the slope-direction
                            !   gravity error on steep terrain)
  !f_exflux_reduction = 1   ! suppression of excessive outgoing fluxes (prevents
                            !   negative depths at dry-bed fronts; on is
                            !   recommended for runs with dry beds)
  !f_hcap_upwind = 1        ! cell-boundary depth (0: mean of both cells,
                            !   1: mean capped by the upstream depth = default,
                            !   2: upstream depth itself; removes the bore-front
                            !   spike, increases front outflow onto dry ground)
  !f_friction_fastmath = 0  ! faster friction term computation (0: exact,
                            !   1-5: table approximation. Larger numbers are
                            !   coarser and faster)
  !p_diagratio = 0.5857864376  ! diagonal share of the 8-direction fluxes
                            !   (= 2/(2+sqrt(2)); normally no need to change)

  ! ---- Advection and diffusion terms ----
  !f_advection_scheme = 3   ! advection scheme (1: cell-centre gradient =
                            !   former default, 2: momentum-conservative
                            !   first-order upwind, 3: momentum-conservative
                            !   + MUSCL = default). Scheme 1 is non-
                            !   conservative: bores lag and on real terrain
                            !   the runoff gets faster and larger with grid
                            !   refinement without converging.
                            !   p_adv_upwind_index below acts on scheme 1
                            !   only
  !f_scalar_scheme = 3      ! advection scheme of the scalar transport
                            !   (suspended sediment, water quality, salt
                            !   layer, driftwood, debris): 3 = link-wise
                            !   MUSCL with van Leer (default), 1 = first-order
                            !   upwind (former default; numerical diffusion
                            !   3-5 times longitudinally, 14 times laterally,
                            !   blunting concentration fronts, peaks and
                            !   density-driven loads; kept for comparison).
                            !   users_guide/advection.md section 7
  !f_advection_donor = 0    ! upwind momentum donors of the momentum-
                            !   conservative advection (schemes 2, 3)
                            !   (0: all wet cells = default, 1: on edges between
                            !   channel cells the streamwise donors are channel
                            !   cells only; needs fn_rw).
                            !   Avoids the artificial loss when the donor on
                            !   the edge line falls on a flooded valley cell
  !f_dry_head_cap = 1       ! cap the edge depth toward a dry cell by the
                            !   energy head eta + u_n^2/2g - z_receiver
                            !   (0: off, 1: on = default). Water cannot climb
                            !   onto dry ground above the water surface plus
                            !   velocity head (fixes leaks at bore fronts and
                            !   in 1-2 cell wide incised channels)
  !f_opening_dynamic = 2    ! dynamic reassignment of blocked openings
                            !   (0: off, 1: edges between channel cells only
                            !   (inactive without fn_rw), 2: all edges =
                            !   default, 3: all edges, with the blockage
                            !   measured above the local plane).
                            !   Passage-width shares that cannot be used
                            !   because the diagonal neighbour's ground is
                            !   above the water surface are reassigned to
                            !   the open edges every step (fixes the reduced
                            !   conveyance of 1-cell channels / wall-side rows
                            !   and the amplified damping of scheme 3). 2 also
                            !   counts the upslope ground of a uniform slope as
                            !   blocking, so use 3 for steep mountain streams
                            !   and sheet flow on hillslopes (developer.md §68.41)
  !f_inflow_edge = 0        ! treatment of the edges touching the outer cells
                            !   of a segment inflow (0: zero advection on all
                            !   of them = default, 1: zero advection only on
                            !   the edges into the inflow section, 2: as 1,
                            !   and the velocity of those edges is capped by
                            !   the energy head of the inflow cell). Use 2 when
                            !   the bed beside or below the inflow moves and
                            !   for narrow inflows (users_guide/advection.md
                            !   section 4.4)
  !p_adv_upwind_index = 0.5 ! upwinding index of the advection term (0-1.
                            !   0: central difference, 1: first-order upwind)
  !f_diffusion_term = 0     ! diffusion term (0: none, 1: constant viscosity,
                            !   2: zero-equation model nu = nu0 + alpha*(u*)*h).
                            !   For a first try, 2 is handy (no coefficient to
                            !   specify, and a physically scaled nu acts)
  !p_diffusion_nu = 0.0     ! kinematic viscosity nu0 (m2/s) (background
                            !   viscosity in model 2). Model 1 requires an
                            !   explicit positive value (unspecified stops
                            !   with an error)
  !p_diffusion_alpha = 0.0683333  ! coefficient alpha of the zero-equation model
                            !   (default 0.41/6 = Elder type)

  ! ---- Non-hydrostatic correction (research use; docs/nonhydrostatic_plan.md) ----
  !f_nonhydrostatic = 0     ! 0: hydrostatic (default), 1: one-layer non-hydrostatic
                            !   correction (beta = h^2/4. For river run-up of tsunamis
                            !   whose soliton fission is essentially dispersive, and
                            !   for near-field generation of landslide tsunamis.
                            !   Grid guideline dx <~ H/2)
  !nh_hmin = 0.1            ! cells shallower than this stay hydrostatic (m)
  !nh_solver = 2            ! iterative solver (1: Jacobi, 2: CG (default). Jacobi
                            !   converges slowly for dx << H; kept for comparison)
  !nh_itmax = 500           ! upper limit of the number of iterations
  !nh_tol = 1.0e-6          ! relative convergence criterion
  !f_nh_adaptive = 0        ! active set (0: solve everywhere inside the NH mask
                            !   (default), 1: solve only on the cells where the
                            !   detector exceeds its threshold plus a margin of
                            !   nh_margin cells around them. Long-wave and steady-
                            !   flow regions then add no cost)
  !nh_detector = 1          ! detector (1: dispersive type chi = |beta D a*|/(|a*|+|beta D a*|)
                            !   ~ (kH)^2/4, independent of the amplitude (default),
                            !   2: absolute type |beta D a*|/g)
  !nh_chi_on = 0.06         ! detection threshold (cells with chi > nh_chi_on are
                            !   seeds. 0.06 is the level of a 3% phase-speed
                            !   difference from the hydrostatic solution)
  !nh_margin = 0            ! margin width (cells. 0: automatic from 2H/dx =
                            !   4 times the decay distance H/2)
  !nh_amin = 1.0e-5         ! lower limit of detection |beta D a*| (m/s2). Cells whose
                            !   correction is below this are not seeds (removes the
                            !   precursor noise of the explicit scheme)
  !nh_arel = 1.0e-3         ! relative lower limit of detection. Cells below this
                            !   ratio of the domain maximum |beta D a*| of the step
                            !   are not seeds
  !f_nh_breaking = 0        ! breaking switch (0: none, 1: cells judged to be breaking
                            !   and nh_break_margin cells around them are made
                            !   hydrostatic and treated as bores. Essential for
                            !   breaking waves running up a slope. Use together
                            !   with f_hcap_upwind = 2)
  !nh_break_type = 3        ! criterion (1: surface rise rate d(eta)/dt > alpha sqrt(gh)
                            !   (SWASH type. Adjacent cells use beta. Carries a
                            !   state that is kept until the crest has passed),
                            !   2: Froude number |V|/sqrt(gh) > nh_break_fr,
                            !   3: surface slope |grad eta| > nh_break_slope (default))
  !nh_break_alpha = 0.6     ! onset threshold alpha of criterion 1
  !nh_break_beta = 0.3      ! threshold beta of criterion 1 for cells adjacent to a breaking cell
  !nh_break_fr = 0.6        ! Froude-number threshold of criterion 2
  !nh_break_slope = 0.3     ! surface-slope threshold of criterion 3
  !nh_break_margin = 0      ! margin made hydrostatic around breaking cells (cells. 0: 2H/dx)
  !nh_break_visc = 0.0      ! eddy-viscosity coefficient delta_b^2 of the breaking zone (0: none.
                            !   1.44 = equivalent to Kennedy et al. 2000. Adds
                            !   nu_b = nh_break_visc*B*h*sqrt(gh)*|grad eta| to the momentum diffusion term)
  !wf_break_visc = 0.0      ! coefficient M of the horizontal mixing by wave breaking of the coastal
                            !   wave field (0: none. 1.0 = the default of nuhfac of XBeach). Adds
                            !   nu_w = M*h*(D/rho)^(1/3) (Battjes 1975; D is the roller dissipation)
                            !   to the momentum diffusion term. Requires fn_wavefield
  !nh_bc_margin = 0         ! margin kept hydrostatic next to forcing boundaries (stage,
                            !   section inflow, outflow, radiation) (cells. 0: 2H/dx)
  !f_nh_slope = 0           ! bottom-slope terms (0: linear NH on a flat bed (Version 1),
                            !   1: Version 2, including the kinematic bottom condition
                            !   w_b = u.grad z_b and the pressure term (phi/h) grad(h+2z_b).
                            !   Use for waves over slopes and shelves)
  !f_nh_bottom = 0          ! acceleration term of a moving bottom (0: none, 1: adds the
                            !   vertical bottom acceleration z_b'' to the NH source term.
                            !   Acts on landslide tsunamis (f_bedslide), landform
                            !   evolution, lava flows and z updates through the API.
                            !   Near-field waveforms of short landslides b/H <~ 2)

  ! ---- Miscellaneous ----
  !f_rivermouth_drop = 0    ! overfall from the river mouth to the sea (old
                            !   method; cannot be combined with fn_tide. Using
                            !   tide + boundary conditions is now recommended)

/
