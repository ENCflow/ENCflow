!======================================================================
!
! 田んぼダム(水田の落水口の流出抑制)の設定 &list_paddy の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_paddy = "list_paddy.txt"
!   のように指定して使用します(fn_paddy 未指定なら完全に不活性)。
!
!   セルの水田面積率の上に降った雨を、種類ごと(1 = 取組なしの通常の落水口、
!   2 = 田んぼダムの落水口)のタンクに貯め、落水口の装置則と畦畔の越流で地表へ
!   戻します。落水口は代表区画面積ごとに 1 か所で、水田面積あたりの流出強度は
!   格子の大きさによりません(users_guide/paddy.md、developer.md §119)。
!   5〜10 m の格子では、区画番号の分布 fn_pd_plot を与えて区画を解像する表現 B も
!   選べます(畦畔を仮想壁面、落水口を壁の切欠きにする。下の例はコメントアウト)。
!
!   必要な入力: 水田面積率(fn_pd_frac か pd_frac)だけ。ほかは省略すると
!   採用値を (default) 付きで表示して使う(区画 30 a・畦畔 0.30 m・減水深
!   5 mm/d・取組なし・水位調整板 0.03 m × 0.30 m・機能分離型 φ50・係数 0.6)
!
!   併用できない機能(指定すると停止): fn_wq・fn_salt・ネスティング・STG 格子。
!   B はさらに fn_geomorph・fn_evap・fn_gwflow・非静水圧とも併用できない。
!   水田のセルは海・河道・境界・ソース・構造物・VZ と貯留関数の区画のセルに
!   置けない。既定の片方向では浸水が水田に入る流れ(逆流)を表さない(双方向
!   f_pd_exchange = 1 か B で表す)
!
!   出力: paddy.csv(雨・落水・越流・減水深の累積、種類ごとの貯留・平均湛水深・
!   満水の割合、収支の残差)、分布 Hpn・Hpd(種類ごとの湛水深。A だけ)。タンクの
!   貯留は Log の S(S_surf)に算入。B の区画の湛水深は水深 H そのもの
!
!======================================================================
&list_paddy
  ! 一時無効化(fn_paddy を書いたまま 0 にすると完全不活性)
  f_paddy = 1

  ! 水田面積率(セル面積に対する割合 0〜1。分布か一様値のどちらか一方が必須)。
  ! 土地利用データから前処理で作る(本体は土地利用を読まない)
  fn_pd_frac = "paddy_frac.asc"
  !pd_frac = 0.5

  ! 取組率(水田面積のうち田んぼダムの落水口の割合 0〜1。分布か一様値。既定 0)
  !fn_pd_adopt = "paddy_adopt.asc"
  pd_adopt = 0.5

  ! 代表区画面積 (m2。落水口 1 か所あたりの水田面積。既定 3000 = 30 a)
  !pd_aplot = 3000.0

  ! 畦畔高 (m。田面から。超えた分は地表へあふれる。既定 0.30)
  !pd_hlev = 0.30

  ! 事前の湛水深 (m。田面から。両方の種類に同じ値。既定 = 種類ごとの流れ始める高さ。
  ! 0 = 中干しなどで空、0.30 = 前の雨で満水)
  !pd_h0 = 0.03

  ! 減水深 (mm/d。系外へ出す。地下水には入らない。既定 5)
  !pd_perc_mmd = 5.0

  ! 地表と水田の水のやり取り(既定 0)
  !   0: 片方向(水田の上の雨だけをタンクに入れる)
  !   1: 双方向(河川・排水路の外水・内水も落水口と畦畔から水田に入る。畦畔より下は水田でない面積だけが
  !      地表の水を貯めて流す。水田でない面積の割合 1 − a/gv ≥ 0.05 が要る)
  !f_pd_exchange = 0

  ! 区画の解像(表現 B。5〜10 m の格子)。区画番号を与えると B になり、上の面積率・取組率・
  ! 代表区画面積・f_pd_exchange は使わない(与えると停止)。区画の幅に 3 セル以上が要る
  !   fn_pd_plot  : 区画番号(0 = 水田でない。整数)。番号が違うセルの間に畦畔の仮想壁面が立つ
  !   fn_pd_outlet: 落水口の向き(出口セルに 1〜8。ラスタの並びで 1 左上・2 上・3 右上・4 左・
  !                 5 右・6 左下・7 下・8 右下。ほかは 0)
  !   fn_pd_class : 落水口の種類(区画のセルに 1 = 通常、2 = 田んぼダム。区画の中で同じ。既定 全て 1)
  !fn_pd_plot = "paddy_plot.asc"
  !fn_pd_outlet = "paddy_outlet.asc"
  !fn_pd_class = "paddy_class.asc"

  ! 落水口の装置(添字 1 = 通常の落水口、2 = 田んぼダム)
  ! 型 0: 水位調整板だけ、1: 機能分離型 = min(水位調整板の堰, 穴 + 流出量調整板の越流)、
  ! 2: 機能一体型 = 穴 + 水位調整板の天端の越流(既定 1 は 0、2 は 1)
  !pd_dev_type(1:2) = 0, 1

  ! 水位調整板の天端の高さ (m。田面から。既定 0.03) と幅 (m。既定 0.30)
  !pd_board_h(1:2) = 0.03, 0.03
  !pd_board_w(1:2) = 0.30, 0.30

  ! 穴の形 1: 円、2: 長方形、3: 三角(V 字の切欠き)。型 1・2 で使う(既定 1)
  !pd_open_type(2) = 1
  ! 円の径 (m。既定 0.05)
  !pd_open_d(2) = 0.05
  ! 長方形の幅・高さ (m。長方形で必須。例は門型の切欠き 12 cm × 5 cm)
  !pd_open_b(2) = 0.12
  !pd_open_e(2) = 0.05
  ! 三角の角度 (度。既定 60)
  !pd_open_ang(2) = 60.0
  ! 穴の下端(三角は頂点)の高さ (m。田面から。既定 0)
  !pd_open_z(2) = 0.0

  ! 機能分離型の流出量調整板の天端の高さ (m) と越流の幅 (m。既定 = 水位調整板の幅)。
  ! 未指定 = 越流しない
  !pd_plate_h(2) = 0.20
  !pd_plate_w(2) = 0.30

  ! 流量係数(堰・穴。既定 0.6 = 手引きの計算プログラムの値。C = 0.525 の堰は本間公式)
  !pd_cw(1:2) = 0.6, 0.6
  !pd_co(1:2) = 0.6, 0.6
/
