!======================================================================
!
! ENC 計算条件 &list_enc の例(全パラメータの一覧)
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_enc = "list_enc.txt"
!   のように指定して使用します(fn_enc 未指定でもコアの浅水流計算は
!   既定値で動きます。このグループは精度・コストの調整用)。
!
!   以下はすべて既定値の明示です。通常の計算で変更を検討するのは
!   適応的ルンゲクッタの閾値 p_adprunge_thresh くらいです
!   (チュートリアル Step 2 参照)。これらのスイッチの多くは計算結果を
!   変えるので、回帰テストの reference とは既定値でのみ比較できます。
!   解説は docs/users_guide/swflow.md。
!
!======================================================================
&list_enc

  ! ---- 時間積分(適応的ルンゲクッタ)----
  !f_adaptive_runge = 1     ! 流速変動の大きいセルだけ4段 RK で再計算
  !p_adprunge_thresh = 1.5  ! 再計算を発動する流速変動率の閾値(1.1〜。
                            !   小さいほど高精度・高コスト)

  ! ---- フラックス・安定化 ----
  !f_gravity_correction = 1 ! 重力の補正(急勾配地形での斜面方向重力の
                            !   誤差を補正)
  !f_exflux_reduction = 1   ! 過大流出フラックスの抑制(乾床前縁の
                            !   負水深防止。乾床のある計算では on 推奨)
  !f_hcap_upwind = 1        ! セル境界水深(0: 両側平均, 1: 上流側水深で頭打ち
                            !   =既定, 2: 上流側水深そのもの。段波先端の
                            !   スパイク抑止。乾床への前縁流出が増える)
  !f_friction_fastmath = 0  ! 摩擦項計算の高速化(0: 厳密, 1〜5: テーブル
                            !   近似。数字が大きいほど粗く速い)
  !p_diagratio = 0.5857864376  ! 8方向フラックスの対角配分比
                            !   (= 2/(2+sqrt(2))。通常変更不要)

  ! ---- 移流項・拡散項 ----
  !f_advection_scheme = 3   ! 移流項のスキーム(1: セル中心勾配=旧既定,
                            !   2: 運動量保存形・1次風上, 3: 運動量保存形
                            !   +MUSCL=既定)。1 は非保存形で段波が遅れ、
                            !   実地形では格子を細かくするほど流出が速く
                            !   大きくなり収束しない。下の
                            !   p_adv_upwind_index は 1 のみに効く
  !f_advection_donor = 0    ! 運動量保存形移流(スキーム 2・3)の風上供給元
                            !   (0: 湿潤セルすべて=既定, 1: 河道セル間の
                            !   エッジでは流下方向の供給元を河道セルのみに。
                            !   fn_rw 必須)。屈曲で
                            !   浸水した谷底セルが供給元になる人工損失の対策
  !f_dry_head_cap = 1       ! 乾燥セルへ向かうエッジ水深をエネルギー頭
                            !   η + u_n²/2g − z_受け手 で頭打ち(0: なし,
                            !   1: 有効=既定)。水面+速度水頭より高い乾いた地盤へ
                            !   水を登らせない(掘込水路・段波先端の漏れ対策)
  !f_opening_dynamic = 1    ! 塞がれた開口の動的振り替え(0: なし,
                            !   1: 河道セル間のエッジのみ=既定(fn_rw が
                            !   なければ無効), 2: 全エッジ)。斜め先の地盤が水面より高くて
                            !   使えない通過幅を開いたエッジへ毎ステップ
                            !   振り替える(1 セル幅河道・壁沿いの通水能
                            !   低下と、スキーム 3 の減衰増幅の対策)
  !p_adv_upwind_index = 0.5 ! 移流項の風上化指数(0〜1。0: 中心差分,
                            !   1: 1次精度の風上差分)
  !f_diffusion_term = 0     ! 拡散項(0: なし, 1: 定数粘性,
                            !   2: ゼロ方程式モデル ν = ν0 + α·u*·h)。
                            !   まず試すなら 2 が手軽(係数指定不要で
                            !   物理的にスケールした ν が働く)
  !p_diffusion_nu = 0.0     ! 動粘性係数 ν0 (m2/s)(モデル2では
                            !   バックグラウンド粘性)。モデル1では
                            !   正値の明示が必須(未指定はエラー停止)
  !p_diffusion_alpha = 0.0683333  ! ゼロ方程式の係数 α(既定 0.41/6 =
                            !   Elder 型)

  ! ---- 非静水圧補正(研究用。docs/nonhydrostatic_plan.md)----
  !f_nonhydrostatic = 0     ! 0: 静水圧(既定), 1: 1 層非静水圧補正
                            !   (β = h²/4。分散性が本質の河川遡上津波の
                            !   ソリトン分裂・地滑り津波の近地造波向け。
                            !   格子は Δx ≲ H/2 が目安)
  !nh_hmin = 0.1            ! この水深未満のセルは静水圧のまま (m)
  !nh_solver = 2            ! 反復ソルバ(1: Jacobi, 2: CG(既定)。Jacobi は
                            !   Δx ≪ H で収束が遅い(比較用))
  !nh_itmax = 500           ! 反復回数の上限
  !nh_tol = 1.0e-6          ! 相対収束判定
  !f_nh_adaptive = 0        ! 活性集合(0: NH マスク内の全域で解く(既定),
                            !   1: 検出量が閾値を超えるセルと、その周り
                            !   nh_margin セルの縁だけで解く。長波・定常流
                            !   の領域にコストが乗らない)
  !nh_detector = 1          ! 検出量(1: 分散型 χ = |βDa*|/(|a*|+|βDa*|)
                            !   ≈ (kH)²/4。振幅に依らない(既定)、
                            !   2: 絶対型 |βDa*|/g)
  !nh_chi_on = 0.06         ! 検出の閾値(χ > nh_chi_on のセルが種。
                            !   0.06 は静水圧との位相速度差 3% の水準)
  !nh_margin = 0            ! 縁の幅(セル数。0: 2H/Δx から自動 =
                            !   減衰距離 H/2 の 4 倍)
  !nh_amin = 1.0e-5         ! 検出の下限 |βDa*| (m/s²)。補正量がこれ未満の
                            !   セルは種にしない(陽解法の前駆ノイズを除く)
  !nh_arel = 1.0e-3         ! 検出の相対下限。そのステップの領域最大の
                            !   |βDa*| に対してこの比未満のセルは種にしない
  !f_nh_breaking = 0        ! 砕波スイッチ(0: なし, 1: 砕波と判定したセルと
                            !   周り nh_break_margin セルを静水圧にして段波と
                            !   して扱う。斜面を遡上する砕波性の波では必須。
                            !   f_hcap_upwind = 2 と併用)
  !nh_break_type = 3        ! 判定量(1: 水面上昇速度 ∂η/∂t > α√(gh)(SWASH 型。
                            !   接するセルは β。波頂通過まで維持する状態を持つ),
                            !   2: フルード数 |V|/√(gh) > nh_break_fr,
                            !   3: 水面勾配 |∇η| > nh_break_slope(既定))
  !nh_break_alpha = 0.6     ! 判定 1 の開始閾値 α
  !nh_break_beta = 0.3      ! 判定 1 の、砕波セルに接するセルの閾値 β
  !nh_break_fr = 0.6        ! 判定 2 のフルード数の閾値
  !nh_break_slope = 0.3     ! 判定 3 の水面勾配の閾値
  !nh_break_margin = 0      ! 砕波セルの周りを静水圧にする縁(セル数。0: 2H/Δx)
  !nh_break_visc = 0.0      ! 砕波域の渦粘性係数 δ_b²(0: なし。1.44 = Kennedy et al. 2000 相当。
                            !   ν_b = nh_break_visc·B·h·√(gh)·|∇η| を運動量の拡散項に加える)
  !nh_bc_margin = 0         ! 強制境界(水位規定・区間流入・流出・放射)から静水圧のままにする縁(セル数。0: 2H/Δx)
  !f_nh_slope = 0           ! 底面勾配項(0: 平坦床の線形 NH(Version 1), 1: 底面の
                            !   運動学条件 w_b = u·∇z_b と圧力項 (φ/h)∇(h+2z_b) を
                            !   含む Version 2。斜面・棚の上の波で使う)
  !f_nh_bottom = 0          ! 動く底面の加速度項(0: なし, 1: 底面の鉛直加速度 z̈_b を
                            !   NH の源項に加える。地滑り津波(f_bedslide)・地形変化・
                            !   溶岩流・API の z 更新に効く。短い地滑り b/H ≲ 2 の近地波形)

  ! ---- その他 ----
  !f_rivermouth_drop = 0    ! 河口から海への段落ち(旧方式。fn_tide とは
                            !   併用不可。現在は潮位+境界条件の利用を推奨)

/
