!======================================================================
!
! ENC 計算条件 &list_enc の例(全パラメータの一覧)
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_enc = "list_enc.txt"
!   のように指定して使用します(fn_enc 未指定でもコアの浅水流計算は
!   既定値で動きます。このグループは精度・コストの調整用)。
!
!   以下はすべて既定値の明示です。通常の計算で変更を検討するのは
!   適応的ルンゲクッタの閾値 p_adprunge_thresh くらいです
!   (チュートリアル Step 2 参照)。これらのスイッチの多くは計算結果を
!   変えるので、回帰テストの reference とは既定値でのみ比較できます。
!   解説は docs/users_guide/swflow.md。
!
!======================================================================
&list_enc

  ! ---- 時間積分(適応的ルンゲクッタ)----
  !f_adaptive_runge = 1     ! 流速変動の大きいセルだけ4段 RK で再計算
  !p_adprunge_thresh = 1.5  ! 再計算を発動する流速変動率の閾値(1.1〜。
                            !   小さいほど高精度・高コスト)

  ! ---- フラックス・安定化 ----
  !f_gravity_correction = 1 ! 重力の補正(急勾配地形での斜面方向重力の
                            !   誤差を補正)
  !f_exflux_reduction = 1   ! 過大流出フラックスの抑制(乾床前縁の
                            !   負水深防止。乾床のある計算では on 推奨)
  !f_hcap_upwind = 1        ! セル境界水深(0: 両側平均, 1: 上流側水深で頭打ち
                            !   =既定, 2: 上流側水深そのもの。段波先端の
                            !   スパイク抑止。乾床への前縁流出が増える)
  !f_friction_fastmath = 0  ! 摩擦項計算の高速化(0: 厳密, 1〜5: テーブル
                            !   近似。数字が大きいほど粗く速い)
  !p_diagratio = 0.5857864376  ! 8方向フラックスの対角配分比
                            !   (= 2/(2+sqrt(2))。通常変更不要)

  ! ---- 移流項・拡散項 ----
  !f_advection_scheme = 3   ! 移流項のスキーム(1: セル中心勾配=旧既定,
                            !   2: 運動量保存形・1次風上, 3: 運動量保存形
                            !   +MUSCL=既定)。1 は非保存形で段波が遅れ、
                            !   実地形では格子を細かくするほど流出が速く
                            !   大きくなり収束しない。下の
                            !   p_adv_upwind_index は 1 のみに効く
  !f_advection_donor = 0    ! 運動量保存形移流(スキーム 2・3)の風上供給元
                            !   (0: 湿潤セルすべて=既定, 1: 河道セル間の
                            !   エッジでは流下方向の供給元を河道セルのみに。
                            !   fn_rw 必須)。屈曲で
                            !   浸水した谷底セルが供給元になる人工損失の対策
  !f_dry_head_cap = 1       ! 乾燥セルへ向かうエッジ水深をエネルギー頭
                            !   η + u_n²/2g − z_受け手 で頭打ち(0: なし,
                            !   1: 有効=既定)。水面+速度水頭より高い乾いた地盤へ
                            !   水を登らせない(掘込水路・段波先端の漏れ対策)
  !f_opening_dynamic = 1    ! 塞がれた開口の動的振り替え(0: なし,
                            !   1: 河道セル間のエッジのみ=既定(fn_rw が
                            !   なければ無効), 2: 全エッジ)。斜め先の地盤が水面より高くて
                            !   使えない通過幅を開いたエッジへ毎ステップ
                            !   振り替える(1 セル幅河道・壁沿いの通水能
                            !   低下と、スキーム 3 の減衰増幅の対策)
  !p_adv_upwind_index = 0.5 ! 移流項の風上化指数(0〜1。0: 中心差分,
                            !   1: 1次精度の風上差分)
  !f_diffusion_term = 0     ! 拡散項(0: なし, 1: 定数粘性,
                            !   2: ゼロ方程式モデル ν = ν0 + α·u*·h)。
                            !   まず試すなら 2 が手軽(係数指定不要で
                            !   物理的にスケールした ν が働く)
  !p_diffusion_nu = 0.0     ! 動粘性係数 ν0 (m2/s)(モデル2では
                            !   バックグラウンド粘性)。モデル1では
                            !   正値の明示が必須(未指定はエラー停止)
  !p_diffusion_alpha = 0.0683333  ! ゼロ方程式の係数 α(既定 0.41/6 =
                            !   Elder 型)

  ! ---- その他 ----
  !f_rivermouth_drop = 0    ! 河口から海への段落ち(旧方式。fn_tide とは
                            !   併用不可。現在は潮位+境界条件の利用を推奨)

/
