!======================================================================
!
! 降雨遮断条件設定 &list_intercept / &list_intercept_fixed の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_intercept = "list_intercept.txt"
!   のように指定して使用します(fn_intercept 未指定なら遮断計算は
!   完全に不活性)。
!
!   共通設定 &list_intercept と、選択したモデルの固有設定
!   (固定遮断率モデルなら &list_intercept_fixed)を同じこのファイルに
!   書きます。固有設定グループは選択されたモデルだけが読むため、
!   使わないモデルのグループが残っていても無視されます。
!
!   遮断は降雨分布の更新直後に一度だけ適用され、降雨強度分布
!   (s%pre / 出力 Pr)が有効雨量(地表到達雨量)になります。
!
!======================================================================
! 遮断共通設定
!======================================================================
&list_intercept
  ! 遮断モデルの選択
  !  0:なし(fn_intercept を書いたまま一時無効化できる)
  !  1:固定遮断率モデル(&list_intercept_fixed を使用)
  f_icmodel = 1

/


!======================================================================
! 固定遮断率モデル固有設定 (f_icmodel = 1)
!======================================================================
!   降雨の一定割合 ic_alpha を遮断し、有効雨量 (1-ic_alpha)P を地表に
!   与える最簡モデル(I = αP)。貯留容量・蒸発・降雨強度依存は
!   表現しません。目安: 樹林地で 0.1〜0.4 程度
!======================================================================
&list_intercept_fixed
  ! 遮断率。0 < ic_alpha < 1 であること
  ic_alpha = 0.2

/
