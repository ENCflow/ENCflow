!======================================================================
!
! 降雨遮断条件設定 &list_intercept /
!   &list_intercept_fixed / &list_intercept_initloss の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_intercept = "list_intercept.txt"
!   のように指定して使用します(fn_intercept 未指定なら遮断計算は
!   完全に不活性)。
!
!   共通設定 &list_intercept と、選択したモデルの固有設定を同じ
!   このファイルに書きます。固有設定グループは選択されたモデルだけが
!   読むため、使わないモデルのグループが残っていても無視されます。
!
!   遮断の適用により、降雨強度分布(s%pre / 出力 Pr)は有効雨量
!   (地表到達雨量)になります。
!
!======================================================================
! 遮断共通設定
!======================================================================
&list_intercept
  ! 遮断モデルの選択
  !  0:なし(fn_intercept を書いたまま一時無効化できる)
  !  1:固定遮断率モデル(&list_intercept_fixed を使用)
  !  2:初期損失モデル(&list_intercept_initloss を使用)
  f_icmodel = 1

/


!======================================================================
! 固定遮断率モデル固有設定 (f_icmodel = 1)
!======================================================================
!   降雨の一定割合を遮断し、有効雨量 (1-α)P を地表に与える最簡モデル
!   (I = αP)。貯留容量・蒸発・降雨強度依存は表現しません。
!   遮断率は一様値または分布ファイルのどちらかで与えます。
!======================================================================
&list_intercept_fixed
  ! 遮断率の一様値。0 < ic_alpha < 1 であること(既定 0.15 = 広葉樹林・
  ! 混交林の代表。目安: 針葉樹林 0.25〜0.4、草地 0.05〜0.1。developer.md §66)
  ic_alpha = 0.15

  ! 遮断率の分布ファイル(dir_data からの相対パス。形式は f_input_mode)。
  ! 指定すると ic_alpha は使われない。値はセルごとに 0 <= α < 1
  ! (0 = 遮断なし。植生のないセルに使う)
  !fn_icalpha = "icalpha.txt"

/


!======================================================================
! 初期損失モデル固有設定 (f_icmodel = 2)
!======================================================================
!   累積遮断量が最大貯留量に達するまで降雨を遮断し(I = min(P, 残容量))、
!   以後は素通しにする Initial Loss モデル。蒸発による貯留の回復
!   (乾き戻り)は表現しません。
!   リスタート時は累積遮断量が save/restore されます
!   (保存ファイル: save/intercept_initloss.dat)。
!======================================================================
&list_intercept_initloss
  ! 最大貯留量の一様値 (mm)。0 より大きいこと(既定 1.5 = 広葉樹林の
  ! 樹冠貯留の代表。目安: 草地 0.3〜0.5、広葉樹林 0.8〜1.5、針葉樹林 1.5〜3、
  ! 市街地の窪地貯留 1〜2.5。developer.md §66)
  ic_smax_mm = 1.5

  ! 最大貯留量の分布ファイル (mm)(dir_data からの相対パス。形式は
  ! f_input_mode)。指定すると ic_smax_mm は使われない。
  ! 値はセルごとに 0 以上(0 = 遮断なし)
  !fn_icsmax = "icsmax.txt"

/
