!======================================================================
! 潮位設定(m_tide)のサンプル
!   有効化: &list_sysparam の fn_tide にこのファイルを指定する
!   (fn_tide 未指定なら潮位なし。海セルは z=0, h=0 の従来挙動)。
!   潮位は海域マスク(list_geoinfo の fn_sw)の領域内セルに水位として
!   強制される: s%z = 潮位 - hsea0, s%h = hsea0(海セルは解かない)。
!   ENC 格子系専用。f_rivermouth_drop とは併用不可。
!   仕様の正本は docs/developer.md §23
!======================================================================
&list_tide
  ! 潮位タイプ
  !   1: 一様固定値(ti0)
  !   2: 一様時系列(tival: 時刻(min), 潮位(m)。線形補間)
  !   3: 分布1枚×倍率時系列(fn_timap(m)× tival: 時刻(min), 倍率)
  !   4: 分布ファイルリスト(fn_timaplist。フレーム i の時刻は
  !      (i-1)*dt_timaplist。隣接2枚を線形補間、範囲外は端値保持)
  titype = 2

  ! 海セルの見かけ水柱厚 (m)。海が供給側になる浸水(高潮・上げ潮)は
  ! この水柱が乾燥判定・セル境界水深・摩擦を賄うため、dd の数十倍以上を
  ! 推奨(小さすぎると浸水流量が過小になる)。dd 以下はエラー
  hsea0 = 1.0

  ! 潮位更新時間間隔 (min)。位相は絶対ステップ番号基準
  ! (リスタート後も中断なしランと同じステップで更新される)
  dt_tiupdate = 1

  ! titype=1: 一様固定潮位 (m)
  !ti0 = 0.5

  ! titype=2, 3: 時系列(時刻(min), 潮位(m)または倍率)
  tival(1:2,1) = 0,    0.0
  tival(1:2,2) = 180,  0.8
  tival(1:2,3) = 360,  0.0
  tival(1:2,4) = 540, -0.8
  tival(1:2,5) = 720,  0.0

  ! titype=3: 潮位分布ファイル (m)
  !fn_timap = 'tide_amp.txt'

  ! titype=4: 潮位分布ファイルリスト(1行1ファイル名)と時間間隔
  !fn_timaplist = 'tide_maps.txt'
  !dt_timaplist = 60        ! (min)
  !dt_timaplist_c = '1h'    ! 文字列指定(h, m, s)。指定時はこちらが優先
/
