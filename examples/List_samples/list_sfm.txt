!======================================================================
!
! 貯留関数法の区画流出の設定 &list_sfm / &list_sfm_zone の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_sfm = "list_sfm.txt"
!   のように指定して使用します(fn_sfm 未指定なら完全に不活性)。
!
!   木村(1961)の貯留関数法の基本形: 区画番号ラスタの区画ごとに 1 つのタンク
!   dS/dt = r − q、S [mm] = K·q[mm/h]^P を解き、区画の流出を出口セル(または
!   系外)へ瞬間的に移します。同じ区画のセルはすべて同じ流出(単位面積あたり)で、
!   セルごとのパラメータはありません。出口と基底流の指定は VZ 流出(list_vz.txt)と
!   同じです(users_guide/sfm.md、developer.md §106)。
!
!   必要な入力:
!     - 区画番号ラスタ fn_sfm_zone(dir_data 下。0 = 適用外)
!     - 区画ごとの貯留係数 sfm_k(必須。流域ごとの校正量で既定なし)
!     - 区画ごとの出口 sfm_out_kind(必須)
!   貯留指数 sfm_p(既定 0.6)と初期の流出強度 sfm_q0_mmh(既定 0.1 mm/h)は
!   省略でき、省略すると採用値を (default) 付きで表示して使う
!
!   初版で併用できない機能(指定すると停止): fn_gwflow・fn_snow・
!   fn_glacier・fn_geomorph・fn_wq・fn_salt・ネスティング・STG 格子。
!   VZ 流出(fn_vz)とは区画が重ならなければ併用できる
!
!   遅れ時間 T_L(sfm_tl_h)を与えると、雨は T_L 遅れてタンクに入る(遅れの間の雨は
!   移動中の水として台帳と sfm_zone.csv の vlag に数える)
!
!   蒸発散(fn_evap)と併用すると、区画セルに残った要求をタンクから引く(sfm_elim_mm で
!   貯留が減ったときに抑えられる。引いた量は sfm_zone.csv の vevap)
!
!   持たないもの: 一次流出率 f1・飽和雨量 R_sa・2 価の貯留関数・深部浸透の損失
!
!   出力: sfm_zone.csv(区画の雨・流出強度・貯留と累積水量・収支残差)。
!   タンクの貯留は Log の S_grnd に算入
!
!======================================================================
&list_sfm
  ! 一時無効化(fn_sfm を書いたまま 0 にすると完全不活性)
  f_sfm = 1

  ! 区画番号ラスタ(整数。0 = 適用外、1, 2, … = 区画番号の連番)
  fn_sfm_zone = "sfm_zone.asc"

  ! タンクを進める同期間隔 (s)。0 = 毎ステップ(既定。タンクの費用は無視できる)
  !dt_sfm = 0.0

  ! タンクの入力(雨)の扱い
  !   0: 区画のセルの地上雨量の一致を要求(違えば停止。既定)
  !   1: 区画の全セルの面積平均(雨に分布があるとき)
  !f_sfm_rain = 0

  ! 数値制御(通常は既定のまま。結果の確認には許容を 1/10 にした再計算と比べる)
  !sfm_tol = 1.0e-4         ! 内部刻みの時間誤差の相対許容(流出強度に対する比)
  !sfm_dt_min = 0.001       ! 内部刻みの下限 (s。下回ると停止)
/

&list_sfm_zone
  ! 区画(区画番号 z で添字。最大 50)
  ! 貯留係数 K(必須。S [mm] = K·q[mm/h]^P。単位 mm^(1−P)·h^P。観測で校正する。
  ! 応答の時間の目安は K·P·q^(P−1) 時間。下の値は形を示すための例)
  sfm_k(1) = 30.0

  ! 貯留指数 P(0 < P ≤ 1。既定 0.6。1 = 時定数 K 時間の線形貯留)
  !sfm_p(1) = 0.6

  ! 初期の流出強度 (mm/h。既定 0.1。初期の貯留は S0 = K·q0^P。0 = 乾いた状態)
  !sfm_q0_mmh(1) = 0.1

  ! 遅れ時間 T_L (h。既定 0 = 遅れなし。雨が T_L 遅れてタンクに入る。時間刻み dt の整数倍に
  ! 丸めて採用値を表示。開始時の遅れの間の水は初期の流出強度の定常)
  !sfm_tl_h(1) = 0.5

  ! 蒸発散(fn_evap と併用時): 樹冠・地表水などの後に残った要求をタンクの貯留から引く。
  ! 貯留がこの値 (mm) より少ないとき S/sfm_elim_mm の割合に減らす(未指定 = 供給制限のみ)
  !sfm_elim_mm(1) = 10.0

  ! 出口(必須) 1: 地表セルへ移す / 2: 系外へ出す
  sfm_out_kind(1) = 1

  ! 出口セル (i, j)(sfm_out_kind = 1。最大 99。区画セル・海セル・水位規定セル・
  ! VZ と併用するときは VZ のセルも不可)
  sfm_out_cell(:,1,1) = 120, 80
  !sfm_out_cell(:,2,1) = 121, 80

  ! 出口の配分率(和 1。1 セルなら省略可。複数セルなら全て必須)
  !sfm_out_w(1:2,1) = 0.6, 0.4

  ! 基底流(省略可。区画面積あたり mm/h の時系列。(時刻(分), mm/h) の組を線形補間し、
  ! 範囲外は端の値を保持。同期区間ごとに積分して流出と同じ出口へ渡す。タンクの
  ! 貯留には入らない外部供給。namelist は 1 区画 100 点まで)
  !sfm_base(:,1,1) = 0, 0.1
  !sfm_base(:,2,1) = 1440, 0.5
  ! 長い系列はファイルで(dir_data 起点。1 行「時刻(分) mm/h」。namelist より優先)
  !fn_sfm_base(1) = 'base_zone1.txt'
/
