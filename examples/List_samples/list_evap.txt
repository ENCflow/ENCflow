&list_evap
  ! ---- 可能蒸発散(PET)の決定法(排他切替)----
  !   1: 一定速度(evap0)
  !   2: 月別気候値(evap_monthly。暦 date0_c が必要)
  !   3: Hamon 式(気温+可照時間。date0_c と lat が必要)
  !   4: Thornthwaite 式(3 に加えて temp_normal が必要)
  f_evmodel = 3

  !evap0 = 3.0                  ! モード1: PET (mm/day)
  !evap_monthly = 0.5, 0.7, 1.2, 2.0, 2.8, 3.2, 3.6, 3.8, 2.8, 1.8, 1.0, 0.6
                                ! モード2: 月別 PET (mm/day)
  evap_kc = 1.0                 ! 換算係数(パン係数・校正用。全モード共通)

  date0_c = "2019-10-11 00:00"  ! シミュレーション時刻 t=0 の暦
                                !   "YYYY-MM-DD" または "YYYY-MM-DD hh:mm"
  lat = 36.0                    ! 代表緯度 (deg。モード3,4)

  !temp_normal = 5.4, 6.1, 9.4, 14.3, 18.8, 21.9, 25.7, 26.9, 23.3, 18.0, 12.5, 7.7
                                ! モード4: 月平均気温の平年値 (℃。熱指数 I 用)

  ! ---- 気温入力(モード3,4。いずれか1つ)----
  !   PET は暦日ごとに「日の先頭」の気温で更新され日内一定。
  !   気温式は日平均気温の式のため、日平均系列(日単位の点)を推奨。
  !   date0_c の時刻が 00:00 なら整数日の点がそのまま各日の値になる
  !temp0 = 15.0                 ! 一様定数 (℃)
  tempval(:,1) = 0.0, 15.0     ! 一様時系列 (t=0 からの経過日, ℃)
  tempval(:,2) = 1.0, 12.0
  tempval(:,3) = 2.0, 10.0
  !fn_tempmap = "temp_maps.txt" ! 気温分布ファイルリスト(1行1ファイル。
                                !   dt_tempmap_c 間隔で順次適用・終端保持)
  !dt_tempmap_c = "1 day"       ! 分布ファイルの時間間隔

  ! ---- 標高による気温減率(オプション。一様気温入力のみ)----
  f_temp_lapse = 1              ! 1 で有効
  temp_lapse = 0.65             ! 減率 (℃/100m)
  !temp_zref = 200.0            ! 基準標高 (m。省略時=領域最低標高)
