! 蒸発散設定サンプル(&list_evap。developer.md §27)
!   有効化はメインパラメータの fn_evap = "このファイル" または "-"。
!   f_evmodel = 0 でファイルを残したまま一時無効化できる。
!   暦は &list_sysparam の date0_c、気温入力は &list_meteo(fn_meteo)を
!   使う(§29。モード2〜4は date0_c、モード3,4はさらに fn_meteo が必須)

&list_evap
  ! ---- 可能蒸発散(PET)の決定法(排他切替)----
  !   1: 一定速度(evap0)
  !   2: 月別気候値(evap_monthly)
  !   3: Hamon 式(気温+可照時間)
  !   4: Thornthwaite 式(3 に加えて temp_normal が必要)
  f_evmodel = 3

  !evap0 = 3.0                  ! モード1: PET (mm/day)
  !evap_monthly = 0.5, 0.7, 1.2, 2.0, 2.8, 3.2, 3.6, 3.8, 2.8, 1.8, 1.0, 0.6
                                ! モード2: 月別 PET (mm/day)
  evap_kc = 1.0                 ! 換算係数(パン係数・校正用。全モード共通)
  lat = 36.0                    ! 代表緯度 (deg。モード3,4)

  !temp_normal = 5.4, 6.1, 9.4, 14.3, 18.8, 21.9, 25.7, 26.9, 23.3, 18.0, 12.5, 7.7
                                ! モード4: 月平均気温の平年値 (℃。熱指数 I 用)
/
