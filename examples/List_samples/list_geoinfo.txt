!======================================================================
!
! 地理情報設定 &list_geoinfo の例(全パラメータの一覧)
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_geoinfo = "list_geoinfo.txt"
!   のように指定して使用します(既定は "-" = 同じファイルから読む。
!   地理情報は唯一の必須機能です)。
!
!   格子は nx, ny(必須)と、方向ごとに dx か lx のどちらかで
!   定義します(両方書くと整合を検査。矛盾は停止)。bil 入力に
!   .hdr サイドカーがあれば nx, ny, dx, dy は補完されます。
!
!   分布ファイルのパスは &list_sysparam の dir_data 起点の相対パス。
!   解説は docs/users_guide/geoinfo.md。
!
!======================================================================
&list_geoinfo

  ! ---- 格子の定義 ----
  nx = 700, ny = 600        ! 格子数(必須)
  dx = 100.0, dy = 100.0    ! 格子間隔 (m)(または lx, ly = 領域の大きさ)
  !epsg = 6677              ! 座標系の EPSG コード(0: 不明。GeoTIFF 出力・
                            !   座標管理に使用。users_guide/coordinates.md)

  ! ---- 地形(地盤高)----
  f_ztype = 1               ! 0: 固定値 z0(既定), 1: ファイル fn_z
  fn_z = "dem.bil"          ! 地盤高分布ファイル
  !z0 = 0.0                 ! 地盤高固定値 (m)(f_ztype=0)
  !mag_z = 1.0              ! 地盤高に乗じる倍率(感度実験・単位換算用)
  !f_user_routine = ""      ! 地形を加工するユーザールーチン識別名
                            !   (登録名は src/user_geoinfo.f90 冒頭)

  ! ---- 粗度(Manning 係数)----
  !f_rntype = 0             ! 0: 固定値 rn0(既定), 1: ファイル fn_rn
  !rn0 = 0.015              ! 粗度係数固定値
  !fn_rn = "roughness.txt"  ! 粗度係数分布ファイル(f_rntype=1)

  ! ---- 領域マスク(計算対象の限定。マスク外はメモリ・CPU ゼロ)----
  !f_masktype = 0           ! 0: 全域を計算(既定), 1: ファイル fn_mask,
                            ! 2: 海域マスクから自動生成(全陸域+海岸1セル)
  !fn_mask = "basin.txt"    ! 領域マスク(1: 対象, 0: 対象外)

  ! ---- 海域マスク・河道マスク ----
  !fn_sw = "sea.txt"        ! 海域マスク(1: 海域。潮位・高潮・津波計算用)
  !fn_rw = "river.txt"      ! 河道マスク(1: 河道セル)
  !depth_rw = 2.0           ! 河道セルの掘り込み深さ (m)(一律値)
  !fn_depth_rw = "depth.txt"  ! 掘り込み深さの分布(一律値の代わり)
  !rn0_rw = 0.03            ! 河道セルの固定粗度(負値: 設定しない。既定 -1。
                            !   正値なら粗度の与え方 f_rntype=0/1/2 によらず
                            !   河道セルの粗度を上書き。掘り込みの有無と独立)

  ! ---- 家屋群の抵抗(市街地の氾濫)----
  !fn_gv = "gv.txt"         ! 家屋の空隙率(建蔽されていない割合)の分布
  !fn_bb = "bb.txt"         ! 家屋の平均寸法 (m) の分布
  !min_gv = 0.001           ! 空隙率の下限
  !min_bb = 0.001           ! 平均寸法の下限 (m)

  ! ---- ため池 ----
  !fn_rscap = "rscap.txt"   ! ため池の限界貯留高 (m) の分布(値のある
                            !   セルがため池。ポンプ取水と連携)

  ! ---- 土層(地下水計算 fn_gwflow 用)----
  !f_sdtype = 0             ! 0: 固定値 sd0(既定), 1: ファイル fn_sd
  !sd0 = 1.0                ! 土層厚固定値 (m)
  !fn_sd = "sd.txt"         ! 土層厚分布ファイル
  !sy0 = 0.2                ! 比湧水量(有効間隙率)

  ! ---- 海岸堤防(仮想壁面。詳細は list_seawall.txt)----
  !fn_seaside = "seaside.txt"   ! 海側マスク(潮位ケースは省略可・津波は必須)
  !fn_seawall = "seawall.txt"   ! 天端高の分布(陸側セル位置。河川堤防
                            !   fn_bank の河道セル持ちとは逆。海側セルと
                            !   角(対角)だけで接する陸セルにも天端を
                            !   与えないと対角リンクから漏水する)
  !seawall0 = 5.0           ! 一律天端値(fn_seawall と排他。既定 -9999 = なし)
  !f_seawall_datum = 2      ! 高さの基準(1: 自セル地盤基準, 2: 絶対標高=既定)
  !f_seawall_mode = 0       ! 水理モード(0: 越流のみ, 1: フラップ, 2: 強制排水)

/
