!======================================================================
!
! 地形変化条件設定 &list_geomorph の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_geomorph = "list_geomorph.txt"
!   のように指定して使用します(fn_geomorph 未指定なら地形変化計算は
!   完全に不活性)。
!
!   プロセスは排他選択ではなく独立フラグの重ね合わせです。有効な
!   プロセスが dt_geomorph 間隔で順に適用され、s%z(計算地形)を
!   更新します(水深 h は保存され、水位 e = z + h が追従します)。
!
!======================================================================
&list_geomorph
  ! 地形変化の更新時間間隔 (s)。0 なら毎ステップ更新。
  ! dt の整数倍に丸められます
  dt_geomorph = 0.0

  ! 加速係数(morphological acceleration factor)。全プロセス共通。
  ! 各更新の地形変化を morfac 倍し、1回の計算で morfac 回の同一
  ! イベントぶんの長期変動を得ます(1 で加速なし)。
  ! 注意: クリープの陽解法安定条件 D*dt_geomorph*morfac*(1/dx^2+1/dy^2)
  ! <= 1/2 は init が検査し、超過は停止します
  morfac = 1.0

  ! 斜面クリープ(線形拡散 dz/dt = D∇²z)(0:無効, 1:有効)
  f_creep = 0
  creep_d = 0.0            ! クリープ拡散係数 D (m2/s)。f_creep=1 で必須
/
