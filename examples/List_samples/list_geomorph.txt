!   のように指定して使用します(fn_geomorph 未指定なら地形変化は
!   完全に不活性)。
!

!   プロセスだけが順に適用され、0 のプロセスは資源を確保しません。
!   時間発展する地形は &list_sysparam の f_out_z = 1 で出力できます。
!   検証済みの例題: test/creep, fluvial, suspend, wash, debris, slide,
!   sedinflow。解説は docs/users_guide/geomorph.md。

&list_geomorph

  ! ---- 実行制御(全プロセス共通)----
  !dt_geomorph = 0.0        ! 地形変化の更新間隔 (s)。0: 毎ステップ(既定)
  !morfac = 1.0             ! 地形加速係数(平衡地形・長期検討用。
                            !   イベント計算では 1 のまま。f_debris は
                            !   morfac = 1 が必須)

  ! ---- 斜面クリープ(線形拡散)----
  f_creep = 1
  creep_d = 1.0e-6          ! クリープ拡散係数 (m2/s)

  ! ---- 掃流砂(Exner。河床の浸食・堆積)----
  !f_fluvial = 1
  !f_qbform = 1             ! 流砂量式(1: 芦田・道上, 2: MPM)
  !fluv_d50 = 0.001         ! 代表粒径 (m)。必須
  !fluv_tausc = 0.05        ! 限界無次元掃流力 τ*c
  !fluv_porosity = 0.4      ! 河床の空隙率
  !fluv_sgrav = 1.65        ! 土粒子の水中比重
  !fluv_dzmax = 0.05        ! 1エッジ・1更新の河床変動上限 (m)(安定化)
  !fluv_bcfeed = 0          ! 開境界の給砂(0: 流入は無給砂, 1: 平衡給砂。
                            !   流出は常に容量輸送)
  !fluv_diagratio = 0.5857864376  ! 斜め方向の配分(通常変更不要)

  ! ---- 浮遊砂(移流+巻き上げ・沈降)----
  !f_suspend = 1
  !f_esform = 1             ! 平衡濃度式(1: 超過掃流力線形(簡易),
                            !   2: 板倉・岸)
  !susp_d50 = 0.0001        ! 代表粒径 (m)。必須
  !susp_wf = 0.0            ! 沈降速度 (m/s)。0: Rubey 式で d50 から導出
  !susp_tausc = 0.05        ! 浮遊の限界無次元掃流力
  !susp_beta = 1.0          ! 沈降の底面濃度係数(c_b = β・C)
  !susp_esa = 0.001         ! 平衡濃度係数(f_esform=1 で必須。校正パラメータ)

  ! ---- 斜面浸食(雨滴+面状。f_suspend が必須)----
  !f_wash = 1
  !wash_kr = 0.001          ! 雨滴侵食係数(無次元。E = kr×降雨強度)
  !wash_kf = 1.0e-6         ! 面状侵食係数 (m/s)
  !wash_tausc = 0.05        ! 面状侵食の限界無次元掃流力

  ! 土石流・地滑り(高橋型平衡濃度の E-D)(0:無効, 1:有効)
  !   地表水 h と土砂柱状量 hs の単層混合体で土石流を表現する
  !   (docs/debris_plan.md、developer.md §28)。有効時は流れ計算が
  !   運動量に hs を算入し(流動深 = h+hs)、抵抗則を f_dbres で切替。
  !   f_suspend とは排他(同一 hs 上の E-D 二重計上防止)。morfac=1 必須。
  !   移動可能範囲・深さは土層厚(&list_geoinfo の sd0/f_sdtype/fn_sd)で
  !   与える(sd=0 は不侵食=岩盤)。空隙率・比重は fluv_* を共有。
  !   推奨: 急勾配ケースは &list_enc の f_gravity_correction = 1 を併用
  f_debris = 0
  db_phi = 0.0             ! 土砂の内部摩擦角 (deg)。f_debris=1 で必須
                           !   (例: 34〜35。Morpho2DH の既定は 34)
  f_dbed = 1               ! E-D 式(0:交換なし — 等価流体モード(移流・停止のみ。
                           !   河床侵食を持たない火砕流・岩屑なだれ用。
                           !   f_dbstop=1 の平衡濃度は 0 扱い=低速セルの hs 全量を
                           !   db_wstop レートで河床へ固定),
                           !   1:平衡濃度への緩和(簡易形。δe/δd・|V| レート),
                           !   2:江頭・芦田1992 — E = |V|・C*・tan(θ−θe)。
                           !   流向勾配・追加パラメータなし(文献照合済み),
                           !   3:高橋・中川1991 — 飽和床侵食(式12)+慣性係数付き
                           !   堆積(式27)。レート δe/δd・|V|h/d(文献照合済み。
                           !   db_d50 必須。洞谷適用値 δe=0.0001, δd=0.5))
  db_delte = 0.0007        ! 侵食速度係数 δe(f_dbed=1,3。校正パラメータ)
  db_deltd = 0.05          ! 堆積速度係数 δd(f_dbed=1,3。校正パラメータ)
  f_dbres = 1              ! 抵抗則(0:マニングのみ, 1:クーロン+マニング合成,
                           !   2:江頭構成則(降伏応力+層流抵抗。文献照合済み。
                           !   db_d50/db_erest 必須),
                           !   3:高橋・中川1991 石礫型(降伏 tanα'=0.45 +
                           !   ダイラタント項。濃度・相対水深で石礫型/掃流状/
                           !   泥流を自動切替。文献照合済み。db_d50 必須),
                           !   4:Voellmy 等価流体(降伏 μ + 乱流項 gV²/(ξh)。
                           !   岩屑なだれ・火砕流の実務標準(RAMMS/Titan2D 系)。
                           !   db_mu/db_xi 必須。f_dbed=0 との組が典型),
                           !   5:一定停止応力 τ_y + マニング合成(VolcFlow 型。
                           !   火砕流のランアウト・堆積再現に実績。db_tauy 必須))
  db_d50 = 0.0             ! 代表粒径 (m)。f_dbres=2,3 / f_dbed=3 で必須
  db_erest = 0.0           ! 粒子の反発係数 e(0〜1]。f_dbres=2 で必須(材料固有値)
  db_cmin = 0.02           ! 江頭層流則の濃度下限(未満はマニングで閉じる)
  f_dbstop = 0             ! 停止条件(0:なし, 1:低速凝集 — vv<db_vstop のセルで
                           !   平衡超過の土砂を db_wstop レートで河床(標高 z)へ
                           !   固定する。停止土砂による天然ダム形成を表現)
  db_vstop = 0.05          ! 停止判定の速度閾値 (m/s)(f_dbstop=1 と f_dbres>=1 の
                           !   降伏判定が共有)
  db_wstop = 0.0           ! 低速凝集の河床転換レート (m/s)。f_dbstop=1 で必須
  f_dbwet = 0              ! 間隙水連行(0:無視, 1:河床交換 Δz に伴う間隙水を
                           !   地表水と授受 — 侵食: h += λ・s_b・|Δz|、
                           !   堆積: h −= λ・s_b・Δz(h 残量でクランプ)。
                           !   高橋・中川1991 式(5)の i{c*+(1-c*)s_b} と同じ台帳。
                           !   侵食卓越では正味の増水(bulking)となる一次効果。
                           !   gwflow とは排他(hg 台帳と二重計上のため))
  db_satbed = 1.0          ! 堆積層の飽和度 s_b(0〜1。f_dbwet=1 で使用。
                           !   洞谷適用値は勾配 21°以上で 0.8、以下で 1.0)
  db_mu = 0.0              ! Voellmy 摩擦係数 μ(無次元)。f_dbres=4 で必須
                           !   (雪崩 0.1〜0.3。大規模崩壊の見かけ摩擦低下
                           !   (H/L)は μ を直接与えて表現する)
  db_xi = 0.0              ! Voellmy 乱流係数 ξ (m/s²)。f_dbres=4 で必須
                           !   (慣用範囲 200〜1000)
  db_tauy = 0.0            ! 一定停止応力 τ_y (Pa)。f_dbres=5 で必須
                           !   (火砕流の適用例 5〜50 kPa)

  ! 発生のさせ方 1: 瞬時流動化(崩壊深の分布を与える)
  !fn_dbinit = "collapse_depth.txt"  ! 崩壊深分布 (m)。指定で有効化
  !db_reltime = 0.0         ! 流動化の発生時刻 (s)。t0 以前なら最初の更新で発火
  !db_relsat = 1.0          ! 崩壊土塊の飽和度 0〜1(gwflow 無効時の間隙水付与)

  !f_slide = 1

  !slide_phi = 30.0         ! 土のせん断抵抗角 (deg)。必須
  !slide_gamma = 18000.0    ! 飽和単位体積重量 (N/m3)。必須

  ! ---- 長期地形発達(千年スケール。morfac・リスタート連鎖と併用)----
  !f_wthr = 1               ! 基岩風化=土層生成
  !wthr_p0 = 100.0          ! 裸岩での土層生成速度 (mm/kyr)。必須
  !wthr_sdstar = 0.5        ! 生成の減衰深 (m)
  !f_uplift = 1             ! 隆起
  !uplift0 = 1.0            ! 隆起速度 (mm/yr)。必須
