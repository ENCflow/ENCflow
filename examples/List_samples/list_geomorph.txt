!======================================================================
!
! 地形変化条件設定 &list_geomorph の例
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_geomorph = "list_geomorph.txt"
!   のように指定して使用します(fn_geomorph 未指定なら地形変化計算は
!   完全に不活性)。
!
!   プロセスは排他選択ではなく独立フラグの重ね合わせです。有効な
!   プロセスが dt_geomorph 間隔で順に適用され、s%z(計算地形)を
!   更新します(水深 h は保存され、水位 e = z + h が追従します)。
!
!======================================================================
&list_geomorph
  ! 地形変化の更新時間間隔 (s)。0 なら毎ステップ更新。
  ! dt の整数倍に丸められます
  dt_geomorph = 0.0

  ! 加速係数(morphological acceleration factor)。全プロセス共通。
  ! 各更新の地形変化を morfac 倍し、1回の計算で morfac 回の同一
  ! イベントぶんの長期変動を得ます(1 で加速なし)。
  ! 注意: クリープの陽解法安定条件 D*dt_geomorph*morfac*(1/dx^2+1/dy^2)
  ! <= 1/2 は init が検査し、超過は停止します
  morfac = 1.0

  ! 斜面クリープ(線形拡散 dz/dt = D∇²z)(0:無効, 1:有効)
  f_creep = 0
  creep_d = 0.0            ! クリープ拡散係数 D (m2/s)。f_creep=1 で必須

  ! 掃流砂 Exner(河床の浸食・堆積)(0:無効, 1:有効)
  !   平衡流砂量による河床変動。土層厚(可動層厚)の初期値は
  !   &list_geoinfo の sd0 / f_sdtype / fn_sd(gwflow と共通)で与える。
  !   浸食・堆積は土層厚 s%sd を標高と共動更新し、gwflow 併用時は
  !   貯留容量 sd*sy0 の減少(飽和・流出の早期化)が自動で結合する。
  !   制約: STG 非対応
  f_fluvial = 0
  f_qbform = 1             ! 流砂量式(1:芦田・道上, 2:MPM)
  fluv_d50 = 0.0           ! 代表粒径 (m)。f_fluvial=1 で必須
  fluv_tausc = 0.05        ! 限界無次元掃流力 τ*c
  fluv_porosity = 0.4      ! 河床の空隙率 λ
  fluv_sgrav = 1.65        ! 土粒子の水中比重 s = (ρs-ρ)/ρ
  fluv_dzmax = 0.05        ! 1エッジ・1更新の河床変動上限 (m)(暴走ガード)
  fluv_bcfeed = 0          ! 開境界の掃流砂給砂(0:流入は無給砂, 1:平衡給砂。
                           ! 開境界からの流出は常に容量輸送)

  ! 浮遊砂(移流+浸食・沈降)(0:無効, 1:有効)
  !   移流は流れ計算の内部(連続式と同一のエッジ流量・係数)で毎ステップ
  !   行われ、浸食・沈降(E-D)が dt_geomorph 間隔で河床と交換される。
  !   河床側の変化は ×morfac(水柱側は水理時間のまま=MORFAC の台帳分離)。
  !   掃流砂と独立に有効化できる(重ね合わせ)。土層厚・空隙率・比重は
  !   掃流砂と共有(sd0 / fluv_porosity / fluv_sgrav)
  f_suspend = 0
  f_esform = 1             ! 平衡濃度式(1:超過掃流力線形(簡易),
                           !   2:板倉・岸(定数は原式固定。susp_esa/tausc 不使用))
  susp_d50 = 0.0           ! 浮遊砂の代表粒径 (m)。f_suspend=1 で必須
  susp_wf = 0.0            ! 沈降速度 (m/s)。0 なら Rubey 式で d50 から導出
  susp_tausc = 0.05        ! 浮遊の限界無次元掃流力 τ*c
  susp_beta = 1.0          ! 沈降の底面濃度係数(c_b = β・C)
  susp_esa = 0.0           ! 平衡濃度係数(C_eq = esa・(τ*/τ*c - 1))。
                           ! f_esform=1 で必須(f_esform=2 では不使用)

  ! 斜面浸食(雨滴+面状侵食)(0:無効, 1:有効)
  !   剥離した土砂を浮遊砂 hs へ注入する(輸送・沈降は浮遊砂の管轄=
  !   f_suspend=1 が必須。河道への土砂流入は移流が自動的に実現する)。
  !   河道セル(rw>0)は対象外(河道内は f_fluvial/f_suspend の管轄)。
  !   E = wash_kr・降雨強度 + wash_kf・(τ*/τ*c - 1)
  f_wash = 0
  wash_kr = 0.0            ! 雨滴侵食係数(無次元。雨量あたりの剥離固体体積)
  wash_kf = 0.0            ! 面状侵食係数 (m/s)
  wash_tausc = 0.05        ! 面状侵食の限界無次元掃流力 τ*c

  ! 土石流・地滑り(高橋型平衡濃度の E-D)(0:無効, 1:有効)
  !   地表水 h と土砂柱状量 hs の単層混合体で土石流を表現する
  !   (docs/debris_plan.md、developer.md §28)。有効時は流れ計算が
  !   運動量に hs を算入し(流動深 = h+hs)、抵抗則を f_dbres で切替。
  !   f_suspend とは排他(同一 hs 上の E-D 二重計上防止)。morfac=1 必須。
  !   移動可能範囲・深さは土層厚(&list_geoinfo の sd0/f_sdtype/fn_sd)で
  !   与える(sd=0 は不侵食=岩盤)。空隙率・比重は fluv_* を共有。
  !   推奨: 急勾配ケースは &list_enc の f_gravity_correction = 1 を併用
  f_debris = 0
  db_phi = 0.0             ! 土砂の内部摩擦角 (deg)。f_debris=1 で必須
                           !   (例: 34〜35。Morpho2DH の既定は 34)
  f_dbed = 1               ! E-D 式(1:高橋型平衡濃度への緩和(δe/δd を使用),
                           !   2:江頭・芦田1992 — E = |V|・C*・tan(θ−θe)。
                           !   流向勾配・追加パラメータなし。文献照合済みの推奨)
  db_delte = 0.0007        ! 侵食速度係数 δe(f_dbed=1 のみ。校正パラメータ)
  db_deltd = 0.05          ! 堆積速度係数 δd(f_dbed=1 のみ。校正パラメータ)
  f_dbres = 1              ! 抵抗則(0:マニングのみ, 1:クーロン+マニング合成,
                           !   2:江頭構成則(降伏応力+層流抵抗。文献照合済み。
                           !   db_d50/db_erest が必須))
  db_d50 = 0.0             ! 代表粒径 (m)。f_dbres=2 で必須(層流抵抗の h/d)
  db_erest = 0.0           ! 粒子の反発係数 e(0〜1]。f_dbres=2 で必須(材料固有値)
  db_cmin = 0.02           ! 江頭層流則の濃度下限(未満はマニングで閉じる)
  f_dbstop = 0             ! 停止条件(0:なし, 1:低速凝集 — vv<db_vstop のセルで
                           !   平衡超過の土砂を db_wstop レートで河床(標高 z)へ
                           !   固定する。停止土砂による天然ダム形成を表現)
  db_vstop = 0.05          ! 停止判定の速度閾値 (m/s)(f_dbstop=1 と f_dbres>=1 の
                           !   降伏判定が共有)
  db_wstop = 0.0           ! 低速凝集の河床転換レート (m/s)。f_dbstop=1 で必須

  ! 瞬時流動化(地滑り型の初期条件。f_debris=1 が必須)
  !   崩壊深分布 D (m) のセル群を指定時刻に一斉に流動化する:
  !   hs += (1-λ)D、標高 z -= D、土層 sd -= D(流出後の地盤低下を表現)。
  !   間隙水は gwflow 有効時は地下水から、無効時は h += λ・db_relsat・D
  fn_dbinit = ""           ! 崩壊深分布ファイル(指定で有効化)
  db_reltime = 0.0         ! 発生時刻 (s)。t0 以前なら計算開始直後に発火
  db_relsat = 1.0          ! 崩壊土塊の飽和度 0〜1(gwflow 無効時のみ使用)

  ! 無限長斜面安定判定による発生の自動判定(f_debris=1 が必須)
  !   dt_geomorph 周期で安全率 Fs を評価し、Fs<1 のセルの土層を全層
  !   流動化する。間隙水圧は gwflow の飽和厚 hg/sy0 から(降雨→浸透→
  !   飽和→崩壊→土石流化の一連が自動で発生する)
  f_slide = 0
  slide_c = 0.0            ! 有効粘着力 c' (Pa)
  slide_phi = 0.0          ! 土のせん断抵抗角 φs (deg)。f_slide=1 で必須(例: 25〜35)
  slide_gamma = 0.0        ! 飽和単位体積重量 γt (N/m3)。f_slide=1 で必須(例: 19000)
/
