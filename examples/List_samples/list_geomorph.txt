!======================================================================
!
! 土砂・地形変化設定 &list_geomorph の例(全パラメータの一覧)
!
!   このファイルはシステムパラメータ &list_sysparam 内で
!     fn_geomorph = "list_geomorph.txt"
!   のように指定して使用します(fn_geomorph 未指定なら地形変化は
!   完全に不活性)。
!
!   プロセスは排他選択ではなく重ね合わせです。f_XXX = 1 にした
!   プロセスだけが順に適用され、0 のプロセスは資源を確保しません。
!   時間発展する地形は &list_sysparam の f_out_z = 1 で出力できます。
!   検証済みの例題: test/creep, fluvial, suspend, wash, debris, slide,
!   sedinflow。解説は docs/users_guide/geomorph.md。
!
!======================================================================
&list_geomorph

  ! ---- 実行制御(全プロセス共通)----
  !dt_geomorph = 0.0        ! 地形変化の更新間隔 (s)。0: 毎ステップ(既定)
  !morfac = 1.0             ! 地形加速係数(平衡地形・長期検討用。
                            !   イベント計算では 1 のまま。f_debris は
                            !   morfac = 1 が必須)

  ! ---- 斜面クリープ(線形拡散)----
  f_creep = 1
  creep_d = 1.0e-6          ! クリープ拡散係数 (m2/s)

  ! ---- 掃流砂(Exner。河床の浸食・堆積)----
  !f_fluvial = 1
  !f_qbform = 1             ! 流砂量式(1: 芦田・道上, 2: MPM)
  !fluv_d50 = 0.001         ! 代表粒径 (m)。必須
  !fluv_tausc = 0.05        ! 限界無次元掃流力 τ*c
  !fluv_porosity = 0.4      ! 河床の空隙率
  !fluv_sgrav = 1.65        ! 土粒子の水中比重
  !fluv_dzmax = 0.05        ! 1エッジ・1更新の河床変動上限 (m)(安定化)
  !fluv_bcfeed = 0          ! 開境界の給砂(0: 流入は無給砂, 1: 平衡給砂。
                            !   流出は常に容量輸送)
  !fluv_diagratio = 0.5857864376  ! 斜め方向の配分(通常変更不要)

  ! ---- 浮遊砂(移流+巻き上げ・沈降)----
  !f_suspend = 1
  !f_esform = 1             ! 平衡濃度式(1: 超過掃流力線形(簡易),
                            !   2: 板倉・岸)
  !susp_d50 = 0.0001        ! 代表粒径 (m)。必須
  !susp_wf = 0.0            ! 沈降速度 (m/s)。0: Rubey 式で d50 から導出
  !susp_tausc = 0.05        ! 浮遊の限界無次元掃流力
  !susp_beta = 1.0          ! 沈降の底面濃度係数(c_b = β・C)
  !susp_esa = 0.001         ! 平衡濃度係数(f_esform=1 で必須。校正パラメータ)

  ! ---- 斜面浸食(雨滴+面状。f_suspend が必須)----
  !f_wash = 1
  !wash_kr = 0.001          ! 雨滴侵食係数(無次元。E = kr×降雨強度)
  !wash_kf = 1.0e-6         ! 面状侵食係数 (m/s)
  !wash_tausc = 0.05        ! 面状侵食の限界無次元掃流力

  ! ---- 土石流(高橋型 E-D。f_suspend と排他・morfac=1 必須)----
  !f_debris = 1
  !db_phi = 35.0            ! 土砂の内部摩擦角 (deg)。必須
  !db_delte = 0.0007        ! 侵食速度係数 δe(既定は文献の一般値)
  !db_deltd = 0.05          ! 堆積速度係数 δd
  !f_dbres = 1              ! 抵抗則(0: Manning のみ,
                            !   1: クーロン+Manning 合成=推奨)
  !f_dbstop = 0             ! 停止条件(0: なし, 1: 低速凝集)
  !db_vstop = 0.05          ! 停止判定の速度閾値 (m/s)(f_dbstop=1)
  !db_wstop = 0.01          ! 低速凝集の河床転換レート (m/s)(f_dbstop=1 で必須)

  ! 発生のさせ方 1: 瞬時流動化(崩壊深の分布を与える)
  !fn_dbinit = "collapse_depth.txt"  ! 崩壊深分布 (m)。指定で有効化
  !db_reltime = 0.0         ! 流動化の発生時刻 (s)。t0 以前なら最初の更新で発火
  !db_relsat = 1.0          ! 崩壊土塊の飽和度 0〜1(gwflow 無効時の間隙水付与)

  ! 発生のさせ方 2: 無限長斜面の安定判定による自動流動化(Fs < 1 の
  ! セルの土層を全層流動化。間隙水圧は gwflow の飽和厚から)
  !f_slide = 1
  !slide_c = 1000.0         ! 有効粘着力 c' (Pa)。必須(0 可)
  !slide_phi = 30.0         ! 土のせん断抵抗角 (deg)。必須
  !slide_gamma = 18000.0    ! 飽和単位体積重量 (N/m3)。必須

  ! ---- 長期地形発達(千年スケール。morfac・リスタート連鎖と併用)----
  !f_wthr = 1               ! 基岩風化=土層生成
  !wthr_p0 = 100.0          ! 裸岩での土層生成速度 (mm/kyr)。必須
  !wthr_sdstar = 0.5        ! 生成の減衰深 (m)
  !f_uplift = 1             ! 隆起
  !uplift0 = 1.0            ! 隆起速度 (mm/yr)。必須

/
