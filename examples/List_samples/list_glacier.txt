!======================================================================
! 氷河設定サンプル(&list_glacier。developer.md §45、docs/glacier_plan.md)
!   有効化はメインパラメータの fn_glacier = "このファイル" または "-"。
!   気温(fn_meteo)と積雪(fn_snow。涵養の供給源)が必須。
!   状態は氷厚 s%hi(m 氷柱)。場出力 Hi(m)、リスタートは私有ファイル
!   glacier.dat。初期条件は「氷なし+降雪からの涵養」が既定で、実在氷河は
!   fn_glacier_hi0(氷厚推定プロダクト等)で与える。
!   時間の台帳: 流動・侵食・氷収支は地形時間(×gl_morfac)、融解水・
!   降雪(SWE)は水文時間(MORFAC 台帳分離。geomorph 併用時は morfac と
!   同値必須)。長期実行は代表年反復(t_cycle)×リスタート連鎖で行う。
!======================================================================
&list_glacier
  f_glacier = 1             ! 0 でファイルを残したまま一時無効化

  ! ---- 更新間隔と時間加速 ----
  dt_glacier_c = "1 day"    ! 氷化・流動・侵食の間欠実行間隔
  gl_morfac = 1.0           ! 氷河=地形時間の加速係数(fn_geomorph 併用時は
                            !   &list_geomorph の morfac と同値にすること)

  ! ---- 質量収支(G1)----
  gl_dens = 900.0           ! 氷の密度 (kg/m3)
  gl_tfirn_yr = 10.0        ! 雪→氷変換(フィルン化)の e-folding 時間 (年。
                            !   温暖氷河で数年〜10年程度)
  gl_ddfi = 8.0             ! 氷面の度日係数 (mm/℃/day。既定 8。雪面より大きく、
                            !   文献値 6〜12 程度。雪に覆われたセルの氷は融けない)
  gl_tmelt = 0.0            ! 氷面融解の気温閾値 (℃)

  ! ---- SIA 氷体流動(G2)----
  f_glflow = 1              ! 0:流動なし(「流れない氷河」= 水文用途の最簡)
  gl_afl = 1.0e-16          ! Glen 流動則の係数 A (Pa^-3 yr^-1。温暖氷 ~1e-16、
                            !   n=3 固定。陽解法の刻みは D_max から自動決定)
  !gl_cfl = 0.4             ! サブサイクルの安全係数 (0-1)
  !gl_nsubmax = 10000       ! 1回の更新のサブサイクル数上限(暴走ガード)
  !gl_diagratio = 0.3333333333   ! SIA 流動の斜めの配分 d (既定 1/3 = 8 近傍の 9 点と、塞がれた斜めの
                            !   配分の動的な振り替え。格子に斜めの細い谷の氷河が向きによらない。0 = 4 近傍
                            !   (2026-11-05 より前の既定)。developer.md §99)

  ! ---- Weertman 底面滑動(G3。f_glflow 必須)----
  f_glslide = 0             ! 1 で有効(us = As τb^3)
  !gl_as = 1.0e-13          ! 滑動係数 As (m yr^-1 Pa^-3。既定 1e-13 =
                            !   τb 100 kPa で 100 m/yr)

  ! ---- 氷河侵食(G3。f_glslide 必須)----
  !   edot = Kg・us^l(us は m/yr で評価)で s%z を削る。s%sd は共動減少
  !   (sd=0 まで基岩固定、以後は基岩侵食)。侵食物は系外排出とみなす
  !   (台帳を dispose で報告。glacier_plan.md §6 の既知の妥協)
  f_glero = 0               ! 1 で有効
  !gl_kg = 1.0e-4           ! 侵食係数 Kg(l=1 で無次元。既定 1e-4 =
                            !   Humphrey & Raymond 1994 の実測)
  !gl_lexp = 1.0            ! 滑動速度べき指数 l (1〜2)
  !gl_lc = 500.0            ! 侵食の結合長 (m)。既定 max(500 m, 2Δx)(谷氷河の実測の結合長)。
                            !   侵食に使う駆動応力を標準偏差 gl_lc のガウス核で氷の中だけ平均する。
                            !   目安は氷厚の 2 倍かつ格子間隔の 2 倍以上(カールの薄い氷河では
                            !   小さく)。0 = 局所の τb(長期の侵食は格子規模の溝・穴が育つ。
                            !   developer.md §45.3・§45.4)

  ! ---- 雪の重力再配分=雪崩(G4)----
  !   氷面 z+hi の最急勾配が閾値を超えるセルの SWE を、最急降下の向きを挟む
  !   2 つの近傍へ移送(D∞。急崖から圏谷への涵養集中)
  f_glava = 0               ! 1 で有効
  !gl_ava_tanc = 0.6        ! 限界勾配 tan(θc)(0.6 ≈ 31°)

  ! ---- 初期氷厚(未指定なら 0 = 無氷から開始)----
  !fn_glacier_hi0 = "hi0.txt"  ! 初期氷厚分布 (m 氷厚。Farinotti 等の全球
                            !   氷厚推定プロダクトの切り出し等。restore 時は
                            !   復元値が勝つ)
