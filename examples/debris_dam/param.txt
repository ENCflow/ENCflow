!======================================================================
! examples/debris_dam: 豪雨 → 斜面崩壊 → 土石流 → 天然ダム → 越流 → 決壊 → 下流の洪水(基本ケース)
!   支流の谷頭の上の斜面(崩積土 3 m・15 万 m³)が、豪雨の浸透で地下水位が上がって崩れ、土石流に
!   なって支流を下り、本川(流量 40 m³/s)をせき止める。上流の盆地に水がたまり、堤体を越えると流れが
!   堤体を削って決壊し、下流の集落へ洪水が出る。崩壊の発生から家屋の被害までを 1 回の計算でつなぐ。
!   入力は make(make_init.py)が data_g10/ に作る。ほかのケースの param_*.txt は、make_init.py が
!   このファイルの行を差し替えて書く。README を参照
!======================================================================

  dt = 0.1                  ! 時間刻み (s)。dt = 0.01 Δx
  t0 = 0

  dt_disp = 600
  dt_file = 120             ! 分布出力 2 分おき
  dt_recrd = 10             ! 計測点・流量測線 10 秒おき
  fn_geoinfo = '-'

  f_out_z = 1               ! 地盤高 Z(堆積・侵食)
  f_out_e = 1               ! 水位 E
  f_out_hs = 1              ! 土砂柱状量 Hs(流動深 = H + Hs)
  f_out_hg = 1              ! 地下貯留 Hg(土層の飽和の進み)
  f_out_fs = 1              ! 斜面の安全率 Fs と期間最小 Fs9999
  f_out_dmax = 1            ! 最大流動深 D9999
  f_out_fmax = 1            ! 最大流体力 F9999
  f_out_bds = 1             ! 家屋の破壊率 Bs
  f_out_wbd = 1             ! 堆積した瓦礫 Bd

  f_sdtype = 1

  sy0 = 0.2                 ! 比湧水量(水の出入りする間隙)。空隙率 0.4 との差 0.2 は土と動く保持水
  fn_gv = 'gv.txt'          ! 建物の空隙率(家屋のセル 0.5)

&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ更新
  morfac = 1.0              ! f_debris は 1 必須
  f_debris = 1
  f_dbed = 3                ! 侵食・堆積: 高橋・中川 (1991)
  f_dbres = 3               ! 抵抗則: 高橋・中川 (1991)
  db_phi = 35.0             ! 内部摩擦角 (deg)
  db_d50 = 0.03             ! 代表粒径 d_L (m)
  db_delte = 0.0007         ! 飽和床の侵食係数 δe
  db_deltd = 0.5            ! 堆積係数 δd
  f_dbunsat = 1             ! 河床が不飽和のセルは不飽和床の侵食式(K = 0.01)
  f_dbimm = 1               ! 緩い勾配は掃流状集合流動の平衡濃度 6.7 c∞² と堆積式
  db_immexp = 2             ! 掃流状集合流動の抵抗を (d_L/h)² で
  f_dbstop = 1              ! 低速で止まった土砂を河床へ戻す
  db_vstop = 0.05
  db_wstop = 0.05
  f_dbwet = 3               ! 間隙水: 排水可能な水と保持水を土砂と一緒に授受する

  f_fluvial = 1             ! 掃流砂を併用する(緩い勾配の区間の土砂輸送を受け持つ)
  f_qbform = 1              ! 流砂量式: 芦田・道上
  fluv_d50 = 0.03           ! 掃流砂の粒径 (m)。db_d50 とそろえる
  fluv_taufac = 0.6         ! 有効掃流力の比(粒子の粗度 0.027〜0.031 と河道の粗度 0.04 の比の 1.5 乗)
  f_slide = 1               ! 斜面安定判定(安全率 < 1 の土層を流動化)
  slide_c = 0.0             ! 粘着力 (Pa)
  slide_phi = 35.0          ! せん断抵抗角 (deg)。db_phi とそろえる
  slide_gamma = 19500.0     ! 飽和単位体積重量 (N/m3) = ρw g (1 + s C*)
  f_slide_seep = 1          ! 浸透力を全水頭の勾配で入れる(湖の下の斜面を過小に見積もらない)
  f_slide_wet = 1           ! 間隙水圧で起きる崩壊だけを流動化(段差は安息角崩落に任せる)
  f_slide_noflow = 1        ! 濃い混合体が載るセルは判定しない(載っている土砂が薄いセルは判定する。既定)
  slide_dmax = 3.0          ! 判定・流動化する深さの上限 (m)。崩積土の厚さ(すべり面は土層の底)
  f_repose = 1              ! 安息角崩落(決壊口の側壁)
  repose_phi = 35.0
/
