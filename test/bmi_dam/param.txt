!======================================================================
! bmi_dam: ダムの量の BMI 公開と外部運転(f_dam_mode = 6)の受け入れ試験
! (developer.md §133)
!   孤立した 1 セルの捕捉帯 4 つ(i = 2。周りは領域外)に内部ソースで同じ
!   ハイドログラフを入れ、放流は i = 5〜24 の平坦な閉じた池へ出す。
!     ダム 1: 放流量の時系列(段状。モデルが自分で放流する)
!     ダム 2: 外部運転。Python が毎歩、ダム 1 と同じ段状の放流量を set する
!             → ダム 1 と全行一致することを確かめる
!     ダム 3: 外部運転。Python の見本の制御器(bmi/python/dam_control.py。
!             貯水位を保つ・放流の上限・変化の速さの制限)で運転する
!     ダム 4: 一定量放流 + 制限水位(モデルの運転則)。目標放流量の set の
!             要素は無視されることを確かめる
!   検定は bmi/python/test_dam.py(Run.sh が実行する)
!======================================================================

!======================================================================

!======================================================================
&list_struct_dam

  dam_in_cell(:,1,1) = 2, 2
  dam_out_cell(:,1,1) = 10, 2
  dam_hv(:,1,1) = 100.0, 0.0
  dam_hv(:,2,1) = 105.0, 2.0e4
  dam_hv(:,3,1) = 112.0, 6.0e4

  dam_qout_val(:,1,1) = 0, 0.3
  dam_qout_val(:,2,1) = 20, 0.3
  dam_qout_val(:,3,1) = 20, 0.6
  dam_qout_val(:,4,1) = 60, 0.6
  dam_qout_val(:,5,1) = 60, 0.4
  dam_qout_val(:,6,1) = 400, 0.4
  dam_h_init(1) = 104.0
  ! ダム 2: 外部運転(Python がダム 1 と同じ放流量を与える。初期値 0.3)
  dam_in_cell(:,1,2) = 2, 4
  dam_out_cell(:,1,2) = 10, 4
  dam_hv(:,1,2) = 100.0, 0.0
