!======================================================================
! salt: 淡塩2層の模式実験 C2(地下→地表の湧出の淡塩配分 f_salt_exch)
!
!   東へ下る斜面(z = 1 → 0 m、i = 1..17)の先に池(底 −0.5 m、i = 18..20、
!   初期水位 −0.1 m の淡水 = 水深 0.4 m)。20 × 8 セル、Δx = 2 m、四周壁。
!   帯水層(底 −1 m、sy = 0.2)は初期に容量の 8 割まで全層が塩水
!   (fn_salt_hgs0 = 0.8·sd·sy。淡水なし)。地下水は斜面を下って池の下で
!   容量を超え、池の底へ湧き出る。帯水層に淡水がないので、淡水優先の
!   規則でも湧き出るのは塩水(f_salt_exch=1。developer.md §47.5)。確認事項:
!     - 池の底に塩水層ができる(Hss > 0。湧出量の分だけ)。f_salt_exch=0
!       なら湧出は暗黙に淡水扱いで Hss = 0 のまま(塩水は整合クランプで消える)
!     - 閉領域の塩水体積 Σ(Hss + Hgs) の保存。残る減少は加算方式の時間
!       分割誤差(側方流が先に合計を動かす分。§47.2)で、dt に比例する
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 7200                 ! 計算終了時刻 (s)
  dt_disp = 600             ! 画面表示時間刻み (s)
  dt_file = 3600            ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_gwflow = '-'
  fn_salt = '-'
  dir_data = 'data_salt'
  dir_result = 'result_exch_out'
  dir_save = 'save_exch_out'
  f_state_save = 1          ! 終状態を倍精度で保存(Check_exch.py が読む)
  f_disp_debug = 1          ! S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_hg = 1              ! Hg / Hgs 場の出力
/

&list_enc
  f_adaptive_runge = 1
/

&list_initial

&list_geoinfo

  f_rntype = 0              ! 粗度係数タイプ (0:固定値)
  rn0 = 0.03                ! 粗度係数固定値
  f_sdtype = 1              ! 土層厚タイプ (1: ファイル)
  fn_sd = 'sd_slope.txt'    ! 土層厚 = z + 1(帯水層底 −1 m)
  sy0 = 0.2                 ! 比湧水量
/

&list_bound
/

!======================================================================
! 地下水設定(Green-Ampt の鉛直浸透+層1側方流)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! Green-Ampt(池の淡水が池の下の帯水層へ浸透する)
  f_gwlateral = 1           ! 非線形 Boussinesq
/

&list_gwflow_greenampt
  gw_ksv_mmh = 360.0        ! 鉛直飽和透水係数 (mm/h) = 1e-4 m/s
  gw_psif = 0.0             ! 一定浸透能
/

&list_gwflow_lateral
  gw_ksh_mmh = 36000.0      ! 側方飽和透水係数 (mm/h) = 0.01 m/s(模式値)
/

!======================================================================
! 淡塩2層設定(地表+地下。交換の淡塩配分あり)
!======================================================================
&list_salt

  f_salt_gw = 1             ! 地下塩水 zone
  f_salt_exch = 1           ! 浸透は底層(塩水)から先、湧出は淡水から先(既定)
  fn_salt_hgs0 = 'hgs0_slope.txt'   ! 初期の地下塩水 = 0.8·sd·sy(淡水なし)
