!======================================================================
! salt: 淡塩2層の模式実験 C1(地表→地下の浸透の淡塩配分 f_salt_exch)
!
!   平坦・閉領域。水深 h0 = 1.0 m のうち底層 0.4 m が塩水(salt_hss0)。
!   帯水層は初期空(土層厚 1.5 m × 比湧水量 0.2 = 容量 0.3 m)。
!   Green-Ampt(K_sv = 3600 mm/h、psi_f = 0 = 一定浸透能)で 0.3 m が
!   浸透して満水になる。鋭利界面では河床に接するのは底層の塩水なので、
!   浸透は塩水から先(f_salt_exch=1。developer.md §47.5)。解析的な終状態:
!     H = 0.7、Hss = 0.4 − 0.3 = 0.1、Hg = 0.3、Hgs = 0.3(全セル一様)
!   (f_salt_exch=0 なら浸透は暗黙に淡水 = Hss 0.4・Hgs 0 のまま)
!   一様なので浅水流・側方流は動かず、塩水体積 Hss + Hgs = 0.4 が厳密に保存。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  dt = 1.0                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 600                  ! 計算終了時刻 (s。300 s で満水)
  dt_disp = 60              ! 画面表示時間刻み (s)
  dt_file = 600             ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_gwflow = '-'
  fn_salt = '-'
  dir_data = 'data_salt'
  dir_result = 'result_exch_in'
  dir_save = 'save_exch_in'
  f_state_save = 1          ! 終状態を倍精度で保存(Check_exch.py が読む)
  f_disp_debug = 1          ! S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_hg = 1              ! Hg / Hgs 場の出力
/

&list_enc
  f_adaptive_runge = 1
/

&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 1.0                  ! 初期水深 (m)
/

&list_geoinfo
  lx = 80.0
  ly = 80.0
  nx = 8
  ny = 8
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値 (m)
  rn0 = 0.03      ! 粗度係数固定値
  f_sdtype = 0    ! 土層厚タイプ (0:固定値)
  sd0 = 1.5       ! 土層厚 (m)
  sy0 = 0.2       ! 比湧水量(容量 = 0.3 m)
/

&list_bound
/

!======================================================================
! 地下水設定(Green-Ampt の鉛直浸透+層1側方流。f_salt_gw=1 の前提)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! Green-Ampt
  f_gwlateral = 1           ! 側方流(一様なので流れない)
/

&list_gwflow_greenampt
  gw_ksv_mmh = 3600.0       ! 鉛直飽和透水係数 (mm/h) = 1e-3 m/s
  gw_psif = 0.0             ! 毛管圧力水頭 0 = 一定浸透能
/

&list_gwflow_lateral
/

!======================================================================
! 淡塩2層設定(地表+地下。交換の淡塩配分あり)
!======================================================================
&list_salt
  f_salt_surf = 1           ! 地表塩水層
  f_salt_gw = 1             ! 地下塩水 zone
  f_salt_exch = 1           ! 浸透は底層(塩水)から先、湧出は淡水から先(既定)
  salt_hss0 = 0.4           ! 初期の地表塩水層厚 (m。h0 の内数)
/
