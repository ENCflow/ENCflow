!======================================================================
! salt: 淡塩2層の模式実験 D2(地表の底層重力流と海岸堤防の仮想壁)
!
!   6 × 4 セル(Δx = 2 m)、四周壁。西の 3 列が陸(z = 0、塩水)、東の 3 列が
!   海側(z = −3 m、fn_seaside、淡水)。海岸堤防 seawall0 = 2 m(絶対標高)。
!   初期水位は全域 1.2 m(f_htype=2)で、水面が水平・天端より低いので浅水流は
!   動かない。陸の水は全層塩水(界面 +1.2 m)、海側は淡水。
!   塩水を動かしうるのは底層重力流だけで、界面が天端より低いので壁を越えない
!   (developer.md §47.8)。解析的な終状態: 陸 Hss = 1.2、海側 Hss = 0、
!   H は全セル不変(2026-10-29 以前は重力流が壁を見ず、塩水が海側へ流れ込んだ)。
!======================================================================
&list_sysparam

  tt = 600                  ! 計算終了時刻 (s)
  dt_disp = 60              ! 画面表示時間刻み (s)
  dt_file = 600             ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_salt = '-'

  f_state_save = 1          ! 終状態を倍精度で保存(Check_exch.py が読む)
  f_disp_debug = 1          ! S 列を全有効桁で表示
  f_check_cfl = 0
/

/

&list_geoinfo

  fn_seaside = 'seaside_wall.txt'  ! 海側マスク(東の 3 列。海域マスクなしの海)
  seawall0 = 2.0            ! 海岸堤防の天端 (m。絶対標高)
  f_seawall_datum = 2       ! 絶対標高
  f_rntype = 0              ! 粗度係数タイプ (0:固定値)

/

/
