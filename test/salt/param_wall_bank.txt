!======================================================================
! salt: 淡塩2層の模式実験 D1(地表の底層重力流と堤防の仮想壁)
!
!   6 × 4 セル(Δx = 2 m)、四周壁。西の 3 列が堤内地(z = 0、塩水)、東の 3 列が
!   河道(z = −3 m、fn_rw、淡水)。堤防 bank0 = 2 m(堤内地の地盤基準 = 天端 +2 m)。

!   動かない。堤内地の水は全層塩水(界面 +1.2 m)、河道は淡水。
!   塩水を動かしうるのは底層重力流だけで、界面が天端より低いので壁を越えない
!   (developer.md §47.8)。解析的な終状態: 堤内地 Hss = 1.2、河道 Hss = 0、
!   H は全セル不変(2026-10-29 以前は重力流が壁を見ず、塩水が河道へ流れ込んだ)。
!======================================================================

  tt = 600                  ! 計算終了時刻 (s)
  dt_disp = 60              ! 画面表示時間刻み (s)
  dt_file = 600             ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_channel = '-'
  fn_salt = '-'
  dir_data = 'data_salt'

  f_state_save = 1          ! 終状態を倍精度で保存(Check_exch.py が読む)
  f_disp_debug = 1          ! S 列を全有効桁で表示
  f_check_cfl = 0
/

  fn_rw = 'rw_wall.txt'     ! 河道マスク(東の 3 列)
  f_rntype = 0              ! 粗度係数タイプ (0:固定値)
  rn0 = 0.03                ! 粗度係数固定値
/

&list_bound

  f_bank_datum = 1          ! 堤内地セルの地盤高基準
