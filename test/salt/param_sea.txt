!======================================================================
! salt: 淡塩2層の模式実験 B(海水浸入の静的平衡 = Ghyben-Herzberg の縮退形)
!
!   平坦な陸(z = 1 m、土層厚 sd = 3 m → 帯水層底 = −2 m)の西縁 1 列を
!   海域(data_salt/sw.txt)とし、潮位を一定 0.5 m に固定(titype=1)。
!   帯水層は初期空(hg = 0)。涵養なし。
!   海側境界から塩水が浸入し、塩水 zone の水頭 Φs = η + εζ が海側
!   Φs_sea = (1+ε)·0.5 と平衡するまで充填される。淡水がないため
!   平衡は ζ = η = 潮位、すなわち
!     hgs = hg → sy·(潮位 − 帯水層底) = 0.2 × 2.5 = 0.5 m(全陸セル一様)
!   確認事項: hg の終状態 ≈ 0.500(水頭平衡)、hgs ≈ 0.493〜0.500、
!   地表は乾いたまま(h = 0)、np=1,2,4 の逐次一致。
!   注意: hg − hgs(見かけの淡水)は加算方式の時間分割誤差(lateral が
!   先に η を動かし塩水 zone が更新後の η を読む逐次分割。dt に比例する
!   1 次誤差で、本ケースは K = 0.01 m/s と極端に速いため可視化される。
!   dt=1.0 で最大 5%、dt=0.25 で 1.5%。developer.md §47 検証記録)。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  dt = 0.25                 ! 時間刻み (s。分割誤差を抑えるため小さめに)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 40000                ! 計算終了時刻 (s)
  dt_disp = 4000            ! 画面表示時間刻み (s)
  dt_file = 20000           ! ファイル出力時間刻み (s)
  dt_recrd = 4000           ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_gwflow = '-'
  fn_tide = '-'
  fn_salt = '-'
  dir_data = 'data_salt'
  dir_result = 'result_sea'
  f_disp_debug = 1          ! S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_e = 1
  f_out_hg = 1              ! Hg / Hgs 場の出力
/

&list_enc
  f_adaptive_runge = 1
/

&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 0.0                  ! 陸は乾いた状態から
/

&list_geoinfo
  lx = 24.0
  ly = 16.0
  nx = 12
  ny = 8
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 1.0        ! 地盤高固定値 (m)
  rn0 = 0.05      ! 粗度係数固定値
  f_masktype = 2  ! 領域マスクを海域マスクから自動生成
  fn_sw = 'sw.txt'
  f_sdtype = 0    ! 土層厚タイプ (0:固定値)
  sd0 = 3.0       ! 土層厚 (m)。帯水層底 = z − sd = −2 m
  sy0 = 0.2       ! 比湧水量
/

&list_bound
/

!======================================================================
! 潮位設定(一様固定 0.5 m)
!======================================================================
&list_tide
  titype = 1           ! 一様固定潮位
  ti0 = 0.5            ! 潮位 (m)
  hsea0 = 1.0          ! 海セルの見かけ水柱厚 (m)
/

!======================================================================
! 地下水設定(層1側方流のみ。鉛直なし)
!======================================================================
&list_gwflow
  f_gwvertical = 0
  f_gwlateral = 1           ! 非線形 Boussinesq(バロトロピック応答)
  f_gwlayer2 = 0
  f_gwconduit = 0
  dt_gwflow = 0             ! 毎ステップ更新
/

&list_gwflow_lateral
  gw_ksh_mmh = 36000.0      ! 側方飽和透水係数 (mm/h) = 0.01 m/s(模式値)
/

!======================================================================
! 淡塩2層設定(地下のみ)
!======================================================================
&list_salt
  f_salt_surf = 0           ! 地表は不使用
  f_salt_gw = 1             ! 地下塩水 zone + 海側境界
  salt_rhof = 1000.0        ! 淡水密度 (kg/m3)
  salt_rhos = 1025.0        ! 塩水密度 (kg/m3)
/
