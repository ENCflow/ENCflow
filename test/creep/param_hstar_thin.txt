!======================================================================
! creep: 深さ依存クリープ(creep_hstar)の薄い土層(developer.md §88.8)
!   param.txt と同じガウス丘に一様な土層 sd0 = 2 cm を置き、減衰厚 H_c = 0.1 m で
!   動かす(利用可能率 1 − e^{−0.2} = 0.18 から始まる)。丘の頂部の土層が剥がれて
!   脚部に溜まり、岩盤は動かない。岩盤面 z − sd の不変・土層の総量の保存・sd >= 0 を
!   Check_hstar.py で検定する。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE, 1:DifWE, 2:KinWE)
  dt = 0.1                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 50                   ! 計算終了時刻 (s)
  dt_disp = 5               ! 画面表示時間刻み (s)
  dt_file = 50              ! ファイル出力時間刻み (s)
  dt_recrd = 1              ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  fn_record = '-'
  dir_result = 'result_hstar_thin'
  f_state_save = 1           ! 終了時の状態を保存(Check_hstar.py が z・sd を読む)
  dir_save = 'save_hstar_thin'
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(既定のまま。水は無いため流れは実質不活性)
!======================================================================
&list_enc
/

!======================================================================
! 初期条件設定(乾燥状態)
!======================================================================
&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 0.0                  ! 初期水深固定値 (m)
/

!======================================================================
! 地理情報設定(dx = lx/nx = 1.0 m。地形はユーザールーチンで作る)
!======================================================================
&list_geoinfo
  lx = 101.0
  ly = 101.0
  nx = 101
  ny = 101
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値(m)。ガウス丘で上書きされる
  rn0 = 0.03      ! 粗度係数固定値
  sd0 = 0.02      ! 土層厚固定値 (m)
  f_user_routine = "gaussian_hill"   ! 中央のガウス丘(A=1m, sigma=10dx)
/

!======================================================================
! 地形変化設定(深さ依存クリープのみ)
!   安定条件: D*dts*(1/dx^2+1/dy^2) <= 1/2 → dts <= 0.25 s
!======================================================================
&list_geomorph
  dt_geomorph = 0.0         ! 0 = 毎ステップ更新(dts = dt = 0.1 s)
  morfac = 1.0              ! 加速係数
  f_creep = 1               ! 斜面クリープ有効
  creep_d = 1.0             ! クリープ拡散係数 (m2/s)
  creep_hstar = 0.1           ! 深さ依存クリープの減衰厚 H_c (m)
/

!======================================================================
! 計測設定(プローブ: 丘の中心 r=0 と r=10, 20(x方向)、r=20(y方向))
!   セル番号指定(pbxytype=0)。丘の中心はセル (51,51)
!======================================================================
&list_record
  pbxytype = 0
  pbxy(1:2,1) = 51, 51
  pbxy(1:2,2) = 61, 51
  pbxy(1:2,3) = 71, 51
  pbxy(1:2,4) = 51, 71
/
