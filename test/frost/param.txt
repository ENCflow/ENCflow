!======================================================================
! frost: 凍土による浸透抑制の模式実験(線形低減の解析検証)
!
!   平坦・閉領域 21×21(dx=dy=2)。初期表面水 h0 = 0.05 m。Green-Ampt
!   (psi_f=0 = 一定浸透能 K_sv=36 mm/h=1e-5 m/s)。気温は一様定数
!   Ta = −8.64 ℃(減率なし)なので凍結指数は毎ステップ
!     ΔFI = 8.64·dt/86400 = 5e-5 °C·day(dt=0.5 s)
!   ずつ線形に増え、fro_fifull = 0.36 で FI/fifull = k/7200(k: ステップ)。
!   低減係数 fac_k = 1 − k/7200 は t=3600 s(k=7200)でちょうど 0
!   (完全凍結 = 不浸透)に達する。表面水は全時間で残る(浸透可能量
!   < h0)ため、浸透の離散和は厳密に
!     S_grnd(3600) = K·dt·Σ_{k=1..7200}(1 − k/7200)
!                  = 1e-5·0.5·(7200 − 7201/2) = 1e-5·0.5·3599.5
!                  = 0.0179975 m
!     S_surf(3600) = 0.05 − 0.0179975 = 0.0320025 m(S_total = 0.05 不変)
!   確認事項: S 列の解析値一致、Ff 場の終値 0(完全凍結)、
!   np=1,2,4 の逐次一致。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 3600                 ! 計算終了時刻 (s)
  dt_disp = 300             ! 画面表示時間刻み (s)
  dt_file = 1800            ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_gwflow = '-'
  fn_meteo = '-'
  dir_data = '.'
  dir_result = 'result'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_hg = 1
/

&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 0.05                 ! 初期水深固定値 (m)
/

&list_geoinfo
  lx = 42.0
  ly = 42.0
  nx = 21
  ny = 21
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値 (m)
  rn0 = 0.05      ! 粗度係数固定値
  f_sdtype = 0    ! 土層厚タイプ (0:固定値)
  sd0 = 1.0       ! 土層厚固定値 (m)
  sy0 = 0.2       ! 比湧水量(有効間隙率)
/

&list_bound
/

!======================================================================
! 気象強制場(一様定数気温。減率なし)
!======================================================================
&list_meteo
  temp0 = -8.64             ! 気温 (℃)。ΔFI = 8.64·dt/86400 = 5e-5 °C·day/step
/

!======================================================================
! 地下水設定(鉛直 Green-Ampt+凍土。側方なし)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! 鉛直モデル(2:Green-Ampt)
  f_gwlateral = 0           ! 側方なし
  f_gwfrost = 1             ! 凍土による浸透抑制
/

&list_gwflow_greenampt
  gw_ksv_mmh = 36.0         ! 1e-5 m/s
  gw_psif = 0.0             ! 一定浸透能に退化
/

&list_gwflow_frost
  fro_fifull = 0.36         ! FI = 0.36 °C·day で完全凍結(t=3600 s)
  fro_fmin = 0.0            ! 完全凍結 = 不浸透
/
