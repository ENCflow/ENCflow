!======================================================================
! saltledger / sea: 海から入った塩水と、海へ出た塩水の台帳
!   陸(24 × 8 セル、dx = 10 m、地盤高 1 m)の西端の 1 列が海。潮位を 0 → 2 m に上げて陸を浸し、
!   1.3 m に下げて水の一部を海へ戻す。陸の水はすべて海から入った塩水なので、塩水層の厚さは水深と
!   同じで、台帳は 地表の体積 = 海からの流入 − 海への流出 − 整合クランプ で閉じる。
!======================================================================
&list_sysparam
  dt = 0.2
  tt = 1800.0
  dt_disp = 300.0
  dt_file = 600.0
  dt_recrd = 60.0
  fn_geoinfo = '-'
  fn_tide = '-'
  fn_salt = '-'
  dir_data = 'data_saltledger'
  dir_result = 'result_sea'
  f_output_mode = 2
  f_disp_debug = 1
/

&list_geoinfo
  nx = 24
  ny = 8
  dx = 10.0
  dy = 10.0
  f_ztype = 0
  z0 = 1.0                  ! 陸の地盤高 (m)
  rn0 = 0.03
  f_masktype = 2            ! 領域マスクを海域マスクから作る
  fn_sw = 'sw_sea.txt'      ! 海域マスク(西端の 1 列が海)
/

&list_tide
  titype = 2                ! 一様な潮位の時系列
  hsea0 = 1.0               ! 海セルの見かけの水柱の厚さ (m)
  dt_tiupdate = 0.1         ! 潮位の更新間隔 (min)
  tival(1:2,1) =  0, 0.0    ! (min, m)
  tival(1:2,2) =  5, 2.0
  tival(1:2,3) = 15, 2.0
  tival(1:2,4) = 20, 1.3
  tival(1:2,5) = 30, 1.3
/

&list_salt
  f_salt_surf = 1           ! 地表の塩水層
  f_salt_gw = 0
/
