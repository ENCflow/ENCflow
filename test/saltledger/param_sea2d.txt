!======================================================================
! saltledger / sea2d: 海から入った塩水の台帳(地盤高が x・y の両方で変わる陸)
!   sea と同じ潮位で、地盤高 0.8〜1.2 m の凹凸のある陸(24 × 16 セル、dx = 10 m)を浸す。流れの向きが
!   場所で変わり、1 つのセルへ複数のエッジから塩水が入る。陸の水はすべて海から入った塩水なので、
!   塩水層の厚さは水深と同じままで、整合クランプで消える量は 0 になる。MPI では、帯の境をまたぐ
!   底層重力流の受け側の上限が、ランク数によらず同じ値になることを確かめる(Run_MPI.sh)。
!======================================================================
&list_sysparam
  dt = 0.2
  tt = 1800.0
  dt_disp = 300.0
  dt_file = 600.0
  dt_recrd = 60.0
  fn_geoinfo = '-'
  fn_tide = '-'
  fn_salt = '-'
  dir_data = 'data_saltledger'
  dir_result = 'result_sea2d'
  f_output_mode = 2
  f_disp_debug = 1
/

&list_geoinfo
  nx = 24
  ny = 16
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z_sea2d.txt'      ! 陸の地盤高 0.8〜1.2 m の凹凸 (m)
  rn0 = 0.03
  f_masktype = 2            ! 領域マスクを海域マスクから作る
  fn_sw = 'sw_sea2d.txt'    ! 海域マスク(西端の 1 列が海)
/

&list_tide
  titype = 2                ! 一様な潮位の時系列
  hsea0 = 1.0               ! 海セルの見かけの水柱の厚さ (m)
  dt_tiupdate = 0.1         ! 潮位の更新間隔 (min)
  tival(1:2,1) =  0, 0.0    ! (min, m)
  tival(1:2,2) =  5, 2.0
  tival(1:2,3) = 15, 2.0
  tival(1:2,4) = 20, 1.3
  tival(1:2,5) = 30, 1.3
/

&list_salt
  f_salt_surf = 1           ! 地表の塩水層
  f_salt_gw = 0
/
