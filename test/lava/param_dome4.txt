!======================================================================
! lava 構成4': 構成4 と同じ入力を、流動の 4 近傍(lv_diagratio = 0。2026-11-05 より前の既定。
!   developer.md §99)で計算する対照(検定は構成4 と同じ Check_lava.py dome。以下は構成4 の説明)
! lava 構成4: 格子の向きによらない広がり(塑性ドームの静止形。developer.md §51.4)
!   平坦床(61m x 61m)の中央1セルから 900 s 定率噴出(計 1798 m3)し、
!   降伏応力で止まった静止形を、平坦床上の Bingham 流体の静止形
!   (塑性ドーム。Blake 1990)と比べる。固化なし。
!   解析解: 降伏の釣り合い ρ g h |∇h| = τ_y から h(r)^2 = a (R − r)、
!     a = 2 τ_y/(ρ g) = 0.7849 m、体積 V = (8π/15) √a R^{5/2} から
!     R = 17.11 m、中心厚 H = √(a R) = 3.665 m。
!   検定(Check_lava.py dome):
!     (1) 台帳 Σhl = 注入量(機械精度。固化なし)
!     (2) 等方性: 格子軸の方向(±22.5°)と斜めの方向(±22.5°)の扇形の
!         面積等価半径の比が 1 に 5% 以内(降伏判定を法線勾配だけで行う
!         修正前の方式では 0.86 = 格子軸の方向に伸びたひし形)
!     (3) 解析解: 中心厚が H に、面積等価半径が R に、それぞれ相対 5% 以内。
!         r < R の断面の RMS 誤差が H の 5% 以内
!   粘度は静止形に効かない(時間尺度だけを決める)ので、検定時間内に
!   止まる値を選んでいる(3000 s と 6000 s で静止形がほぼ同じ)。
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 1.0
  t0 = 0
  tt = 6000
  dt_disp = 1000
  dt_file = 3000
  dt_recrd = 6000
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_lavaflow = '-'
  dir_result = 'result_dome4'
  dir_save = 'save_dome4'
  f_state_save = 1
  f_check_cfl = 0
/

&list_initial

  h0 = 0.0                  ! 水なし(溶岩のみの検定)
/

&list_geoinfo
  lx = 61.0
  ly = 61.0
  nx = 61
  ny = 61
  f_ztype = 0
  z0 = 0.0                  ! 平坦床
  f_rntype = 0
  rn0 = 0.03
/

&list_lavaflow
  lv_rho = 2600.0
  lv_visc = 1.0e6           ! η (Pa s)。静止形には効かない(止まるまでの時間尺度)
  lv_tauy = 1.0e4           ! 降伏応力 (Pa)。a = 2τ_y/(ρg) = 0.7849 m
  lv_wsol = 0.0             ! 固化なし(静止形を解析解と比べる)
  lv_diagratio = 0.0        ! 4 近傍(対照。既定は 8 近傍の 9 点 d = 1/3。§99)
  lv_cell(1:2,1,1) = 31, 31 ! 中央1セルの噴火口
  lv_val(1:2,1,1) = 0, 2.0  ! (時刻(min), 噴出率(m3/s))
  lv_val(1:2,2,1) = 15, 2.0
  lv_val(1:2,3,1) = 15, 0.0 ! 重複時刻 = t=900s ちょうどで階段状に停止
/
