!======================================================================
! lava 構成8: 熱の移流(温度モード f_lvtemp=1、冷却なし。lava_plan.md §8.10)
!   平坦な床(41 m x 21 m)の 2 つの噴火口 (11,11)・(31,11) から、噴出温度
!   1150 °C と 1050 °C の溶岩を 0.5 m3/s ずつ 600 s 噴出させる(降伏応力 0 =
!   Newton 流体。温度感度 β = 0 なので流れは等温と同じ)。冷却なし
!   (ε = 0、h_c = 0)。2 つの溶岩は中央で接して混ざる。
!   検定(Check_lava.py adv):
!     (1) 溶岩の台帳 Σhl·A = V_in(機械精度)
!     (2) 熱の保存 Σhl·Tl·A = Σ_k Q_k T_k t(相対 1e-9)と、画面出力の熱の
!         台帳の残差(噴出の相対 1e-12 以内)
!     (3) 範囲: 溶岩のある全セルで 1050 ≤ Tl ≤ 1150(丸めの分の余裕つき)、
!         混ざった帯(両方の噴出温度から 1 K 以上離れたセル)がある
!   param_adv0.txt は同じ入力で f_lvtemp = 0 にした等温の計算(温度の
!   パラメータは書いたまま = 読まれない)。Run.sh が lavaflow.dat・state.dat の
!   バイト一致を確かめる(β = 0・冷却なしなら流れは等温と同じ)
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.5
  t0 = 0
  tt = 900
  dt_disp = 300
  dt_file = 900
  dt_recrd = 900
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_lavaflow = '-'
  dir_result = 'result_adv'
  dir_save = 'save_adv'
  f_state_save = 1
  f_check_cfl = 0
/

&list_initial
  f_htype = 0
  h0 = 0.0                  ! 水なし(溶岩のみの検定)
/

&list_geoinfo
  lx = 41.0
  ly = 21.0
  nx = 41
  ny = 21
  f_ztype = 0
  z0 = 0.0                  ! 平坦床
  f_rntype = 0
  rn0 = 0.03
/

&list_lavaflow
  lv_rho = 2600.0
  lv_visc = 2.0e4           ! η (Pa s)(構成1 と同じ)
  lv_tauy = 0.0             ! Newton 流体
  lv_cell(1:2,1,1) = 11, 11 ! 噴火口1(西)
  lv_val(1:2,1,1) = 0, 0.5  ! (時刻(min), 噴出率(m3/s))
  lv_val(1:2,2,1) = 10, 0.5
  lv_val(1:2,3,1) = 10, 0.0 ! 重複時刻 = t=600s ちょうどで階段状に停止
  lv_cell(1:2,1,2) = 31, 11 ! 噴火口2(東)
  lv_val(1:2,1,2) = 0, 0.5
  lv_val(1:2,2,2) = 10, 0.5
  lv_val(1:2,3,2) = 10, 0.0
  f_lvtemp = 1              ! 温度を計算する
  lv_temp0(1) = 1150.0      ! 噴火口1の噴出温度 (°C)
  lv_temp0(2) = 1050.0      ! 噴火口2の噴出温度 (°C)
  lv_tref = 1100.0          ! 基準温度 (°C)
  lv_bvisc = 0.0            ! 温度感度 0 = 流れは等温と同じ
  lv_btau = 0.0
  lv_tsol = 700.0           ! 固化温度 (°C)(冷却なしなので固まらない)
  lv_emis = 0.0             ! 冷却なし
  lv_hconv = 0.0
  lv_cp = 1150.0            ! 比熱 (J/kg/K)
  lv_tair = 20.0            ! 気温 (°C)
/
