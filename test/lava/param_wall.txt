!======================================================================
! lava 構成5: 溶岩より高い壁の前にたまった溶岩の固化(developer.md §51.5)
!   平坦な床(40m x 8m)の東端に高さ 10 m の乾いた段(i >= 31。zwall.txt)を
!   置き、壁から 3 m の列(i=28, j=1..8)に計 0.2 m3/s を 300 s 与えて止める。
!   溶岩は壁の前にたまり、西へ広がって止まる。停止判定は、溶岩が動けない
!   エッジ(溶岩面の高い側 = 送り手に溶岩がない流れ込みのエッジ。壁へ向かう
!   エッジがこれ)を数えないので、壁に接する列も含めて全量が固化する。
!   修正前の判定は壁へ向かう上り勾配のエッジを大きな流速とみなし、壁に
!   接する列(i=30)が溶けたまま残った(噴出量の約 14%)。
!   検定(Check_lava.py wall):
!     (1) 台帳 Σhl + Σ(z−z0) = 注入量(機械精度。固化は z への内部移転)
!     (2) 全量の固化: Σdz / 注入量 > 0.999
!     (3) 壁に接する列: 全行で hl = 0 かつ dz が溶岩の最大厚の半分以上
!         (壁の前にたまった溶岩が固化して地形になった)
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 0.5
  t0 = 0

  dt_disp = 800
  dt_file = 4800

  fn_geoinfo = '-'
  fn_initial = '-'
  fn_lavaflow = '-'
  dir_result = 'result_wall'
  dir_save = 'save_wall'
  f_state_save = 1
  f_check_cfl = 0
/

&list_initial

  h0 = 0.0                  ! 水なし(溶岩のみの検定)
/

&list_geoinfo
  lx = 40.0
  ly = 8.0
  nx = 40
  ny = 8
  f_ztype = 1
  fn_z = 'zwall.txt'        ! 平坦な床と高さ 10 m の段(make_init.py wall)
  f_rntype = 0
  rn0 = 0.03
/

&list_lavaflow
  lv_rho = 2600.0
  lv_visc = 2.0e3           ! η (Pa s)(構成2 と同じ)
  lv_tauy = 2000.0          ! 降伏応力 (Pa)
  lv_wsol = 0.005           ! 固化レート (m/s)
  lv_vsol = 5.0e-4          ! 停止判定の速度閾値 (m/s)
  lv_cell(1:2,1,1) = 28, 1  ! 壁から 3 m の列噴火口
  lv_cell(1:2,2,1) = 28, 2
  lv_cell(1:2,3,1) = 28, 3
  lv_cell(1:2,4,1) = 28, 4
  lv_cell(1:2,5,1) = 28, 5
  lv_cell(1:2,6,1) = 28, 6
  lv_cell(1:2,7,1) = 28, 7
  lv_cell(1:2,8,1) = 28, 8
  lv_val(1:2,1,1) = 0, 0.2  ! (時刻(min), 総噴出率(m3/s))
  lv_val(1:2,2,1) = 5, 0.2
  lv_val(1:2,3,1) = 5, 0.0  ! 重複時刻 = t=300s ちょうどで階段状に停止
/
