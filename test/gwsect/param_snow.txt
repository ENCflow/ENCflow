!======================================================================
! gwsect: 融雪の水が σ のセルに加わるときの体積(§26.2)
!
!   param.txt と同じ谷(σ の河道 + 河道幅 2 m)で、地表は不浸透(地下水なし)。
!   全セルに初期積雪 20 mm(水当量)を置き、気温 10 ℃ 一定・度日係数
!   20 mm/℃/day(1 時間に 8.3 mm)で 2.4 時間かけて融かす。融雪水は斜面を
!   流れて谷底の σ の河道へ集まり、河道のセルの上の雪は河道へ直接融け込む。
!   σ のセルでは融けた水を体積(矩形換算水深)で足す(足さないと σ(h) 倍しか
!   地表の体積が増えない)。閉領域なので総水量(地表 + 積雪)は一定。
!======================================================================
&list_sysparam
  dt = 0.5                  ! 時間刻み (s)
  tt_c = "6h"               ! 計算終了時刻
  dt_disp_c = "30m"         ! 画面表示の間隔
  dt_file_c = "1h"          ! 分布ファイル出力の間隔
  fn_geoinfo = '-'
  fn_channel = '-'
  fn_snow = '-'
  fn_meteo = '-'
  dir_data = 'data_gwsect'
  dir_result = 'result_snow'
  f_check_cfl = 0
  f_disp_debug = 1          ! S 列を全有効桁で表示
  f_output_mode = 2         ! 分布は bil
/

&list_geoinfo
  nx = 12
  ny = 8
  dx = 10.0
  dy = 10.0
  f_ztype = 1               ! 地盤高ファイル
  fn_z = 'z.txt'
  rn0 = 0.05
  fn_rw = 'rw.txt'          ! 河道マスク(列 10)
  depth_rw = 0.5            ! 河道の掘り込み深さ = σ の遷移深さ D (m)
/

&list_channel
  fn_width = 'width.txt'    ! サブグリッド河道幅 2 m(wfrac = 0.2)
  p_sect_m = 0.5            ! 断面形 σ(h) = (h/D)^0.5(放物線的)
/

&list_snow
  snow_t_snow = 0.0         ! 全て雪になる気温 (℃)
  snow_t_rain = 2.0         ! 全て雨になる気温 (℃)
  snow_t_melt = 0.0         ! 融雪閾値 (℃)
  snow_ddf = 20.0           ! 度日係数 (mm/℃/day)
  snow_swe0 = 20.0          ! 一様初期積雪水量 (mm)
/

&list_meteo
  tempval(:,1) = 0.0, 10.0  ! 一様時系列 (t=0 からの経過日, ℃)。一定
  tempval(:,2) = 1.0, 10.0
/
