!======================================================================
! dissol: 石灰岩の溶食(f_dissol)の解析検証・回帰テスト(developer.md §86、
!         docs/karst_plan.md §3)。構成 A = 面的溶食(浸透量に比例)
!   平坦地形(21×21、Δx = 2 m)に一様湛水 h0 = 0.1 m。バケツ浸透
!   36 mm/h(= 1e-5 m/s。容量 0.05 m)で毎ステップ 5e-7 m が浸透し、
!   100 s で I = 1e-3 m(容量・湛水とも余る)。溶食は dt_geomorph = 5 s
!   (100 ステップ)ごとに、その間隔に積算された浸透量 5e-5 m を消費する:
!     w = k_d × F × morfac = 4e-5 × 5e-5 × 1e4 = 2e-5 m / 更新、20 回で 4e-4 m
!     k_d = c_eq/ρ_r = (100 g/m3)/(2500 kg/m3) = 4e-5 m/m
!   残渣率 0.25 なので地表は (1−0.25) 倍だけ下がり、土層は 0.25 倍だけ増える:
!     Δz = −3e-4 m、Δsd = +1e-4 m(z0 = 0、sd0 = 1 → z = −3e-4、sd = 1.0001)
!   全セル同一(流れなし)。Check_analytic.py が Z9998 / Sd9998 を検定する。
!======================================================================

  dt_disp = 10              ! 画面表示時間刻み (s)
  dt_file = 50              ! ファイル出力時間刻み (s)

  fn_geoinfo = '-'

  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_output_mode = 3         ! text + bil(解析検証は bil の全精度を読む)
