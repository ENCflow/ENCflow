!     F = 浸透 I + fsurf × fb × (雨 P − I) = 0.01 + 0.3 × 0.99 × 0.02 = 0.01594 m
!   溶食は dt_geomorph = 60 s(60 ステップ)ごとに積算を消費、morfac = 1000:
!     Δz = −k_d × F × morfac = −4e-5 × 0.01594 × 1000 = −6.376e-4 m(全セル同一)
!   Check_analytic.py が Z9998 を検定する。
!======================================================================

  dt_file = 3600            ! ファイル出力時間刻み (s)
  dt_recrd = 3600           ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_gwflow = '-'

  dir_data = 'data_dissol'

  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_output_mode = 3         ! text + bil(解析検証は bil の全精度を読む)
  f_out_z = 1               ! 地盤高の出力(解析検証は Z9998 / Sd9998 を読む)
  f_out_hg = 1

&list_geoinfo
