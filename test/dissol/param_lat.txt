!======================================================================
! dissol: 構成 D = 側方溶食(dissol_lat。karst_plan.md §2.5)
!   平坦 0 m の中央 3×3 だけ −1 m の窪みに水位 −0.5 m の湛水を置く
!   (f_htype = 2)。周囲は乾いているので流れはなく、バケツ浸透 36 mm/h が
!   窪みの 9 セルだけに同じ記録 F = 5e-5 m / 更新(5 s)を与える。
!   1 更新の溶解能 W = k_d × F × morfac = 4e-5 × 5e-5 × 1e4 = 2e-5 m、
!   20 更新で 4e-4 m。p = dissol_lat = 0.4 を壁(窪みを囲む 16 セル。
!   床 0 m > 水面 −0.5 m)へ接触幅比 L_k(軸 1、斜め 2/(2+√2))で配り、
!   残り 0.6 を床へ。中央の窪みセルは壁がないので床に全量:
!     角の窪みセル: 壁 = 軸 2 + 斜め 3、辺の窪みセル: 壁 = 軸 1 + 斜め 2
!   Check_analytic.py が配分式を独立に実装して Z9998 と比較し、
!   総溶解能 9 W = 床 + 壁 の和(保存)を見る。水位は浸透で 1e-3 m、床は
!   2.4e-4 m 下がるが、壁判定(0 > −0.5)は変わらない。
!======================================================================

  dt = 0.05                 ! 時間刻み (s)

  dt_disp = 10              ! 画面表示時間刻み (s)

  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'

  dir_result = 'result_lat'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_output_mode = 3         ! text + bil(解析検証は bil の全精度を読む)
  f_out_z = 1               ! 地盤高の出力(解析検証は Z9998 / Sd9998 を読む)
  f_out_hg = 1
/

  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  rn0 = 0.05      ! 粗度係数固定値

&list_bound
