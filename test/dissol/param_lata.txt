!======================================================================
! dissol: 構成 E = 側方溶食の自動配分(dissol_lat = −1。濡れ面の面積比)
!   構成 D と同じ窪み(中央 3×3 が −1 m、水位 −0.5 m)に、地下水の代わりに
!   一様な微小降雨 0.36 mm/h(= 1e-7 m/s)を降雨源(dissol_src = 2)として
!   与え、全セルに同じ記録 F = 1e-5 m(100 s)を作る。morfac = 1e6 で
!   W = 4e-4 m(構成 D と同じ)。壁の分け前はセルごとに
!     p_w = h ΣL / (h ΣL + Δx²)、ΣL = 壁の接触幅の和(軸 Δx、斜め Δx·2/(2+√2))
!   角の窪みセル: ΣL = (2 + 3 × 0.5858) × 2 m → p = 0.5 × 7.515 /(0.5 × 7.515 + 4)
!   辺の窪みセル: ΣL = (1 + 2 × 0.5858) × 2 m、中央: 壁なし(床に全量)。
!   乾いたセルも降雨で W だけ床が下がる。湛水深は降雨で 1e-5 m 増えるので
!   p は 2e-5 の相対変化をもつ(Check は相対 1e-4 で比較)。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 0.05                 ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 100                  ! 計算終了時刻 (s)
  dt_disp = 10              ! 画面表示時間刻み (s)
  dt_file = 50              ! ファイル出力時間刻み (s)
  dt_recrd = 10             ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_geomorph = '-'
  dir_data = 'data_dissol'
  dir_result = 'result_lata'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_output_mode = 3         ! text + bil(解析検証は bil の全精度を読む)
  f_out_z = 1               ! 地盤高の出力(解析検証は Z9998 / Sd9998 を読む)
  f_out_hg = 1
/

&list_enc
  f_adaptive_runge = 1
/

&list_initial
  f_htype = 2               ! 初期水位タイプ (2: 水位固定値)
  e0 = -0.5                 ! 初期水位 (m)。窪み(z = −1)の 9 セルだけ 0.5 m 湛水
/

&list_geoinfo
  lx = 42.0
  ly = 42.0
  nx = 21
  ny = 21
  f_ztype = 1     ! 地盤高タイプ (1:ファイル)
  fn_z = 'zpit.txt'  ! 平坦 0 m、中央 3×3 だけ −1 m の窪み
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  rn0 = 0.05      ! 粗度係数固定値
/

&list_bound
/

!======================================================================
! 降雨設定(一様 0.36 mm/h = 1e-7 m/s。100 s で 1e-5 m。全セルに同じ記録)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 0.36
  prval(1:2,2) = 9999, 0.36
/

!======================================================================
! 地形変化設定(溶食のみ)
!======================================================================
&list_geomorph
  dt_geomorph = 5.0         ! 更新間隔 (s) = 100 ステップ(積算の検証)
  morfac = 1.0e6            ! 加速係数(W = 4e-5 × 1e-5 × 1e6 = 4e-4 m)
  f_dissol = 1              ! 石灰岩の溶食
  dissol_ceq = 100.0        ! 有効溶解能 (mg/L CaCO3)
  dissol_rho = 2500.0       ! 岩の密度 (kg/m3)
  dissol_src = 2            ! 地上降雨量を水量にする(全セル同じ F = 1e-5 m)
  dissol_fres = 0.0         ! 残渣なし(sd を使わない)
  dissol_lat = -1.0         ! 側方溶食: 濡れ面の面積比で自動配分
/
