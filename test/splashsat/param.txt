!======================================================================
! splashsat: 乾式斜面侵食の凹地形増幅の飽和(spl_ksat)の解析検証・回帰テスト
!         (developer.md §48.6)
!   x 方向に一様な放物線の谷 z = a y'²(a = 0.05、y' = y − ly/2。ユーザールーチン
!   parabolic_trough。曲率 κ = 2a = 0.1 1/m が全セルで一定(2 次式の 5 点差分は
!   厳密))に一様降雨 30 mm/h、Green-Ampt 72 mm/h(全時間 h = 0)。
!   リル項を切り(spl_kt = 0)、雨滴項だけに増幅 spl_ca = 10 m を掛ける:
!     E = P kr (1 + ca κ̃)、κ̃ = κ / (1 + κ/κ_s)
!   κ_s = 0.1 → κ̃ = 0.05、Φ_A = 1.5(飽和なしなら Φ_A = 2.0)。
!   解析形 Δz = P kr Φ_A T_geo = 8.333e-6 × 0.004 × 1.5 × 18000 = 0.9e-3 m
!   (飽和なしなら 1.2e-3 m。Sd 出力の印字桁 1e-4 m の整数倍)。
!   境界行(j = 1, ny)は鏡像で κ が半分になるため解析対象は内部行。
!   Check_analytic.py が Sd0000 − Sd9998 を照合する(T_geo = 3600 × 5 = 18000 s)。
!======================================================================
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 1.0                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 3600                 ! 計算終了時刻 (s)
  dt_disp = 600             ! 画面表示時間刻み (s)
  dt_file = 3600            ! ファイル出力時間刻み (s)
  dt_recrd = 3600           ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_precip = '-'
  fn_gwflow = '-'
  fn_geomorph = '-'
  fn_enc = '-'
  dir_result = 'result'
  f_out_z = 1               ! 地盤高の出力(解析検証は Z9998 を読む)
  f_check_cfl = 0
/

&list_enc
/

!======================================================================
! 地理情報設定(dx = dy = 1 m。地形はユーザールーチン parabolic_trough)
!======================================================================
&list_geoinfo
  lx = 20.0
  ly = 40.0
  nx = 20
  ny = 40
  f_ztype = 0     ! 地盤高タイプ(0:固定値。ユーザールーチンが上書き)
  f_rntype = 0    ! 粗度係数タイプ(0:固定値)
  z0 = 0.0
  rn0 = 0.03
  sd0 = 5.0       ! 土層厚 (m)。総侵食深(~1e-3 m)より十分厚い
  sy0 = 0.2       ! 比湧水量(浸透水の貯留 0.03 m < sd0*sy0 = 1 m)
  f_user_routine = "parabolic_trough"   ! z = 0.05 (y − ly/2)²(κ = 0.1 1/m 一定)
/

!======================================================================
! 降雨設定(一様 30 mm/h)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 30
  prval(1:2,2) = 9999, 30
/

!======================================================================
! 地下水設定(Green-Ampt。ksv=72 mm/h > 降雨強度 → 地表流なし)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! 鉛直モデル(2:Green-Ampt)
  f_gwlateral = 0           ! 側方なし
/
&list_gwflow_greenampt
  gw_ksv_mmh = 72.0         ! 2e-5 m/s
/

!======================================================================
! 地形変化設定(乾式斜面侵食の雨滴項+増幅の飽和=解析検証)
!======================================================================
&list_geomorph
  dt_geomorph = 10.0        ! 更新間隔 (s)
  morfac = 5.0              ! 加速係数(T_geo = 18000 s)
  f_splash = 1              ! 乾式斜面侵食有効
  spl_kr = 0.004            ! 雨滴侵食係数
  spl_kt = 0.0              ! リル項なし(kr だけ指定 → kt = 0)
  spl_h = 1.0               ! 勾配べき指数(不使用)
  spl_ca = 10.0             ! 雨滴項の凹地形増幅長 (m)(κ̃ = 0.05 → Φ_A = 1.5)
  spl_cb = 0.0              ! リル項の増幅(不使用)
  spl_ksat = 0.1            ! 増幅の飽和曲率 κ_s (1/m)(κ = 0.1 → κ̃ = 0.05)
  fluv_porosity = 0.0       ! 空隙率(poroi = 1 で解析式が単純)
/
