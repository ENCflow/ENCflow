!======================================================================
! diffusion: 運動方程式の拡散項(定数渦動粘性)のガウス形ジェット・ベンチマーク
!   平坦・無摩擦・閉じた水路(200 m × 101 m、Δx = Δy = 1 m、h0 = 1 m)に
!   x 方向に一様なガウス形ジェット u(y) = U0 exp(−(y−yc)²/(2σ²))
!   (U0 = 0.05 m/s、σ = 10 m。user_initial `jet_gaussian`)を置き、
!   定数 ν = 4 m²/s(f_diffusion_term = 1)で 12.5 s 拡散させる(σ² + 2νt が
!   2 倍 = 中心の流速が 1/√2 になる時間)。解析解
!     u(y, t) = U0 σ/√(σ² + 2νt) exp(−(y−yc)²/(2(σ² + 2νt)))
!   と中央列(ix = 100)のプローブ u(t) を比較する(Check_analytic.py)。
!   粗度 0・流れは x 方向に一様なので、x の壁から出る波(√(gh) ≈ 3.1 m/s。
!   12.5 s で 39 m)が中央列に届く前は、運動方程式は ∂u/∂t = ∇·(ν∇u) だけ。
!   壁のセルの水深は U0 h t/Δx = 0.63 m 減るが乾かない。
!   陽解法の安定上界 ν_max = Δx²/(2.83 dt) = 7.1 m²/s > ν。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE, 1:DifWE, 2:KinWE)
  dt = 0.05                 ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 12.5                 ! 計算終了時刻 (s)
  dt_disp = 2.5             ! 画面表示時間刻み (s)
  dt_file = 6.25            ! ファイル出力時間刻み (s)
  dt_recrd = 0.25           ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_record = '-'
  dir_result = 'result'
  f_check_cfl = 0
  f_out_u = 1               ! x 方向流速の分布(Fig.py の断面図に使う)
/

!======================================================================
! ENC計算条件(拡散項: 定数粘性 ν = 4 m²/s。他は既定)
!======================================================================
&list_enc
  f_diffusion_term = 1      ! 拡散項(1: 定数粘性)
  p_diffusion_nu = 4.0      ! 動粘性係数 ν (m²/s)
  f_adaptive_runge = 0      ! 適応的ルンゲクッタ無効。初回ステップは前ステップ流量 mn = 0 のため
                            ! 大半のエッジが RK で再計算され、段内の仮水深更新が偽の水位差を
                            ! 作って初回ステップで流速が落ちる(dt² に比例。dt = 0.05 s で 1%、
                            ! 0.1 s で 3.5%。README「既知の問題」)。拡散項の検定には
                            ! 関係ない要因なので外す
/

!======================================================================
! 初期条件設定(静水深 1 m にガウス形ジェットを重ねる)
!======================================================================
&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 1.0                  ! 初期水深固定値 (m)
  f_user_routine = "jet_gaussian"   ! u(y) = U0 exp(−(y−yc)²/(2σ²))、v = 0
/

!======================================================================
! 地理情報設定(平坦・無摩擦。dx = dy = 1 m。四周は壁)
!======================================================================
&list_geoinfo
  lx = 200.0
  ly = 101.0
  nx = 200
  ny = 101
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値(m)
  rn0 = 0.0       ! 粗度係数固定値(無摩擦)
/

!======================================================================
! 計測設定(プローブ: 中央列 ix = 100 の y = yc(iy = 51)、r = 10, 20, 30 m)
!   セル番号指定(pbxytype=0)。ジェットの中心はセル行 51
!======================================================================
&list_record
  pbxytype = 0
  pbxy(1:2,1) = 100, 51
  pbxy(1:2,2) = 100, 61
  pbxy(1:2,3) = 100, 71
  pbxy(1:2,4) = 100, 81
/
