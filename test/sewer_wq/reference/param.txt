!======================================================================
! sewer_wq: 回帰テスト(下水噴出の衛生リスク = 水質×管路連続体層)
!
!   examples/sewer_hybrid のラン D(幹線1本+枝管網・セル別 sy)を
!   短縮した自己完結ケース。降雨 100 mm/h x 30 min で管内貯留
!   (~16 mm)を超えさせてサーチャージ・噴出を起こし、噴出・吐口水を
!   「下水 10^6 CFU/100mL」(単位読み替え g = 10^6 CFU →
!   wq_gwc_conc = 1e4)として大腸菌の拡散・死滅を追う(§30 の
!   供給側固定濃度近似)。dt = 0.1 s は管路層の陽解法上限を超え、
!   サブサイクリング(§46.5 (4))が自動で入る(この経路も検査対象)。
!
!   検証対象: Log.txt(水収支)+ wq.csv(物質収支。in_gwc/to_gwc の
!   連携列を含む)。閉境界+降雨のみなので質量収支は機械精度で閉合する。
!   実務手順の正本は users_guide/wq.md「下水噴出の衛生リスク評価」。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 0.1                  ! 時間刻み (s)(管路層はサブサイクルで追随)
  t0 = 0
  tt = 3600                 ! 1 時間(降雨 30 min + 排水 30 min)
  dt_disp = 300             ! 画面表示間隔 (s)
  dt_file = 1800            ! 分布ファイル出力間隔 (s)
  dt_recrd = 60             ! 計測出力間隔 (s)
  fn_geoinfo = '-'
  fn_enc = '-'
  fn_gwflow = '-'
  fn_precip = '-'
  fn_record = '-'
  fn_wq = '-'
  dir_data = 'data_sewer'
  dir_result = 'result'
  f_disp_debug = 1          ! S 列を全有効桁で表示(水収支の照合用)
  f_out_e = 1
  f_out_hg = 1              ! Hgc 場の出力
/

&list_enc
  f_adaptive_runge = 1
/

&list_geoinfo
  nx = 40
  ny = 21
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z.txt'            ! 勾配 0.005 の東向き斜面
  f_rntype = 0
  rn0 = 0.03                ! 市街地相当の粗度
/

&list_bound
/                           ! 全周閉境界(質量保存の検証)

&list_precip
  prtype = 1                ! 一様時系列
  dt_prupdate = 1
  prval(:,1) =  0, 100      ! 100 mm/h
  prval(:,2) = 30, 100
  prval(:,3) = 31,   0      ! 30 分で終了
/

!======================================================================
! 地下水設定: 管路連続体層のみ(都市プリセット: 不浸透+下水道)
!======================================================================
&list_gwflow
  f_gwvertical = 0          ! 不浸透(都市域)
  f_gwlateral = 0
  f_gwlayer2 = 0
  f_gwconduit = 1
  dt_gwflow = 0             ! 毎ステップ更新
/

&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! sqrt 乱流則
  fn_gwc_cnd = 'cnd_T.txt'  ! 通水能密度(幹線 2.40 / 枝管 0.208 m2/s)
  fn_gwc_cap = 'cap_T.txt'  ! 貯留容量(幹線セル 0.091 / 枝管 0.0126 m)
  fn_gwc_bot = 'bot_T.txt'  ! 管底(幹線 z-5 / 枝管 z-2)
  fn_gwc_sy = 'sy_T.txt'         ! セル別 sy(幹線 0.0911 / 枝管 0.0314)
  fn_gwc_slot_sy = 'slot_T.txt'  ! セル別 slot(= sy/5。硬い圧力応答)
  gwc_sat0 = 0.0            ! 初期は空管
  fn_gwc_inlet = 'inlet.txt' ! 枡 1 個/セル、幹線末端は 10 個(吐口代用)
  gwc_cw = 2.66
  gwc_co = 0.15
  gwc_leak_layer = 0        ! 層間交換なし(都市プリセット)
  gwc_eps_h = 0.1           ! sqrt 則の線形化幅 (m)
/

!======================================================================
! 計測: 幹線末端(吐口)・幹線中間・幹線から遠い枝管域
!======================================================================
&list_record
  pbxytype = 0
  pbxy(:,1) = 38, 11        ! 吐口セル
  pbxy(:,2) = 20, 11        ! 幹線中間
  pbxy(:,3) = 20, 5         ! 幹線から遠い枝管域
/

!======================================================================
! 水質: 下水噴出の大腸菌(管路連携・供給側固定濃度近似。§30)
!   単位の読み替え: cq の「g」を「10^6 CFU」と読む。下水の大腸菌
!   10^6 CFU/100mL = 1e10 CFU/m3 = 1e4 単位/m3 → wq_gwc_conc = 1e4
!   減衰: T90 = 1 日 → k = ln(10)/T90 = 2.303 /day
!======================================================================
&list_wq
  wq_gwc_conc = 1.0e4       ! 噴出・吐口水の固定濃度(名目 mg/L = 単位/m3)
  f_wq_gwc_in = 1           ! 枡流入は濃度同伴で管路台帳へ(既定)
  wq_k20 = 2.303            ! 一次減衰 (1/day) = ln(10)/T90(T90=1日)
/
