!======================================================================
! stageside: 湧き出しの列から流れ出す水と、側方 Neumann の辺(developer.md §113.9 (9))
!   長さ 600 m・幅 40 m(120 × 8 セル、Δx = 5 m)の水路。河床は西端 −3 m から東端 −1.5 m へ上がる。
!   東端の 1 列(8 セル)に、湧き出しで 48 m3/s(単位幅 1.2 m2/s。10 分で立ち上げて一定)を与える。東の辺は壁、
!   西の辺は長波放射(基準水位 0)、南北の辺は側方 Neumann(岸沿いに一様)。静水(水位 0)から始めると、
!   湧き出しの列から西へ流れて、定常になる(西端の流速 0.37 m/s、東端のすぐ内側 0.69 m/s。test/wfdoppler の
!   逆流と同じ流れ)。
!   移流項の規則は、古い値の組み合わせ(f_advection_donor = 3・f_advection_bend = 1・f_advection_lateral = 1・
!   f_dry_head_cap = 3)。
!   2 つの構成(param_q3.txt = 3・1・1・3、param_q4.txt = 4・2・2・4)で、最後まで計算して定常になること、
!   岸沿いに一様なこと、2 つの結果が一致することを確かめる(Check_stageside.py)。東端の列を水位規定セルに
!   した構成は param_s3.txt・param_s4.txt
!======================================================================
&list_sysparam
  f_gridsystem = 0
  dt = 0.5
  t0 = 0
  tt = 3600
  dt_disp = 600
  dt_file = 1200
  dt_recrd = 600
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_boundary = '-'
  dir_result = 'result_q3'
  f_check_cfl = 1
  f_output_mode = 3         ! テキストとバイナリ(検定はバイナリ = 単精度を読む)
  f_out_u = 1
  f_out_v = 1
/
&list_enc
  f_advection_donor = 3
  f_advection_bend = 1
  f_advection_lateral = 1
  f_dry_head_cap = 3
/
&list_initial
  f_htype = 2
  e0 = 0.0
/
&list_geoinfo
  nx = 120
  ny = 8
  dx = 5.0
  dy = 5.0
  f_ztype = 1
  fn_z = 'ss_z.txt'
  f_rntype = 0
  rn0 = 0.02
/
&list_bound_edge
  f_bc_w = 2                ! 西は長波放射(流出。基準水位 0)
  f_bc_e = 0                ! 東は壁(湧き出しの列の背)
  f_bc_n = 3                ! 南北は側方 Neumann(岸沿いに一様)
  f_bc_s = 3
/
&list_bound_source
  src_cell(:,1,1) = 120, 1  ! 東端の列に 48 m3/s(8 セルに等分)。時刻は分
  src_cell(:,2,1) = 120, 2
  src_cell(:,3,1) = 120, 3
  src_cell(:,4,1) = 120, 4
  src_cell(:,5,1) = 120, 5
  src_cell(:,6,1) = 120, 6
  src_cell(:,7,1) = 120, 7
  src_cell(:,8,1) = 120, 8
  src_val(:,1,1) = 0, 0.0
  src_val(:,2,1) = 10, 48.0   ! 10 分で立ち上げて、あとは一定
/
