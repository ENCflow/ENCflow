!======================================================================
! stageside: 水位規定セルの列から流れ出す水と、側方 Neumann の辺(developer.md §113.9 (7))
!   長さ 600 m・幅 40 m(120 × 8 セル、Δx = 5 m)の水路。河床は西端 −3 m から東端 −1.5 m へ上がる。
!   東端の 1 列(8 セル)が水位規定セル(規定水位 +0.2 m)、東の辺は壁、西の辺は長波放射(基準水位 0)、
!   南北の辺は側方 Neumann(岸沿いに一様)。静水(水位 0)から始めると、水位規定セルの列から西へ
!   流れ出して、定常になる(西端の流速 0.35 m/s、東端のすぐ内側 0.64 m/s)。
!   移流項の規則は、4 つの値の組み合わせ(f_advection_donor = 4・f_advection_bend = 2・f_advection_lateral = 2・
!   f_dry_head_cap = 4。深くて遅い隣のセルの扱い)。
!   2 つの構成(param_s3.txt = 3・1・1・3、param_s4.txt = 4・2・2・4)で、最後まで計算して定常になること、
!   岸沿いに一様なこと、2 つの結果が一致することを確かめる(Check_stageside.py)。東端の列を湧き出しに替えた
!   構成は param_q3.txt・param_q4.txt
!======================================================================
&list_sysparam
  f_gridsystem = 0
  dt = 0.5
  t0 = 0
  tt = 3600
  dt_disp = 600
  dt_file = 1200
  dt_recrd = 600
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_boundary = '-'
  dir_result = 'result_s4'
  f_check_cfl = 1
  f_output_mode = 3         ! テキストとバイナリ(検定はバイナリ = 単精度を読む)
  f_out_u = 1
  f_out_v = 1
/
&list_enc
  f_advection_donor = 4
  f_advection_bend = 2
  f_advection_lateral = 2
  f_dry_head_cap = 4
/
&list_initial
  f_htype = 2
  e0 = 0.0
/
&list_geoinfo
  nx = 120
  ny = 8
  dx = 5.0
  dy = 5.0
  f_ztype = 1
  fn_z = 'ss_z.txt'
  f_rntype = 0
  rn0 = 0.02
/
&list_bound_edge
  f_bc_w = 2                ! 西は長波放射(流出。基準水位 0)
  f_bc_e = 0                ! 東は壁(水位規定セルの列の背)
  f_bc_n = 3                ! 南北は側方 Neumann(岸沿いに一様)
  f_bc_s = 3
/
&list_bound_stage
  stage_cell(:,1,1) = 120, 1  ! 東端の列を、規定水位 +0.2 m に強制する
  stage_cell(:,2,1) = 120, 2
  stage_cell(:,3,1) = 120, 3
  stage_cell(:,4,1) = 120, 4
  stage_cell(:,5,1) = 120, 5
  stage_cell(:,6,1) = 120, 6
  stage_cell(:,7,1) = 120, 7
  stage_cell(:,8,1) = 120, 8
  stage_eta(1) = 0.2
/
