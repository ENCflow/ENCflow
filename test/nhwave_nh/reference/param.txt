!======================================================================
! nhwave(非静水圧): 閉じた正方水槽の定在波に 1 層 NH 補正を掛ける(
!   docs/nonhydrostatic_plan.md §11 Phase 4)。目標周期は ω² = gHΛ/(1+(H²/4)Λ)
!   平坦床・無摩擦・壁境界。初期水位は x 方向モード (4, 0) の余弦
!   (user routine wave_standing_x)。h0 を変えると kH が変わる。
!   分散関係の走査は Dispersion.py(h0・モード・dt・RK を振って周期を測る)
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.02                 ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 40                   ! 計算終了時刻 (s)(NH の周期 ≈ 6.3 s の 6 周期)
  dt_disp = 1               ! 画面表示時間刻み (s)
  dt_file = 40              ! ファイル出力時間刻み (s)
  dt_recrd = 0.02           ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_record = '-'
  dir_result = 'result'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_disp_qq = 1             ! 回帰テスト用: Q_max 列も表示
  f_check_cfl = 0
  f_out_h = 1
/

!======================================================================
! ENC計算条件
!======================================================================
&list_enc
  f_gravity_correction = 0         ! 平坦床なので無効果
  f_exflux_reduction = 0           ! 微小振幅なので不要
  f_hcap_upwind = 0                ! 両側平均(連続式の風上化なし)
  f_friction_fastmath = 0
  f_adaptive_runge = 1             ! 適応的ルンゲクッタ
  p_adprunge_thresh = 1.2
  f_nonhydrostatic = 1             ! 1 層非静水圧補正
  nh_solver = 2                    ! CG
  nh_tol = 1.0e-6
/

!======================================================================
! 初期条件設定
!======================================================================
&list_initial
  f_htype = 0             ! 初期水深タイプ (0: 固定値)
  f_uvtype = 0            ! 初期流速タイプ (0: 固定値)
  h0 = 8.0                ! 水深 H (m)。kH = 4π h0/lx = 1.0
  u0 = 0
  v0 = 0
  f_user_routine = "wave_standing_x"   ! 定在波 (4, 0)。対角は wave_standing_xy
/

!======================================================================
! 地理情報設定
!======================================================================
&list_geoinfo
  lx = 100.0
  ly = 100.0
  nx = 100                ! Δx = 1 m(モード 4 の波長 50 m = 50 セル)
  ny = 100
  f_ztype = 0             ! 地盤高タイプ (0:固定値)
  f_rntype = 0            ! 粗度係数タイプ (0:固定値)
  z0 = 0.0
  rn0 = 0.0               ! 無摩擦
/

!======================================================================
! 境界条件設定(すべて壁)
!======================================================================
&list_bound
/

!======================================================================
! 計測(プローブ: 南西隅のセル = 両モードの腹、中央セル)
!======================================================================
&list_record
  pbxytype = 0
  pbxy(:,1) = 1, 1
  pbxy(:,2) = 50, 50
/
