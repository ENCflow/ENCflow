!======================================================================
! driftwood 構成3a: リスタート分割継続の前半(0 → 15 s。save_rt へ保存)
!   構成1と同一の設定で前半だけ計算して保存する。Run.sh が構成3b の
!   継続結果と通し実行(構成1)の driftwood.dat を数値比較する
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.02                 ! 時間刻み (s)
  t0 = 0
  tt = 15                   ! 前半の終了時刻 (s)
  dt_disp = 5
  dt_file = 30
  dt_recrd = 5
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  fn_precip = '-'
  fn_driftwood = '-'
  dir_result = 'result_rt1'
  dir_save = 'save_rt'
  f_state_save = 1          ! 検定は save/driftwood.dat を読む
  f_out_hd = 1
  f_out_wd = 1
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(急勾配の重力補正を有効化 = 土石流ケースの標準)
!======================================================================
&list_enc
  f_gravity_correction = 1
/

!======================================================================
! 初期条件(乾燥。水は降雨で供給)
!======================================================================
&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報(dx = 1.0 m。遷緩点付き斜面 + 厚い可動層)
!======================================================================
&list_geoinfo
  lx = 60.0
  ly = 20.0
  nx = 60
  ny = 20
  f_ztype = 0     ! 固定値 0 の上に user routine が斜面を構築
  z0 = 0.0
  f_user_routine = "slope_break"   ! tanθ=0.4, 遷緩点 x=30m
  f_rntype = 0
  rn0 = 0.03
  sd0 = 5.0       ! 可動層厚(床クリップ不発の厚さ)
  sy0 = 0.2
/

!======================================================================
! 降雨(一様 200 mm/h。斜面の薄層流 = 侵食の駆動)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 200
  prval(1:2,2) = 9999, 200
/

!======================================================================
! 地形変化(土石流 E-D + クーロン抵抗 + 低速凝集。test/debris と同一)
!======================================================================
&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ更新(E-D の時定数は速い)
  morfac = 1.0              ! f_debris・fn_driftwood は 1 必須
  f_debris = 1
  db_phi = 35.0             ! 内部摩擦角 (deg)
  db_delte = 0.0007         ! 侵食係数 δe(高橋の一般値【要文献照合】)
  db_deltd = 0.05           ! 堆積係数 δd
  f_dbres = 1               ! クーロン+マニング合成抵抗
  f_dbstop = 1              ! 低速凝集(停止 → 河床へ固定)
  db_vstop = 0.05
  db_wstop = 0.05
  fluv_porosity = 0.4       ! 空隙率 λ(C* = 1-λ = 0.6)
/

!======================================================================
! 流木(侵食連行のみ。ストックは一様 0.01 m3/m2)
!======================================================================
&list_driftwood
  dw_stock0 = 0.01          ! 立木ストック (m3/m2)
  dw_dlog = 0.02            ! 代表直径 (m) → 喫水 hf = 0.01 m
  dw_sglog = 0.5            ! 材比重
  dw_droot = 0.002          ! 根系深 (m。疎通検定用の浅い値 — 30 s の
                            !   渓床侵食 ~3 mm で連行が発火する設定)
  dw_wstop = 0.005          ! 接地堆積レート (m/s)
  dw_vstop = 0.02           ! 低速堆積の流速閾値 (m/s)
/
