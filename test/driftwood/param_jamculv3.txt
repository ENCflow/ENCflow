!======================================================================
! driftwood 構成6a: 小さい集積と 3 セルの取水セル群(param_jamculv.txt と同じ排水試験)
!   取水セル群を (10,6)・(10,5)・(10,7) の 3 セルにし、3 セルの立木 0.06 m3 が流入口に集積する。
!   出来上がった集積の C_A = 2 κ0 V_j/(B D) = 0.74。上流水深が高い間は満管(損失の和に C_A を足した管路の流量)、
!   下がると堰(倍率 (1 + C_A/3)^(−3/2))が支配し、両方のレジームを通る。3 セルの実材積の合計は、
!   MPI では帯をまたいで集める(np = 4 で 2 つのランクに分かれる)。
!   検定(Check_woodjam.py culv)は param_jamculv.txt と同じ
!======================================================================
&list_sysparam
  f_gridsystem = 0
  dt = 0.5
  t0 = 0
  tt = 21600
  dt_disp = 600
  dt_file = 21600
  dt_recrd = 300            ! 流木の台帳(driftwood.csv)を dt_disp の半分で記録(検定が区間の中点の集積量を読む)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_boundary = '-'
  fn_structure = '-'
  fn_driftwood = '-'
  dir_result = 'result_jamculv3'
  dir_save = 'save_jamculv3'
  f_state_save = 1
  f_disp_debug = 1          ! S 列を全有効桁で表示(流量の復元に使う)
  f_check_cfl = 0
  f_out_vv = 1              ! 流速 V0001…(6b・6c と 6d の流れの比較に使う)
  f_out_wj = 1
/

&list_initial
  f_htype = 1
  fn_hinit = 'jam_ch.txt'   ! 上流槽 3.0 m、その他 0
/

&list_geoinfo
  lx = 120.0
  ly = 110.0
  nx = 12
  ny = 11
  f_ztype = 1
  fn_z = 'jam_cz.txt'       ! 上流槽 0 / 盛土 10 / 下流列 0
  f_rntype = 0
  rn0 = 0.03
/

&list_bound_edge
/

&list_bound_stage
  fn_stage_cell(1) = 'jam_cstage.txt'   ! (12, 1..11)
  stage_eta(1) = -999.0                 ! 河床より低い水位 = 完全排水口
/

&list_struct_culvert
  culv_in_cell(:,1,1)  = 10, 6     ! 代表セル(先頭)
  culv_in_cell(:,2,1)  = 10, 5
  culv_in_cell(:,3,1)  = 10, 7
  culv_out_cell(:,1,1) = 12, 6
  culv_width(1)  = 1.0
  culv_height(1) = 1.0
  culv_zin(1)  = 0.0
  culv_zout(1) = 0.0
  culv_length(1) = 0.0
  culv_ce(1) = 0.5
  culv_jam(1) = 1                  ! 流入口の集積流木の損失を流量則に入れる
/

&list_driftwood
  fn_dwstock = 'jam_cstock3.txt'   ! 立木ストック(取水セル群の 3 セルに 0.0002 m3/m2 ずつ = 0.06 m3)
  dw_dlog = 0.3
  dw_sglog = 0.5
  dw_wrec = 1.0                    ! 水理的流失(最初のステップで全量)
  dw_hrec = 0.5
  dw_vrec = 0.0
  dw_wstop = 0.01
  dw_vstop = 0.0
/
