!======================================================================
! bldgdebris 構成3b: リスタート分割継続の後半(save_rt を復元し 30 → 60 s)
!   構成1と同一の設定で前半だけ計算して保存する。Run.sh が構成3b の
!   継続結果と通し実行(構成1)の bldgdebris.dat をバイト比較する
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.02                 ! 時間刻み (s)
  t0 = 0
  tt = 60                   ! 計算終了時刻 (s)
  dt_disp = 10
  dt_file = 60
  dt_recrd = 5
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_bldgdebris = '-'
  dir_result = 'result_rt2'
  dir_save = 'save_rt'
  f_state_save = 1
  f_state_restore = 1       ! save_rt から復元して継続
  f_out_hbd = 1
  f_out_wbd = 1
  f_out_bds = 1
  f_out_fdmax = 1
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(急勾配の重力補正を有効化)
!======================================================================
&list_enc
  f_gravity_correction = 1
/

!======================================================================
! 初期条件(乾燥。水は降雨で供給)
!======================================================================
&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報(dx = 1.0 m。遷緩点付き斜面)
!======================================================================
&list_geoinfo
  lx = 60.0
  ly = 20.0
  nx = 60
  ny = 20
  f_ztype = 0     ! 固定値 0 の上に user routine が斜面を構築
  z0 = 0.0
  f_user_routine = "slope_break"   ! tanθ=0.4, 遷緩点 x=30m
  f_rntype = 0
  rn0 = 0.03
/

!======================================================================
! 降雨(一様 200 mm/h)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 200
  prval(1:2,2) = 9999, 200
/

!======================================================================
! 家屋破壊・瓦礫(浸水深判定。ストックは一様 0.01 m3/m2)
!======================================================================
&list_bldgdebris
  bd_stock0 = 0.01          ! 家屋ストック (m3/m2)
  bd_dlog = 0.005           ! 瓦礫の代表寸法 (m) → 喫水 hf = 0.0025 m
  bd_sg = 0.5               ! 瓦礫の見かけ比重
  f_bdcrit = 1              ! 浸水深判定
  bd_hcrit = 0.004          ! 浸水深閾値 (m。平坦部の湛水で超える)
  bd_hcrit2 = 0.008         ! 第2閾値(線形ランプの上端)
  bd_wdes = 2.0e-5          ! 破壊レート (m/s)
  bd_fsink = 0.3            ! 沈下率(破壊量の 3 割はその場で堆積)
  bd_wstop = 0.005          ! 接地堆積レート (m/s)
  bd_vstop = 0.01           ! 低速堆積の流速閾値 (m/s)
  f_bdgv = 0                ! 空隙率への帰還なし(片方向)
/
