!======================================================================
! bldgdebris 構成2: 流木+家屋瓦礫の同時有効化(荷重判定 f_bdcrit=2)
!   構成1と同じ斜面・降雨に流木モジュール(水理的流失。test/driftwood 構成2
!   と同一設定)を重ね、家屋の破壊判定を荷重 (h+(1+s)hs+sg_log·hd+sg_bd·hbd)·V²
!   にする。流木 hd が荷重に算入される(一方向依存)。
!   検定: (1)〜(3) は構成1と同じ(save_dw/bldgdebris.dat)。加えて Run.sh が
!   流木のみの構成2'(param_dwonly.txt)と state.dat・driftwood.dat の
!   バイト一致を確認する = 本モジュール(片方向)は流れと流木を変えない。
!   ※ 閾値・レートは疎通検定用の設定であり実地の推奨値ではない
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.02                 ! 時間刻み (s)
  t0 = 0
  tt = 60                   ! 平坦部の湛水が発達するまで
  dt_disp = 10
  dt_file = 60
  dt_recrd = 5
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_bldgdebris = '-'
  dir_result = 'result_dw'
  dir_save = 'save_dw'
  fn_driftwood = '-'
  f_out_hd = 1
  f_out_wd = 1
  f_state_save = 1          ! 検定は save/bldgdebris.dat を読む
  f_out_hbd = 1
  f_out_wbd = 1
  f_out_bds = 1
  f_out_fdmax = 1
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(急勾配の重力補正を有効化)
!======================================================================
&list_enc
  f_gravity_correction = 1
/

!======================================================================
! 初期条件(乾燥。水は降雨で供給)
!======================================================================
&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報(dx = 1.0 m。遷緩点付き斜面)
!======================================================================
&list_geoinfo
  lx = 60.0
  ly = 20.0
  nx = 60
  ny = 20
  f_ztype = 0     ! 固定値 0 の上に user routine が斜面を構築
  z0 = 0.0
  f_user_routine = "slope_break"   ! tanθ=0.4, 遷緩点 x=30m
  f_rntype = 0
  rn0 = 0.03
/

!======================================================================
! 降雨(一様 200 mm/h)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 200
  prval(1:2,2) = 9999, 200
/

!======================================================================
! 家屋破壊・瓦礫(浸水深判定。ストックは一様 0.01 m3/m2)
!======================================================================
&list_bldgdebris
  bd_stock0 = 0.01          ! 家屋ストック (m3/m2)
  bd_dlog = 0.005           ! 瓦礫の代表寸法 (m) → 喫水 hf = 0.0025 m
  bd_sg = 0.5               ! 瓦礫の見かけ比重
  f_bdcrit = 2              ! 荷重判定
  bd_fcrit = 2.0e-4         ! 荷重閾値 (m3/s2)
  bd_fcrit2 = 6.0e-4        ! 第2閾値(線形ランプの上端)
  bd_wdes = 2.0e-5          ! 破壊レート (m/s)
  bd_fsink = 0.3            ! 沈下率(破壊量の 3 割はその場で堆積)
  bd_wstop = 0.005          ! 接地堆積レート (m/s)
  bd_vstop = 0.01           ! 低速堆積の流速閾値 (m/s)
  f_bdgv = 0                ! 空隙率への帰還なし(片方向)
/

!======================================================================
! 流木(水理的流失。test/driftwood 構成2 と同一)
!======================================================================
&list_driftwood
  dw_stock0 = 0.01          ! 立木ストック (m3/m2)
  dw_dlog = 0.005           ! 代表直径 (m) → 喫水 hf = 0.0025 m
  dw_sglog = 0.5            ! 材比重
  dw_wrec = 2.0e-5          ! 水理的流失レート (m/s)
  dw_hrec = 0.002           ! 流失の水深閾値 (m)
  dw_vrec = 0.2             ! 流失の流速閾値 (m/s)
  dw_wstop = 0.005          ! 接地堆積レート (m/s)
  dw_vstop = 0.01           ! 低速堆積の流速閾値 (m/s)
/
