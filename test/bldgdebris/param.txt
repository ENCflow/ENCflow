!======================================================================
! bldgdebris 構成1: 氾濫流による家屋破壊・瓦礫の疎通・保存則検定(浸水深判定)
!   test/driftwood 構成2 と同じ遷緩点付き斜面(slope_break: tanθ=0.4,
!   遷緩点 x=30m)への一様降雨によるシート流出。平坦部(i>=30)の湛水が
!   浸水深閾値 bd_hcrit を超えたセルの家屋ストックが破壊 → 浮遊分は
!   流動瓦礫として移流、沈下分(bd_fsink)はその場で堆積 → 低速域で堆積。
!   空隙率への帰還なし(f_bdgv=0 = 片方向結合)。
!   検定(Check_bldgdebris.py, 最終 save/bldgdebris.dat から):
!     (1) 材積保存: Σwbs + Σhbd + Σwbd = 初期ストック総量(閉領域・ダムなし)
!     (2) 活性確認: 破壊の発生(Σwbs < 初期)と堆積(Σwbd > 0)
!     (3) 到達確認: 平坦部(i>=30)の瓦礫 max(hbd+wbd) > 0
!   ランク数不変(state.dat / bldgdebris.dat のバイト一致)は Run_MPI.sh、
!   リスタート往復(save→restore→save バイト一致)は Run.sh が検定する。
!   ※ 閾値・レートは疎通検定用の設定であり実地の推奨値ではない
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.02                 ! 時間刻み (s)
  t0 = 0
  tt = 60                   ! 平坦部の湛水が発達するまで
  dt_disp = 10
  dt_file = 60
  dt_recrd = 5
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_bldgdebris = '-'
  dir_result = 'result'
  f_state_save = 1          ! 検定は save/bldgdebris.dat を読む
  f_out_hbd = 1
  f_out_wbd = 1
  f_out_bds = 1
  f_out_fdmax = 1
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(急勾配の重力補正を有効化)
!======================================================================
&list_enc
  f_gravity_correction = 1
/

!======================================================================
! 初期条件(乾燥。水は降雨で供給)
!======================================================================
&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報(dx = 1.0 m。遷緩点付き斜面)
!======================================================================
&list_geoinfo
  lx = 60.0
  ly = 20.0
  nx = 60
  ny = 20
  f_ztype = 0     ! 固定値 0 の上に user routine が斜面を構築
  z0 = 0.0
  f_user_routine = "slope_break"   ! tanθ=0.4, 遷緩点 x=30m
  f_rntype = 0
  rn0 = 0.03
/

!======================================================================
! 降雨(一様 200 mm/h)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 200
  prval(1:2,2) = 9999, 200
/

!======================================================================
! 家屋破壊・瓦礫(浸水深判定。ストックは一様 0.01 m3/m2)
!======================================================================
&list_bldgdebris
  bd_stock0 = 0.01          ! 家屋ストック (m3/m2)
  bd_dlog = 0.005           ! 瓦礫の代表寸法 (m) → 喫水 hf = 0.0025 m
  bd_sg = 0.5               ! 瓦礫の見かけ比重
  f_bdcrit = 1              ! 浸水深判定
  bd_hcrit = 0.004          ! 浸水深閾値 (m。平坦部の湛水で超える)
  bd_hcrit2 = 0.008         ! 第2閾値(線形ランプの上端)
  bd_wdes = 2.0e-5          ! 破壊レート (m/s)
  bd_fsink = 0.3            ! 沈下率(破壊量の 3 割はその場で堆積)
  bd_wstop = 0.005          ! 接地堆積レート (m/s)
  bd_vstop = 0.01           ! 低速堆積の流速閾値 (m/s)
  f_bdgv = 0                ! 空隙率への帰還なし(片方向)
/
