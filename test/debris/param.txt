!======================================================================
! debris: 土石流(浸食起動型)の疎通・保存則検定
!   遷緩点付き斜面(slope_break: tanθ=0.4, 遷緩点 x=30m)への一様降雨で
!   斜面上の薄層流が高橋型 E-D により渓床を侵食し、混合体の流下と
!   遷緩点直下への堆積(扇状地型)が起きる。クーロン+マニング抵抗
!   (f_dbres=1)と低速凝集の停止条件(f_dbstop=1)を併用。
!   検定(Check_debris.py, 最終 state.dat から):
!     (1) 固体総量保存: Σhs + (1-λ)Σ(z-z0) ≈ 0(E-D 反対称の帰結)
!     (2) 共動更新恒等: (sd - sd0) ≈ (z - z0)
!     (3) 活性確認: 斜面部の侵食(min dz)と遷緩点下の堆積(max dz)
!   ランク数不変(state.dat のバイト一致)は Run_MPI.sh が検定する。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.02                 ! 時間刻み (s)
  t0 = 0
  tt = 30                   ! 計算終了時刻 (s)
  dt_disp = 5
  dt_file = 30
  dt_recrd = 30
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  fn_precip = '-'
  dir_result = 'result'
  f_state_save = 1          ! 検定は save/state.dat を読む
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(急勾配の重力補正を有効化 = 土石流ケースの標準)
!======================================================================
&list_enc
  f_gravity_correction = 1
/

!======================================================================
! 初期条件(乾燥。水は降雨で供給)
!======================================================================
&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報(dx = 1.0 m。遷緩点付き斜面 + 厚い可動層)
!======================================================================
&list_geoinfo
  lx = 60.0
  ly = 20.0
  nx = 60
  ny = 20
  f_ztype = 0     ! 固定値 0 の上に user routine が斜面を構築
  z0 = 0.0
  f_user_routine = "slope_break"   ! tanθ=0.4, 遷緩点 x=30m
  f_rntype = 0
  rn0 = 0.03
  sd0 = 5.0       ! 可動層厚(床クリップ不発の厚さ)
  sy0 = 0.2
/

!======================================================================
! 降雨(一様 200 mm/h。斜面の薄層流 = 侵食の駆動)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 200
  prval(1:2,2) = 9999, 200
/

!======================================================================
! 地形変化(土石流 E-D + クーロン抵抗 + 低速凝集)
!======================================================================
&list_geomorph
  dt_geomorph = 0.0         ! 毎ステップ更新(E-D の時定数は速い)
  morfac = 1.0              ! f_debris は 1 必須
  f_debris = 1
  db_phi = 35.0             ! 内部摩擦角 (deg)
  db_delte = 0.0007         ! 侵食係数 δe(高橋の一般値【要文献照合】)
  db_deltd = 0.05           ! 堆積係数 δd
  f_dbres = 1               ! クーロン+マニング合成抵抗
  f_dbstop = 1              ! 低速凝集(停止 → 河床へ固定)
  db_vstop = 0.05
  db_wstop = 0.05
  fluv_porosity = 0.4       ! 空隙率 λ(C* = 1-λ = 0.6)
/
