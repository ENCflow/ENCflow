  fn_geoinfo = '-'
  fn_initial = '-'

&list_geoinfo
  nx = 68                   ! 格子数(流域 1000 × 1620 m + 下流の同じ断面の区間 200 m + 水路の延長と海 6 セル + 余白 1 セル)
  ny = 83
  dx = 20

&list_precip

&list_record

  flxytype = 0
  flxy(:,1) = 51, 82, 51, 2 ! 出口の計測面: 流域の下流端の列を右岸側の尾根から左岸側の尾根まで(全幅)
  flxy(:,2) = 2, 41, 51, 41 ! 斜面の脚: 左岸側の斜面の水路に接する列を上流端から出口まで
