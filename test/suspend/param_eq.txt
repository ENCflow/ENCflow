! suspend 構成 3: 等流の水路での巻き上げと沈降の釣り合い(平衡濃度式 1 = 線形)
!   南北に長い幅 1 セルの直線水路(長さ 1000 m、Δy = 1 m、勾配 0.002、n = 0.03。両側は
!   河床 +2 m の乾いた地盤。make_eq.py が書く)に、北端の区間流入で Manning の等流
!   (水深 0.5 m、単位幅流量 q = 0.469545 m²/s)を流し、南辺は自由流出。流入は清水。
!   浮遊砂は E − D = w_f (C_eq − β C) で河床と交換し、流下につれて平衡濃度 C_eq/β へ
!   近づく。Check_suspend.py eq が
!     (1) 平衡: 適応を終えた区間(250〜350 m)の C = hs/h が、その場の流速と水深で
!         求めた C_eq(τ*)/β に一致する
!     (2) 適応: 平衡からの不足が流下方向に一定の比で減り、その比が MUSCL(既定の
!         スカラー輸送)と時間の分割を含む離散の予測 (1 − k)/(1 + k)、
!         k = b / (2c (1 − b))(b = w_f β Δt/h、c = q Δt/(h Δy))に一致する
!   を確かめる(連続の解は exp(−w_f β Δy/q) のセルあたりの比)。河床の変化が流れを
!   変えないよう morfac を非常に小さくする(水柱側の交換は水理時間で進む。§19.3)
!======================================================================

  f_output_mode = 2         ! bil(float32)

  f_out_hs = 1

  fluv_porosity = 0.4
