!======================================================================
! suspend: 浮遊砂(移流+浸食・沈降)の疎通・保存則検定(閉領域ダムブレイク型)
!   平坦床(z0=0、可動層 sd0=10m)の閉じた箱に円形コサイン水塊
!   (wave_hump)を置き、放射状のダムブレイク流で河床を浸食・堆積させる。
!   検定(Check_suspend.py, 最終 state.dat から):
!     (1) 固体総量保存: Σhs + (1-λ)Σ(z-z0) ≈ 0(機械精度。E-D の反対称の帰結)
!     (2) 共動更新恒等: (sd - sd0) ≈ (z - z0)(帯水層底 z-sd の不変性)
!     (3) 活性確認: max|z - z0| がしきい値以上(空回りの検出)
!   ランク数不変(state.dat のバイト一致)は Run_MPI.sh が検定する。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.1                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 8                    ! 計算終了時刻 (s)
  dt_disp = 1               ! 画面表示時間刻み (s)
  dt_file = 8               ! ファイル出力時間刻み (s)
  dt_recrd = 8              ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  dir_result = 'result'
  f_state_save = 1          ! 終了時に状態保存(検定は save/state.dat を読む)
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(既定)
!======================================================================
&list_enc
/

!======================================================================
! 初期条件設定(基準水深 + 中央の円形コサイン水塊)
!======================================================================
&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 0.5                  ! 初期水深固定値 (m)
  f_user_routine = "wave_hump"   ! 円形コサイン型の初期水位(+1m at center)
/

!======================================================================
! 地理情報設定(dx = lx/nx = 1.0 m。平坦床・厚い可動層)
!======================================================================
&list_geoinfo
  lx = 101.0
  ly = 101.0
  nx = 101
  ny = 101
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値(m)
  rn0 = 0.03      ! 粗度係数固定値
  sd0 = 10.0      ! 土層厚(=可動層厚)固定値 (m)。床クリップ不発の厚さ
  sy0 = 0.2       ! 比湧水量(gwflow 無効のため本ケースでは未使用)
/

!======================================================================
! 地形変化設定(掃流砂 Exner のみ)
!======================================================================
&list_geomorph
  dt_geomorph = 0.0         ! 0 = 毎ステップ更新
  morfac = 1.0              ! 加速係数
  f_suspend = 1             ! 浮遊砂有効
  f_esform = 1              ! 平衡濃度式(1:超過掃流力線形)
  susp_d50 = 0.0002         ! 浮遊砂の代表粒径 (m。細砂)
  susp_tausc = 0.05         ! 浮遊の限界無次元掃流力
  susp_esa = 1.0e-4         ! 平衡濃度係数
  susp_beta = 1.0           ! 沈降の底面濃度係数
  fluv_porosity = 0.4       ! 空隙率
/
