!======================================================================
! tracer: スカラー輸送(1 次風上)の数値拡散の計測 — 構成 A(縦断。dx = 20 m、dt = 2 s)
!   一様流の直線水路(幅 3 セル、南北向き、勾配 9e-4、n = 0.03、水深 1 m、
!   u = 1 m/s)に初期濃度 100 mg/L の水を満たし、北辺から清水を流入させる。清水の前線
!   (階段)が流下しながらなまる幅から、等価拡散係数 D = σ²/(2t) を求め、
!   1 次風上の理論値 u·dx·(1 − Cr)/2(Cr = u·dt/dx)と比べる(Check_tracer.py)。
!   減衰・沈降・物理分散はなし(水質モジュールの既定)。流れの移流項は
!   スキーム 1(セル中心勾配)で、区間流入 + 保存形移流の自励振動(handoff 00a)
!   を避けて流れを一様に保つ。スカラーのスキームはこれと独立。水路を南北向きに
!   するのは、帯分割(j 方向)の MPI 検証をこのケースでできるようにするため。
!   回帰: Log.txt + wq.csv(reference)。検定: Check_tracer.py
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 2.0
  t0 = 0
  tt = 5400                 ! 90 分(前線は 5.4 km まで進む。水路長 6 km)
  dt_disp = 600
  dt_file = 600
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_boundary = '-'
  fn_enc = '-'
  fn_wq = '-'
  dir_result = 'result_m_a20'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_out_h = 1
  f_check_cfl = 0
/

&list_enc
  f_advection_scheme = 1    ! 流れを一様に保つ(ヘッダ参照)
  f_scalar_scheme = 3       ! スカラー輸送をリンクごとの MUSCL に
/

&list_initial
  f_htype = 0
  h0 = 1.0                  ! 等流水深(Q = 60 m3/s、B = 60 m)
/

&list_geoinfo
  nx = 3
  ny = 300
  dx = 20.0
  dy = 20.0
  f_ztype = 1
  fn_z = 'z_a20.txt'
  f_rntype = 0
  rn0 = 0.03
/

&list_bound_edge
  f_bc_s = 1                ! 南辺: 自由流出
/
&list_bound_inflow
  inflow_cell(:,1,1) = 1, 1
  inflow_cell(:,2,1) = 2, 1
  inflow_cell(:,3,1) = 3, 1
  inflow_val(:,1,1) =   0, 60.0
  inflow_val(:,2,1) = 999, 60.0
  inflow_dist(1) = 0
/

&list_wq
  wq_c0 = 100.0             ! 初期濃度(一様)。流入は清水(濃度 0)
/
