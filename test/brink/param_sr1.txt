!======================================================================
! brink 構成 sr1: 東と北東のリンクが交互に並ぶ 1 セル幅の水路の等流。f_advection_bend = 1
!   勾配 0.005(路長に沿って)、n = 0.03、Δx = 50 m。西端の区間流入 200 m3/s、東辺は自由流出。
!   斜めのリンクは通過幅が 0.707 倍で、等流でも流速が 1.41 倍になる。f_advection_bend = 1 がこの幅の
!   違いを加減速に数えないこと(0 と同じ水深になること)を確かめる。
!   検定(Check_brink.py): 終了時の水深(路の 40〜80 % の区間)の平均と幅を、0 と 1 で比べる。
!   入力 br_z_stair.txt は make_brink.py が作る
!======================================================================
&list_sysparam
  f_gridsystem = 0
  dt = 2.0
  t0 = 0
  tt = 4500
  dt_disp = 1500
  dt_file = 4500
  fn_geoinfo = '-'
  fn_boundary = '-'
  fn_enc = '-'
  dir_result = 'result_sr1'
  f_out_h = 1
  f_out_hmax = 0
/

&list_enc
  f_advection_bend = 1      ! 供給元が壁・乾床の流向面の風上値(0: u_e = 既定、1: 経路の値)
/

&list_geoinfo
  nx = 80
  ny = 39
  dx = 50.0
  dy = 50.0
  f_ztype = 1
  fn_z = 'br_z_stair.txt'
  f_rntype = 0
  rn0 = 0.03
/

&list_bound_edge
  f_bc_e = 1                ! 東辺は自由流出
/

&list_bound_inflow            ! 西端の水路のセルに区間流入
  inflow_cell(:,1,1) = 1, 37
  inflow_val(:,1,1) = 0, 200
  inflow_val(:,2,1) = 9999, 200
  inflow_dist(1) = 0
/
