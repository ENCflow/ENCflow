!======================================================================
! gwwidth: サブグリッド河道幅のセルでの地下⇔地表の水の受け渡し(§26.1)
!
!   閉じた谷(12 × 8 セル、10 m 格子)。勾配 5 % の両斜面が列 10 の谷底へ
!   下り、列 10 は幅 2 m のサブグリッド河道(平面積率 wfrac = 0.2)。
!   1 時間の一様降雨 50 mm/h が浸透し(Green-Ampt。土層 0.5 m × 有効間隙率
!   0.1 = 0.05 m で飽和)、地下水の側方流で谷へ集まって、飽和した谷底から
!   地表へ湧き出す(容量超過の引き渡し)。
!   - 閉領域なので、雨が止んだ後の総水量 S_total は一定。地下の水深を
!     河道の平面に換算しないと、湧き出すたびに水が河道幅の割合しか地表に
!     出ず、総水量が減る(修正前は 6 時間で約 6 割が消えた)
!   - 雨に濃度 1 mg/L のトレーサを持たせる。浸透・地下の流れ・湧出の
!     どこを通っても濃度は 1 のまま(湧出の水が縮むと 1/wfrac 倍に見える)
!   Check_gwwidth.py が S_total の保存・河道幅なし(param_nowidth.txt)との
!   総水量の一致・濃度の上限・水質の台帳の閉合を検定する。
!======================================================================
&list_sysparam
  dt = 0.5                  ! 時間刻み (s)
  tt_c = "6h"               ! 計算終了時刻
  dt_disp_c = "30m"         ! 画面表示の間隔
  dt_file_c = "1h"          ! 分布ファイル出力の間隔
  dt_recrd_c = "10m"        ! wq.csv の記録間隔
  fn_geoinfo = '-'
  fn_channel = '-'
  fn_gwflow = '-'
  fn_precip = '-'
  fn_wq = '-'
  dir_data = 'data_gwwidth'
  dir_result = 'result'
  f_check_cfl = 0
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_output_mode = 2         ! 分布は bil(単精度。濃度の検定に桁を残す)
/

&list_geoinfo
  nx = 12
  ny = 8
  dx = 10.0
  dy = 10.0
  f_ztype = 1               ! 地盤高ファイル
  fn_z = 'z.txt'
  rn0 = 0.05
  fn_rw = 'rw.txt'          ! 河道マスク(列 10)
  sd0 = 0.5                 ! 土層厚 (m)
  sy0 = 0.1                 ! 有効間隙率(地下の容量 0.05 m)
/

&list_channel
  fn_width = 'width.txt'    ! サブグリッド河道幅 2 m(wfrac = 0.2)
/

&list_precip
  prtype = 1                ! 一様時系列
  dt_prupdate = 1           ! 更新間隔 (min)
  prval(:,1) =    0, 50.0   ! (時刻 min, 雨量強度 mm/h)
  prval(:,2) =   60, 50.0
  prval(:,3) =   61,  0.0
  prval(:,4) = 9999,  0.0
/

&list_gwflow
  f_gwvertical = 2          ! Green-Ampt の浸透
  f_gwlateral = 1           ! 地下水の側方流動(谷底で湧き出す)
/

&list_gwflow_greenampt
  gw_ksv_mmh = 200.0        ! 浸透能 (mm/h)。雨はすべて浸透する
/

&list_gwflow_lateral
  gw_ksh_mmh = 36000.0      ! 側方透水係数 (mm/h) = 1e-2 m/s(数時間で谷へ集まる)
/

&list_wq
  wq_rain_conc = 1.0        ! 雨の濃度 (mg/L)。全経路で 1 のまま保たれるはず
/
