!======================================================================
! tide: 潮位境界(m_tide)の疎通テスト
!   平坦な陸(z=1m)の西縁1列を海域マスクにする(f_masktype=2 で領域
!   マスクを自動生成)。初期水深 0.5m の陸水がまず海へ排水され
!   (S 減少)、潮位が 0→2m に上がると陸へ浸水する(S 増加)。
!   海セルの水位は s%z = 潮位 - hsea0、s%h = hsea0 で強制される。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE, 1:DifWE, 2:KinWE)
  dt = 0.2                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 240                  ! 計算終了時刻 (s)
  dt_disp = 10              ! 画面表示時間刻み (s)
  dt_file = 60              ! ファイル出力時間刻み (s)
  dt_recrd = 10             ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_tide = '-'
  fn_enc = '-'
  dir_result = 'result'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_disp_qq = 1             ! 回帰テスト用: Q_max 列も表示
  f_check_cfl = 0
  f_out_e = 1
/

!======================================================================
! ENC計算条件
!======================================================================
&list_enc
  f_gravity_correction = 0
  f_exflux_reduction = 1
  f_hcap_upwind = 1
  p_adv_upwind_index = 0.0  ! reference 作成時の既定値を明示固定
  f_adaptive_runge = 1
  p_adprunge_thresh = 1.2
/

!======================================================================
! 初期条件設定
!======================================================================
&list_initial
    f_htype = 0             ! 初期水深タイプ (0: 固定値)
    f_uvtype = 0            ! 初期流速タイプ (0: 固定値)
    h0 = 0.5                ! 初期水深固定値 (m)
    u0 = 0
    v0 = 0
/

!======================================================================
! 地理情報設定
!======================================================================
&list_geoinfo
  nx = 21
  ny = 21
  dx = 10.0
  dy = 10.0

  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 1.0        ! 地盤高固定値 (m)
  rn0 = 0.02      ! 粗度係数固定値

  f_masktype = 2  ! 領域マスクを海域マスクから自動生成
  fn_sw = 'sw.txt'
/

!======================================================================
! 潮位設定(一様時系列: 0min:0m → 2min:2m → 4min:2m)
!======================================================================
&list_tide
  titype = 2           ! 一様時系列
  hsea0 = 1.0          ! 海セルの見かけ水柱厚 (m)
  dt_tiupdate = 0.1    ! 潮位更新時間間隔 (min)
  tival(1:2,1) = 0, 0.0
  tival(1:2,2) = 2, 2.0
  tival(1:2,3) = 4, 2.0
/

!======================================================================
! 境界条件設定(全辺 wall)
!======================================================================
&list_bound
