!======================================================================
! fluvial(creep 併用): 掃流砂 + 斜面クリープの帯界面の検定(param.txt と同じ箱)
!   クリープは近傍の z を読むので、先に掃流砂が帯内の z を更新したあとハロを
!   交換しないと、MPI の帯界面のエッジが両ランクで食い違い、ランク数依存と
!   土砂体積の誤差が出る(2026-10-27 実検出・修正。developer.md §19.9)。
!   検定(Check_fluvial.py nosd):
!     (1) 土砂体積保存: Σ(z - z0) ≈ 0(掃流砂・クリープとも反対称フラックス)
!     (3) 活性確認
!   クリープは z だけを動かす(sd は不変)ので (2) 共動更新恒等は対象外。
!   ランク数不変(state.dat のバイト一致)は Run_MPI.sh が検定する。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 0.1                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 8                    ! 計算終了時刻 (s)
  dt_disp = 1               ! 画面表示時間刻み (s)
  dt_file = 8               ! ファイル出力時間刻み (s)
  dt_recrd = 8              ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_geomorph = '-'
  dir_result = 'result_creep'
  dir_save = 'save_creep'
  f_state_save = 1          ! 終了時に状態保存(検定は save/state.dat を読む)
  f_check_cfl = 0
/

!======================================================================
! ENC計算条件(既定)
!======================================================================
&list_enc
/

!======================================================================
! 初期条件設定(基準水深 + 中央の円形コサイン水塊)
!======================================================================
&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 0.5                  ! 初期水深固定値 (m)
  f_user_routine = "wave_hump"   ! 円形コサイン型の初期水位(+1m at center)
/

!======================================================================
! 地理情報設定(dx = lx/nx = 1.0 m。平坦床・厚い可動層)
!======================================================================
&list_geoinfo
  lx = 101.0
  ly = 101.0
  nx = 101
  ny = 101
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値(m)
  rn0 = 0.03      ! 粗度係数固定値
  sd0 = 10.0      ! 土層厚(=可動層厚)固定値 (m)。床クリップ不発の厚さ
  sy0 = 0.2       ! 比湧水量(gwflow 無効のため本ケースでは未使用)
/

!======================================================================
! 地形変化設定(掃流砂 Exner のみ)
!======================================================================
&list_geomorph
  dt_geomorph = 0.0         ! 0 = 毎ステップ更新
  morfac = 1.0              ! 加速係数
  f_fluvial = 1             ! 掃流砂 Exner 有効
  f_qbform = 1              ! 流砂量式(1:芦田・道上)
  fluv_d50 = 0.001          ! 代表粒径 (m)
  fluv_tausc = 0.05         ! 限界無次元掃流力
  fluv_porosity = 0.4       ! 空隙率
  fluv_sgrav = 1.65         ! 水中比重
  fluv_dzmax = 0.05         ! 1エッジ・1更新の変動上限 (m)
  f_creep = 1               ! 斜面クリープ(4 近傍の線形拡散。z だけを動かす)
  creep_d = 0.01            ! クリープ拡散係数 (m2/s)。dts = 0.1 s で安定条件 D dts (2/dx²) = 0.002 <= 0.5
/
