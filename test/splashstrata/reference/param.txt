!======================================================================
! splashstrata: 乾式斜面侵食の水平層理(spl_strata。硬軟差の倍率)の
!         解析検証・回帰テスト(developer.md §48.5)
!   test/splash と同じ遷緩点付き斜面(slope_break: z = 0.4(30 − x)、x >= 30 m
!   で平坦)・同じ降雨 30 mm/h に、増幅なしの係数(kr = 0.004、kt = 0.02。
!   Δz が Sd 出力の印字桁 1e-4 m の整数倍になる値)と、標高だけで決まる
!   3 層の侵食性倍率を与える:
!     z >= 10 m: 倍率 0(削れない硬層 = キャップロック。x <= 4.5 m の列)
!     6 <= z < 10: 倍率 0.5(x = 5.5〜14.5 m の列)
!     0 <= z < 6:  倍率 1(x = 15.5〜29.5 m の斜面と平坦部)
!   侵食深(~1.8 mm)は層境界までの余裕(>= 0.2 m)より十分小さいので、各列は
!   同じ層に留まり、解析形 Δz = f(z)・P・(spl_kr + spl_kt・S)・T_geo が成り立つ:
!     斜面内部(S=0.4): 1.8e-3 m(倍率 1)、0.9e-3 m(倍率 0.5)、0(倍率 0)
!     遷緩点の列(x=29.5、面の勾配 0.2): 1.2e-3 m、平坦部(S=0): 0.6e-3 m
!   Check_analytic.py が Sd0000 − Sd9998 を列クラス別に照合する。
!   (T_geo = tt・morfac = 3600・5 = 18000 s、poroi = 1)
!======================================================================
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 1.0                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 3600                 ! 計算終了時刻 (s)
  dt_disp = 600             ! 画面表示時間刻み (s)
  dt_file = 3600            ! ファイル出力時間刻み (s)
  dt_recrd = 3600           ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_precip = '-'
  fn_gwflow = '-'
  fn_geomorph = '-'
  fn_enc = '-'
  dir_result = 'result'
  f_out_z = 1               ! 地盤高の出力(解析検証は Z9998 を読む)
  f_check_cfl = 0
/

&list_enc
/

!======================================================================
! 地理情報設定(dx = dy = 1 m。地形はユーザールーチン slope_break)
!======================================================================
&list_geoinfo
  lx = 50.0
  ly = 12.0
  nx = 50
  ny = 12
  f_ztype = 0     ! 地盤高タイプ(0:固定値。ユーザールーチンが上書き)
  f_rntype = 0    ! 粗度係数タイプ(0:固定値)
  z0 = 0.0
  rn0 = 0.03
  sd0 = 5.0       ! 土層厚 (m)。総侵食深(~1.5e-3 m)より十分厚い
  sy0 = 0.2       ! 比湧水量(浸透水の貯留 0.03 m < sd0*sy0 = 1 m)
  f_user_routine = "slope_break"   ! tanθ=0.4, 遷緩点 x=30m
/

!======================================================================
! 降雨設定(一様 30 mm/h)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 30
  prval(1:2,2) = 9999, 30
/

!======================================================================
! 地下水設定(Green-Ampt。ksv=72 mm/h > 降雨強度 → 地表流なし)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! 鉛直モデル(2:Green-Ampt)
  f_gwlateral = 0           ! 側方なし
/
&list_gwflow_greenampt
  gw_ksv_mmh = 72.0         ! 2e-5 m/s
/

!======================================================================
! 地形変化設定(乾式斜面侵食+水平層理。凹地形増幅は無効=解析検証)
!======================================================================
&list_geomorph
  dt_geomorph = 10.0        ! 更新間隔 (s)
  morfac = 5.0              ! 加速係数(T_geo = 18000 s)
  f_splash = 1              ! 乾式斜面侵食有効
  spl_kr = 0.004            ! 雨滴侵食係数(平坦部 Δz = 0.6 mm)
  spl_kt = 0.02             ! サブグリッドリル侵食係数(斜面 Δz = 1.8 mm)
  spl_h = 1.0               ! 勾配べき指数
  spl_ca = 0.0              ! 凹地形増幅(雨滴項)無効
  spl_cb = 0.0              ! 凹地形増幅(リル項)無効
  fluv_porosity = 0.0       ! 空隙率(poroi = 1 で解析式が単純)
  spl_strata(1:2,1) = 0.0, 1.0    ! 層 1: z >= 0 m、倍率 1(軟層)
  spl_strata(1:2,2) = 6.0, 0.5    ! 層 2: z >= 6 m、倍率 0.5(やや硬い層)
  spl_strata(1:2,3) = 10.0, 0.0   ! 層 3: z >= 10 m、倍率 0(削れないキャップロック)
/
