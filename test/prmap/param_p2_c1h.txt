!   検定は Check_prmap.py(構成の一覧は README.md)。

&list_sysparam
  f_gridsystem = 0
  dt = 10.0
  t0 = 0

  fn_initial = '-'
  fn_precip = '-'

  dir_result = 'result_p2_c1h'

&list_initial

  dy = 10.0
  f_ztype = 0
  z0 = 0.0
  f_rntype = 0
  rn0 = 0.03
/

&list_precip
  prtype = 2
  fn_prmap = 'prmap.txt'    ! 全セル 24 mm
  prval(1:2,1) = 0, 1       ! (min, 倍率)。倍率は 1 で一定
