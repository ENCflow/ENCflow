!======================================================================
! bankcap 構成 5: 河道の水位に、河岸の高さを横切らせる(f_dry_head_cap = 2 の境目のつなぎ)
!   構成 1 と同じ配置で、流量を 137.3 → 330 → 137.3 m3/s と 10 分ずつで上げ下げする。河道の水位は
!   河岸の地盤(河床 + 3.5 m)を下から上へ、上から下へ 1 回ずつ横切り、その間は氾濫原へ水が載る。
!   河岸の線(河道と氾濫原の間)を横切る流量を毎ステップ記録する。水位が河岸の高さにあるとき、
!   頭打ちが効く・効かないを数ステップごとに行き来しないことを確かめる(developer.md §108.9)。
!   検定(Check_bankcap.py): 毎ステップの流量の 2 階差分が小さい、水位が河岸より高い間は水が載る、
!   立ち上がりで水位が河岸より低い間は水膜が増えない。入力(bc_*.txt)は make_bankcap.py が作る
!======================================================================
&list_sysparam
  f_gridsystem = 0
  dt = 0.2
  t0 = 0
  tt = 1800
  dt_disp = 300
  dt_file = 300
  dt_recrd = 0.2            ! プローブと測線を毎ステップ記録する
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_boundary = '-'
  fn_enc = '-'
  fn_record = '-'
  dir_result = 'result_cross'
  f_out_h = 1
/

&list_enc
  f_dry_head_cap = 2
/

&list_initial
  f_htype = 1
  fn_hinit = 'bc_h_wet.txt'   ! 河道 2.50 m、氾濫原 2 cm の水膜
  u0 = 2.745949             ! 等流の流速(流れを最初から定常に近づける)
/

&list_geoinfo
  nx = 80
  ny = 16
  dx = 5.0
  dy = 5.0
  f_ztype = 1
  fn_z = 'bc_z.txt'         ! 河道(行 9〜12)10 − 0.002 x、氾濫原(行 1〜8)+3.5 m、台地(行 13〜16)+7 m
  f_rntype = 1
  fn_rn = 'bc_rn.txt'       ! 河道 0.03、氾濫原と台地 0.05
/

&list_bound_edge
  f_bc_e = 1                ! 東辺は自由流出
/

&list_bound_inflow
  inflow_cell(:,1,1) = 1, 9
  inflow_cell(:,2,1) = 1, 10
  inflow_cell(:,3,1) = 1, 11
  inflow_cell(:,4,1) = 1, 12
  inflow_val(:,1,1) = 0, 137.3      ! 時刻 (min), 流量 (m3/s)
  inflow_val(:,2,1) = 10, 330.0     ! 10 分で満杯の流量(約 240 m3/s)を超える
  inflow_val(:,3,1) = 20, 137.3
  inflow_val(:,4,1) = 9999, 137.3
  inflow_dist(1) = 2
/

&list_record
  pbxytype = 0
  pbxy(:,1) = 40, 9         ! 河岸に接する河道のセル
  pbxy(:,2) = 40, 8         ! 河岸の列のセル
  flxytype = 0
  flxy(:,1) = 20, 8, 60, 8  ! 河岸の線(行 8 と 9 の間。x = 100〜300 m)。河道から氾濫原へが正
/
