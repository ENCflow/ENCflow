!======================================================================
! conduit: 管路連続体層の模式実験 A(一様・解析平衡)
!
!   平坦地形・一様湛水 h0 = 0.1 m を、一様な枡密度の管路層が
!   堰式→オリフィス式で飲み込み、満管後は疑似スロットの被圧水頭が
!   立ち上がって地表水位と平衡する。全セルが同一の常微分方程式に
!   従う(側方流なし)ため、平衡値が解析的に決まる:
!     水頭底  zbot = z0 - gwc_depth = -2 m
!     管頂水位 hcrown = zbot + gwc_cap/gwc_sy = -1 m
!     平衡条件 h = hcrown + (hgc - cap)/slot_sy かつ h + hgc = 0.1
!       → hgc_eq = 51.1/1001 = 0.0510489510... m
!          h_eq  = 0.1 - hgc_eq = 0.0489510489... m
!   Log の S_surf / S_grnd / S_total(f_disp_debug=1 で全有効桁)が
!   それぞれ h_eq / hgc_eq / 0.1 に収束し、S_total は初期値のまま
!   一定であること(質量保存)を確認する。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  dt = 0.05                 ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 150                  ! 計算終了時刻 (s)
  dt_disp = 10              ! 画面表示時間刻み (s)
  dt_file = 50              ! ファイル出力時間刻み (s)
  dt_recrd = 10             ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_gwflow = '-'
  dir_result = 'result'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_e = 1
  f_out_hg = 1              ! Hg / Hgc 場の出力
/

&list_enc
  f_adaptive_runge = 1
/

&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 0.1                  ! 初期水深固定値 (m)
/

&list_geoinfo
  lx = 42.0
  ly = 42.0
  nx = 21
  ny = 21
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値 (m)
  rn0 = 0.05      ! 粗度係数固定値
/

&list_bound
/

!======================================================================
! 地下水設定(管路連続体層のみ有効)
!======================================================================
&list_gwflow
  f_gwvertical = 0          ! 鉛直モデルなし(不浸透地表の都市域相当)
  f_gwlateral = 0
  f_gwlayer2 = 0
  f_gwconduit = 1           ! 管路連続体層
  dt_gwflow = 0             ! 毎ステップ更新
/

&list_gwflow_conduit
  f_gwc_fluxlaw = 2         ! sqrt 乱流則(本ケースでは側方無効のため未使用)
  gwc_cnd_m2s = 0.0         ! 側方通水なし(セル独立の貯留・交換のみ)
  gwc_cap = 0.05            ! 貯留容量 (m 柱状)
  gwc_depth = 2.0           ! 埋設深: zbot = z - 2 (m)
  gwc_sy = 0.05             ! 不圧貯留係数(管内充水相当)
  gwc_slot_sy = 0.001       ! 疑似スロット貯留係数(被圧枝)
  gwc_inlet = 0.01          ! 枡密度 (個/m2) = 100 m2 に 1 個
  gwc_cw = 2.66             ! 堰係数 (m3/s/個 at h=1m)
  gwc_co = 0.15             ! オリフィス係数 Cd·A (m2)
/
