!======================================================================
! glacier: SIA 氷体流動の Halfar ドーム・ベンチマーク
!   平坦床・質量収支ゼロ(気温 -30℃ で融解なし・降水なし)の氷帽が
!   浅氷近似の相似解(Halfar 1983)に従って広がる。初期氷厚は t=t0 の
!   相似解プロファイル(Make_input.py が data/hi0.txt を生成)。
!   地形時間 tt×gl_morfac だけ流動させ、解析解と比較する
!   (Check_halfar.py)。S_total 列(氷の水当量)で体積保存も検定する。
!   定数の対応: Γ = 2A(ρi g)^3/5, A=1e-16 Pa^-3 yr^-1, ρi=900, g=9.8
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 60                   ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 86400                ! 計算終了時刻 (s)(地形時間 = tt×80 ≈ 2 t0)
  dt_disp = 3600            ! 画面表示時間刻み (s)
  dt_file = 86400           ! ファイル出力時間刻み (s)
  dt_recrd = 86400          ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_meteo = '-'
  fn_snow = '-'
  fn_glacier = '-'
  dir_data = 'data'
  dir_result = 'result'
  f_disp_debug = 1          ! S 列を全有効桁で表示(体積保存の検定用)
  f_check_cfl = 0
/

&list_enc
/

!======================================================================
! 初期条件設定(乾燥状態。水は無し)
!======================================================================
&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報設定(dx = 25 m の平坦床)
!======================================================================
&list_geoinfo
  lx = 1525.0
  ly = 1525.0
  nx = 61
  ny = 61
  f_ztype = 0
  f_rntype = 0
  z0 = 0.0
  rn0 = 0.03
/

!======================================================================
! 気象強制場(一様 -30℃ = 融解なし)
!======================================================================
&list_meteo
  temp0 = -30.0
/

!======================================================================
! 積雪・融雪(氷河の必須前提。降水なしのため実質不活性)
!======================================================================
&list_snow
  snow_ddf = 3.0            ! 度日係数 (mm/℃/day)
/

!======================================================================
! 氷河(SIA 流動のみ。滑動・侵食・雪崩なし)
!======================================================================
&list_glacier
  dt_glacier_c = "1 hour"   ! 更新間隔
  gl_morfac = 80.0          ! 地形時間 = 86400 s × 80 ≈ 6.9e6 s ≈ 2 t0
  gl_ddfi = 8.0             ! 氷面度日係数 (mm/℃/day。-30℃では不活性)
  f_glflow = 1              ! SIA 流動
  gl_afl = 1.0e-16          ! Glen A (Pa^-3 yr^-1)
  fn_glacier_hi0 = 'hi0.txt'   ! Halfar ドームの初期氷厚(m)
/
