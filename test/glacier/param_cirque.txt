!======================================================================
! glacier: カール(圏谷)形成のスモークテスト(円錐山体)
!   気温減率で山頂側が雪、降水は標高勾配で山頂側ほど多い。雪崩再配分・
!   氷化・SIA 流動・Weertman 滑動・氷河侵食の全プロセスを有効化し、
!   1日(地形時間 ×500)で「氷河が形成され山体上部が侵食される」ことを
!   検定する(Check_cirque.py。定量ベンチマークではなく全経路の疎通と
!   保存量の健全性の確認)。氷化は加速のため gl_tfirn_yr を極小にする
!======================================================================
&list_sysparam
  f_gridsystem = 0
  f_govequation = 0
  dt = 30                   ! 時間刻み (s)(水は微量のため CFL に余裕)
  t0 = 0
  tt = 86400                ! 1 日(地形時間 = ×500 ≈ 1.4 年)
  dt_disp = 3600
  dt_file = 86400
  dt_recrd = 86400
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  fn_precip = '-'
  fn_meteo = '-'
  fn_snow = '-'
  fn_glacier = '-'
  dir_data = 'data'
  dir_result = 'result_cirque'
  f_disp_debug = 1
  f_out_z = 1               ! 地形の時間変化(侵食の検定用)
  f_check_cfl = 0
/

&list_enc
/

&list_initial
  f_htype = 0
  h0 = 0.0
/

!======================================================================
! 地理情報設定(dx = 50 m の円錐山体。Make_input.py が生成)
!======================================================================
&list_geoinfo
  lx = 4050.0
  ly = 4050.0
  nx = 81
  ny = 81
  f_ztype = 1
  fn_z = 'zcone.txt'
  f_rntype = 0
  rn0 = 0.05
/

!======================================================================
! 降水(一様 5 mm/h。標高勾配は &list_meteo の f_prec_lapse で付与)
!======================================================================
&list_precip
  prtype = 1
  dt_prupdate = 60.0
  prval(1:2,1) = 0.0, 5.0
  prval(1:2,2) = 1440.0, 5.0
/

!======================================================================
! 気象強制場(気温 0.5℃ + 減率 → 山体のほぼ全域が雪。融解は麓の
! 帯のみ = 水理側を微量に保ち、氷のプロセスを主役にする)
!======================================================================
&list_meteo
  temp0 = 0.5
  f_temp_lapse = 1
  temp_lapse = 0.65         ! ℃/100m
  temp_zref = 0.0
  f_prec_lapse = 1
  prec_lapse = 5.0          ! %/100m
  prec_zref = 0.0
/

!======================================================================
! 積雪・融雪
!======================================================================
&list_snow
  snow_ddf = 3.0            ! 度日係数 (mm/℃/day)
/

!======================================================================
! 氷河(全プロセス有効)
!======================================================================
&list_glacier
  dt_glacier_c = "1 hour"
  gl_morfac = 500.0
  gl_tfirn_yr = 1.0e-3      ! 氷化を加速(スモークテスト用 ≈ 8.8 時間)
  gl_ddfi = 8.0             ! 氷面度日係数 (mm/℃/day)
  f_glflow = 1
  gl_afl = 1.0e-16          ! Glen A (Pa^-3 yr^-1)
  f_glslide = 1
  gl_as = 1.0e-13           ! Weertman 滑動係数 (m yr^-1 Pa^-3)
  f_glero = 1
  gl_kg = 1.0e-4            ! 侵食係数(l=1 で無次元)
  gl_lexp = 1.0
  f_glava = 1
  gl_ava_tanc = 0.30        ! 雪崩の限界勾配(円錐斜面 0.35 が対象になる)
/
