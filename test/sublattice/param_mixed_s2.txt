!======================================================================
! sublattice: 運動量保存形移流の斜め副格子(市松)モードの減衰率ベンチマーク
!   平坦・無摩擦・閉じた正方水槽(1 km × 1 km、Δx = Δy = 10 m、h0 = 5 m)に
!   1 セル渦 ψ = A sin(πx/L) sin(πy/L)(最大流速 V0 = 0.2 m/s)を
!   (−1)^(i+j) で偶奇の副格子ごとに逆回転にして置く(user_initial
!   `gyre_alternating`)。法線エッジの初期流速は両セル平均で 0 になり、斜め
!   エッジだけが副格子ごとの渦を持つ。この「斜め副格子モード」は発散ゼロで
!   水位を変えず、移流項(運動量保存形)は同じ偶奇のセル・リンクだけを見る
!   ため作用しない。減衰は数値散逸(スキーム依存)だけで、無摩擦なので
!   その減衰率を直接測れる(developer.md §68.36)。
!   (param_mixed.txt: 滑らかな渦(振幅 1)に副格子モード(振幅 0.5)を重ねる。
!   背景流の流束で副格子モードが線形に減衰する率を測る = 実ケースの状況)スキーム比較: param_s1.txt / param_s2.txt。
!   判定は Check_sublattice.py(初期パターンへの射影振幅の時間変化)。
!   Cn = √(g h0)·dt/Δx = 0.35。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  f_govequation = 0         ! 基礎方程式(0:DynWE, 1:DifWE, 2:KinWE)
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 3600                 ! 計算終了時刻 (s)
  dt_disp = 300             ! 画面表示時間刻み (s)
  dt_file = 300             ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_enc = '-'
  dir_result = 'result_mixed_s2'
  f_check_cfl = 0
  f_out_u = 1               ! x 方向流速の分布(Check_sublattice.py が読む)
  f_out_v = 1               ! y 方向流速の分布
/

!======================================================================
! ENC計算条件(既定 = 運動量保存形+MUSCL。拡散項なし)
!======================================================================
&list_enc
  f_advection_scheme = 2
/

!======================================================================
! 初期条件設定(静水深 5 m に副格子ごとに逆回転の渦を重ねる)
!======================================================================
&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 5.0                  ! 初期水深固定値 (m)
  f_user_routine = "gyre_mixed"
/

!======================================================================
! 地理情報設定(平坦・無摩擦。dx = dy = 10 m。四周は壁)
!======================================================================
&list_geoinfo
  lx = 1000.0
  ly = 1000.0
  nx = 100
  ny = 100
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値(m)
  rn0 = 0.0       ! 粗度係数固定値(無摩擦)
/
