!======================================================================
! splash: 乾式斜面侵食(f_splash。無流水の雨滴・サブグリッドリル)の
!         解析検証・回帰テスト(developer.md §48)
!   遷緩点付き斜面(slope_break: tanθ=0.4, 遷緩点 x=30m)に一様降雨
!   30 mm/h。浸透能(Green-Ampt ksv=72 mm/h)が降雨を上回るため
!   地表流は生じず(全セル乾燥のまま)、侵食は f_splash のみが担う。
!   凹地形増幅は無効(spl_ca = spl_cb = 0)なので侵食深は解析的:
!     Δz = P・(spl_kr + spl_kt・S)・poroi・T_geo(porosity=0 → poroi=1)
!     斜面部(S=0.4): Δz = (30/3.6e6)・0.01・18000 = 1.5e-3 m
!     平坦部(S=0):   Δz = (30/3.6e6)・0.003・18000 = 4.5e-4 m
!   (T_geo = tt・morfac = 3600・5 = 18000 s。境界列 i=1, nx と遷緩点
!    近傍は鏡像・曲率の影響があるため解析対象は内部セル)
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  f_govequation = 0         ! 基礎方程式(0:DynWE)
  dt = 1.0                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 3600                 ! 計算終了時刻 (s)
  dt_disp = 600             ! 画面表示時間刻み (s)
  dt_file = 3600            ! ファイル出力時間刻み (s)
  dt_recrd = 3600           ! プローブ出力時間刻み (s)
  fn_geoinfo = '-'
  fn_precip = '-'
  fn_gwflow = '-'
  fn_geomorph = '-'
  fn_enc = '-'
  dir_result = 'result'
  f_out_z = 1               ! 地盤高の出力(解析検証は Z9998 を読む)
  f_check_cfl = 0
/

&list_enc
/

!======================================================================
! 地理情報設定(dx = dy = 1 m。地形はユーザールーチン slope_break)
!======================================================================
&list_geoinfo
  lx = 50.0
  ly = 12.0
  nx = 50
  ny = 12
  f_ztype = 0     ! 地盤高タイプ(0:固定値。ユーザールーチンが上書き)
  f_rntype = 0    ! 粗度係数タイプ(0:固定値)
  z0 = 0.0
  rn0 = 0.03
  sd0 = 5.0       ! 土層厚 (m)。総侵食深(~1.5e-3 m)より十分厚い
  sy0 = 0.2       ! 比湧水量(浸透水の貯留 0.03 m < sd0*sy0 = 1 m)
  f_user_routine = "slope_break"   ! tanθ=0.4, 遷緩点 x=30m
/

!======================================================================
! 降雨設定(一様 30 mm/h)
!======================================================================
&list_precip
  prtype = 1
  prval(1:2,1) = 0, 30
  prval(1:2,2) = 9999, 30
/

!======================================================================
! 地下水設定(Green-Ampt。ksv=72 mm/h > 降雨強度 → 地表流なし)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! 鉛直モデル(2:Green-Ampt)
  f_gwlateral = 0           ! 側方なし
/
&list_gwflow_greenampt
  gw_ksv_mmh = 72.0         ! 2e-5 m/s
/

!======================================================================
! 地形変化設定(乾式斜面侵食のみ。凹地形増幅は無効=解析検証)
!======================================================================
&list_geomorph
  dt_geomorph = 10.0        ! 更新間隔 (s)
  morfac = 5.0              ! 加速係数(T_geo = 18000 s)
  f_splash = 1              ! 乾式斜面侵食有効
  spl_kr = 0.003            ! 雨滴侵食係数
  spl_kt = 0.0175           ! サブグリッドリル侵食係数
  spl_h = 1.0               ! 勾配べき指数
  spl_ca = 0.0              ! 凹地形増幅(雨滴項)無効
  spl_cb = 0.0              ! 凹地形増幅(リル項)無効
  fluv_porosity = 0.0       ! 空隙率(poroi = 1 で解析式が単純)
/
