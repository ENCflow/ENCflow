!======================================================================
! pump: 井戸揚水シンクの模式実験(線形水位低下の解析検証)
!
!   平坦・閉領域 21×21(dx=dy=2, A=1764 m2)。初期表面水 h0 = 0.05 m が
!   Green-Ampt(psi_f=0 = 一定浸透能 K_sv=360 mm/h=1e-4 m/s)で土層
!   (sd=1.0, sy0=0.2, cap=0.2 m)へ浸透し、t=500 s で表面が乾く。
!   井戸1 = 全 441 セル(data_pump/wells.txt)へ Q1=0.01 m3/s を等分配、
!   井戸2 = 中央セル (11,11) へ Q2=2e-5 m3/s(インライン時系列の定数)。
!   側方流なし。全セルの hg は計算終了まで正(供給律速にならない)。
!   解析解: S_total(t) = 0.05 - (Q1+Q2)·t/A(取水はセルの涸れが
!   ない限り需要どおり)
!     S_total(3600) = 0.05 - 0.01002*3600/1764 = 0.02955102040816...
!   確認事項: S_total の線形低下と終値、S_surf が t=500 で 0、
!   np=1,2,4 の逐次一致(井戸セルが全帯に分散 = owner-compute の検証)。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 3600                 ! 計算終了時刻 (s)
  dt_disp = 300             ! 画面表示時間刻み (s)
  dt_file = 1800            ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_gwflow = '-'
  dir_data = 'data_pump'
  dir_result = 'result'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_hg = 1
/

&list_initial
  f_htype = 0               ! 初期水深タイプ (0: 固定値)
  h0 = 0.05                 ! 初期水深固定値 (m)
/

&list_geoinfo
  lx = 42.0
  ly = 42.0
  nx = 21
  ny = 21
  f_ztype = 0     ! 地盤高タイプ (0:固定値)
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  z0 = 0.0        ! 地盤高固定値 (m)
  rn0 = 0.05      ! 粗度係数固定値
  f_sdtype = 0    ! 土層厚タイプ (0:固定値)
  sd0 = 1.0       ! 土層厚固定値 (m)
  sy0 = 0.2       ! 比湧水量(有効間隙率)
/

&list_bound
/

!======================================================================
! 地下水設定(鉛直 Green-Ampt+井戸揚水。側方なし)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! 鉛直モデル(2:Green-Ampt)
  f_gwlateral = 0           ! 側方なし
  f_gwpump = 1              ! 井戸揚水
/

&list_gwflow_greenampt
  gw_ksv_mmh = 360.0        ! 1e-4 m/s
  gw_psif = 0.0             ! 一定浸透能に退化
/

&list_gwflow_pump
  fn_gwp_cell(1) = 'wells.txt'   ! 井戸1: 全 441 セル(等分配)
  gwp_q0(1) = 0.01               ! 一定流量 (m3/s)

  gwp_cell(1:2,1,2) = 11, 11     ! 井戸2: 中央セル(インライン)
  gwp_val(1:2,1,2) = 0.0, 2e-5   ! 時系列の定数退化 (min, m3/s)
/
