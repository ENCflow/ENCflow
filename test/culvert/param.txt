!======================================================================
! culvert: カルバート流量則の 3 レジーム通過(排水試験)
!
!   平坦な上流槽 10×11 セル(dx=dy=10 m、A_up = 11000 m2)に初期水深
!   3.0 m を置き、盛土(i=11、z=10 m)を横断する 1 m×1 m のボックス
!   カルバート (10,6)→(12,6) だけで、下流列 i=12(水位規定 η=-999 の
!   完全排水口 = 常に乾燥)へ排水する。上流水深 h1 は 3D から下がり、
!   満管(流入口水没 h1 >= D)→ 遷移(管頂付近)→ 自由水面(堰)の順に
!   3 レジームを通過する。下流が乾いているので Q は h1 だけの関数で、
!   Log の S_surf から Q = -A_total dS/dt、h1 ≈ S_surf A_total / A_up が
!   復元できる(Fig.py)。
!   確認事項: Q(h1) が単調(管頂直上の落ち込みがない。2026-10-12 の
!   下流水頭 (dc+D)/2 基準への改修)、満管での Q が C_p√(h1 - D) に、
!   堰支配での Q が 0.35 B h1 √(2g h1) に乗ること、np=1,2,4 の逐次一致。
!   A_total = 132 セル × 100 m2 = 13200 m2(盛土列・下流列を含む)。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC, 1:STG)
  dt = 0.5                  ! 時間刻み (s)
  t0 = 0                    ! 計算開始時刻 (s)
  tt = 14400                ! 計算終了時刻 (s)
  dt_disp = 120             ! 画面表示時間刻み (s)
  dt_file = 14400           ! ファイル出力時間刻み (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_boundary = '-'
  fn_structure = '-'
  dir_data = 'data_culvert'
  dir_result = 'result'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
/

&list_initial
  f_htype = 1               ! 初期水深タイプ (1: ファイル)
  fn_hinit = 'hinit.txt'    ! 上流槽 3.0 m、その他 0
/

&list_geoinfo
  lx = 120.0
  ly = 110.0
  nx = 12
  ny = 11
  f_ztype = 1     ! 地盤高タイプ (1:ファイル)
  fn_z = 'z.txt'  ! 上流槽 0 / 盛土 10 / 下流列 0
  f_rntype = 0    ! 粗度係数タイプ (0:固定値)
  rn0 = 0.03      ! 粗度係数固定値
/

!======================================================================
! 境界: 下流列 i=12 を完全排水口(水位規定 η=-999)。四周は壁(既定)
!======================================================================
&list_bound_edge
/

&list_bound_stage
  fn_stage_cell(1) = 'stage.txt'   ! (12, 1..11)
  stage_eta(1) = -999.0            ! 河床より低い水位 = 完全排水口
/

!======================================================================
! 構造物: 盛土を横断するボックスカルバート(摩擦なし、流入損失 0.5)
!======================================================================
&list_struct_culvert
  culv_in_cell(:,1,1)  = 10, 6     ! 上流側セル(上流槽の東端)
  culv_out_cell(:,1,1) = 12, 6     ! 下流側セル(排水口列)
  culv_width(1)  = 1.0             ! 断面幅 B (m)
  culv_height(1) = 1.0             ! 断面高 D (m)
  culv_zin(1)  = 0.0               ! 上流側敷高 (m)
  culv_zout(1) = 0.0               ! 下流側敷高 (m)
  culv_length(1) = 0.0             ! 管路長 (m。0 = 摩擦損失なし)
  culv_ce(1) = 0.5                 ! 流入損失係数(既定と同じ)
/
