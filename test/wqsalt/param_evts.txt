!======================================================================
! wqsalt / evts: 可能蒸発散の時系列入力(f_evmodel = 5)
!   平らな閉じた水域(10 × 10 セル、水深 1 m)。可能蒸発散を時系列 evap_val で直接与える:
!   0 → 96 mm/day(0.25 日)→ 0(0.5 日)の三角形。周期強制 t_cycle = 0.5 日で折り返すので、
!   1 日の計算では同じ三角形が 2 回繰り返される。1 周期の蒸発量は 0.5 × 0.5 日 × 96 mm/day =
!   24 mm、1 日で 48 mm(折れ点が時間刻みに乗るので、ステップごとの和が積分に一致する)。
!======================================================================
&list_sysparam
  dt = 60.0
  tt_c = "1d"
  t_cycle_c = "12h"         ! 強制の反復周期(蒸発散の時系列も折り返す)
  dt_disp_c = "3h"
  dt_file_c = "1d"
  dt_recrd_c = "3h"
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_evap = '-'
  dir_result = 'result_evts'
  f_output_mode = 2
  f_disp_debug = 1
  f_check_cfl = 0           ! 水が動かない検定なので、クーラン数の検査を外す
/

&list_geoinfo
  nx = 10
  ny = 10
  dx = 10.0
  dy = 10.0
  f_ztype = 0
  z0 = 0.0
  rn0 = 0.03
/

&list_initial
  f_htype = 0
  h0 = 1.0                  ! 初期水深 (m)
/

&list_evap
  f_evmodel = 5             ! 時系列で直接与える
  evap_val(1:2,1) = 0.0,   0.0     ! (経過日, mm/day)
  evap_val(1:2,2) = 0.25, 96.0
  evap_val(1:2,3) = 0.5,   0.0
/
