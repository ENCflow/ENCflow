!======================================================================
! wqsalt / mix: 水平の混合係数 wq_diff の検定(静水の中の物質の広がり)
!   平らな閉じた水域(81 × 81 セル、dx = 10 m、水深 1 m、流れなし)の中心に、
!   物質を 10 秒だけ投入する。水は動かないので、物質は混合係数 D = 0.5 m2/s の
!   拡散だけで広がる。雲が壁に届かない間、分布の分散(x・y 方向)は 1 秒あたり
!   2D ずつ増え、x と y の共分散は 0 のまま(8 近傍の離散化でも丸めの範囲で
!   厳密。developer.md §139)。質量は投入の後は変わらない。
!======================================================================
&list_sysparam
  dt = 1.0                  ! 時間刻み (s)
  tt = 600.0                ! 計算終了時刻 (s)
  dt_disp = 100.0           ! Log.txt の間隔 (s)
  dt_file = 200.0           ! 分布出力の間隔 (s)。200 s と 600 s の分布で分散の増え方を測る
  dt_recrd = 100.0          ! wq.csv の間隔 (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_wq = '-'
  dir_result = 'result_mix'
  f_output_mode = 2         ! 分布は bil(float32)
  f_disp_debug = 1
/

&list_geoinfo
  nx = 81
  ny = 81
  dx = 10.0
  dy = 10.0
  f_ztype = 0               ! 地盤高一様(平ら)
  z0 = 0.0
  rn0 = 0.03
/

&list_initial
  f_htype = 0
  h0 = 1.0                  ! 初期水深 (m)。全域で静水
/

&list_wq
  wq_diff = 0.5             ! 水平の混合係数 D (m2/s)
  wq_pt_cell(:,1,1) = 41, 41          ! 中心のセルに投入
  wq_pt_series(:,1,1) = 0.0, 1000.0   ! (経過日, g/s)。10 秒だけ投入して止める
  wq_pt_series(:,2,1) = 1.1574074e-4, 1000.0
  wq_pt_series(:,3,1) = 1.2731481e-4, 0.0
  wq_pt_series(:,4,1) = 1.0, 0.0
/
