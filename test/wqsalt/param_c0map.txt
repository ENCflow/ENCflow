!======================================================================
! wqsalt / c0map: 初期濃度の分布 fn_wq_c0
!   閉じた箱(40 × 20 セル、dx = 10 m)の列 21 に高さ 2 m の乾いた尾根を置き、両側に水位 1 m の
!   静水を張る。初期濃度を分布ファイルで、尾根の西 100 mg/L・東 300 mg/L と与える。
!   水は動かず、水平の混合は乾いた尾根を通さないので、両側の濃度は与えた値のまま変わらない。
!======================================================================
&list_sysparam
  dt = 1.0
  tt = 600.0
  dt_disp = 200.0
  dt_file = 300.0
  dt_recrd = 200.0
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_wq = '-'
  dir_data = 'data_wqsalt'
  dir_result = 'result_c0map'
  f_output_mode = 2
  f_disp_debug = 1
/

&list_geoinfo
  nx = 40
  ny = 20
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z_ridge.txt'      ! 列 21 が高さ 2 m の尾根
  rn0 = 0.03
/

&list_initial
  f_htype = 2               ! 水位の固定値
  e0 = 1.0                  ! 両側の水位 1 m(尾根は乾いている)
/

&list_wq
  fn_wq_c0 = 'c0_ridge.txt' ! 初期濃度の分布 (mg/L)。尾根の西 100、東 300
  wq_diff = 20.0            ! 水平の混合係数 D (m2/s)
/
