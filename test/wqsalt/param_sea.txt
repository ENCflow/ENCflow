!======================================================================
! wqsalt / sea: 海から陸へ入る水の濃度 wq_sea_conc
!   西端の 1 列が海域マスクの海(24 × 8 セル、dx = 10 m)。陸は標高 1 m の平らな地面で、
!   はじめは乾いている。潮位を 5 分で 0 → 2 m に上げて 10 分保ち、5 分で 1.3 m に下げる。
!   満ち潮で陸に入る水は濃度 35 g/L(wq_sea_conc)を持つので、陸の水はどこでも 35 g/L。
!   引き潮で海へ戻る水が持ち出す分は台帳の out_sea_g、海から入った分は in_sea_g
!   (wq_sea_conc を指定した計算だけ wq.csv の末尾に加わる)で、差が陸の水の質量になる。
!======================================================================
&list_sysparam
  dt = 0.2
  tt = 1800.0
  dt_disp = 300.0
  dt_file = 600.0
  dt_recrd = 60.0
  fn_geoinfo = '-'
  fn_tide = '-'
  fn_wq = '-'
  dir_data = 'data_wqsalt'
  dir_result = 'result_sea'
  f_output_mode = 2
  f_disp_debug = 1
/

&list_geoinfo
  nx = 24
  ny = 8
  dx = 10.0
  dy = 10.0
  f_ztype = 0
  z0 = 1.0                  ! 陸の地盤高 (m)
  rn0 = 0.03
  f_masktype = 2            ! 領域マスクを海域マスクから作る
  fn_sw = 'sw_sea.txt'      ! 海域マスク(西端の 1 列が海)
/

&list_tide
  titype = 2                ! 一様な潮位の時系列
  hsea0 = 1.0               ! 海セルの見かけの水柱の厚さ (m)
  dt_tiupdate = 0.1         ! 潮位の更新間隔 (min)
  tival(1:2,1) =  0, 0.0    ! (min, m)
  tival(1:2,2) =  5, 2.0
  tival(1:2,3) = 15, 2.0
  tival(1:2,4) = 20, 1.3
  tival(1:2,5) = 30, 1.3
/

&list_wq
  wq_sea_conc = 35000.0     ! 海から入る水の濃度 (mg/L) = 35 g/L
/
