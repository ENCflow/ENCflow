!======================================================================
! wqsalt / inf: 浸透する雨が析出物を溶かして地下へ運ぶ。地下水と一緒に動く濃度の上限
!   平らな閉じた領域(10 × 10 セル)。地表は乾いていて、初期の析出物(表面蓄積プール)が
!   西半分に 100 kg/m2、東半分に 0.5 kg/m2 ある。36 mm/h の雨を 1 時間降らせ(その後の
!   1 分で 0 へ下げる分を含めて 36.5 mm)、バケツ型の浸透ですべてしみ込ませる。浸透水は
!   飽和(溶解度 300 g/L)まで析出物を溶かして地下の質量プールへ運ぶので、西は
!   300 g/L × 36.5 mm = 10.95 kg/m2(析出物は 89.05 kg/m2 残る)、東は 0.5 kg/m2 の全量が
!   地下へ入る。
!   2〜3 時間の間、西と東の 1 セルずつから井戸で地下水を半分(18 mm)汲む。遅延化係数
!   R = 2 なので、汲む水が運ぶのは濃度の 1/2。西は地下水の濃度がちょうど溶解度で、水だけが
!   先に減ると濃度が溶解度を超えるが、水と一緒に動く濃度は溶解度で頭打ちなので、運び出すのは
!   300 g/L × 1/2 × 18 mm = 2.7 kg/m2。東は溶解度より薄いので、濃度に比例して運び出す。
!======================================================================
&list_sysparam
  dt = 10.0
  tt_c = "4h"
  dt_disp_c = "30m"
  dt_file_c = "4h"
  dt_recrd_c = "30m"
  fn_geoinfo = '-'
  fn_precip = '-'
  fn_gwflow = '-'
  fn_wq = '-'
  dir_data = 'data_wqsalt'
  dir_result = 'result_inf'
  f_output_mode = 2
  f_out_hg = 1              ! 地下貯留 Hg と地下の質量プール Cg を出力
  f_disp_debug = 1
  f_check_cfl = 0           ! 水が動かない検定なので、クーラン数の検査を外す
/

&list_geoinfo
  nx = 10
  ny = 10
  dx = 10.0
  dy = 10.0
  f_ztype = 0
  z0 = 0.0
  rn0 = 0.03
/

&list_precip
  prtype = 1
  dt_prupdate = 1
  prval(:,1) =    0, 36.0   ! (min, mm/h)。1 時間 36 mm/h、次の 1 分で 0 へ
  prval(:,2) =   60, 36.0
  prval(:,3) =   61,  0.0
  prval(:,4) = 9999,  0.0
/

&list_gwflow
  f_gwvertical = 1          ! バケツ型(側方流なし)
  f_gwpump = 1              ! 井戸揚水
/

&list_gwflow_bucket
  gw_infil_mmh = 200.0      ! 浸透能 (mm/h)。雨はすべてしみ込む
  gw_capacity = 0.2         ! 容量 (m)
/

&list_gwflow_pump
  gwp_cell(1:2,1,1) = 3, 5            ! 井戸 1: 西半分の 1 セル
  gwp_val(1:2,1,1) =   0.0, 0.0       ! (分, m3/s)。2〜3 時間に 5e-4 m3/s = 18 mm を汲む
  gwp_val(1:2,2,1) = 120.0, 0.0
  gwp_val(1:2,3,1) = 120.01, 5.0e-4
  gwp_val(1:2,4,1) = 180.0, 5.0e-4
  gwp_val(1:2,5,1) = 180.01, 0.0
  gwp_val(1:2,6,1) = 9999.0, 0.0
  gwp_cell(1:2,1,2) = 8, 5            ! 井戸 2: 東半分の 1 セル
  gwp_val(1:2,1,2) =   0.0, 0.0
  gwp_val(1:2,2,2) = 120.0, 0.0
  gwp_val(1:2,3,2) = 120.01, 5.0e-4
  gwp_val(1:2,4,2) = 180.0, 5.0e-4
  gwp_val(1:2,5,2) = 180.01, 0.0
  gwp_val(1:2,6,2) = 9999.0, 0.0
/

&list_wq
  wq_csat = 300000.0        ! 溶解度 (mg/L) = 300 g/L
  wq_rg = 2.0               ! 地下輸送の遅延化係数 R
  fn_wq_bd0 = 'pool_inf.txt'   ! 初期の析出物 (kg/ha)。西 1e6(100 kg/m2)、東 5e3(0.5 kg/m2)
/
