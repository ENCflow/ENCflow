!======================================================================
! wqsalt / mixwd: サブグリッド河道幅のセルをまたぐ水平の混合
!   平らな閉じた箱(24 × 12 セル、dx = 10 m)に水位 1 m の静水を張る。列 13 を幅 2 m の
!   サブグリッド河道(平面積率 0.2。河道の両端のセルは 0.1)にして、箱を西と東に分ける。西側の
!   1 セルに物質を 10 秒だけ投入する。物質は河道のセルを通って東へ広がり、質量を保存したまま、
!   最後は濃度が一様になる(濃度 = 投入量 ÷ 水の体積 27820 m3)。
!======================================================================
&list_sysparam
  dt = 1.0
  tt = 43200.0
  dt_disp = 10800.0
  dt_file = 21600.0
  dt_recrd = 10800.0
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_wq = '-'
  fn_channel = '-'
  dir_data = 'data_wqsalt'
  dir_result = 'result_mixwd'
  f_output_mode = 2
  f_disp_debug = 1
/

&list_geoinfo
  nx = 24
  ny = 12
  dx = 10.0
  dy = 10.0
  f_ztype = 0               ! 地盤高一様(平ら)
  z0 = 0.0
  rn0 = 0.03
  fn_rw = 'rw_col.txt'      ! 河道マスク(列 13)
/

&list_channel
  fn_width = 'width_col.txt'  ! サブグリッド河道幅 2 m(平面積率 0.2)
/

&list_initial
  f_htype = 2               ! 水位の固定値
  e0 = 1.0                  ! 水位 1 m の静水
/

&list_wq
  wq_diff = 20.0            ! 水平の混合係数 D (m2/s)
  wq_pt_cell(:,1,1) = 4, 6            ! 西側のセルに投入
  wq_pt_series(:,1,1) = 0.0, 1000.0   ! (経過日, g/s)。10 秒だけ(計 10 kg)
  wq_pt_series(:,2,1) = 1.1574074e-4, 1000.0
  wq_pt_series(:,3,1) = 1.2731481e-4, 0.0
  wq_pt_series(:,4,1) = 1.0, 0.0
/
