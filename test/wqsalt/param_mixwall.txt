!======================================================================
! wqsalt / mixwall: 乾いた尾根は水平の混合を通さない
!   閉じた箱(40 × 20 セル、dx = 10 m)の列 21 に高さ 2 m の尾根を置き、両側に水位 1 m の
!   静水を張る(尾根は水面の上で乾いている)。西側の 1 セルに物質を 10 秒だけ投入する。
!   混合は水のあるセルの間だけに働くので、物質は西側の水域の中だけに広がって一様に近づき、
!   東側の濃度は厳密に 0 のまま。
!======================================================================
&list_sysparam
  dt = 1.0
  tt = 7200.0
  dt_disp = 1200.0
  dt_file = 3600.0
  dt_recrd = 1200.0
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_wq = '-'
  dir_data = 'data_wqsalt'
  dir_result = 'result_mixwall'
  f_output_mode = 2
  f_disp_debug = 1
/

&list_geoinfo
  nx = 40
  ny = 20
  dx = 10.0
  dy = 10.0
  f_ztype = 1
  fn_z = 'z_ridge.txt'      ! 列 21 が高さ 2 m の尾根
  rn0 = 0.03
/

&list_initial
  f_htype = 2               ! 水位の固定値
  e0 = 1.0                  ! 両側の水位 1 m(尾根は乾いている)
/

&list_wq
  wq_diff = 20.0            ! 水平の混合係数 D (m2/s)。西側の水域(200 m)が 2 時間でほぼ一様になる
  wq_pt_cell(:,1,1) = 5, 10           ! 西側のセルに投入
  wq_pt_series(:,1,1) = 0.0, 1000.0   ! (経過日, g/s)。10 秒だけ
  wq_pt_series(:,2,1) = 1.1574074e-4, 1000.0
  wq_pt_series(:,3,1) = 1.2731481e-4, 0.0
  wq_pt_series(:,4,1) = 1.0, 0.0
/
