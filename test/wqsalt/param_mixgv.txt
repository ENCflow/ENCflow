!======================================================================
! wqsalt / mixgv: 家屋の空隙率の違うセルの間の水平の混合
!   平らな閉じた箱(24 × 12 セル、dx = 10 m)に水位 1 m の静水を張る。家屋の空隙率を西半分 1.0、
!   東半分 0.5 にする(東半分は水の入る面積が半分)。西側の 1 セルに物質を 10 秒だけ投入する。
!   混合の流束は濃度の差に比例し、リンクの両側から同じ値で数えるので、質量を保存し、最後は
!   空隙率によらず濃度が一様になる(濃度 = 投入量 ÷ 水の体積 21600 m3)。
!======================================================================
&list_sysparam
  dt = 1.0
  tt = 14400.0
  dt_disp = 3600.0
  dt_file = 7200.0
  dt_recrd = 3600.0
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_wq = '-'
  dir_data = 'data_wqsalt'
  dir_result = 'result_mixgv'
  f_output_mode = 2
  f_disp_debug = 1
/

&list_geoinfo
  nx = 24
  ny = 12
  dx = 10.0
  dy = 10.0
  f_ztype = 0               ! 地盤高一様(平ら)
  z0 = 0.0
  rn0 = 0.03
  fn_gv = 'gv_half.txt'     ! 家屋の空隙率(西半分 1.0、東半分 0.5)
/

&list_initial
  f_htype = 2               ! 水位の固定値
  e0 = 1.0                  ! 水位 1 m の静水
/

&list_wq
  wq_diff = 20.0            ! 水平の混合係数 D (m2/s)。箱(240 m)が 4 時間で一様になる
  wq_pt_cell(:,1,1) = 4, 6            ! 西側のセルに投入
  wq_pt_series(:,1,1) = 0.0, 1000.0   ! (経過日, g/s)。10 秒だけ(計 10 kg)
  wq_pt_series(:,2,1) = 1.1574074e-4, 1000.0
  wq_pt_series(:,3,1) = 1.2731481e-4, 0.0
  wq_pt_series(:,4,1) = 1.0, 0.0
/
