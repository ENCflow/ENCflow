!======================================================================
! wqsalt / sat: 蒸発で濃くなった水から、溶解度を超えた分が析出する
!   平らな閉じた水域(10 × 10 セル、dx = 10 m)に、水深 0.5 m・濃度 100 g/L の水を張り、
!   2400 mm/day(0.1 m/h)で蒸発させる。水は 5 時間で干上がる。溶解度 wq_csat = 300 g/L なので、
!   水深が 0.5 × 100 / 300 = 0.1667 m を下回る 3.33 時間から析出が始まり、水柱の物質は
!   「溶解度 × 水深」、残りは表面蓄積プール(析出物)になる。干上がると全量が析出物になる。
!   水は動かないので、台帳(wq.csv)の値が式と一致する。
!======================================================================
&list_sysparam
  dt = 10.0
  tt_c = "6h"
  dt_disp_c = "30m"
  dt_file_c = "3h"
  dt_recrd_c = "10m"
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_evap = '-'
  fn_wq = '-'
  dir_result = 'result_sat'
  f_output_mode = 2
  f_disp_debug = 1
  f_check_cfl = 0           ! 水が動かない検定なので、クーラン数の検査を外す
/

&list_geoinfo
  nx = 10
  ny = 10
  dx = 10.0
  dy = 10.0
  f_ztype = 0
  z0 = 0.0
  rn0 = 0.03
/

&list_initial
  f_htype = 0
  h0 = 0.5                  ! 初期水深 (m)
/

&list_evap
  f_evmodel = 1             ! 一定速度
  evap0 = 2400.0            ! 蒸発 (mm/day) = 0.1 m/h
/

&list_wq
  wq_c0 = 100000.0          ! 初期濃度 (mg/L) = 100 g/L
  wq_csat = 300000.0        ! 溶解度 (mg/L) = 300 g/L
/
