!======================================================================
! gwbed: 河床交換能(f_gwbed)の解析検証 — 構成 1「堤内地湛水から河道への
!        地下水流入(得水)」
!
!   2 列 × 11 行(dx=dy=2 m)。行は [堤内地, 河道, 堤内地, 領域外] の
!   3 セル縦列を領域外行で区切って 3 組並べる(全組が同一 = MPI の
!   帯分割の検査にも使う)。河道セル(rw=1)は z = −3 m、堤内地は 0 m。
!   堤防(bank0=2、堤内地基準 → 天端 +2 m)が地表水を隔てるので、地表の
!   流れは起きず、地表水の移動は地下経路だけになる。初期水深 1.2 m 一様。
!
!   Green-Ampt(K_sv = 1e-2 m/s、psif=0)で堤内地は 20 s で飽和し
!   (容量 sd·sy = 0.2 m)、毎ステップ失った分だけ地表水から補充されて
!   側方計算の時点では常に飽和(界面飽和帯厚 h_e = sd = 1 m)。河道セルは
!   鉛直 k_bed と側方流入で約 200 s で飽和し、以後は流入分がそのまま地表へ
!   湧出する。堤内地→河道の水頭差は常に ≥ 0.9 m で Darcy 流量が上限を
!   大きく超えるため、エッジ流量は厳密に上限
!     q_k = k_bed · l_k · h_e      (k_bed = 1e-3 m/s)
!   に等しい。各堤内地セルは軸 1 + 対角 1 の 2 エッジで河道に接し、
!   各河道セルは 4 エッジで受けるので、t ≥ 300 s では
!     堤内地の地表水深:  dh/dt = −k_bed·sd·(l_ax + l_diag)/A   = −4.142e-4 m/s
!     河道の地表水深:    dh/dt = +k_bed·sd·2(l_ax + l_diag)/A  = +8.284e-4 m/s
!     河道の Hg = 0.2(飽和)、堤内地の Hg = 0.2 − 1 ステップ分の損失(定数)
!   (l_ax = (1−r)·dy、l_diag = (r/2)·√(dx²+dy²)、r = 2/(2+√2)、A = dx·dy)
!   確認事項: プローブ(河道 (1,2)・堤内地 (1,1))の h の増分と Hg、
!   Log の S_grnd 一定・S_total = 1.2 不変、np=1,2,4 の逐次一致。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 0.25                 ! 時間刻み (s)。側方流の安定上限は 1.41 s
  t0 = 0
  tt = 1800
  dt_disp = 100
  dt_file = 900
  dt_recrd = 100            ! プローブ出力間隔 (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_gwflow = '-'
  fn_channel = '-'
  fn_record = '-'
  dir_data = 'data_gwbed'
  dir_result = 'result'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_hg = 1
/

&list_initial
  f_htype = 0               ! 一様水深
  h0 = 1.2                  ! 堤内地 1.2 m(飽和に 0.2 m、残り 1.0 m が 1800 s で 0.75 m 流入)
/

&list_geoinfo
  nx = 2
  ny = 11
  dx = 2.0
  dy = 2.0
  f_ztype = 1
  fn_z = 'z.txt'            ! 堤内地 0 m、河道 −3 m
  f_masktype = 1
  fn_mask = 'mask.txt'      ! 3 セル縦列を区切る行(4, 8)は領域外
  fn_rw = 'rw.txt'          ! 河道マスク(行 2, 6, 10)
  f_rntype = 0
  rn0 = 0.03
  f_sdtype = 0
  sd0 = 1.0                 ! 土層厚 (m)
  sy0 = 0.2                 ! 比湧水量
/

&list_bound
/

!======================================================================
! 堤防(地表水を隔てる仮想壁。天端 = 堤内地 0 m + 2 m > 水位 1.2 m)
!======================================================================
&list_channel
  bank0 = 2.0
  f_bank_datum = 1
/

!======================================================================
! プローブ(1: 河道セル、2: 堤内地セル)
!======================================================================
&list_record
  pbxytype = 0
  pbxy(:,1) = 1, 2
  pbxy(:,2) = 1, 1
/

!======================================================================
! 地下水(Green-Ampt + 側方 + 河床交換能)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! Green-Ampt
  f_gwlateral = 1           ! 非線形 Boussinesq
  f_gwbed = 1               ! 河床交換能
  dt_gwflow = 0
/

&list_gwflow_greenampt
  gw_ksv_mmh = 36000.0      ! 1e-2 m/s(堤内地を即座に飽和・毎ステップ補充)
  gw_psif = 0.0
/

&list_gwflow_lateral
  gw_ksh_mmh = 360000.0     ! 1e-1 m/s(Darcy 流量 ≫ 上限 となるように大きく)
/

&list_gwflow_bed
  gwbed_k_mmh = 3600.0      ! k_bed = 1e-3 m/s
/
