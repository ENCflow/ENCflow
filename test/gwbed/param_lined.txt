!======================================================================
! gwbed: 河床交換能(f_gwbed)の解析検証 — 構成 2「ライニング水路
!        (k_bed = 0)」
!
!   param.txt(構成 1)と同一の配置・初期条件で k_bed = 0 とする。
!   河道セルは鉛直浸透も河道・堤内地間の側方交換も持たないので、河道の
!   Hg = 0・h = 1.2 m が全時間で厳密に不変、堤内地は飽和(Hg = 0.2)した
!   のち変化しない(S_grnd = 12 × 0.2 / 18 一定、S_total = 1.2 不変)。
!   k_bed なし(f_gwbed = 0)なら堤内地の地下水が河道へ流入して湧出する
!   ので、この不変性が「河床交換能による遮断」の検定になる。
!======================================================================
&list_sysparam
  f_gridsystem = 0          ! 格子システム(0:ENC)
  dt = 0.25                 ! 時間刻み (s)。側方流の安定上限は 1.41 s
  t0 = 0
  tt = 1800
  dt_disp = 100
  dt_file = 900
  dt_recrd = 100            ! プローブ出力間隔 (s)
  fn_geoinfo = '-'
  fn_initial = '-'
  fn_gwflow = '-'
  fn_channel = '-'
  fn_record = '-'
  dir_data = 'data_gwbed'
  dir_result = 'result_lined'
  f_disp_debug = 1          ! 回帰テスト用: S 列を全有効桁で表示
  f_check_cfl = 0
  f_out_hg = 1
/

&list_initial
  f_htype = 0               ! 一様水深
  h0 = 1.2                  ! 堤内地 1.2 m(飽和に 0.2 m、残り 1.0 m が 1800 s で 0.75 m 流入)
/

&list_geoinfo
  nx = 2
  ny = 11
  dx = 2.0
  dy = 2.0
  f_ztype = 1
  fn_z = 'z.txt'            ! 堤内地 0 m、河道 −3 m
  f_masktype = 1
  fn_mask = 'mask.txt'      ! 3 セル縦列を区切る行(4, 8)は領域外
  fn_rw = 'rw.txt'          ! 河道マスク(行 2, 6, 10)
  f_rntype = 0
  rn0 = 0.03
  f_sdtype = 0
  sd0 = 1.0                 ! 土層厚 (m)
  sy0 = 0.2                 ! 比湧水量
/

&list_bound
/

!======================================================================
! 堤防(地表水を隔てる仮想壁。天端 = 堤内地 0 m + 2 m > 水位 1.2 m)
!======================================================================
&list_channel
  bank0 = 2.0
  f_bank_datum = 1
/

!======================================================================
! プローブ(1: 河道セル、2: 堤内地セル)
!======================================================================
&list_record
  pbxytype = 0
  pbxy(:,1) = 1, 2
  pbxy(:,2) = 1, 1
/

!======================================================================
! 地下水(Green-Ampt + 側方 + 河床交換能)
!======================================================================
&list_gwflow
  f_gwvertical = 2          ! Green-Ampt
  f_gwlateral = 1           ! 非線形 Boussinesq
  f_gwbed = 1               ! 河床交換能
  dt_gwflow = 0
/

&list_gwflow_greenampt
  gw_ksv_mmh = 36000.0      ! 1e-2 m/s(堤内地を即座に飽和・毎ステップ補充)
  gw_psif = 0.0
/

&list_gwflow_lateral
  gw_ksh_mmh = 360000.0     ! 1e-1 m/s(Darcy 流量 ≫ 上限 となるように大きく)
/

&list_gwflow_bed
  gwbed_k_mmh = 0.0         ! k_bed = 0 = 不透(河道セルは地下水と遮断される)
/
