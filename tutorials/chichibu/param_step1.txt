! システムパラメータ設定

&list_sysparam
  dt = 6.0                       ! 時間刻み (s)
  tt_c = "6 hour"                ! 計算終了時刻
  dt_disp_c = "30 min"           ! 画面表示時間刻み
  dt_file_c = "30 min"           ! ファイル出力時間刻み

  fn_geoinfo = "-"               ! 地理情報設定ファイル
  fn_precip  = "-"               ! 降水条件設定ファイル

  dir_data   = "data_chichibu"   ! 入力データディレクトリ

  fn_enc = "-"               ! ENC 条件設定ファイル(下の &list_enc をこのファイルから読む)
/

!======================================================================
! 地理情報設定
!======================================================================
&list_geoinfo
  nx = 280
  ny = 150
  dx = 200.0
  dy = 200.0

  f_ztype = 1          ! 地盤高タイプ (0:デフォルト固定値, 1:ファイル)
  f_masktype = 1       ! 領域マスクの有無 (0:なし, 1:流域マスク)
  rn0 = 0.05           ! 粗度係数固定値(山地斜面の目安。既定 0.015 は水路向け)

  fn_z = "Chichibu_200m_filled.txt"        ! 地盤高ファイル名
  fn_mask = "Chichibu_200m_basin.txt"      ! 領域マスクファイル名
/

!======================================================================
! 降水条件設定
!======================================================================
&list_precip
  ! 降水データのタイプ (0:降水なし, 1:一様分布で時系列)
  prtype = 1
  ! 降水の時系列 (時刻(min) 雨量強度(mm/h))
  prval(:,1) =   0,   0
  prval(:,2) =  30, 200
  prval(:,3) =  60,   0
/

! 塞がれた斜め開口の動的振り替え。この教材の param はすべて 2026-10-18〜31 の既定 2 を明示する。
! 2026-10-31 からの既定 3 では、流域出口を閉じて深く湛水させる Step 1〜5 と補足の構成で、5 時間
! 過ぎに湛水域の谷底で 18 m/s を超える流れが出てクーラン数が 1 を超え計算が止まる(2 では
! 止まらない。原因は調査中。developer.md §68.41)。教材の本文の数値と図は 2 で作られている。
&list_enc
  f_opening_dynamic = 2
/
