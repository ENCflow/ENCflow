!======================================================================
!
! Supplement: effect of resolution (200 m)
!       The Step 4 configuration (no losses, no outlet, dd = dv = 0.0001) moved to the 200 m grid.
!       Text input; distributed output is depth only, every hour
!
!======================================================================
! System parameter settings
!======================================================================
&list_sysparam
  f_govequation = 0          ! governing equation (0: DynWE, 1: DifWE, 2: KinWE)

  dt = 6.0    ! time step (s): 200 m: 6 s / 100 m: 1.5 s / 50 m: 0.75 s
                             ! (finer grids resolve steeper tributaries and faster flow, so dt
                             !  must shrink faster than the grid ratio (1/2, 1/4) to keep Cn below 1)
  tt_c = "6 hour"            ! end time of the computation
  dt_disp_c = "30 min"       ! screen display interval
  dt_file_c = "1 hour"       ! file output interval
  dt_recrd_c = "1 min"       ! probe output interval

  fn_geoinfo = "-"           ! geographic information settings file
  fn_precip  = "-"           ! precipitation settings file
  fn_record  = "-"           ! record settings file

  dir_data = "data_chichibu"   ! input data directory

  f_input_mode = 1           ! matrix input format (1: text, 2: bil, 4: geotiff)
  f_output_mode = 1          ! matrix output format (1: text, 2: bil, 4: geotiff)

  dd = 0.0001                ! water moves only above this depth (threshold depth)
  dv = 0.0001                ! depths below this are forced to this value (virtual depth)
/

!======================================================================
! Geographic information settings
!======================================================================
&list_geoinfo
  nx = 280
  ny = 150
  dx = 200.0
  dy = 200.0

  f_ztype = 1          ! ground elevation type (0: default constant, 1: file)
  f_masktype = 1       ! domain mask (0: none, 1: catchment mask)

  f_rntype = 0         ! roughness type (0: default constant, 1: file)
  rn0 = 0.05           ! fixed roughness
  rn0_rw = 0.02        ! fixed roughness on channel-mask cells

  fn_z = "Chichibu_200m_filled.txt"        ! ground elevation file name
  fn_mask = "Chichibu_200m_basin.txt"      ! domain mask file name
  fn_rw = "Chichibu_200m_river.txt"        ! channel mask file name
/

!======================================================================
! Precipitation settings
!======================================================================
&list_precip
  ! precipitation data type (0: no precipitation, 1: uniform, time series)
  prtype = 1
  ! precipitation time series (time (min), intensity (mm/h))
  prval(:,1) =   0,   0
  prval(:,2) =  30, 200
  prval(:,3) =  60,   0
/

!======================================================================
! Record settings (the 4 transects and 4 probes of Step 3 at the same places, as cell numbers of the 200 m grid)
!======================================================================
&list_record
  ! probe position type (0: cell indices, 1: real coordinates (m))
  ! probe positions (x, y)
  pbxytype = 0
  pbxy(:,1) =   82,  105
  pbxy(:,2) =  107,  102
  pbxy(:,3) =  147,  104
  pbxy(:,4) =  216,   39

  ! flux transect coordinate type (0: cell indices, 1: real coordinates (m))
  ! flux transect endpoints (right-bank x, right-bank y, left-bank x, left-bank y)
  flxytype = 0
  flxy(:,1) =   82,  107,    82,  103
  flxy(:,2) =  107,  103,   107,  100
  flxy(:,3) =  147,  105,   147,  102
  flxy(:,4) =  216,   43,   216,   37
/
