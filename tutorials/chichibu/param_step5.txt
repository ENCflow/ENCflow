!======================================================================
!
! Step 5: 降雨遮断と地下浸透の導入
!
!======================================================================
! システムパラメータ設定
!======================================================================
&list_sysparam
  f_govequation = 0          ! 基礎方程式(0:DynWE, 1:DifWE, 2:KinWE)

  dt = 6.0                   ! 時間刻み (s)
  tt_c = "6 hour"            ! 計算終了時刻
  dt_disp_c = "30 min"       ! 画面表示時間刻み
  dt_file_c = "30 min"       ! ファイル出力時間刻み
  dt_recrd_c = "1 min"       ! プローブ出力時間刻み

  fn_geoinfo = "-"           ! 地理情報設定ファイル
  fn_precip  = "-"           ! 降水条件設定ファイル
  fn_record  = "-"           ! 降水条件設定ファイル
  fn_enc     = "-"           ! ENC条件設定ファイル
  fn_intercept = "-"         ! 降雨遮断条件設定ファイル
  fn_gwflow = "-"            ! 地下水条件設定ファイル

  dir_data = "data_chichibu" ! 入力データディレクトリ

  f_input_mode = 4           ! matrix入力形式(1:text, 2:bil, 4:geotiff)
  f_output_mode = 5          ! matrix出力形式(1:text, 2:bil, 4:geotiff)

  f_out_h = 1                ! ファイル出力(水深H0001)
  f_out_vv = 1               ! ファイル出力(流速絶対値V0001)
  f_out_qq = 1               ! ファイル出力(流量絶対値Q0001)
  f_out_cn = 1               ! ファイル出力(クーラン数Cn0001)
  f_out_hmax = 1             ! ファイル出力(最大水深H9999)
  f_out_vvmax = 1            ! ファイル出力(最大流速V9999)

  dd = 0.0001                ! この水深以上なら水の移動を計算する(限界水深)
  dv = 0.0001                ! これ以下なら強制的にこの水深にする(仮想水深)
/

!======================================================================
! 地理情報設定
!======================================================================
&list_geoinfo
  dx = 200.0
  dy = 200.0

  f_ztype = 1          ! 地盤高タイプ (0:デフォルト固定値, 1:ファイル)
  f_masktype = 1       ! 領域マスクの有無 (0:なし, 1:流域マスク)

  f_rntype = 0         ! 粗度係数タイプ (0:デフォルト固定値, 1:ファイル, 2:土地利用から)
  rn0 = 0.05           ! 祖度係数固定値
  rn0_rw = 0.02        ! 河道マスク部の固定祖度係数

  fn_z = "Chichibu_200m_filled.tif"        ! 地盤高ファイル名
  fn_mask = "Chichibu_200m_basin.tif"      ! 領域マスクファイル名
  fn_rw = "Chichibu_200m_river.tif"        ! 河道マスクファイル名

  !f_sdtype = 0         ! 0: 固定値 sd0(既定), 1: ファイル fn_sd
  !sd0 = 1.0            ! 土層厚固定値 (m)
  !sy0 = 0.2            ! 比湧水量(有効間隙率)
/

!======================================================================
! 降水条件設定
!======================================================================
&list_precip
  ! 降水データのタイプ (0:降水なし, 1:一様分布で時系列)
  prtype = 1
  ! 降水の時系列 (時刻(min) 雨量強度(mm/h))
  prval(:,1) =   0,   0
  prval(:,2) =  30, 200
  prval(:,3) =  60,   0
/

!======================================================================
! 記録設定
!======================================================================
&list_record
  ! プローブの位置のタイプ (0:セル番号, 1:実座標(m))
  ! プローブの位置 (x座標, y座標)
  pbxytype = 0
  pbxy(:,1) =  82, 105
  pbxy(:,2) = 107, 102
  pbxy(:,3) = 147, 103
  pbxy(:,4) = 216,  40

  ! フラックス測線の座標のタイプ (0:セル番号, 1:実座標(m))
  ! フラックス測線の始点終点 (右岸x, 右岸y, 左岸x, 左岸y)
  flxytype = 0
  flxy(:,1) =  82, 106,   82, 103
  flxy(:,2) = 107, 103,  107, 100
  flxy(:,3) = 147, 105,  147, 102
  flxy(:,4) = 216,  43,  216,  37
/

!======================================================================
! ENC計算条件設定
!======================================================================
&list_enc
  p_adv_upwind_index = 0.0        ! 風上差分の割合 (0.0 ~ 1.0)
  p_adprunge_thresh = 1.5         ! 適応的ルンゲクッタの閾値 (1.1 ~)
/

!======================================================================
! 遮断共通設定
!======================================================================
&list_intercept
  ! 遮断モデルの選択
  !  0:なし(fn_intercept を書いたまま一時無効化できる)
  !  1:固定遮断率モデル(&list_intercept_fixed を使用)
  !  2:初期損失モデル(&list_intercept_initloss を使用)
  f_icmodel = 1
/

!======================================================================
! 固定遮断率モデル固有設定 (f_icmodel = 1)
!======================================================================
!   降雨の一定割合を遮断し、有効雨量 (1-α)P を地表に与える最簡モデル
!   (I = αP)。貯留容量・蒸発・降雨強度依存は表現しません。
!   遮断率は一様値または分布ファイルのどちらかで与えます。
!======================================================================
&list_intercept_fixed
  ! 遮断率の一様値。0 < ic_alpha < 1 であること
  ! (目安: 樹林地で 0.1〜0.4 程度)
  ic_alpha = 0.2
/

!======================================================================
! 地下水共通設定
!======================================================================
&list_gwflow
  ! 鉛直モデルの選択
  !  0:なし(fn_gwflow を書いたまま一時無効化できる)
  !  1:バケツモデル(&list_gwflow_bucket を使用)
  !  2:Green-Ampt モデル(&list_gwflow_greenampt を使用。
  !    &list_geoinfo の土層厚 sd・比湧水量 sy0 が必要)
  f_gwvertical = 1
/

!======================================================================
! バケツモデル固有設定 (f_gwvertical = 1)
!======================================================================
!   セルごとの地下貯留(容量 gw_capacity)へ一定浸透能 gw_infil_mmh で
!   表面水を移す最簡モデル。水平流なし・鉛直交換のみ。
!   各ステップの浸透量は「浸透能×dts・表面水深・貯留残容量」の最小値。
!   側方モデル(f_gwlateral=1)とは併用不可(容量の二重定義になるため
!   エラー停止。土層厚を容量とする定数浸透能が必要なら Green-Ampt の
!   gw_psif=0 を使うこと)。
!======================================================================
&list_gwflow_bucket
  ! 浸透能 (mm/h)。0 より大きいこと
  gw_infil_mmh = 5.0

  ! 地下貯留容量(柱状換算水深) (m)。0 より大きいこと
  gw_capacity = 0.2
/
