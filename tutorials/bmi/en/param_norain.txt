!======================================================================
!
! BMI tutorial Step 6: parameters for supplying the rain from Python
!
!   Identical to param.txt except that fn_precip and &list_precip are
!   removed and the results go to result_norain. Because no precipitation
!   is given in the parameters, rain can be supplied through BMI set.
!
!======================================================================
! System parameter settings
!======================================================================
&list_sysparam
  dt = 6.0                   ! time step (s)
  tt_c = "6 hour"            ! end time of the computation
  dt_disp_c = "30 min"       ! screen display interval
  dt_file_c = "30 min"       ! file output interval

  fn_geoinfo = "-"           ! geographic information settings file ("-" = in this file)
  fn_boundary = "-"          ! boundary condition settings file

  dir_data = "../chichibu/data_chichibu"  ! input data directory (shared with chichibu)
  dir_result = "result_norain"  ! result output directory (separate, so the param.txt results stay)

  f_input_mode = 4           ! matrix input format (1: text, 2: bil, 4: geotiff)
  f_output_mode = 5          ! matrix output format (1: text, 2: bil, 4: geotiff)
/

!======================================================================
! Geographic information settings
!======================================================================
&list_geoinfo
  dx = 200.0
  dy = 200.0

  f_ztype = 1          ! ground elevation type (0: default constant, 1: file)
  f_masktype = 1       ! domain mask (0: none, 1: catchment mask)

  f_rntype = 0         ! roughness coefficient type (0: default constant, 1: file)
  rn0 = 0.05           ! constant roughness coefficient (mountain slopes)
  rn0_rw = 0.02        ! constant roughness coefficient on channel-mask cells

  fn_z = "Chichibu_200m_filled.tif"        ! ground elevation file
  fn_mask = "Chichibu_200m_basin.tif"      ! domain mask file
  fn_rw = "Chichibu_200m_river.tif"        ! channel mask file
/

!======================================================================
! Stage-prescribed cell group: make the catchment outlet a perfect drain
!======================================================================
&list_bound_stage
  ! group 1: catchment outlet (stage below the bed = perfect drain)
  stage_cell(:,1,1) = 239, 19
  stage_cell(:,2,1) = 240, 19
  stage_eta(1) = 111.0
/
