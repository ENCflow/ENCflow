!======================================================================
!
! BMI チュートリアル: 秩父流域(200 m 格子)に 200 mm/h × 30 分の雨
!
!   tutorials/chichibu の Step 2(GeoTIFF 入力・河道マスクと粗度)に、
!   Step 6 の流域出口の排水口だけを足した構成です。入力データは
!   tutorials/chichibu/data_chichibu を共有します(複製しません)。
!   BMI でも通常実行(./encflow param.txt)でも同じ答えになります。
!
!======================================================================
! システムパラメータ設定
!======================================================================
&list_sysparam
  dt = 6.0                   ! 時間刻み (s)
  tt_c = "6 hour"            ! 計算終了時刻
  dt_disp_c = "30 min"       ! 画面表示時間刻み
  dt_file_c = "30 min"       ! ファイル出力時間刻み

  fn_geoinfo = "-"           ! 地理情報設定ファイル("-" = このファイル内)
  fn_precip  = "-"           ! 降水条件設定ファイル
  fn_boundary = "-"          ! 境界条件設定ファイル

  dir_data = "../chichibu/data_chichibu"  ! 入力データディレクトリ(chichibu と共有)
  dir_result = "result"      ! 結果出力ディレクトリ

  f_input_mode = 4           ! matrix入力形式(1:text, 2:bil, 4:geotiff)
  f_output_mode = 5          ! matrix出力形式(1:text, 2:bil, 4:geotiff)
/

!======================================================================
! 地理情報設定
!======================================================================
&list_geoinfo
  dx = 200.0
  dy = 200.0

  f_ztype = 1          ! 地盤高タイプ (0:デフォルト固定値, 1:ファイル)
  f_masktype = 1       ! 領域マスクの有無 (0:なし, 1:流域マスク)

  f_rntype = 0         ! 粗度係数タイプ (0:デフォルト固定値, 1:ファイル)
  rn0 = 0.05           ! 粗度係数固定値(山地斜面)
  rn0_rw = 0.02        ! 河道マスク部の固定粗度係数

  fn_z = "Chichibu_200m_filled.tif"        ! 地盤高ファイル名
  fn_mask = "Chichibu_200m_basin.tif"      ! 領域マスクファイル名
  fn_rw = "Chichibu_200m_river.tif"        ! 河道マスクファイル名
/

!======================================================================
! 降水条件設定(Step 6 では、このグループを持たない param_norain.txt を
! 使い、同じ雨を Python から set で与えます)
!======================================================================
&list_precip
  ! 降水データのタイプ (0:降水なし, 1:一様分布で時系列)
  prtype = 1
  ! 降水の時系列 (時刻(min) 雨量強度(mm/h))
  prval(:,1) =   0,   0
  prval(:,2) =  30, 200
  prval(:,3) =  60,   0
/

!======================================================================
! 水位規定セル群: 流域出口を河床より低い水位にして完全排水口にする
! (tutorials/chichibu Step 6 と同じ。これがないと出口に水が溜まり続ける)
!======================================================================
&list_bound_stage
  ! 群1: 流域出口(河床より低い水位で完全排水)
  stage_cell(:,1,1) = 239, 19
  stage_cell(:,2,1) = 240, 19
  stage_eta(1) = 111.0
/
